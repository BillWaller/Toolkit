95/** @brief Wait for an input event on @p target (or stdscr if NULL).
96 @param ui UI runtime context (unused — event comes from the window).
97 @param target Surface to read from, or NULL for stdscr.
98 @param ev Output UiEvent structure.
99 @param timeout_ms Milliseconds to wait; -1 = block indefinitely.
100 @return 0 on success, -1 if @p ev is NULL.
101*/
102int ui_get_event(UiRuntime *ui, UiSurface *sfc, int w, UiEvent *ev,
103 int timeout_ms) {
104 (void)ui;
105 if (!ev)
106 return -1;
107 memset(ev, 0, sizeof(*ev));
108 qiflush();
109 curs_set(1);
110 if (timeout_ms <= 0) {
111 timeout_ms = -1;
112 } else
113 wtimeout(sfc->mwin[w], timeout_ms);
114 mousemask(ALL_MOUSE_EVENTS | REPORT_MOUSE_POSITION | BUTTON_SHIFT | BUTTON_CTRL | BUTTON_ALT, NULL);
115 int ch = wgetch(sfc->mwin[w]);
116 ev->key = translate_key(ch);
117 if (ev->key == UI_KEY_CHAR) {
118 ev->ch = (uint32_t)ch;
119 } else if (ev->key == UI_KEY_MOUSE) {
120 MEVENT me;
121 if (getmouse(&me) == OK) {
122 ev->y = me.y;
123 ev->x = me.x;
124#ifdef BUTTON4_PRESSED
125 if (me.bstate & BUTTON4_PRESSED)
126 ev->mouse_action = UI_MOUSE_SCROLL_UP;
127 else
128#endif
129#ifdef BUTTON5_PRESSED
130 if (me.bstate & BUTTON5_PRESSED)
131 ev->mouse_action = UI_MOUSE_SCROLL_DOWN;
132 else
133#endif
134 if (me.bstate & BUTTON1_PRESSED)
135 ev->mouse_action = UI_MOUSE_PRESS;
136 else if (me.bstate & BUTTON1_RELEASED)
137 ev->mouse_action = UI_MOUSE_RELEASE;
138 if (wenclose(sfc->mwin[w], me.y, me.x) &&
139 wmouse_trafo(sfc->mwin[w], &me.y, &me.x, false)) {
140 ev->mouse_inside = true;
141 ev->y = me.y;
142 ev->x = me.x;
143 } else {
144 ev->mouse_inside = false;
145 }
146 }
147 }
148 return ch;
149}

151int ui_get_event_multi(UiRuntime *ui, UiSurface *sfc, int w, UiEvent *ev, int timeout_ms) {
152 (void)ui;
153 int i;
154 if (!ev)
155 return -1;
156 memset(ev, 0, sizeof(*ev));
157 keypad(sfc->mwin[w], true);
158 if (timeout_ms <= 0) {
159 timeout_ms = -1;
160 } else
161 wtimeout(sfc->mwin[w], timeout_ms);
162 mousemask(ALL_MOUSE_EVENTS | REPORT_MOUSE_POSITION | BUTTON_SHIFT | BUTTON_CTRL | BUTTON_ALT, NULL);
163 ev->chyron = -1;
164 qiflush();
165 cbreak();
166 curs_set(1);
167 int ch = wgetch(sfc->mwin[w]);
168 ev->key = translate_key(ch);
169 if (ev->key == UI_KEY_CHAR) {
170 ev->ch = (uint32_t)ch;
171 } else if (ev->key == UI_KEY_MOUSE) {
172 MEVENT me;
173 if (getmouse(&me) == OK) {
174 ev->y = me.y;
175 ev->x = me.x;
176 if (me.bstate & BUTTON4_PRESSED)
177 ev->mouse_action = UI_MOUSE_SCROLL_UP;
178 else if (me.bstate & BUTTON5_PRESSED)
179 ev->mouse_action = UI_MOUSE_SCROLL_DOWN;
180 else if (me.bstate & BUTTON1_CLICKED || me.bstate & BUTTON1_DOUBLE_CLICKED) {
181 ev->mouse_action = UI_MOUSE_PRESS;
182 for (i = 0; i < 4; i++) {
183 if (sfc->mwin[i] != NULL &&
184 wenclose(sfc->mwin[i], me.y, me.x) &&
185 wmouse_trafo(sfc->mwin[i], &me.y, &me.x, false)) {
186 ev->in_win = i;
187 ev->key = 0;
188 break;
189 }
190 }
191 ev->y = me.y;
192 ev->x = me.x;
193 ev->key = 0;
194 curs_set(0);
195 return 0;
196 }
197 }
198 }
199 curs_set(0);
200 return ch;
201}
