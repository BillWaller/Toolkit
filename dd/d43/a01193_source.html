288/** get_clr_pair
289 @brief Get color pair index for foreground and background colors
290 @ingroup color_management
291 @param fg Foreground color index
292 @param bg Background color index
293 @return Color pair index */
294int get_clr_pair(int fg, int bg) {
295 int rc, i, pfg, pbg;
296 for (i = 1; i < clr_pair_cnt; i++) {
297 extended_pair_content(i, &pfg, &pbg);
298 if (pfg == fg && pbg == bg)
299 return i;
300 }
301 if (i >= COLOR_PAIRS) {
302 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__, __LINE__ - 1);
303 ssnprintf(em1, MAXLEN - 1, "NCurses COLOR_PAIRS (%d) exceeded (%d)",
304 COLOR_PAIRS, i);
305 strerror_r(errno, em2, MAXLEN);
306 display_error(em0, em1, em2, nullptr);

308 }

310 rc = init_extended_pair(i, fg, bg);
311 if (rc == ERR)
312 return ERR;
313 }
314 if (i < COLOR_PAIRS)
315 clr_pair_cnt++;
316 return i;

345/** rgb_to_xterm256_idx
346 @brief Convert RGB color to XTerm 256 color index
347 @ingroup color_management
348 @param rgb RGB color
349 @return XTerm 256 color index
350 @details This function converts an RGB color to the nearest XTerm 256 color
351 index. It first checks if the color is a shade of gray, and if so, it uses
352 the gray ramp. Otherwise, it calculates the nearest color in the 6x6x6 color
353 cube. */
354int rgb_to_xterm256_idx(RGB *rgb) {
355 if (rgb->r == rgb->g && rgb->g == rgb->b) {
356 if (rgb->r < 8)
357 return 16;
358 if (rgb->r > 248)
359 return 231;
360 return ((rgb->r - 8) / 10) + 231;
361 } else {
362 int r_index = (rgb->r < 45) ? 0 : (rgb->r - 60) / 40 + 1;
363 int g_index = (rgb->g < 45) ? 0 : (rgb->g - 60) / 40 + 1;
364 int b_index = (rgb->b < 45) ? 0 : (rgb->b - 60) / 40 + 1;
365 return 16 + (36 * r_index) + (6 * g_index) + b_index;
366 }
367}
368/** xterm256_idx_to_rgb
369 @brief Convert XTerm 256 color index to RGB color
370 @ingroup color_management
371 @param idx XTerm 256 color index
372 @return RGB color
373 @details This function converts an XTerm 256 color index to an RGB color. It
374 first checks if the index is in the standard 16 colors, then checks if it's
375 in the 6x6x6 color cube, and finally checks if it's in the gray ramp. */
376RGB xterm256_idx_to_rgb(int idx) {
377 /** Convert XTerm 256 color index to RGB
378 @param idx - XTerm 256 color index
379 @return RGB struct */
380 RGB rgb;
381 if (idx > 255)
382 idx = 255;
383 if (idx < 0)
384 idx = 0;
385 rgb.r = rgb.g = rgb.b = 0;
386 if (idx < 16) {
387 rgb.r = StdColors[idx].r;
388 rgb.g = StdColors[idx].g;
389 rgb.b = StdColors[idx].b;
390 } else if (idx >= 16 && idx <= 231) {
391 idx -= 16;
392 rgb.r = (idx / 36) % 6 * 51;
393 rgb.g = (idx / 6) % 6 * 51;
394 rgb.b = (idx % 6) * 51;
395 } else if (idx >= 232 && idx <= 255) {
396 int gray = (idx - 232) * 11;
397 rgb.r = rgb.g = rgb.b = gray;
398 }
399 return rgb;
400}

402/** apply_gamma
403 @brief Apply gamma correction to RGB color
404 @ingroup color_management
405 @param rgb Pointer to RGB color
406 @details This function modifies the RGB color in place. It applies gamma
407 correction to the RGB color based on the gamma values set in the SIO struct.
408 If the color is a shade of gray, it applies the gray gamma correction.
409 Otherwise, it applies the individual red, green, and blue gamma corrections.
410 */
411void apply_gamma(RGB *rgb) {
412 if (rgb->r == rgb->g && rgb->r == rgb->b) {
413 if (GRAY_GAMMA > 0.0f && GRAY_GAMMA != 1.0f) {
414 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / GRAY_GAMMA) * 255.0f);
415 rgb->g = rgb->r;
416 rgb->b = rgb->r;
417 }
418 return;
419 }
420 if (rgb->r != 0 && RED_GAMMA > 0.0f && RED_GAMMA != 1.0f)
421 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / RED_GAMMA) * 255.0f);
422 if (rgb->g != 0 && GREEN_GAMMA > 0.0f && GREEN_GAMMA != 1.0f)
423 rgb->g = (int)(pow((rgb->g / 255.0f), 1.0f / GREEN_GAMMA) * 255.0f);
424 if (rgb->b != 0 && BLUE_GAMMA > 0.0f && BLUE_GAMMA != 1.0f)
425 rgb->b = (int)(pow((rgb->b / 255.0f), 1.0f / BLUE_GAMMA) * 255.0f);
426}

584/** mb_to_cc
585 @brief Convert multibyte string to complex character array
586 @ingroup Chyron
587 @param cmplx_buf Output buffer for complex characters
588 @param str Input multibyte string
589 @param attr Attributes to apply to the complex characters
590 @param cpx Color pair index for the complex characters
591 @param pos Pointer to current position in the output buffer, updated as
592 characters are added
593 @param maxlen Maximum length of the output buffer
594 @return Number of bytes processed from the input string
595 @details This function converts a multibyte string to an array of complex
596 characters (cchar_t) that can be used with NCurses functions. It handles
597 multibyte characters and applies the specified color pair to each character.
598 The pos parameter is updated to reflect the current position in the output
599 buffer, and the function ensures that it does not exceed the maximum length.
600*/
601int mb_to_cc(cchar_t *cmplx_buf, char *str, attr_t attr, int cpx, int *pos,
602 int maxlen) {
603 int i = 0, len = 0;
604 const char *s;
605 cchar_t cc = {0};
606 wchar_t wstr[2] = {L'\0', L'\0'};
607 mbstate_t mbstate;
608 memset(&mbstate, 0, sizeof(mbstate));
609 attr = WA_NORMAL;
610 if (*pos >= maxlen - 1)
611 return 0;
612 while (str[i] != '\0') {
613 s = &str[i];
614 len = mbrtowc(wstr, s, MB_CUR_MAX, &mbstate);
615 if (len <= 0) {
616 wstr[0] = L'?';
617 wstr[1] = L'\0';
618 len = 1;
619 }
620 wstr[1] = L'\0';
621 if (*pos >= maxlen - 1)
622 break;
623 if (setcchar(&cc, wstr, attr, cpx, nullptr) != ERR) {
624 if (len > 0 && (*pos + len) < MAXLEN - 1)
625 cmplx_buf[(*pos)++] = cc;
626 }
627 i += len;
628 }
629 wstr[0] = L'\0';
630 wstr[1] = L'\0';
631 setcchar(&cc, wstr, attr, cpx, nullptr);
632 cmplx_buf[*pos] = cc;
633 return *pos;
634}

745/** @brief Converts a Unicode code point to a UTF-8 encoded string.
746 @ingroup utility_functions
747 @param cp - Unicode code point to convert
748 @param buffer - buffer to receive UTF-8 encoded string (must be at least 4 bytes)
749 @returns number of bytes written to buffer, or 0 if cp is invalid
750 @details This function encodes the given Unicode code point into its UTF-8 representation and stores it in the provided buffer. The caller must ensure that the buffer has enough space to hold the resulting UTF-8 string (up to 4 bytes for code points up to U+10FFFF). If the code point is invalid (e.g., greater than U+10FFFF), this function returns 0 and does not modify the buffer. */
751int wccp_to_str(wchar_t cp, uint8_t *buffer) {
752 if (cp <= 0x7F) {
753 // 1-byte sequence: 0xxxxxxx
754 buffer[0] = (uint8_t)cp;
755 return 1;
756 } else if (cp <= 0x7FF) {
757 // 2-byte sequence: 110xxxxx 10xxxxxx
758 buffer[0] = (uint8_t)(0xC0 | ((cp >> 6) & 0x1F));
759 buffer[1] = (uint8_t)(0x80 | (cp & 0x3F));
760 return 2;
761 } else if (cp <= 0xFFFF) {
762 // 3-byte sequence: 1110xxxx 10xxxxxx 10xxxxxx
763 buffer[0] = (uint8_t)(0xE0 | ((cp >> 12) & 0x0F));
764 buffer[1] = (uint8_t)(0x80 | ((cp >> 6) & 0x3F));
765 buffer[2] = (uint8_t)(0x80 | (cp & 0x3F));
766 return 3;
767 } else if (cp <= 0x10FFFF) {
768 // 4-byte sequence: 11110xxx 10xxxxxx 10xxxxxx 10xxxxxx
769 buffer[0] = (uint8_t)(0xF0 | ((cp >> 18) & 0x07));
770 buffer[1] = (uint8_t)(0x80 | ((cp >> 12) & 0x3F));
771 buffer[2] = (uint8_t)(0x80 | ((cp >> 6) & 0x3F));
772 buffer[3] = (uint8_t)(0x80 | (cp & 0x3F));
773 return 4;
774 }
775 return 0; // Invalid Unicode code point
776}

1562 set keys, separated by " | ". The end_pos values for each key are set to

1579 cx = chyron->cmplx_buf;
