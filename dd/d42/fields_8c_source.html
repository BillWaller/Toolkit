518 if (is_valid_time(Time.hh, Time.mm, Time.ss))

520 break;
521 case FF_APR:
522 sscanf(input_s, "%lf", &double_n);
523 sprintf(accept_s, "%lf", double_n);

525 right_justify(display_s, fl);
526 break;
527 default:
528 Perror("form_fmt_field() invalid format");
529 break;
530 }
531 strnz(accept_s, fl);
532 left_justify(accept_s);

607bool is_valid_date(int yyyy, int mm, int dd) {
608 if (yyyy < 1 || mm < 1 || mm > 12 || dd < 1)
609 return false;
610 int days_in_month[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
611 if ((yyyy % 4 == 0 && yyyy % 100 != 0) || (yyyy % 400 == 0))
612 days_in_month[2] = 29;
613 if (dd > days_in_month[mm])
614 return false;
615 return true;
616}
