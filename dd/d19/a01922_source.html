94/** @brief Draw a text string at a specific position in a surface window, filling the remaining space with spaces if the string is shorter than n.
95 @param s The surface to modify.
96 @param w The window index within the surface.
97 @param y The y-coordinate (row) to draw the text.
98 @param x The x-coordinate (column) to draw the text.
99 @param text The text string to draw.
100 @param n The number of characters to draw, padding with spaces if necessary.
101 @return 0 on success, -1 on error.
102*/
103int ui_draw_text_fill(UiSurface *s, ss_t w, uint y, uint x, const char *text, int n) {
104 if (!s || !text)
105 return -1;
106 uint l = strlen(text);
107 if (l < (uint)n) {
108 char *tmp_str = (char *)malloc(n + 1);
109 if (!tmp_str)
110 return -1;
111 strcpy(tmp_str, text);
112 for (int i = l; i < (int)n; i++) {
113 tmp_str[i] = ' ';
114 }
115 tmp_str[n] = '\0';
116 mvwaddstr(s->mwin[w], y, x, tmp_str);
117 free(tmp_str);
118 } else {
119 mvwaddnstr(s->mwin[w], y, x, text, (int)n);
120 }
121 return 0;
122}

195/** @brief Draw a text string at a specific position in a surface window, filling the remaining space with spaces if the string is shorter than m.
196 @param s The surface to modify.
197 @param w The window index within the surface.
198 @param y The y-coordinate (row) to draw the text.
199 @param x The x-coordinate (column) to draw the text.
200 @param str The text string to draw.
201 @param m The number of characters to draw, padding with spaces if necessary.
202 @return 0 on success, -1 on error.
203*/
204int ui_mvwaddstr_fill(UiSurface *s, ss_t w, uint y, uint x, const char *str, int m) {
205 char *d, *e;
206 uint maxy, maxx;
207 char tmp_str[MAXLEN];
208 getmaxyx(s->mwin[w], maxy, maxx);
209 y = min(y, maxy);
210 m = min(m, maxx);
211 m = min(m, (uint)MAXLEN - 1);
212 e = d = tmp_str;
213 e += m;
214 while (d < e) {
215 if (*str == '\0' || *str == '\n')
216 *d++ = ' ';
217 else
218 *d++ = *str++;
219 }
220 *d = '\0';
221 m = strlen(tmp_str);
222 mvwaddnstr(s->mwin[w], y, x, tmp_str, m);
223 return 0;
224}
