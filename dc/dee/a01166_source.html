278/** @brief Initializes pick interface, calculates window size and position, and
279 enters picker loop
280 * @ingroup pick_engine
281 @param init Pointer to Init structure containing pick information
282 @return Count of selected objects on success, -1 if user cancels
283 @details Initializes key command strings for chyron display and calculates
284 pick window size and position based on terminal size and pick parameters. Opens
285 pick window and displays first page of objects. Enters picker loop to handle
286 user input and interactions. If user cancels selection, returns -1. If user
287 accepts selection, returns count of selected objects. */
288int pick_engine(Init *init) {

291 int tbl_max_cols, pg_max_objs;
292 bool f_processed = false;

294 getmaxyx(stdscr, maxy, maxx);
295 /** Calculate pick window size and position based on terminal size and pick
296 parameters */
297 win_maxy = min(((maxy * 8) / 10), (maxy - pick->begy - 1));
298 win_maxx = min(((maxx * 9) / 10), (maxx - pick->begx - 1));
299 pick->tbl_col_width = max(pick->tbl_col_width, 4);
300 pick->tbl_col_width = min(pick->tbl_col_width, win_maxx - 2);
301 if (pick->d_cnt <= win_maxy) {
302 pick->tbl_lines = pick->d_cnt;
303 pick->tbl_cols = 1;
304 } else {
305 tbl_max_cols = (win_maxx / (pick->tbl_col_width + 1));
306 pg_max_objs = win_maxy * tbl_max_cols;
307 if (pick->d_cnt > pg_max_objs)
308 pick->tbl_cols = tbl_max_cols;
309 else
310 pick->tbl_cols = pick->d_cnt / win_maxy;
311 pick->tbl_lines = pick->d_cnt / tbl_max_cols;
312 }
313 pick->tbl_pages = (pick->tbl_lines / (win_maxy - 1)) + 1;
314 pick->pg_lines = (pick->tbl_lines / pick->tbl_pages);
315 pick->win_lines = pick->pg_lines + 1;
316 pick->tbl_page = 0;
317 if (pick->begy == 0)
318 pick->begy = (LINES - pick->win_lines) / 5;
319 else if (pick->begy + pick->win_lines > LINES - 4)
320 pick->begy = LINES - pick->win_lines - 2;
321 pick->win_width = (pick->tbl_col_width + 1) * pick->tbl_cols;
322
323 pick->chyron = new_chyron();
324 set_chyron_key(pick->chyron, 1, "F1 Help", KEY_F(1));
325 set_chyron_key(pick->chyron, 2, "F9 Cancel", KEY_F(9));
326 set_chyron_key(pick->chyron, 3, "F10 Accept", KEY_F(10));
327 set_chyron_key(pick->chyron, 4, "Sp Toggle", KEY_F(13));
328 set_chyron_key(pick->chyron, 5, "Tab Edit", '\t');
329 if (pick->tbl_pages > 1) {
330 set_chyron_key(pick->chyron, 11, "PgUp", KEY_PPAGE);
331 set_chyron_key(pick->chyron, 2, "PgDn", KEY_NPAGE);
332 }
333 compile_chyron(pick->chyron);
334 if (pick->chyron->l > win_maxx)
335 pick->chyron->l = strnz(pick->chyron->s, win_maxx);
336 if (pick->win_width < pick->chyron->l)
337 pick->win_width = pick->chyron->l;
338 pick->win_width = max(pick->win_width, pick->chyron->l) + 5;
339 if (pick->begx + pick->win_width > COLS - 2)
340 pick->begx = COLS - pick->win_width - 2;
341 else if (pick->begx == 0)
342 pick->begx = (COLS - pick->win_width) / 2;
343
344 rc = open_pick_win(init);
345#ifdef DEBUG_IMMEDOK
346 immedok(pick->box, true);
347 immedok(pick->win, true);
348 immedok(pick->win2, true);
349#endif
350 if (rc)
351 return (rc);
352
353 /** Enter picker loop to handle user input and interactions */
354 pick->d_idx = 0;
355 pick->x = 1;
356 char field[MAXLEN]; /**< Buffer for user input in the field */
357 field[0] = '\0';
358 display_page(pick);
359 do {
360 rc = picker(init, field);
361 if (rc == KEY_F(9))
362 break;
363 if (rc == -1)
364 break;
365 else {
366 if (pick->select_cnt > 0) {
367 if (pick->f_out_spec && pick->out_spec[0]) {
368 output_objects(pick);
369 f_processed = true;
370 }
371 if (pick->f_cmd && pick->cmd[0]) {
372 exec_objects(init);
373 f_processed = true;
374 }
375 if (pick->f_read_theme) {
376 read_theme(init);
377 f_processed = true;
378 }
379 if (f_processed) {
380 mvwaddstr(pick->win2, 0, 0, "Selection Processed");
381 wclrtoeol(pick->win2);
382 wrefresh(pick->win2);
383 }
384 }
385 }
386 deselect_object(pick);
387 } while (1);
388 destroy_chyron(pick->chyron);
389 return (rc);
390}

415/** @brief Displays current page of objects in pick window
416 * @ingroup pick_engine
417 @param pick Pointer to Pick structure containing objects and display

439 }
440 pick->d_idx -= 1;

600 @details Parses command string and appends selected objects as arguments to

612 memory leaks.

616 proceeding and restores terminal settings
617 If the command is a pager or editor, it is executed within the pick
618 interface using popup_view instead of execvp
619 The base name of the command is extracted to check if it is a pager or
620 editor
621 If the command is a pager or editor, the pick interface is used to
622 display the command output instead of executing it in a separate terminal
623 This allows the user to view the command output without leaving the pick
624 interface and provides a more seamless user experience.
625 If the command is not a pager or editor, it is executed in a separate
626 terminal and the pick interface is restored after execution

628 needed, instead the popup_view function can be used to display the output

747 free(eargv[i]);

749 }

778 restore_wins();
