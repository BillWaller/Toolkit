354 int rc, i, pfg, pbg;
355 for (i = 1; i < clr_pair_cnt; i++) {
356 extended_pair_content(i, &pfg, &pbg);
357 if (pfg == fg && pbg == bg)
358 return i;
359 }
360 if (i >= COLOR_PAIRS) {
361 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__, __LINE__ - 1);
362 ssnprintf(em1, MAXLEN - 1, "NCurses COLOR_PAIRS (%d) exceeded (%d)",
363 COLOR_PAIRS, i);
364 strerror_r(errno, em2, MAXLEN);
365 display_error(em0, em1, em2, nullptr);
366 return (EXIT_FAILURE);
367 }
368 if (i < COLOR_PAIRS) {
369 rc = init_extended_pair(i, fg, bg);
370 if (rc == ERR)
371 return ERR;
372 }
373 if (i < COLOR_PAIRS)
374 clr_pair_cnt++;
375 return i;

413int rgb_to_xterm256_idx(RGB *rgb) {
414 if (rgb->r == rgb->g && rgb->g == rgb->b) {
415 if (rgb->r < 8)
416 return 16;
417 if (rgb->r > 248)
418 return 231;
419 return ((rgb->r - 8) / 10) + 231;
420 } else {
421 int r_index = (rgb->r < 45) ? 0 : (rgb->r - 60) / 40 + 1;
422 int g_index = (rgb->g < 45) ? 0 : (rgb->g - 60) / 40 + 1;
423 int b_index = (rgb->b < 45) ? 0 : (rgb->b - 60) / 40 + 1;
424 return 16 + (36 * r_index) + (6 * g_index) + b_index;
425 }
426}

435RGB xterm256_idx_to_rgb(int idx) {
439 RGB rgb;
440 if (idx > 255)
441 idx = 255;
442 if (idx < 0)
443 idx = 0;
444 rgb.r = rgb.g = rgb.b = 0;
445 if (idx < 16) {
446 rgb.r = StdColors[idx].r;
447 rgb.g = StdColors[idx].g;
448 rgb.b = StdColors[idx].b;
449 } else if (idx >= 16 && idx <= 231) {
450 idx -= 16;
451 rgb.r = (idx / 36) % 6 * 51;
452 rgb.g = (idx / 6) % 6 * 51;
453 rgb.b = (idx % 6) * 51;
454 } else if (idx >= 232 && idx <= 255) {
455 int gray = (idx - 232) * 11;
456 rgb.r = rgb.g = rgb.b = gray;
457 }
458 return rgb;
459}

470void apply_gamma(RGB *rgb) {
471 if (rgb->r == rgb->g && rgb->r == rgb->b) {
472 if (GRAY_GAMMA > 0.0f && GRAY_GAMMA != 1.0f) {
473 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / GRAY_GAMMA) * 255.0f);
474 rgb->g = rgb->r;
475 rgb->b = rgb->r;
476 }
477 return;
478 }
479 if (rgb->r != 0 && RED_GAMMA > 0.0f && RED_GAMMA != 1.0f)
480 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / RED_GAMMA) * 255.0f);
481 if (rgb->g != 0 && GREEN_GAMMA > 0.0f && GREEN_GAMMA != 1.0f)
482 rgb->g = (int)(pow((rgb->g / 255.0f), 1.0f / GREEN_GAMMA) * 255.0f);
483 if (rgb->b != 0 && BLUE_GAMMA > 0.0f && BLUE_GAMMA != 1.0f)
484 rgb->b = (int)(pow((rgb->b / 255.0f), 1.0f / BLUE_GAMMA) * 255.0f);
485}

814int wccp_to_str(wchar_t cp, uint8_t *buffer) {
815 if (cp <= 0x7F) {
816 // 1-byte sequence: 0xxxxxxx
817 buffer[0] = (uint8_t)cp;
818 return 1;
819 } else if (cp <= 0x7FF) {
820 // 2-byte sequence: 110xxxxx 10xxxxxx
821 buffer[0] = (uint8_t)(0xC0 | ((cp >> 6) & 0x1F));
822 buffer[1] = (uint8_t)(0x80 | (cp & 0x3F));
823 return 2;
824 } else if (cp <= 0xFFFF) {
825 // 3-byte sequence: 1110xxxx 10xxxxxx 10xxxxxx
826 buffer[0] = (uint8_t)(0xE0 | ((cp >> 12) & 0x0F));
827 buffer[1] = (uint8_t)(0x80 | ((cp >> 6) & 0x3F));
828 buffer[2] = (uint8_t)(0x80 | (cp & 0x3F));
829 return 3;
830 } else if (cp <= 0x10FFFF) {
831 // 4-byte sequence: 11110xxx 10xxxxxx 10xxxxxx 10xxxxxx
832 buffer[0] = (uint8_t)(0xF0 | ((cp >> 18) & 0x07));
833 buffer[1] = (uint8_t)(0x80 | ((cp >> 12) & 0x3F));
834 buffer[2] = (uint8_t)(0x80 | ((cp >> 6) & 0x3F));
835 buffer[3] = (uint8_t)(0x80 | (cp & 0x3F));
836 return 4;
837 }
838 return 0; // Invalid Unicode code point
839}
