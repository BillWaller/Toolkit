26/** * @brief Create a new UI surface.
27 * @ingroup ui_ncurses
28 * @param ui The UI runtime instance.
29 * @param parent The parent surface, or NULL for a top-level surface.
30 * @param rect The rectangle defining the position and size of the surface.
31 * @return A pointer to the newly created UiSurface, or NULL on failure.
32 */
33UiSurface *ui_surface_new(UiRuntime *ui, UiSurface *parent, UiRect rect) {
34 UiSurface *s = calloc(1, sizeof(*s));
35 if (!s)
36 return NULL;
37
38 s->runtime = ui;
39 s->parent = parent;
40 s->y = rect.y;
41 s->x = rect.x;
42 s->rows = rect.rows;
43 s->cols = rect.cols;
44
45 if (parent && parent->win) {
46 s->win = derwin(parent->win, rect.rows, rect.cols, rect.y, rect.x);
47 } else {
48 s->win = newwin(rect.rows, rect.cols, rect.y, rect.x);
49 }
50
51 if (!s->win) {
52 free(s);
53 return NULL;
54 }
55
56 s->pan = new_panel(s->win);
57 if (!s->pan) {
58 delwin(s->win);
59 free(s);
60 return NULL;
61 }
62
63 return s;
64}

171/** @brief Initialize NCUrses
172 * @ingroup ui_ncurses
173 * @param cfg The UI runtime instance.
174 * @return 0 on success, or -1 on failure.
175 */
176UiRuntime *ui_init(const UiConfig *cfg) {
177 UiRuntime *ui = calloc(1, sizeof(*ui));
178 if (!ui)
179 return NULL;
180 char tty_name[XLEN];
181
182 // Get the name of the terminal device
183 if (ttyname_r(STDERR_FILENO, tty_name, sizeof(tty_name)) != 0) {
184 Perror("ui_init: ttyname_r() failed");
185 exit(EXIT_FAILURE);
186 }
187 // open the terminal device for NCurses input and output
188 tty_fp = fopen(tty_name, "r+");
189 if (tty_fp == nullptr) {
190 Perror("ui_init: fopen() failed");
191 exit(EXIT_FAILURE);
192 }
193 // newterm() allows us to specify a tty device for NCurses input and
194 // output. nullptr for the first argument means to use the default terminal type.
195 screen = newterm(nullptr, tty_fp, tty_fp);
196 if (screen == nullptr) {
197 Perror("ui_init: newterm() failed");
198 exit(EXIT_FAILURE);
199 }
200 set_term(screen);
201 f_curses_open = true;
202 if (!has_colors()) {
203 ui_shutdown(ui);
204 Perror("Terminal does not support colors");
205 exit(EXIT_FAILURE);
206 }
207 if (!can_change_color()) {
208 ui_shutdown(ui);
209 Perror("Terminal cannot change colors");
210 exit(EXIT_FAILURE);
211 }
212 start_color();
213 use_default_colors();
214 cbreak();
215 noecho();
216 keypad(stdscr, TRUE);
217 if (cfg) {
218 ui->mouse_enabled = cfg->enable_mouse;
219 ui->alt_screen = cfg->enable_alt_screen;
220 ui->cursor_visible = cfg->cursor_visible;
221 } else {
222 ui->cursor_visible = true;
223 }
224 if (ui->mouse_enabled)
225 mousemask(ALL_MOUSE_EVENTS | REPORT_MOUSE_POSITION, NULL);
226 curs_set(ui->cursor_visible ? 1 : 0);
227 getmaxyx(stdscr, ui->rows, ui->cols);
228 for (int i = 0; i < MAXWIN; i++)
229 win_flags[i] = 0;
230 win_ptr = -1;
231 return ui;
232}
