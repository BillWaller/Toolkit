363/** @brief Convert a multibyte string to a complex character array. */
364int mbstr_to_cc(char *in_str, cchar_t *cmplx_buf_s) {
365 char ansi_tok[MAXLEN];
366 int i = 0, j = 0, x = 0;
367 int len = 0;
368 int tab_stop = 4;
369 int tab_spaces = 0;
370 int char_width;
371 attr_t attr = WA_NORMAL;
372 int cpx = cp_nt;
373 cchar_t cc = {0};
374 wchar_t wstr[2] = {L'\0', L'\0'};
375 cchar_t *cmplx_buf = cmplx_buf_s;
376 mbstate_t mbstate;
377 memset(&mbstate, 0, sizeof(mbstate));
378 while (in_str[i] != '\0') { // line
379 while (1) { // ANSI SGR, Character, and Word
380 if (in_str[i] == '\033') { // ANSI SGR
381 if (in_str[i + 1] == '[') {
382 len = strcspn(&in_str[i], "mK ") + 1;
383 memcpy(ansi_tok, &in_str[i], len + 1);
384 ansi_tok[len] = '\0';
385 if (ansi_tok[0] == '\0') {
386 if (i + 2 < MAXLEN)
387 i += 2;
388 continue;
389 }
390 if (len == 0 || in_str[i + len - 1] == ' ') {
391 i += 2;
392 continue;
393 } else if (in_str[i + len - 1] == 'K') {
394 i += len;
395 continue;
396 }
397 parse_ansi(ansi_tok, &attr, &cpx);
398 i += len;
399 } else {
400 i++;
401 continue;
402 }
403 } else { // Characters
404 if (in_str[i] == ' ') {
405 wstr[0] = L' ';
406 wstr[1] = L'\0';
407 setcchar(&cc, wstr, attr, cpx, nullptr);
408 cmplx_buf[j++] = cc;
409 x++;
410 i++;
411 continue;
412 }
413 if (in_str[i] == '\0') {
414 break;
415 }
416 if (in_str[i] == '\t') {
417 tab_spaces = tab_stop - j % tab_stop;
418 wstr[0] = L' ';
419 wstr[1] = L'\0';
420 setcchar(&cc, wstr, attr, cpx, nullptr);
421 for (int z = 0; z < tab_spaces; z++) {
422 cmplx_buf[j++] = cc;
423 x++;
424 }
425 i++;
426 continue;
427 }
428 wstr[1] = L'\0';
429 len = mbrtowc(wstr, &in_str[i], MB_CUR_MAX, &mbstate);
430 if (len <= 0) {
431 wstr[0] = L'?';
432 wstr[1] = L'\0';
433 len = 1;
434 }
435 char_width = wcwidth(wstr[0]);
436 x += char_width;
437 setcchar(&cc, wstr, attr, cpx, nullptr);
438 cmplx_buf[j++] = cc;
439 i += len;
440 continue;
441 } // END Character and Word
442 } // END WHILE - ANSI SGR, Character, and Word
443 } // END WHILE - line
444 wstr[0] = '\0';
445 wstr[1] = '\0';
446 setcchar(&cc, wstr, WA_NORMAL, cpx, nullptr);
447 cmplx_buf[j] = cc;
448 return x;
449}
450/** @brief Parse an ANSI SGR sequence and update attributes and color pair index. */
451void parse_ansi(char *ansi_str, attr_t *attr, int *cpx) {
452 char *tok;
453 char t0, t1;
454 char tstr[3];
455 int len, x_idx;
456 int fg, bg;
457 int fg_clr, bg_clr;
458 char *ansi_p = ansi_str + 2;
459 extended_pair_content(*cpx, &fg_clr, &bg_clr);
460 fg = fg_clr;
461 bg = bg_clr;
462 RGB rgb;
463 tok = strtok((char *)ansi_p, ";m");
464 bool a_toi_error = false;
465 while (1) {
466 if (tok == nullptr || *tok == '\0')
467 break;
468 len = strlen(tok);
469 if (len == 2) {
470 t0 = tok[0];
471 t1 = tok[1];
472 if (t0 == '3' || t0 == '4') {
473 if (t1 == '8') {
474 tok = strtok(nullptr, ";m");
475 if (tok != nullptr) {
476 if (*tok == '5') {
477 tok = strtok(nullptr, ";m");
478 if (tok != nullptr) {
479 x_idx = a_toi(tok, &a_toi_error);
480 rgb = xterm256_idx_to_rgb(x_idx);
481 }
482 } else if (*tok == '2') {
483 tok = strtok(nullptr, ";m");
484 rgb.r = a_toi(tok, &a_toi_error);
485 tok = strtok(nullptr, ";m");
486 rgb.g = a_toi(tok, &a_toi_error);
487 tok = strtok(nullptr, ";m");
488 rgb.b = a_toi(tok, &a_toi_error);
489 }
490 }
491 if (t0 == '3')
492 fg_clr = rgb_to_curses_clr(&rgb);
493 else if (t0 == '4')
494 bg_clr = rgb_to_curses_clr(&rgb);
495 } else if (t1 == '9') {
496 if (t0 == '3')
497 fg_clr = CLR_NT_FG;
498 else if (t0 == '4')
499 bg_clr = CLR_NT_BG;
500 } else if (t1 >= '0' && t1 <= '7') {
501 if (t0 == '3') {
502 tstr[0] = t1;
503 tstr[1] = '\0';
504 x_idx = a_toi(tstr, &a_toi_error);
505 rgb = xterm256_idx_to_rgb(x_idx);
506 fg_clr = rgb_to_curses_clr(&rgb);
507 } else if (t0 == '4') {
508 tstr[0] = t1;
509 tstr[1] = '\0';
510 x_idx = a_toi(tstr, &a_toi_error);
511 rgb = xterm256_idx_to_rgb(x_idx);
512 bg_clr = rgb_to_curses_clr(&rgb);
513 }
514 }
515 } else if (t0 == '0') {
516 *tok = t1;
517 len = 1;
518 }
519 }
520 if (len == 1) {
521 if (*tok == '0') {
522 *attr = WA_NORMAL;
523 fg_clr = CLR_NT_FG;
524 bg_clr = CLR_NT_BG;
525 } else {
526 switch (a_toi(tok, &a_toi_error)) {
527 case 1:
528 *attr |= WA_BOLD;
529 break;
530 case 2:
531 *attr |= WA_DIM;
532 break;
533 case 3:
534 *attr |= WA_ITALIC;
535 break;
536 case 4:
537 *attr |= WA_UNDERLINE;
538 break;
539 case 5:
540 *attr |= WA_BLINK;
541 break;
542 case 7:
543 *attr |= WA_REVERSE;
544 break;
545 case 8:
546 *attr |= WA_INVIS;
547 break;
548 default:
549 break;
550 }
551 }
552 } else if (len == 0) {
553 *attr = WA_NORMAL;
554 fg_clr = CLR_NT_FG;
555 bg_clr = CLR_NT_BG;
556 }
557 tok = strtok(nullptr, ";m");
558 }
559 if (!a_toi_error && (fg_clr != fg || bg_clr != bg)) {
560 clr_pair_idx = get_clr_pair(fg_clr, bg_clr);
561 *cpx = clr_pair_idx;
562 }
563 return;
564}
