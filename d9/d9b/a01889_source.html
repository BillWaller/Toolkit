666static inline int unicode_to_utf8_gcluster(uint32_t cp, GCluster *g) {
667 if (cp <= 0x7F) {
668 g->c[0] = (char)cp;
669 g->c[1] = '\0';
670 return 1;
671 } else if (cp <= 0x7FF) {
672 g->c[0] = (char)(0xC0 | (cp >> 6));
673 g->c[1] = (char)(0x80 | (cp & 0x3F));
674 g->c[2] = '\0';
675 return 2;
676 } else if (cp <= 0xFFFF) {
677 g->c[0] = (char)(0xE0 | (cp >> 12));
678 g->c[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
679 g->c[2] = (char)(0x80 | (cp & 0x3F));
680 g->c[3] = '\0';
681 return 3;
682 } else if (cp <= 0x10FFFF) {
683 g->c[0] = (char)(0xF0 | (cp >> 18));
684 g->c[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
685 g->c[2] = (char)(0x80 | ((cp >> 6) & 0x3F));
686 g->c[3] = (char)(0x80 | (cp & 0x3F));
687 return 4;
688 }
689 return 0;
690}
