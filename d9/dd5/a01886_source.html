302/** ui_mbstr_to_wcstr
303 @brief Convert a multibyte string to a wide character string
304 @ingroup window_support
305 @param mb_str Input multibyte string
306 @return Pointer to newly allocated wide character string, or nullptr on error
307 @details This function converts a null-terminated multibyte string to a wide
308 character string. It first calculates the required length for the wide
309 character string, allocates memory for it, and then performs the conversion.
310 The caller is responsible for freeing the allocated memory. If the conversion
311 fails, it returns nullptr. */
312wchar_t *ui_mbstr_to_wcstr(const char *mb_str) {
313 const char *src_ptr = mb_str;
314 mbstate_t state = {};
315 size_t wc_count = mbsrtowcs(NULL, &src_ptr, 0, &state);
316 if (wc_count == (size_t)-1)
317 return nullptr;
318 src_ptr = mb_str;
319 wmemset((wchar_t *)&state, 0, sizeof(state) / sizeof(wchar_t));
320 wchar_t *wc_str = malloc((wc_count + 1) * sizeof(wchar_t));
321 if (!wc_str)
322 return nullptr;
323 mbsrtowcs(wc_str, &src_ptr, wc_count + 1, &state);
324 return wc_str;
325}

993/** ui_rgb_to_xterm256_idx
994 @brief Convert RGB color to XTerm 256 color index
995 @ingroup color_management
996 @param rgb RGB color
997 @return XTerm 256 color index
998 @details This function converts an RGB color to the nearest XTerm 256 color
999 index. It first checks if the color is a shade of gray, and if so, it uses
1000 the gray ramp. Otherwise, it calculates the nearest color in the 6x6x6 color
1001 cube. */
1002uint ui_rgb_to_xterm256_idx(RGB *rgb) {
1003 if (rgb->r == rgb->g && rgb->g == rgb->b) {
1004 if (rgb->r < 8)
1005 return 16;
1006 if (rgb->r > 248)
1007 return 231;
1008 return ((rgb->r - 8) / 10) + 231;
1009 } else {
1010 uint r_index = (rgb->r < 45) ? 0 : (rgb->r - 60) / 40 + 1;
1011 uint g_index = (rgb->g < 45) ? 0 : (rgb->g - 60) / 40 + 1;
1012 uint b_index = (rgb->b < 45) ? 0 : (rgb->b - 60) / 40 + 1;
1013 return 16 + (36 * r_index) + (6 * g_index) + b_index;
1014 }
1015}
1016/** ui_xterm256_idx_to_rgb
1017 @brief Convert XTerm 256 color index to RGB
1018 @ingroup color_management
1019 @param idx XTerm 256 color index
1020 @return RGB color
1021 @details This function converts an XTerm 256 color index to an RGB color. It
1022 first checks if the index is in the standard 16 colors, then checks if it's
1023 in the 6x6x6 color cube, and finally checks if it's in the gray ramp. */
1024RGB ui_xterm256_idx_to_rgb(uint idx) {
1025 /** Convert XTerm 256 color index to RGB
1026 @param idx - XTerm 256 color index
1027 @return RGB struct */
1028 RGB rgb;
1029 if (idx > 255)
1030 idx = 255;
1031 rgb.r = rgb.g = rgb.b = 0;
1032 if (idx < 16) {
1033 rgb.r = std_color[idx].r;
1034 rgb.g = std_color[idx].g;
1035 rgb.b = std_color[idx].b;
1036 } else if (idx >= 16 && idx <= 231) {
1037 idx -= 16;
1038 rgb.r = (idx / 36) % 6 * 51;
1039 rgb.g = (idx / 6) % 6 * 51;
1040 rgb.b = (idx % 6) * 51;
1041 } else if (idx >= 232 && idx <= 255) {
1042 uint gray = (idx - 232) * 11;
1043 rgb.r = rgb.g = rgb.b = gray;
1044 }
1045 return rgb;
1046}

1048/** ui_apply_gamma
1049 @brief Apply gamma correction to RGB color
1050 @ingroup color_management
1051 @param rgb Pointer to RGB color
1052 @details This function modifies the RGB color in place. It applies gamma
1053 correction to the RGB color based on the gamma values set in the SIO struct.
1054 If the color is a shade of gray, it applies the gray gamma correction.
1055 Otherwise, it applies the individual red, green, and blue gamma corrections.
1056 */
1057void ui_apply_gamma(RGB *rgb) {
1058 if (rgb->r == rgb->g && rgb->r == rgb->b) {
1059 if (GRAY_GAMMA > 0.0f && GRAY_GAMMA != 1.0f) {
1060 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / GRAY_GAMMA) * 255.0f);
1061 rgb->g = rgb->r;
1062 rgb->b = rgb->r;
1063 }
1064 return;
