94static error_t parse_opt(int key, char *arg, struct argp_state *state) {
95 struct lf_opts *lf_opts = state->input;
96 int i = 0;
97 switch (key) {
98 case 'a':
99 lf_opts->flags |= LF_ALL;
100 break;
101 case 'd':
102 lf_opts->max_depth = atoi(arg);
103 break;
104 case 'e':
105 lf_opts->exclude_p = arg;
106 lf_opts->flags |= LF_EXC_REGEX;
107 break;
108 case 'h':
109 lf_opts->help = true;
110 break;
111 case 'i':
112 lf_opts->flags |= LF_ICASE;
113 break;
114 case 't':
115 lf_opts->file_types_p = arg;
116 while (lf_opts->file_types_p[i]) {
117 switch (lf_opts->file_types_p[i++]) {
118 case 'b':
119 lf_opts->flags |= FT_BLK << 8;
120 break;
121 case 'c':
122 lf_opts->flags |= FT_CHR << 8;
123 break;
124 case 'd':
125 lf_opts->flags |= FT_DIR << 8;
126 break;
127 case 'p':
128 lf_opts->flags |= FT_FIFO << 8;
129 break;
130 case 'l':
131 lf_opts->flags |= FT_LNK << 8;
132 break;
133 case 'f':
134 lf_opts->flags |= FT_REG << 8;
135 break;
136 case 's':
137 lf_opts->flags |= FT_SOCK << 8;
138 break;
139 case 'u':
140 lf_opts->flags |= FT_UNKNOWN << 8;
141 break;
142 default:
143 break;
144 }
145 }
146 break;
147 case 'V':
148 lf_opts->f_version = true;
149 break;
150 case 'v':
151 lf_opts->f_verbose = true;
152 break;
153 case ARGP_KEY_ARG:
154 if (state->arg_num == 0 || state->arg_num == 1) {
155 lf_opts->args[state->arg_num] = arg;
156 } else {
157 argp_usage(state);
158 }
159 break;
160 case ARGP_KEY_END:
161 break;
162 default:
163 return ARGP_ERR_UNKNOWN;
164 }
165 return 0;
166}
