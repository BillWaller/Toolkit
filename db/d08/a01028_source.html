57/** @brief Main initialization function for MAPP - Menu Application
58 @ingroup init
59 @param init - pointer to Init struct to be initialized
60 @param argc - argument count from main()
61 @param argv - argument vector from main()
62 @code
63 1. Read environment variables and set defaults
64 2. Parse configuration file
65 3. Parse command-line options
66 4. Set up SIO struct with colors and other settings
67 5. Handle special options like help and version
68 @endcode

70void mapp_initialization(Init *init, int argc, char **argv) {
71 char term[MAXLEN];
72 char tmp_str[MAXLEN];
73 char *e;
74 setlocale(LC_ALL, "en_US.UTF-8");
75 SIO *sio = init->sio;
76 if (!init) {
77 ssnprintf(tmp_str, sizeof(tmp_str), "%s",
78 "init struct not allocated on entry");
79 abend(-1, tmp_str);

81 }
82
83 e = getenv("CMENU_HOME");
84 if (!e || *e == '\0')
85 strnz__cpy(init->mapp_home, "~/menuapp", MAXLEN);
86 else
87 strnz__cpy(init->mapp_home, e, MAXLEN);
88 e = getenv("CMENU_RC");
89 if (!e || *e == '\0')
90 strnz__cpy(init->minitrc, "~/menuapp/.minitrc", MAXLEN);
91 else
92 strnz__cpy(init->minitrc, e, MAXLEN);

94 strnz__cpy(init->minitrc, "~/.minitrc", MAXLEN - 1);

100 strnz__cpy(sio->ln_clr_x, "#0070ff",
101 COLOR_LEN - 1); /**< line number olor */
102 strnz__cpy(sio->ln_bg_clr_x, "#101010",
103 COLOR_LEN - 1); /**< line number background */
104 init->f_erase_remainder = true; /**< erase remainder on enter */
105 init->brackets[0] = '\0'; /**< field enclosure brackets */
106 strnz__cpy(init->fill_char, "_", MAXLEN - 1); /**< field fill character */
107 init->mapp_spec[0] = '\0'; /**< menu specification file */
108 strnz__cpy(init->mapp_home, "~/menuapp", MAXLEN - 1);

119 strnz__cpy(term, e, MAXLEN - 1);

123 else
124 strnz__cpy(init->editor, e, MAXLEN - 1);
125 parse_config(init); /**< generally /home/user/.minitrc */

127 if (f_dump_config) {
128 dump_config(init, "Configuration after parse_config and "
129 "parse_opt_args");

131 write_config(init);

174/** @brief Parse command-line options and set Init struct values accordingly
175 @ingroup init
176 @param init - pointer to Init struct
177 @param argc - argument count
178 @param argv - argument vector
179 Return index of first non-option argument
180 @note Accepts both short and long options
181 */
182int parse_opt_args(Init *init, int argc, char **argv) {
183 int i;
184 int opt;
185 int longindex = 0;
186 int flag = 0;
187 char *optstring = "a:b:c:d:ef:g:hi:j:k:m:n:o:p:r:st:u:vw:xzA:B:C:DF:G:H:L:"
188 "MNO:P:Q:R:S:T:VWX:Y:Z";
189 struct option long_options[] = {{"help", 0, &flag, MAPP_HELP},
190 {"mapp_spec", 1, &flag, MAPP_SPEC},
191 {"mapp_data", 1, &flag, MAPP_DATA_DIR},
192 {"mapp_help", 1, &flag, MAPP_HELP_DIR},
193 {"help_spec", 1, &flag, HELP_SPEC},
194 {"mapp_spec", 1, &flag, MAPP_SPEC},
195 {"in_spec", 1, &flag, IN_SPEC},
196 {"out_spec", 1, &flag, OUT_SPEC},
197 {"mapp_msrc", 1, &flag, MAPP_MSRC},
198 {0, 0, 0, 0}};
199 char mapp[MAXLEN];
200 SIO *sio = init->sio;
201 base_name(mapp, argv[0]);
202 optind = 1;
203 while ((opt = getopt_long(argc, argv, optstring, long_options,
204 &longindex)) != -1) {
205 switch (opt) {
206 case 0:
207 switch (flag) {
208 case MAPP_DATA_DIR:
209 strnz__cpy(init->mapp_data, optarg, MAXLEN - 1);
210 break;
211 case MAPP_HELP_DIR:
212 strnz__cpy(init->mapp_help, optarg, MAXLEN - 1);
213 break;
214 case MAPP_SPEC:
215 strnz__cpy(init->mapp_spec, optarg, MAXLEN - 1);
216 break;
217 case HELP_SPEC:
218 strnz__cpy(init->help_spec, optarg, MAXLEN - 1);
219 break;
220 case IN_SPEC:
221 strnz__cpy(init->in_spec, optarg, MAXLEN - 1);
222 break;
223 case OUT_SPEC:
224 strnz__cpy(init->out_spec, optarg, MAXLEN - 1);
225 break;
226 case MAPP_MSRC:
227 strnz__cpy(init->mapp_msrc, optarg, MAXLEN - 1);
228 break;
229 default:
230 break;
231 }
232 break;
233 case 'a':
234 strnz__cpy(init->minitrc, optarg, MAXLEN - 1);
235 break;
236 case 'b':
237 sio->blue_gamma = str_to_double(optarg);
238 break;
239 case 'c':
240 strnz__cpy(init->cmd, optarg, MAXLEN - 1);
241 break;
242 case 'd':
243 strnz__cpy(init->mapp_spec, optarg, MAXLEN - 1);
244 break;
245 case 'e':
246 init->f_erase_remainder = true;
247 break;
248 case 'f': // fill char
249 strnz__cpy(init->fill_char, optarg, 1);
250 break;
251 case 'g':
252 sio->green_gamma = str_to_double(optarg);
253 break;
254 case 'h':
255 f_help = true;
256 break;
257 case 'i':
258 strnz__cpy(init->in_spec, optarg, MAXLEN - 1);
259 break;
260 case 'j':
261 init->f_strip_ansi = true;
262 break;
263 case 'k':
264 strnz__cpy(init->parent_cmd, optarg, MAXLEN - 1);
265 break;
266 case 'm':
267 strnz__cpy(init->mapp_home, optarg, MAXLEN - 1);
268 break;
269 case 'n':
270 init->select_max = atoi(optarg);
271 break;
272 case 'o':
273 strnz__cpy(init->out_spec, optarg, MAXLEN - 1);
274 break;
275 case 'r':
276 sio->red_gamma = str_to_double(optarg);
277 break;
278 case 's':
279 init->f_squeeze = true;
280 break;
281 case 't':
282 init->tab_stop = atoi(optarg);
283 if (init->tab_stop < 1)
284 init->tab_stop = 1;
285 break;
286 case 'u': // brackets
287 strnz__cpy(init->brackets, optarg, 2);
288 break;
289 case 'v':
290 case 'V':
291 f_version = true;
292 break;
293 case 'x':
294 init->f_ignore_case = true;
295 break;
296 case 'y':
297 init->f_at_end_remove = true;
298 break;
299 case 'A':
300 strnz__cpy(init->cmd_all, optarg, MAXLEN - 1);
301 break;
302 case 'B':
303 strnz__cpy(sio->bg_clr_x, optarg, MAXLEN - 1);
304 break;
305 case 'C':
306 init->cols = atoi(optarg);
307 break;
308 case 'D':
309 f_dump_config = true;
310 break;
311 case 'F':
312 strnz__cpy(sio->fg_clr_x, optarg, MAXLEN - 1);
313 break;
314 case 'G':
315 sio->gray_gamma = str_to_double(optarg);
316 break;
317 case 'H':
318 strnz__cpy(init->help_spec, optarg, MAXLEN - 1);
319 break;
320 case 'L':
321 init->lines = atoi(optarg);
322 break;
323 case 'M':
324 init->f_multiple_cmd_args = true;
325 break;
326 case 'N':
327 init->f_ln = true;
328 break;
329 case 'O':
330 strnz__cpy(sio->bo_clr_x, optarg, MAXLEN - 1);
331 break;
332 case 'R':
333 strnz__cpy(init->receiver_cmd, optarg, MAXLEN - 1);
334 break;
335 case 'S':
336 strnz__cpy(init->provider_cmd, optarg, MAXLEN - 1);
337 break;
338 case 'T':
339 strnz__cpy(init->title, optarg, MAXLEN - 1);
340 break;
341 case 'U':
342 strnz__cpy(init->mapp_user, optarg, MAXLEN - 1);
343 break;
344 case 'W':
345 f_write_config = true;
346 break;
347 case 'X':
348 init->begx = atoi(optarg);
349 break;
350 case 'Y':
351 init->begy = atoi(optarg);
352 break;
353 default:
354 break;
355 }
356 }
357 init->argv[0] = strdup(argv[0]);
358 i = 0;
359 while (i < argc) {
360 init->argv[i] = strdup(argv[i]);
361 i++;
362 }
363 init->argv[i] = nullptr;
364 init->argc = argc;
365 return optind;
366}

648/** @brief Write the current configuration to a file specified in

659int write_config(Init *init) {

751 @param dir - directory path
752 @param file_name - file name
753 @returns true if file_spec is derived, false otherwise
754 @note If dir is nullptr, use MAPP_DIR environment variable or default
755 directory
756 ~/menuapp
757 @note file_spec should be a pre-allocated char array of size MAXLEN to
758 hold the resulting file specification
759 */
760bool derive_file_spec(char *file_spec, char *dir, char *file_name) {
761 char ts[MAXLEN];
762 char ts2[MAXLEN];
763 char *e;
764
765 if (!file_name || !*file_name) {
766 *file_spec = '\0';
767 return false;
768 }
769 if (dir) {
770 strnz__cpy(ts, dir, MAXLEN - 1);
771 } else {
772 e = getenv("MAPP_DIR");
773 if (e) {
774 strnz__cpy(ts, e, MAXLEN - 1);
775 } else {
776 strnz__cpy(ts, "~/menuapp", MAXLEN - 1);
