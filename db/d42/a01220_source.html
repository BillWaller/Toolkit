418/** @brief Main initialization function for MAPP - Menu Application
419 @ingroup init
420 @param init - pointer to Init struct to be initialized
421 @param argc - argument count from main()
422 @param argv - argument vector from main()
423 @code
424 1. Read environment variables and set defaults
425 2. Parse configuration file
426 3. Parse command-line options
427 4. Set up SIO struct with colors and other settings
428 5. Handle special options like help and version
429 @endcode
430 */
431void mapp_initialization(Init *init, int argc, char **argv) {
432 char term[MAXLEN];
433 char tmp_str[MAXLEN];
434 char *e;
435 setlocale(LC_ALL, "en_US.UTF-8");
436 SIO *sio = init->sio;
437 if (!init) {
438 ssnprintf(tmp_str, sizeof(tmp_str), "%s",
439 "init struct not allocated on entry");
440 abend(-1, tmp_str);
441 exit(-1);
442 }
443#ifdef DEBUG_LOG
444 open_cmenu_log();
445#endif
446 e = getenv("CMENU_HOME");
447 if (!e || *e == '\0')
448 strnz__cpy(init->mapp_home, "~/menuapp", MAXLEN);
449 else
450 strnz__cpy(init->mapp_home, e, MAXLEN);
451
452 if (init->mapp_home[0] != '\0') {
453 expand_tilde(init->mapp_home, MAXLEN - 1);
454 if (!verify_dir(init->mapp_home, R_OK))
455 abend(-1, "MAPP_HOME directory invalid");
456 }
457 // CMENU_RC should be an absolute path
458 e = getenv("CMENU_RC");
459 if (!e || *e == '\0') {
460 strnz__cpy(init->minitrc, init->mapp_home, MAXLEN - 1);
461 strnz__cat(init->minitrc, "/.minitrc", MAXLEN);
462 } else
463 strnz__cpy(init->minitrc, e, MAXLEN);
464 if (init->mapp_user[0] == '\0') {
465 strnz__cpy(init->mapp_user, init->mapp_home, MAXLEN - 1);
466 strnz__cat(init->mapp_user, "/user", MAXLEN - 1);
467 }
468 if (init->mapp_theme[0] == '\0') {
469 strnz__cpy(init->mapp_theme, init->mapp_home, MAXLEN - 1);
470 strnz__cat(init->mapp_theme, "/themes/default", MAXLEN - 1);
471 }
472 if (init->mapp_msrc[0] == '\0') {
473 strnz__cpy(init->mapp_msrc, init->mapp_home, MAXLEN - 1);
474 strnz__cat(init->mapp_msrc, "/msrc", MAXLEN - 1);
475 }
476 if (init->mapp_data[0] == '\0') {
477 strnz__cpy(init->mapp_data, init->mapp_home, MAXLEN - 1);
478 strnz__cat(init->mapp_data, "/data", MAXLEN - 1);
479 }
480 if (init->mapp_help[0] == '\0') {
481 strnz__cpy(init->mapp_help, init->mapp_home, MAXLEN - 1);
482 strnz__cat(init->mapp_help, "/help", MAXLEN - 1);
483 }
484 init->mapp_spec[0] = '\0'; /**< menu specification file */
485 // Set default colors and settings in SIO struct
486 // These can be overridden by the config file or command-line options
487 // Included here to ensure SIO has valid defaults even if config parsing fails
488 strnz__cpy(sio->bg, "#000000",
489 COLOR_LEN - 1); /**< background color */
490 strnz__cpy(sio->fg, "#c0c0c0",
491 COLOR_LEN - 1); /**< foreground color */
492 strnz__cpy(sio->box_fg, "#f00000", COLOR_LEN - 1); /**< bold color */
493 strnz__cpy(sio->box_bg, "#000000", COLOR_LEN - 1); /**< bold color */
494 strnz__cpy(sio->title_fg, "#f0f0f0", COLOR_LEN - 1); /**< title foreground color */
495 strnz__cpy(sio->title_bg, "#000000", COLOR_LEN - 1); /**< title background color */
496 strnz__cpy(sio->nt_fg, "#c0c0c0", COLOR_LEN - 1); /**< normal foreground color */
497 strnz__cpy(sio->nt_bg, "#000000", COLOR_LEN - 1); /**< normal background color */
498 strnz__cpy(sio->nt_rev_fg, "#000000", COLOR_LEN - 1); /**< normal reverse foreground color */
499 strnz__cpy(sio->nt_rev_bg, "#c0c0c0", COLOR_LEN - 1); /**< normal reverse background color */
500 strnz__cpy(sio->nt_hl_fg, "#f00000", COLOR_LEN - 1); /**< normal highlight foreground color */
501 strnz__cpy(sio->nt_hl_bg, "#000000", COLOR_LEN - 1); /**< normal highlight background color */
502 strnz__cpy(sio->nt_hl_rev_fg, "#000000", COLOR_LEN - 1); /**< normal reverse foreground color */
503 strnz__cpy(sio->nt_hl_rev_bg, "#c0c0c0", COLOR_LEN - 1); /**< normal reverse background color */
504 strnz__cpy(sio->ln_fg, "#0070ff",
505 COLOR_LEN - 1); /**< line number olor */
506 strnz__cpy(sio->ln_bg, "#101010",
507 COLOR_LEN - 1); /**< line number background */
508
509 strnz__cpy(sio->cmdln_fg, "#d0d0d0",
510 COLOR_LEN - 1); /**< line number olor */
511 strnz__cpy(sio->cmdln_bg, "#000000",
512 COLOR_LEN - 1); /**< line number background */
513
514 init->f_erase_remainder = true; /**< erase remainder on enter */
515 init->brackets[0] = '\0'; /**< field enclosure brackets */
516 strnz__cpy(init->fill_char, " ", MAXLEN - 1); /**< field fill character */
517 e = getenv("TERM");
518 if (e == nullptr || *e == '\0')
519 strnz__cpy(term, "xterm-256color", MAXLEN);
520 else
521 strnz__cpy(term, e, MAXLEN - 1);
522 e = getenv("EDITOR");
523 if (e && *e != '\0')
524 strnz__cpy(init->editor, "vi", MAXLEN - 1);
525 else
526 strnz__cpy(init->editor, e, MAXLEN - 1);
527 process_config_files(init);
528 init->mapp_spec[0] = '\0';
529 init->argc = argc;
530 argp_parse(&argp, argc, argv, 0, 0, init);
531 if (f_write_config) {
532 write_config(init);
533 exit(EXIT_SUCCESS);
534 }
535}
536/** @brief Parse command-line options and set Init struct values accordingly
537 @ingroup init
538 @param init - pointer to Init struct to be populated with option values
539 @param argc - argument count from main()
540 @param argv - argument vector from main()
541 @returns 0 on success, -1 on failure
542 @details This function uses the argp library to parse command-line
543 options defined in the options array. It updates the Init struct with
544 values from the options and handles any special flags for dumping or

547int parse_opt_args(Init *init, int argc, char **argv) {

550 return 0;
551}
