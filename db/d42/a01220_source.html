399/** @brief Format field according to its format type

406 accept_s and display_s strings for the field based on the formatted
407 value.
408 Handles various format types, including strings, decimal integers,
409 hexadecimal integers, floating-point numbers, currency, dates, and times.
410 Uses helper functions for validation and formatting, such as
411 is_valid_date(), is_valid_time(), numeric(), right_justify(),
412 left_justify(), and strnzcpy(). The function ensures that the formatted
413 output fits within the field's length and creates a filler string for the
414 field as needed. The function also handles error cases, such as invalid
415 formats, and provides feedback through error messages. It is designed to
416 be extensible, allowing for additional format types to be added in the
417 future as needed.
418 Assumes that the input string is well-formed and does not contain

421 The function currently does not handle localization or
422 internationalization of number formats, such as different decimal
423 separators or currency symbols. Future enhancements may include support
424 for locale-specific formatting.
425 Error handling is basic; future
426 versions may include more detailed error reporting and logging
427 mechanisms.
428 Performance optimizations may be considered for
429 handling large volumes of data or high-frequency updates in real-time
430 applications. The following variables and structures are used in
431 this function:
432 @code
433 char field_s[FIELD_MAXLEN];
434 int decimal_int_n = 0;
435 int hex_int_n = 0;
436 float float_n = 0.0;
437 double double_n = 0.0;
438 double currency_n = 0.0;
439 struct Date { int yyyy; int mm; int dd; };
440 struct Time { int hh; int mm; int ss; };
441 @endcode
442 */
443int form_fmt_field(Form *form, char *s) {
444 strnz__cpy(form->field[form->fidx]->input_s, s, FIELD_MAXLEN - 1);
445 char *input_s = form->field[form->fidx]->input_s;
446 char *accept_s = form->field[form->fidx]->accept_s;
447 char *display_s = form->field[form->fidx]->display_s;
448 // char *filler_s = form->field[form->fidx]->filler_s;
449 int ff = form->field[form->fidx]->ff;
450 int fl = form->field[form->fidx]->len;
451
452 char field_s[FIELD_MAXLEN];
453 int decimal_int_n = 0;
454 int hex_int_n = 0;
455 float float_n = 0.0;
456 double double_n = 0.0;
457 double currency_n = 0.0;
458
459 struct {
460 int yyyy;
461 int mm;
462 int dd;
463 } Date;
464 Date.yyyy = Date.mm = Date.dd = 0;
465 struct {
466 int hh;
467 int mm;
468 int ss;
469 } Time;
470 Time.hh = Time.mm = Time.ss = 0;
471 strnz(accept_s, fl);
472 switch (ff) {
473 case FF_STRING:
474 left_justify(s);
475 trim(s);
476 strnz__cpy(input_s, s, FIELD_MAXLEN - 1);
477 strnz__cpy(accept_s, s, FIELD_MAXLEN - 1);
478 strnz__cpy(display_s, s, FIELD_MAXLEN - 1);
479 break;
480 case FF_DECIMAL_INT:
481 sscanf(input_s, "%d", &decimal_int_n);
482 sprintf(accept_s, "%d", decimal_int_n);
483 sprintf(display_s, "%d", decimal_int_n);
484 right_justify(display_s, fl);
485 break;
486 case FF_HEX_INT:
487 sscanf(input_s, "%x", &hex_int_n);
488 sprintf(accept_s, "%d", hex_int_n);
489 sprintf(display_s, "%x", hex_int_n);
490 right_justify(display_s, fl);
491 break;
492 case FF_FLOAT:
493 sscanf(input_s, "%f", &float_n);
494 sprintf(accept_s, "%f", float_n);
495 sprintf(display_s, "%f", float_n);
496 right_justify(display_s, fl);
497 break;
498 case FF_DOUBLE:
499 sscanf(input_s, "%lf", &double_n);
500 sprintf(accept_s, "%lf", double_n);
501 sprintf(display_s, "%lf", double_n);
502 right_justify(display_s, fl);
503 break;
504 case FF_CURRENCY:
505 numeric(field_s, input_s);
506 sscanf(field_s, "%lf", &currency_n);
507 sprintf(accept_s, "%.2lf", currency_n);
508 sprintf(display_s, "%'.2lf", currency_n);
509 right_justify(display_s, fl);
510 break;
511 case FF_YYYYMMDD:
512 Date.yyyy = Date.mm = Date.dd = 0;
513 strnz__cpy(field_s, input_s, FIELD_MAXLEN - 1);

518 break;
519 case FF_HHMMSS:
520 Time.hh = Time.mm = Time.ss = 0;
521 strnz__cpy(field_s, input_s, FIELD_MAXLEN - 1);
522 sscanf(field_s, "%2d%2d%2d", &Time.hh, &Time.mm, &Time.ss);
523 sprintf(accept_s, "%02d%02d%02d", Time.hh, Time.mm, Time.ss);

526 break;
527 case FF_APR:
528 sscanf(input_s, "%lf", &double_n);
529 sprintf(accept_s, "%lf", double_n);

531 right_justify(display_s, fl);

533 default:

540}

587void right_justify(char *s, int fl) {

602/** @brief Check if a given date is valid, including leap years
603 @ingroup field_editor
604 @param yyyy Year
605 @param mm Month
606 @param dd Day
607 @return true if the date is valid, false otherwise
608 @details This function checks if the provided year, month, and day
609 constitute a valid date. It accounts for leap years when determining the
610 number of days in February. The function returns true if the date is valid
611 and false if it is not.
612 */
613bool is_valid_date(int yyyy, int mm, int dd) {
614 if (yyyy < 1 || mm < 1 || mm > 12 || dd < 1)
615 return false;
616 int days_in_month[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
617 if ((yyyy % 4 == 0 && yyyy % 100 != 0) || (yyyy % 400 == 0))
618 days_in_month[2] = 29;
619 if (dd > days_in_month[mm])
620 return false;
621 return true;
622}
