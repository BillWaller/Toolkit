215static error_t parse_opt(int key, char *arg, struct argp_state *state) {
216 LfContext *lf = state->input;
217 int i = 0;
218 bool a_toi_error = false;
219
220 switch (key) {
221 case 'a':
222 parse_local_timestamp(arg, &lf->after);
223 if (lf->after && lf->before && lf->before < lf->after) {
224 fprintf(stderr, "-b time must be greater than -a time.\n");
225 lf->after = 0;
226 }
227 break;
228 case 'b':
229 parse_local_timestamp(arg, &lf->before);
230 if (lf->after && lf->before && lf->before < lf->after) {
231 fprintf(stderr, "-b time must be greater than -a time.\n");
232 lf->before = 0;
233 }
234 break;
235 case 'c':
236 lf->count = true;
237 if (arg && arg[0] != '\0') {
238 if (strcmp(arg, "s") == 0)
239 lf->count_silently |= true;
240 }
241 break;
242 case 'd':
243 if (arg && arg[0] != '\0')
244 lf->max_depth = a_toi(arg, &a_toi_error);
245 break;
246 case 'D':
247 lf->debug = true;
248 if (arg && arg[0] != '\0') {
249 debug_p = strdup(arg);
250 i = 0;
251 while (debug_p[i]) {
252 switch (debug_p[i]) {
253 case '1':
254 lf->report_config = true;
255 break;
256 case '2':
257 lf->report_info = true;
258 break;
259 case '3':
260 lf->report_warnings = true;
261 break;
262 case '4':
263 lf->report_errors = true;
264 break;
265 case '5':
266 lf->report_badlinks = true;
267 break;
268 case '6':
269 lf->report_trace = true;
270 break;
271 case '7':
272 lf->report_config = true;
273 lf->report_info = true;
274 lf->report_warnings = true;
275 lf->report_errors = true;
276 lf->report_badlinks = true;
277 break;
278 case '8':
279 lf->only_errors = true;
280 break;
281 default:
282 break;
283 }
284 i++;
285 }
286 free(debug_p);
287 }
288 break;
289 case 'e':
290 lf->ere = strdup(arg);
291 lf->flags |= LF_EXC_REGEX;
292 break;
293 case 'H':
294 lf->include_hidden = true; // Include hidden files
295 if (arg && arg[0] == 'o') {
296 lf->hidden_only = true;
297 lf->include_hidden = true;
298 }
299 lf->flags &= ~(LF_HIDE); // Turn hide flag off
300 // LF_HIDE = 0 - include hidden files,
301 // LF_HIDE = 1 - suppress hidden files
302 break;
303 case 'i':
304 lf->ignore_case = true;
305 break;
306 case 'L':
307 lf->follow_links = true; // follow symbolic links
308 break;
309 case 'p':
310 perms_p = arg;
311 while (perms_p[i]) {

313 case 'g':

315 break;
316 case 'r':
317 lf->include_perms |= LF_IRUSR;
318 break;
319 case 's':
320 lf->include_perms |= LF_ISUID;
321 break;
322 case 'w':
323 lf->include_perms |= LF_IWUSR;
324 break;
325 case 'x':
326 lf->include_perms |= LF_IXUSR;
327 break;
328 default:
329 break;
330 }
331 }
332 break;
333 case 'R':
334 lf->sort_reverse = true;
335 break;
336 case 'r':
337 lf->re = strdup(arg);
338 lf->flags |= LF_REGEX;
339 break;
340 case 'S':
341 lf->sort = true;
342 break;
343 case 's':
344 lf->file_size_min = (intmax_t)(a_to_ul(arg));
345 break;
346 case 'T':
347 if (arg && arg[0] != '\0')
348 lf->nthreads = a_toi(arg, &a_toi_error);
349 break;
350 case 't':
351 file_types_p = arg;
352 while (file_types_p[i]) {
353 switch (file_types_p[i++]) {
354 case 'b':
355 lf->include_types |= LF_BLK;
356 break;
357 case 'c':
358 lf->include_types |= LF_CHR;
359 break;
360 case 'd':
361 lf->include_types |= LF_DIR;
362 break;
363 case 'p':
364 lf->include_types |= LF_FIFO;
365 break;
366 case 'l':
367 lf->include_types |= LF_LNK;
368 break;
369 case 'f': // for regular files, 'f' is more intuitive than 'r'
370 case 'r': // regular files are the most common type, so 'r' is also
371 // accepted
372 lf->include_types |= LF_REG;
373 break;
374 case 's':
375 lf->include_types |= LF_SOCK;
376 break;
377 case 'u':
378 lf->include_types |= LF_UNKNOWN;
379 break;
380 default:
381 break;
382 }
383 }
384 break;
385 case 'u':
386 lf->user_name = strdup(arg);
387 struct passwd *pwd = getpwnam(arg);
388 if (pwd) {
389 lf->user_id = (uintmax_t)pwd->pw_uid;

396 case ARGP_KEY_ARG:
397 if (state->arg_num == 0 || state->arg_num == 1) {

400 } else {
401 argp_usage(state);
402 }
403 break;
404 case ARGP_KEY_END:
405 break;
406 default:
407 return ARGP_ERR_UNKNOWN;
408 }
409 return 0;

501// Initialize and transfer control to the finder
502/** If sorting is requested, execute the finder and pipe its output
503 * to the sort command. We can achieve this by creating a child
504 * process that runs the sort command, and redirecting the output of
505 * the file finder to the input of the sort command using a pipe.
506 * The parent process will run the finder and write its output to
507 * the pipe, while the child process will read from the pipe and
508 * execute the sort command. */
509void sort_lf_output(LfContext *lf, int argc, char **argv) {
510 char tmp_str[MAXLEN];
511 char *eargv[MAXARGS];
512 int eargc = 0;
513 eargv[eargc++] = strdup("sort");
514 snprintf(tmp_str, MAXLEN - 1, "--parallel=%d", lf->nthreads);
515 eargv[eargc++] = strdup(tmp_str);
516 snprintf(tmp_str, MAXLEN - 1, "--buffer-size=%s", lf->sbuffer);
517 eargv[eargc++] = strdup(tmp_str);
518 if (lf->sort_reverse)
519 eargv[eargc++] = strdup("-r");
520 eargv[eargc] = nullptr;
521
522 setenv("LC_ALL", "C", 1); // Set locale to C for consistent sorting
523 int wstatus;
524
525 int fds[2];
526 pipe(fds); // Create the pipes
527
528 pid_t pid1 = fork();
529 if (pid1 == 0) { // Child process
530 // Child's STDIN will be redirected to the read end of the pipe, so that
531 // the sort process will read the output of the finder from the pipe.
532 close(fds[1]); // Close the write end of the pipe in the child
533 dup2(fds[0], STDIN_FILENO); // Clone child's read pipe to STDIN_FILENO
534 close(fds[0]); // Close the original read end of the pipe
535 execvp(eargv[0], eargv); // Execute the sort command
536 fprintf(stderr, "Failed to execute sort: %s\n", strerror(errno));
537 exit(termination_status);
538 }
539 // fclose(stdout);
540 dup2(fds[1], STDOUT_FILENO); // Clone write pipe to STDOUT_FILENO
541 stdout = fdopen(STDOUT_FILENO, "w"); // Reopen STDOUT as a stream
542 setvbuf(stdout, NULL, _IONBF, 0); // line buffering
543 init_find(lf, argc, argv); // Initialize and transfer control to the finder
544 fclose(stdout);
545 close(fds[1]);
546 wait(&wstatus);
547 for (int i = 0; i < eargc; i++)
548 free(eargv[i]);
549}

553/** @brief Initialize the file finder and start the search process.
554 @param lf A pointer to the LfContext struct containing the search parameters.
555 @param argc The number of command-line arguments.
556 @param argv The array of command-line argument strings.
557 @return true if initialization and search were successful, false otherwise.
558 @details This function initializes the file finder by setting up the necessary
559 parameters, compiling regular expressions if specified, and creating the initial
560 queue payload for the search. It also initializes threads for concurrent searching
561 and handles any errors that may occur during initialization. If sorting is requested,
562 it will pipe the output to the sort command. The function returns true if the
563 initialization and search were successful, or false if an error occurred.
564 */
565bool init_find(LfContext *lf, int argc, char **argv) {
566 /** suppress file types that aren't included */
567 if (!lf->include_types)
568 lf->include_types = 0xff;
569 if (lf->include_types)
570 lf->suppress_types = lf->include_types ^ 0xff;
571 // LF_HIDE = 0 - include hidden files,
572 // LF_HIDE = 1 - suppress hidden files
573 lf->include_hidden = !(lf->flags & LF_HIDE);
574 int reti = 0;
575 lf->reg_flags = REG_EXTENDED;
576 if (lf->ignore_case)
577 lf->reg_flags |= REG_ICASE;
578 if (lf->flags & LF_REGEX) {
579 reti = regcomp(&lf->compiled_re, lf->re, lf->reg_flags);
580 if (reti) {
581 fprintf(stderr, "lf: '%s' Invalid pattern\n", lf->re);
582 regfree(&lf->compiled_re);
583 return false;
584 }
585 }
586 if (lf->flags & LF_EXC_REGEX) {
587 reti = regcomp(&lf->compiled_ere, lf->ere, lf->reg_flags);
588 if (reti) {
589 fprintf(stderr, "lf: '%s' Invalid exclude pattern\n", lf->ere);
590 regfree(&lf->compiled_ere);
591 return false;
592 }
593 }
594 debug_out(lf, argc, argv);
595 //--------------------------------------------------------------------
596 // Create and enqueue the first QueuePayload
597 termination_status = TS_SUCCESS;
598 int rc = 0;
599 struct stat st;
600 rc = pthread_mutex_init(&lf->output_mutex, NULL);
601 if (rc != 0)
602 perror("Mutex initialization failed");
603 if (stat(lf->base_path, &st) == 0) {
604 lf->q = queue_init();
605 QueuePayload child_node;
606 if (S_ISDIR(st.st_mode)) {
607 strnz__cpy(child_node.path, lf->base_path, MAX_PATH_LEN - 1);
608#ifdef CYCLIC
609 child_node.depth = 0;
610 child_node.dev_ino[0].dev = st.st_dev;
611 child_node.dev_ino[0].ino = st.st_ino;
612#endif
613 if (!enqueue_dir(lf, &child_node)) {
614 fprintf(stderr, "Failed to enqueue initial directory\n");
615 return false;
616 }
617 atomic_fetch_add_explicit(&lf->q->active_tasks, 1, memory_order_relaxed);
618 //------------------------------------------------------------
619 // INITIALIZE THREADS
620 //------------------------------------------------------------
621 lf->threads = calloc(lf->nthreads, sizeof(pthread_t));
622 if (!lf->threads) {
623 fprintf(stderr, "Out of memory allocating threads\n");
624 return false;
625 }
626 for (unsigned int i = 0; i < lf->nthreads; i++) {
627 rc = pthread_create(
628 &lf->threads[i],
629 NULL,
630 worker,
631 lf);
632
633 if (rc != 0) {
634 fprintf(stderr, "Error: Unable to create thread %d\n", rc);
635 lf->q->shut_down = 1;
636 termination_status = TS_ERROR;
637 return false;
638 }
639 }
640 for (unsigned int i = 0; i < lf->nthreads; i++)
641 pthread_join(lf->threads[i], NULL);
642 return true;
643 } else {
644 fprintf(stderr,
645 "Warning: Base path '%s' is not a directory. No "
646 "files will be found.\n",
647 lf->base_path);
648 termination_status = TS_ERROR;
649 return false;
650 }
651 }
652 // ------------------------------------------------------------------
653 // END PROGRAM
654 // ------------------------------------------------------------------
655 rc = pthread_mutex_destroy(&lf->output_mutex);
656 if (rc != 0)
657 perror("Mutex destroy failed");
658 if (lf->flags & LF_REGEX) {
659 regfree(&lf->compiled_re);
660 }
661 if (lf->flags & LF_EXC_REGEX) {
662 regfree(&lf->compiled_ere);
663 }
664 free(lf->q);
665 free(lf->base_path);
666 free(lf->user_name);
667 free(lf->re);
668 free(lf->ere);
669 free(lf->threads);
670 if (reti)
671 return false;
672 return true;
673}

826/** @brief Build a full path by concatenating a directory path and a file name.
827 @param dst A pointer to the destination buffer where the full path will be stored.

829 @param dir_path The directory path to be concatenated.
830 @param name The file name to be concatenated.
831 @param out_len A pointer to a size_t variable where the length of the resulting full path will be stored (optional).
832 @return true if the full path was successfully built, false if an error occurred (e.g., null pointers or insufficient buffer size).

If the buffer is not large enough to hold the full path, the function returns false.
834 */
835bool build_full_path(char *dst, size_t dst_size, const char *dir_path,
836 const char *name, size_t *out_len) {
837 if (dst == nullptr || dir_path == nullptr || name == nullptr || dst_size == 0)
838 return false;
839 size_t dir_len = strlen(dir_path);
840 size_t name_len = strlen(name);
841 bool need_sep = dir_len > 0 && dir_path[dir_len - 1] != '/';
842 size_t full_len = dir_len + (need_sep ? 1 : 0) + name_len;
843 if (full_len >= dst_size)
844 return false;
845 memcpy(dst, dir_path, dir_len);
846 size_t pos = dir_len;
847 if (need_sep)
848 dst[pos++] = '/';
849 memcpy(dst + pos, name, name_len);
850 dst[full_len] = '\0';
851 if (out_len)
852 *out_len = full_len;
853 return true;

855/** @brief Flush the output buffer to stdout in a thread-safe manner.

858 @details This function writes the contents of the output buffer to stdout using fwrite_unlocked for efficiency. 860void flush_output_buffer(LfContext *lf, OutputBuffer *output) {
861 if (lf == nullptr || output == nullptr || output->len == 0)
862 return;

864 fwrite_unlocked(output->data, 1, output->len, stdout);
865 pthread_mutex_unlock(&lf->output_mutex);

902// ----------------------------------------------------------------------
903/** @brief Enqueue a directory payload into the lock-free queue.
904 @param lf A pointer to the LfContext struct containing the queue.
905 @param dir A pointer to the QueuePayload struct representing the directory to enqueue.
906 @return true if the directory was successfully enqueued, false if the queue is full or an error occurred.
907 @details This function attempts to enqueue a directory payload into a lock-free multi-producer, multi-consumer (MPMC) queue. It uses atomic operations to ensure thread-safe access to the queue. If the queue is full, it returns false, allowing the caller to handle the situation (e.g., by using a fallback mechanism). The function employs a spin-wait loop with a limit to prevent excessive CPU usage during contention.
908 */
909bool enqueue_dir(LfContext *lf, const QueuePayload *dir) {
910 size_t pos = atomic_load_explicit(&lf->q->enqueue_pos,
911 memory_order_relaxed);
912 int spin_count = 0;
913 const int SPIN_LIMIT = 16;
914 while (true) {
915 // Fast tracking check using the clean tracking array
916 size_t seq = atomic_load_explicit(&lf->q->sequences[pos & QUEUE_MASK], memory_order_acquire);
917 intptr_t diff = (intptr_t)seq - (intptr_t)pos;
918 if (diff == 0) {
919 // Weak CAS allows the CPU to immediately fail out if a conflict occurs
920 if (atomic_compare_exchange_weak_explicit(&lf->q->enqueue_pos, &pos, pos + 1,
921 memory_order_relaxed, memory_order_relaxed)) {
922 break;
923 }
924 } else if (diff < 0) {
925 return false; // Queue is full, hit your local recursive fallback
926 } else {
927 pos++;
928 }
929 // Drop out early if threads are colliding heavily to save context
930 // switching
931 if (++spin_count > SPIN_LIMIT) {
932 return false;
933 }
934// Relieve hardware bus pressure during contention
935#if defined(__x86_64__)
936 __builtin_ia32_pause();
937#endif
938 }
939 // This section is now guaranteed unique to our thread
940 // Only copy the small data footprint
941 memcpy(&lf->q->nodes[pos & QUEUE_MASK], dir, sizeof(QueuePayload));
942 // Release the block slot to workers
943 atomic_store_explicit(&lf->q->sequences[pos & QUEUE_MASK], pos + 1, memory_order_release);
944 return true;
945}

947// DEQUEUE
948// ----------------------------------------------------------------------
949/** @brief Dequeue a directory payload from the lock-free queue.
950 @param lf A pointer to the LfContext struct containing the queue.
951 @param task A pointer to the QueuePayload struct where the dequeued directory will be stored.
952 @return true if a directory was successfully dequeued, false if the queue is empty or a shutdown was detected.
953 @details This function attempts to dequeue a directory payload from a lock-free multi-producer, multi-consumer (MPMC) queue. It uses atomic operations to ensure thread-safe access to the queue. If the queue is empty or a shutdown signal is detected, it returns false, allowing the caller to handle the situation (e.g., by terminating the worker thread). The function employs a spin-wait loop with checks for shutdown conditions to prevent excessive CPU usage during contention.
954 */
955bool dequeue_dir(LfContext *lf, QueuePayload *task) {
956 QueuePayload *node;
957 size_t pos = atomic_load_explicit(&lf->q->dequeue_pos, memory_order_relaxed);
958 while (true) {
959 if (atomic_load_explicit(&lf->q->shut_down, memory_order_relaxed))
960 return false;
961 node = &lf->q->nodes[pos & QUEUE_MASK];
962 size_t seq = atomic_load_explicit(&lf->q->sequences[pos & QUEUE_MASK], memory_order_acquire);
963 intptr_t diff = (intptr_t)seq - (intptr_t)(pos + 1);
964 if (diff == 0) {
965 // Full slot, try to claim it
966 // Upgraded to strong CAS to avoid spurious fails under high contention
967 if (atomic_compare_exchange_strong_explicit(&lf->q->dequeue_pos, &pos, pos + 1,
968 memory_order_relaxed, memory_order_relaxed)) {
969 break;
970 }
971 } else if (diff < 0) { // Queue is temporarily empty
972 // If the global shutdown was triggered while we were waiting, exit smoothly
973 if (atomic_load_explicit(&lf->q->shut_down, memory_order_relaxed)) {
974 return false;
975 }
976 // Back off to prevent high-core cache bus starvation
977 sched_yield();
978 // Reload pos because another thread might have advanced it
979 pos = atomic_load_explicit(&lf->q->dequeue_pos, memory_order_relaxed);
980 } else {
981 pos = atomic_load_explicit(&lf->q->dequeue_pos, memory_order_relaxed);
982 }
983 }
984 node = &lf->q->nodes[pos & QUEUE_MASK];
985 *task = *node; // node becomes a task
986 // Release the slot back to producers (advancing sequence by capacity)
987 atomic_store_explicit(&lf->q->sequences[pos & QUEUE_MASK], pos + QUEUE_CAPACITY, memory_order_release);
988 return true;
989}

995 @return NULL upon completion.
996 @details This function is executed by each worker thread. It continuously dequeues directory payloads from the lock-free queue and processes them using the finder function. 1225 //-------------------------------------------------------
1226 // We duplicate the current history of dev/inode pairs and
1227 // add the current directory's dev/inode to the new history
1228 // for the child task. This allows us to maintain a record
1229 // of the directories we've visited in the current path,
1230 // which is essential for cycle detection. By checking this
1231 // history for each new directory we encounter, we can
1232 // effectively prevent infinite loops caused by symbolic
1233 // links or hard links that create cycles in the directory
1234 // structure.

1237 // --------------------------------------------------------

1239 continue;

1244 for (i = 0; i < current_node->depth; i++) {
1245 child_node.dev_ino[i].dev = current_node->dev_ino[i].dev;
1246 child_node.dev_ino[i].ino = current_node->dev_ino[i].ino;
1247 }

1249 child_node.dev_ino[i + 1].ino = st.st_ino;
1250 } else {
1251 child_node.dev_ino[0].dev = 0;
1252 child_node.dev_ino[0].ino = 0;
1253 }
1254#endif

1256 // Success! Track it as a new outstanding global task
1257 atomic_fetch_add_explicit(&lf->q->active_tasks, 1, memory_order_relaxed);
1258 } else {
1259 // QUEUE FULL: Run inline recursively.
1260 // We DO NOT touch the counter here because the thread is
1261 // staying within its current execution bubble.
1262 finder(lf, &child_node);
1263 }
1264
1265 // --------------------------------------------------------
1266 // CONDITIONALLY PRINT THE DIRECTORY
1267 // --------------------------------------------------------
1268 if (is_hidden(entry->d_name)) {
1269 if (!lf->include_hidden && !lf->hidden_only)
1270 continue;
1271 } else {
1272 if (lf->hidden_only)
1273 continue;
1274 }
1275 scan_file(child_node.path, lf, effective_type, &st, &output);
1276 } else {
1277 // --------------------------------------------------------
1278 // NOT DIRECTORY - CONDITIONALLY PRINT THE ENTRY
1279 // --------------------------------------------------------
1280 if (is_hidden(entry->d_name)) {
1281 if (!lf->include_hidden && !lf->hidden_only)
1282 continue;
1283 } else {
1284 if (lf->hidden_only)
1285 continue;
1286 }
1287 if (full_path[0] == '\0') {
1288 if (!build_full_path(full_path, sizeof(full_path),
1289 current_node->path,
1290 entry->d_name, nullptr)) {
1291 atomic_fetch_add(&lf->error_count, 1);
1292 if (lf->report_errors) {
1293 pthread_mutex_lock(&lf->output_mutex);
1294 fprintf(stderr, "PATH_TOO_LONG,%s/%s\n",
1295 current_node->path, entry->d_name);
1296 pthread_mutex_unlock(&lf->output_mutex);
1297 }
1298 continue;
1299 }
1300 }
1301 scan_file(full_path, lf, effective_type, &st, &output);
1302 }
1303 }
1304 }
1305 close(dir_fd);
1306 flush_output_buffer(lf, &output);
1307 return NULL;
1308}
