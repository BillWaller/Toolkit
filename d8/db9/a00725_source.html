32/** @brief max macro evaluates two expressions, returning greatest result.
33 @details These macros use compound statements to create local scopes for the
34 temporary variables _x and _y, which store the values of x and y,
35 respectively. This ensures that if x or y have side effects (such as being
36 incremented), they will only be evaluated once when the macro is expanded.
37 @note The use of typeof allows the macro to work with any data type.
38 @note The line (void)(&_x == &_y) is a compile-time check to ensure that the
39 types of the arguments are compatible.
40 @note This implementation of the min and max macros provides a safer and
41 more robust way to determine the minimum and maximum values between two
42 expressions without risking unintended consequences from multiple
43 evaluations.
44 */
45#define max(a, b)
46 ({
47 typeof(a) _a = (a);
48 typeof(b) _b = (b);
49 _a > _b ? _a : _b;
50 })
51/** @brief min macro evaluates two expressions, returning least result */
52#define min(x, y)
53 ({
54 typeof(x) _x = (x);
55 typeof(y) _y = (y);
56 (void)(&_x == &_y);
57 _x < _y ? _x : _y;
58 })
