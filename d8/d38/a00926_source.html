27/** @brief Initialize C-Menu View in full screen mode.
28 @ingroup init_view
29 @note This function sets up the view structure for full screen mode and
30 creates a new pad for the view.
31 @param init Pointer to the Init structure containing view settings.
32 @return 0 on success, -1 on failure.
33 */
34int init_view_full_screen(Init *init) {
35 int scr_lines, scr_cols;
36 view = init->view;
37
38 if (view->tab_stop <= 0)
39 view->tab_stop = TABSIZE;
40 set_tabsize(view->tab_stop);
41 view->f_full_screen = true;
42 getmaxyx(stdscr, scr_lines, scr_cols);
43 if (view->lines == 0 || view->lines > scr_lines)
44 view->lines = scr_lines;
45 if (view->cols == 0 || view->cols > scr_cols)
46 view->cols = scr_cols;
47 if (view->begy + view->lines > scr_lines)
48 view->begy = scr_lines - view->lines - 2;
49 if (view->begx + view->cols > scr_cols)
50 view->begx = scr_cols - view->cols - 2;
51 view->ln_win_lines = scr_lines;
52 view->ln_win_cols = 7;
53 view->ln_win = newwin(view->ln_win_lines - 1, view->ln_win_cols, 0, 0);
54 wsetscrreg(view->ln_win, 0, view->scroll_lines - 1);
55 scrollok(view->ln_win, true);
56#ifdef DEBUG
57 immedok(view->ln_win, true);
58#endif
59 keypad(view->ln_win, true);
60 idlok(view->ln_win, false);
61 idcok(view->ln_win, false);
62 wbkgrnd(view->ln_win, &CCC_LN);
63 wbkgrndset(view->ln_win, &CCC_LN);
64 wmove(view->ln_win, 0, 0);
65 wclear(view->ln_win);
66 wnoutrefresh(view->ln_win);
67 view->pminrow = 0;
68 view->pmincol = 0;
69 view->sminrow = 0;
70 view->smincol = 0;
71 view->scroll_lines = view->lines - 1;
72 view->cmd_line = view->lines - 1;
73 view->smaxrow = view->lines - 1;
74 view->smaxcol = view->cols - 1;
75 view->pad = newpad(view->lines, PAD_COLS);

79 PAD_COLS);
80 em2[0] = '\0';
81 display_error(em0, em1, em2, nullptr);
82 abend(-1, "init_view_full_screen: newpad() failed");

85#ifdef DEBUG

87#endif
88 keypad(view->pad, true);
89 idlok(view->pad, false);
90 idcok(view->pad, false);
91 wbkgrnd(view->pad, &CCC_NORM);
92 wbkgrndset(view->pad, &CCC_NORM);
93 wclear(view->pad);
94 wsetscrreg(view->pad, 0, view->scroll_lines - 1);
95 return 0;
96}

175/** @brief Initialize the input for a C-Menu View.
176 @ingroup init_view
177 @details This function initializes the input for view, which can be a file,
178 standard input, or a provider command to be initiated by view. It handles
179 different input sources and sets up the necessary file descriptors and memory
180 mapping for efficient access.
181 @param view Pointer to the View structure to be initialized.
182 @param file_name Name of the input file or "-" for standard input.
183 @return true on success, false on failure.
184 @note if a provider command is specified, set up a pipe to read its output.
185 A child process is spawned, and view, the parent process, reads from the
186 pipe.
187 @note If input is from a pipe or standard input, clone it to a temporary
188 file. This allows for memory-mapping the input later. It does not support
189 real-time updates to the input, but it allows for efficient access to the
190 data.
191 */
192bool view_init_input(View *view, char *file_name) {
193 struct stat sb;
194 int idx = 0;
195 pid_t pid;
196 int pipe_fd[2];
197 char *s_argv[MAXARGS];
198 char tmp_str[MAXLEN];
199 view->f_in_pipe = false;
200 if (strcmp(file_name, "-") == 0) {
201 file_name = "/dev/stdin";
202 view->f_in_pipe = true;
203 }
204 if (view->provider_cmd[0] != '\0') {
205 str_to_args(s_argv, view->provider_cmd, MAXARGS - 1);
206 if (pipe(pipe_fd) == -1) {
207 Perror("pipe(pipe_fd) failed in init_view");
208 return (1);
209 }
210 if ((pid = fork()) == -1) {
211 Perror("fork() failed in init_view");
212 return (1);
213 }
214 if (pid == 0) { // Child
215 close(pipe_fd[P_READ]);
216 dup2(pipe_fd[P_WRITE], STDOUT_FILENO);
217 close(pipe_fd[P_WRITE]);
218 execvp(s_argv[0], s_argv);
219 strnz__cpy(tmp_str, "Can't exec view start cmd: ", MAXLEN - 1);
220 strnz__cat(tmp_str, s_argv[0], MAXLEN - 1);
221 Perror(tmp_str);
222 exit(EXIT_FAILURE);
223 }
224 // Back to parent
225 close(pipe_fd[P_WRITE]);
226 dup2(pipe_fd[P_READ], STDIN_FILENO);
227 view->in_fd = dup(STDIN_FILENO);
228 view->f_in_pipe = true;
229 } else {
230 if (view->f_in_pipe)
231 view->in_fd = dup(STDIN_FILENO);
232 else {
233 /** Open the input file for reading and get its size. */
234 expand_tilde(file_name, MAXLEN - 1);
235 view->in_fd = open(file_name, O_RDONLY);
236 if (view->in_fd == -1) {
237 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
238 __LINE__ - 2);
239 ssnprintf(em1, MAXLEN - 1, "open %s", file_name);
240 strerror_r(errno, em2, MAXLEN);
241 display_error(em0, em1, em2, nullptr);
242 return false;
243 }
244 if (fstat(view->in_fd, &sb) == -1) {
245 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
246 __LINE__ - 1);
247 ssnprintf(em1, MAXLEN - 1, "fstat %s", file_name);
248 strerror_r(errno, em2, MAXLEN);
249 display_error(em0, em1, em2, nullptr);
250 close(view->in_fd);
251 return (EXIT_FAILURE);
252 }
253 view->file_size = sb.st_size;
254 if (view->file_size == 0) {
255 close(view->in_fd);
256 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
257 __LINE__ - 1);
258 ssnprintf(em1, MAXLEN - 1, "file %s is empty", file_name);
259 strerror_r(errno, em2, MAXLEN);
260 display_error(em0, em1, em2, nullptr);
261 return (EXIT_FAILURE);
262 }
263 if (!S_ISREG(sb.st_mode))
264 view->f_in_pipe = true;
265 }
266 }
267 if (view->f_in_pipe) {
268 char tmp_filename[] = "/tmp/view_XXXXXX";
269 char buf[VBUFSIZ];
270 ssize_t bytes_read = 0;
271 ssize_t bytes_written = 0;
272 close(view->in_fd);
273 view->in_fd = mkstemp(tmp_filename);
274 if (view->in_fd == -1) {
275 abend(-1, "failed to mkstemp");
276 exit(EXIT_FAILURE);
277 }
278 unlink(tmp_filename);
279 while ((bytes_read = read(STDIN_FILENO, buf, sizeof(buf))) > 0) {
280 if (write(view->in_fd, buf, bytes_read) != bytes_read) {
281 abend(-1, "unable to write tmp");
282 exit(EXIT_FAILURE);
283 }
284 bytes_written += bytes_read;
285 }
286 if (bytes_written == 0) {
287 abend(-1, "unable to read stdin");
288 exit(EXIT_FAILURE);
289 }
290 if (fstat(view->in_fd, &sb) == -1) {
291 abend(-1, "fstat failed");
292 exit(EXIT_FAILURE);
293 }
294 view->file_size = sb.st_size;
295 if (view->file_size == 0) {
296 close(view->in_fd);
297 strnz__cpy(tmp_str, "no standard input", MAXLEN - 1);
298 abend(-1, tmp_str);
299 exit(EXIT_FAILURE);
300 }
301 waitpid(-1, nullptr, 0);
302 }
303 // Memory-map the input file for efficient access.
304 view->buf =
305 mmap(nullptr, view->file_size, PROT_READ, MAP_PRIVATE, view->in_fd, 0);
306 if (view->buf == MAP_FAILED) {
307 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__, __LINE__ - 2);
308 ssnprintf(em1, MAXLEN - 1, "mmap %s", file_name);
309 strerror_r(errno, em2, MAXLEN);
310 display_error(em0, em1, em2, nullptr);
311 close(view->in_fd);
312 return (EXIT_FAILURE);
313 }
314 close(view->in_fd);
315 view->file_size = sb.st_size;
316 view->prev_file_pos = NULL_POSITION;
317 view->buf_curr_ptr = view->buf;
318 if (view->cmd_all[0] != '\0')
319 strnz__cpy(view->cmd, view->cmd_all, MAXLEN - 1);
320 for (idx = 0; idx < NMARKS; idx++)
321 view->mark_tbl[idx] = NULL_POSITION;
322 strnz__cpy(view->cur_file_str, file_name, MAXLEN - 1);
323 base_name(view->file_name, view->cur_file_str);
324 return true;
325}
