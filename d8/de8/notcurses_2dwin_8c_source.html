426int rgb_to_xterm256_idx(RGB *rgb) {
427 if (rgb->r == rgb->g && rgb->g == rgb->b) {
428 if (rgb->r < 8)
429 return 16;
430 if (rgb->r > 248)
431 return 231;
432 return ((rgb->r - 8) / 10) + 231;
433 } else {
434 int r_index = (rgb->r < 45) ? 0 : (rgb->r - 60) / 40 + 1;
435 int g_index = (rgb->g < 45) ? 0 : (rgb->g - 60) / 40 + 1;
436 int b_index = (rgb->b < 45) ? 0 : (rgb->b - 60) / 40 + 1;
437 return 16 + (36 * r_index) + (6 * g_index) + b_index;
438 }

448RGB xterm256_idx_to_rgb(int idx) {
452 RGB rgb;
453 if (idx > 255)
454 idx = 255;
455 if (idx < 0)
456 idx = 0;
457 rgb.r = rgb.g = rgb.b = 0;
458 if (idx < 16) {
459 rgb.r = StdColors[idx].r;
460 rgb.g = StdColors[idx].g;
461 rgb.b = StdColors[idx].b;
462 } else if (idx >= 16 && idx <= 231) {
463 idx -= 16;
464 rgb.r = (idx / 36) % 6 * 51;
465 rgb.g = (idx / 6) % 6 * 51;
466 rgb.b = (idx % 6) * 51;
467 } else if (idx >= 232 && idx <= 255) {
468 int gray = (idx - 232) * 11;
469 rgb.r = rgb.g = rgb.b = gray;
470 }
471 return rgb;

484 if (rgb->r == rgb->g && rgb->r == rgb->b) {
485 if (GRAY_GAMMA > 0.0f && GRAY_GAMMA != 1.0f) {
486 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / GRAY_GAMMA) * 255.0f);
487 rgb->g = rgb->r;
488 rgb->b = rgb->r;
489 }
490 return;
491 }
492 if (rgb->r != 0 && RED_GAMMA > 0.0f && RED_GAMMA != 1.0f)
493 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / RED_GAMMA) * 255.0f);

495 rgb->g = (int)(pow((rgb->g / 255.0f), 1.0f / GREEN_GAMMA) * 255.0f);
496 if (rgb->b != 0 && BLUE_GAMMA > 0.0f && BLUE_GAMMA != 1.0f)
497 rgb->b = (int)(pow((rgb->b / 255.0f), 1.0f / BLUE_GAMMA) * 255.0f);

832int wccp_to_str(wchar_t cp, uint8_t *buffer) {
833 if (cp <= 0x7F) {
834 // 1-byte sequence: 0xxxxxxx
835 buffer[0] = (uint8_t)cp;
836 return 1;
837 } else if (cp <= 0x7FF) {
838 // 2-byte sequence: 110xxxxx 10xxxxxx
839 buffer[0] = (uint8_t)(0xC0 | ((cp >> 6) & 0x1F));
840 buffer[1] = (uint8_t)(0x80 | (cp & 0x3F));
841 return 2;
842 } else if (cp <= 0xFFFF) {
843 // 3-byte sequence: 1110xxxx 10xxxxxx 10xxxxxx
844 buffer[0] = (uint8_t)(0xE0 | ((cp >> 12) & 0x0F));
845 buffer[1] = (uint8_t)(0x80 | ((cp >> 6) & 0x3F));
846 buffer[2] = (uint8_t)(0x80 | (cp & 0x3F));
847 return 3;
848 } else if (cp <= 0x10FFFF) {
849 // 4-byte sequence: 11110xxx 10xxxxxx 10xxxxxx 10xxxxxx
850 buffer[0] = (uint8_t)(0xF0 | ((cp >> 18) & 0x07));
851 buffer[1] = (uint8_t)(0x80 | ((cp >> 12) & 0x3F));
852 buffer[2] = (uint8_t)(0x80 | ((cp >> 6) & 0x3F));
853 buffer[3] = (uint8_t)(0x80 | (cp & 0x3F));
854 return 4;
855 }
856 return 0; // Invalid Unicode code point
857}
