27/** @brief Initialize C-Menu View in full screen mode.
28 @ingroup init_view
29 @note This function sets up the view structure for full screen mode and
30 creates a new pad for the view.
31 @param init Pointer to the Init structure containing view settings.
32 @return 0 on success, -1 on failure.
33 @verbatim
34 The function creates the following windows:
35 1. view->win: Status or Command Line
36 2. view->ln_win: Line Number Window
37 3. view->pad: Main Content Pad
38 @endverbatim
39 */
40int init_view_full_screen(Init *init) {
41 int scr_lines, scr_cols;
42 view = init->view;
43
44 if (view->tab_stop <= 0)
45 view->tab_stop = TABSIZE;
46 set_tabsize(view->tab_stop);
47 view->f_full_screen = true;
48 getmaxyx(stdscr, scr_lines, scr_cols);
49 if (view->lines == 0 || view->lines > scr_lines)
50 view->lines = scr_lines;
51 if (view->cols == 0 || view->cols > scr_cols)
52 view->cols = scr_cols;
53 if (view->begy + view->lines > scr_lines)
54 view->begy = scr_lines - view->lines;
55 if (view->begx + view->cols > scr_cols)
56 view->begx = scr_cols - view->cols;
57 /** view->win: status or command line window */
58 view->win = newwin(1, view->cols, view->begy + view->lines - 1, view->begx);
59 keypad(view->win, true);
60 idlok(view->win, false);
61 idcok(view->win, false);
62 wbkgrnd(view->win, &CCC_NORM);
63 wbkgrndset(view->win, &CCC_NORM);
64 scrollok(view->win, false);
65#ifdef NCDEBUG
66 immedok(view->win, true);
67#endif
68
69 /** view->ln_win: line number window */
70 view->ln_win_lines = scr_lines;
71 view->ln_win_cols = 7;
72 view->ln_win = newwin(view->ln_win_lines - 1, view->ln_win_cols, 0, 0);
73 keypad(view->ln_win, false);
74 idlok(view->ln_win, false);
75 idcok(view->ln_win, false);
76 wbkgrnd(view->ln_win, &CCC_LN);
77 wbkgrndset(view->ln_win, &CCC_LN);
78 scrollok(view->ln_win, true);
79 wsetscrreg(view->ln_win, 0, view->scroll_lines - 1);
80#ifdef NCDEBUG
81 immedok(view->ln_win, true);
82#endif
83
84 /** view->win: status or command line window */
85 view->pminrow = 0;
86 view->pmincol = 0;
87 view->sminrow = 0;
88 view->smincol = 0;
89 view->scroll_lines = view->lines - 1;
90 view->cmd_line = 0;
91 view->smaxrow = view->lines - 1;
92 view->smaxcol = view->cols - 1;
93 view->pad = newpad(view->lines - 1, PAD_COLS);
94 if (view->pad == nullptr) {
95 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__, __LINE__ - 2);
96 ssnprintf(em1, MAXLEN - 1, "newpad(%d, %d) failed", view->lines,
97 PAD_COLS);
98 em2[0] = '\0';
99 display_error(em0, em1, em2, nullptr);
100 abend(-1, "init_view_full_screen: newpad() failed");
101 }
102 keypad(view->pad, false);
103 idlok(view->pad, false);
104 idcok(view->pad, false);
105 wbkgrnd(view->pad, &CCC_NORM);
106 wbkgrndset(view->pad, &CCC_NORM);
107 scrollok(view->pad, true);
108 wsetscrreg(view->pad, 0, view->scroll_lines - 1);
109#ifdef NCDEBUG
110 immedok(view->pad, true);
111#endif
112 return 0;
113}

206/** @brief Initialize the input for a C-Menu View.
207 @ingroup init_view
208 @details This function initializes the input for view, which can be a file,
209 standard input, or a provider command to be initiated by view. It handles
210 different input sources and sets up the necessary file descriptors and memory
211 mapping for efficient access.
212 @param view Pointer to the View structure to be initialized.
213 @param file_name Name of the input file or "-" for standard input.
214 @return true on success, false on failure.
215 @note if a provider command is specified, set up a pipe to read its output.
216 A child process is spawned, and view, the parent process, reads from the
217 pipe.
218 @note If input is from a pipe or standard input, clone it to a temporary
219 file. This allows for memory-mapping the input later. It does not support
220 real-time updates to the input, but it allows for efficient access to the
221 data.
222 */
223int view_init_input(View *view, char *file_name) {
224 struct stat sb;
225 int idx = 0;
226 pid_t pid = -1;
227 int pipe_fd[2];
228 char *s_argv[MAXARGS];
229 char tmp_str[MAXLEN];
230 view->f_in_pipe = false;
231 if (strcmp(file_name, "-") == 0) {
232 file_name = "/dev/stdin";
233 view->f_in_pipe = true;
234 }
235 if (view->provider_cmd[0] != '\0') {
236 str_to_args(s_argv, view->provider_cmd, MAXARGS - 1);
237 if (pipe(pipe_fd) == -1) {
238 Perror("pipe(pipe_fd) failed in init_view");
239 return -1;
240 }
241 if ((pid = fork()) == -1) {
242 Perror("fork() failed in init_view");
243 return -1;
244 }
245 if (pid == 0) { // Child
246 close(pipe_fd[P_READ]);
247 dup2(pipe_fd[P_WRITE], STDOUT_FILENO);
248 close(pipe_fd[P_WRITE]);
249 execvp(s_argv[0], s_argv);
250 strnz__cpy(tmp_str, "Can't exec view start cmd: ", MAXLEN - 1);
251 strnz__cat(tmp_str, s_argv[0], MAXLEN - 1);
252 Perror(tmp_str);
253 exit(EXIT_FAILURE);
254 }
255 // Back to parent
256 close(pipe_fd[P_WRITE]);
257 dup2(pipe_fd[P_READ], STDIN_FILENO);
258 view->in_fd = dup(STDIN_FILENO);
259 view->f_in_pipe = true;
260 } else {
261 if (view->f_in_pipe)
262 view->in_fd = dup(STDIN_FILENO);
263 else {
264 /*----------------------------------------------------------------------*/
265 /** Open the input file for reading and get its size. */
266 expand_tilde(file_name, MAXLEN - 1);
267 view->in_fd = open(file_name, O_RDONLY);
268 if (view->in_fd == -1) {
269 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
270 __LINE__ - 2);
271 ssnprintf(em1, MAXLEN - 1, "open %s", file_name);
272 strerror_r(errno, em2, MAXLEN);
273 display_error(em0, em1, em2, nullptr);
274 return -1;
275 }
276 if (fstat(view->in_fd, &sb) == -1) {
277 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
278 __LINE__ - 1);
279 ssnprintf(em1, MAXLEN - 1, "fstat %s", file_name);
280 strerror_r(errno, em2, MAXLEN);
281 display_error(em0, em1, em2, nullptr);
282 close(view->in_fd);
283 return -1;
284 }
285 view->file_size = sb.st_size;
286 if (view->file_size == 0) {
287 close(view->in_fd);
288 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
289 __LINE__ - 1);
290 ssnprintf(em1, MAXLEN - 1, "file %s is empty", file_name);
291 strerror_r(errno, em2, MAXLEN);
292 display_error(em0, em1, em2, nullptr);
293 return -1;
294 }
295 if (!S_ISREG(sb.st_mode))
296 view->f_in_pipe = true;
297 }
298 /*----------------------------------------------------------------------*/
299 }
300 if (view->f_in_pipe) {
301 char tmp_filename[] = "/tmp/view_XXXXXX";
302 char buf[VBUFSIZ];
303 ssize_t bytes_read = 0;
304 ssize_t bytes_written = 0;
305 close(view->in_fd);
306 view->in_fd = mkstemp(tmp_filename);
307 if (view->in_fd == -1) {
308 abend(-1, "failed to mkstemp");
309 exit(EXIT_FAILURE);
310 }
311 unlink(tmp_filename);
312 /*-----------------------------------------------------------------*/
313 bool f_wait = false;
314 int ready;
315 fd_set read_fds;
316 struct timeval timeout;
317 Chyron *wait_chyron = nullptr;
318 WINDOW *wait_win = nullptr;
319 int remaining;
320 FD_ZERO(&read_fds);
321 FD_SET(STDIN_FILENO, &read_fds);
322 timeout.tv_sec = 0;
323 timeout.tv_usec = 200000; /**< 200ms timeout to check for input */
324 ready = select(STDIN_FILENO + 1, &read_fds, nullptr, nullptr, &timeout);
325 if (ready == 0) {
326 f_wait = true;
327 remaining = wait_timeout;
328 wait_chyron = wait_mk_chyron();
329 wait_win = wait_mk_win(wait_chyron, "WAITING for VIEW INPUT");
330 }
331 cmd_key = 0;
332 while (ready == 0 && remaining > 0 && cmd_key != KEY_F(9)) {
333 cmd_key = wait_continue(wait_win, wait_chyron, remaining);
334 if (cmd_key == KEY_F(9))
335 break;
336 FD_ZERO(&read_fds);
337 FD_SET(STDIN_FILENO, &read_fds);
338 timeout.tv_sec = 0;
339 timeout.tv_usec = 0;
340 ready =
341 select(STDIN_FILENO + 1, &read_fds, nullptr, nullptr, &timeout);
342 remaining--;
343 }
344 if (f_wait) {
345 if (wait_chyron != nullptr)
346 wait_destroy(wait_chyron);
347 }
348 if (cmd_key == KEY_F(9)) {
349 if (view->f_in_pipe && pid > 0) {
350 /** If user cancels while waiting for view input, kill
351 * provider_cmd child process and close pipe */
352 kill(pid, SIGKILL);
353 waitpid(pid, nullptr, 0);
354 close(pipe_fd[P_READ]);
355 }
356 Perror("No view input available");
357 return -1;
358 }
359 if (ready == -1) {
360 Perror("Error waiting for view input");
361 if (view->f_in_pipe && pid > 0) {
362 /** If error occurs while waiting for view input, kill
363 * provider_cmd child process and close pipe */
364 kill(pid, SIGKILL);
365 waitpid(pid, nullptr, 0);
366 close(pipe_fd[P_READ]);
367 }
368 return -1;
369 }
370 if (ready == 0) {
371 Perror("Timeout waiting for view input");
372 if (view->f_in_pipe && pid > 0) {
373 /** If timeout occurs while waiting for view input, kill
374 * provider_cmd child process and close pipe */
375 kill(pid, SIGKILL);
376 waitpid(pid, nullptr, 0);
377 close(pipe_fd[P_READ]);
378 }
379 return -1;
380 }
381 if (ready == 1 && !FD_ISSET(STDIN_FILENO, &read_fds)) {
382 Perror("Unexpected error waiting for view input");
383 if (view->f_in_pipe && pid > 0) {
384 /** If unexpected error occurs while waiting for view input,
385 * kill provider_cmd child process and close pipe */
386 kill(pid, SIGKILL);
387 waitpid(pid, nullptr, 0);
388 close(pipe_fd[P_READ]);
389 }
390 return -1;
391 }
392 /*-----------------------------------------------------------------*/
393 while ((bytes_read = read(STDIN_FILENO, buf, sizeof(buf))) > 0) {
394 if (write(view->in_fd, buf, bytes_read) != bytes_read) {
395 abend(-1, "unable to write tmp");
396 exit(EXIT_FAILURE);
397 }
398 bytes_written += bytes_read;
399 }
400 if (bytes_written == 0) {
401 abend(-1, "unable to read stdin");
402 exit(EXIT_FAILURE);
403 }
404 if (fstat(view->in_fd, &sb) == -1) {
405 abend(-1, "fstat failed");
406 exit(EXIT_FAILURE);
407 }
408 view->file_size = sb.st_size;
409 if (view->file_size == 0) {
410 close(view->in_fd);
411 strnz__cpy(tmp_str, "no standard input", MAXLEN - 1);
412 abend(-1, tmp_str);
413 exit(EXIT_FAILURE);
414 }
415 waitpid_with_timeout(pid, wait_timeout);
416 // waitpid(-1, nullptr, 0);
417 }
418 // Memory-map the input file for efficient access.
419 view->buf =
420 mmap(nullptr, view->file_size, PROT_READ, MAP_PRIVATE, view->in_fd, 0);
421 if (view->buf == MAP_FAILED) {
422 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__, __LINE__ - 2);
423 ssnprintf(em1, MAXLEN - 1, "mmap %s", file_name);
424 strerror_r(errno, em2, MAXLEN);
425 display_error(em0, em1, em2, nullptr);
426 close(view->in_fd);
427 return -1;
428 }
429 close(view->in_fd);
430 view->file_size = sb.st_size;
431 view->prev_file_pos = NULL_POSITION;
432 view->buf_curr_ptr = view->buf;
433 if (view->cmd_all[0] != '\0')
434 strnz__cpy(view->cmd, view->cmd_all, MAXLEN - 1);
435 for (idx = 0; idx < NMARKS; idx++)
436 view->mark_tbl[idx] = NULL_POSITION;
437 strnz__cpy(view->cur_file_str, file_name, MAXLEN - 1);
438 base_name(view->file_name, view->cur_file_str);
439 return 0;
