60/** @brief Main initialization function for MAPP - Menu Application
61 @ingroup init
62 @param init - pointer to Init struct to be initialized
63 @param argc - argument count from main()
64 @param argv - argument vector from main()
65 @code
66 1. Read environment variables and set defaults
67 2. Parse configuration file
68 3. Parse command-line options
69 4. Set up SIO struct with colors and other settings
70 5. Handle special options like help and version
71 @endcode
72 */
73void mapp_initialization(Init *init, int argc, char **argv) {
74 char term[MAXLEN];
75 char tmp_str[MAXLEN];
76 char *t;
77 setlocale(LC_ALL, "en_US.UTF-8");
78 SIO *sio = init->sio;
79 if (!init) {
80 ssnprintf(tmp_str, sizeof(tmp_str), "%s",
81 "init struct not allocated on entry");
82 abend(-1, tmp_str);
83 exit(-1);
84 }
85 if (init->minitrc[0] == '\0')
86 strnz__cpy(init->minitrc, "~/.minitrc", MAXLEN - 1);
87 strnz__cpy(sio->bg_clr_x, "#000007",
88 COLOR_LEN - 1); /**< background color */
89 strnz__cpy(sio->fg_clr_x, "#c0c0c0",
90 COLOR_LEN - 1); /**< foreground color */
91 strnz__cpy(sio->bo_clr_x, "#f00000", COLOR_LEN - 1); /**< bold color */
92 strnz__cpy(sio->ln_clr_x, "#0070ff",
93 COLOR_LEN - 1); /**< line number olor */
94 strnz__cpy(sio->ln_bg_clr_x, "#101010",
95 COLOR_LEN - 1); /**< line number background */
96 init->f_at_end_clear = true; /**< clear screen on exit (obsolete) */
97 init->f_erase_remainder = true; /**< erase remainder on enter */
98 init->brackets[0] = '\0'; /**< field enclosure brackets */
99 strnz__cpy(init->fill_char, "_", MAXLEN - 1); /**< field fill character */
100 init->prompt_type = PT_LONG; /**< view prompt type */
101 init->mapp_spec[0] = '\0'; /**< menu specification file */
102 strnz__cpy(init->mapp_home, "~/menuapp", MAXLEN - 1);
103 strnz__cpy(init->mapp_user, "~/menuapp/user", MAXLEN - 1);
104 strnz__cpy(init->mapp_msrc, "~/menuapp/msrc", MAXLEN - 1);
105 strnz__cpy(init->mapp_data, "~/menuapp/data", MAXLEN - 1);
106 strnz__cpy(init->mapp_help, "~/menuapp/help", MAXLEN - 1);
107 // Priority-4 - cfg_args
108
109 t = getenv("TERM");
110 if (!t || *t == '\0')
111 strnz__cpy(term, "xterm-256color", MAXLEN);
112 t = getenv("EDITOR");
113 if (t && *t != '\0')
114 strnz__cpy(init->editor, "vi", MAXLEN - 1);
115
116 parse_config(init); /**< generally /home/user/.minitrc */
117 if (f_debug)
118 dump_config(init, "Configuration after parse_config");
119 parse_opt_args(init, argc,
120 argv); /**< command-line options override config file */
121 if (f_dump_config) {
122 dump_config(init, "Configuration after parse_config and "
123 "parse_opt_args");
124 } else if (f_debug)
125 dump_config(init, "Configuration after parse_opt_args");
126 if (f_write_config) {
127 write_config(init);
128 }
129 if (f_help) {
130 dump_config(init, "Current Configuration");
131 }
132 if (f_version) {
133 display_version();
134 }
135 if (init->mapp_home[0] != '\0') {
136 expand_tilde(init->mapp_home, MAXLEN - 1);
137 if (!verify_dir(init->mapp_home, R_OK))
138 abend(-1, "MAPP_HOME directory invalid");
139 }
140}

167/** @brief Parse command-line options and set Init struct values accordingly

170 @param argc - argument count
171 @param argv - argument vector
172 Return index of first non-option argument
173 @note Accepts both short and long options
174 */
175int parse_opt_args(Init *init, int argc, char **argv) {
176 int i;
177 int opt;
178 int longindex = 0;
179 int flag = 0;
180 char tmp_str[MAXLEN];
181 char *optstring = "a:b:c:d:ef:g:hi:j:k:m:n:o:p:r:st:u:vw:xzA:B:C:DF:G:H:L:"
182 "MNO:P:Q:R:S:T:VWX:Y:Z";
183 struct option long_options[] = {{"help", 0, &flag, MAPP_HELP},
184 {"mapp_spec", 1, &flag, MAPP_SPEC},
185 {"mapp_data", 1, &flag, MAPP_DATA_DIR},
186 {"mapp_help", 1, &flag, MAPP_HELP_DIR},
187 {"help_spec", 1, &flag, HELP_SPEC},
188 {"mapp_spec", 1, &flag, MAPP_SPEC},
189 {"in_spec", 1, &flag, IN_SPEC},
190 {"out_spec", 1, &flag, OUT_SPEC},
191 {"mapp_msrc", 1, &flag, MAPP_MSRC},
192 {0, 0, 0, 0}};
193 char mapp[MAXLEN];
194 SIO *sio = init->sio;
195 base_name(mapp, argv[0]);
196 optind = 1;
197 while ((opt = getopt_long(argc, argv, optstring, long_options,
198 &longindex)) != -1) {
199 switch (opt) {
200 case 0:
201 switch (flag) {
202 case MAPP_DATA_DIR:
203 strnz__cpy(init->mapp_data, optarg, MAXLEN - 1);
204 break;
205 case MAPP_HELP_DIR:
206 strnz__cpy(init->mapp_help, optarg, MAXLEN - 1);
207 break;
208 case MAPP_SPEC:
209 strnz__cpy(init->mapp_spec, optarg, MAXLEN - 1);
210 break;
211 case HELP_SPEC:

219 break;
220 case MAPP_MSRC:

222 break;

225 }
226 break;
227 case 'a':
228 strnz__cpy(init->minitrc, optarg, MAXLEN - 1);
229 break;
230 case 'b':
231 sio->blue_gamma = str_to_double(optarg);
232 break;
233 case 'c':

236 case 'd':

245 case 'g':

248 case 'h':

253 break;
254 case 'j':

256 break;
257 case 'k':
258 strnz__cpy(init->parent_cmd, optarg, MAXLEN - 1);
259 break;
260 case 'm':
261 strnz__cpy(init->mapp_home, optarg, MAXLEN - 1);
262 break;
263 case 'n':
264 init->select_max = atoi(optarg);
265 break;
266 case 'o':
267 strnz__cpy(init->out_spec, optarg, MAXLEN - 1);
268 break;
269 case 'p':
270 strnz__cpy(init->prompt_str, optarg, MAXLEN - 1);
271 break;
272 case 'r':
273 sio->red_gamma = str_to_double(optarg);
274 break;
275 case 's':
276 init->f_squeeze = true;
277 break;
278 case 't':
279 init->tab_stop = atoi(optarg);
280 if (init->tab_stop < 1)
281 init->tab_stop = 1;
282 break;
283 case 'u': // brackets
284 strnz__cpy(init->brackets, optarg, 2);
285 break;
286 case 'v':
287 case 'V':
288 f_version = true;
289 break;
290 case 'x':
291 init->f_ignore_case = true;
292 break;
293 case 'y':
294 init->f_at_end_remove = true;
295 break;
296 case 'z':
297 init->f_at_end_clear = true;
298 break;
299 case 'A':
300 strnz__cpy(init->cmd_all, optarg, MAXLEN - 1);
301 break;
302 case 'B':
303 strnz__cpy(sio->bg_clr_x, optarg, MAXLEN - 1);
304 break;
305 case 'C':
306 init->cols = atoi(optarg);
307 break;
308 case 'D':
309 f_dump_config = true;
310 break;
311 case 'F':
312 strnz__cpy(sio->fg_clr_x, optarg, MAXLEN - 1);
313 break;
314 case 'G':
315 sio->gray_gamma = str_to_double(optarg);
316 break;
317 case 'H':
318 strnz__cpy(init->help_spec, optarg, MAXLEN - 1);
319 break;
320 case 'L':
321 init->lines = atoi(optarg);
322 break;
323 case 'M':
324 init->f_multiple_cmd_args = true;
325 break;
326 case 'N':
327 init->f_ln = true;
328 break;
329 case 'O':
330 strnz__cpy(sio->bo_clr_x, optarg, MAXLEN - 1);
331 break;
332 case 'P':
333 strnz__cpy(tmp_str, optarg, MAXLEN - 1);
334 init->prompt_type = prompt_str_to_int(tmp_str);
335 break;
336 case 'R':
337 strnz__cpy(init->receiver_cmd, optarg, MAXLEN - 1);
338 break;
339 case 'S':
340 strnz__cpy(init->provider_cmd, optarg, MAXLEN - 1);
341 break;
342 case 'T':
343 strnz__cpy(init->title, optarg, MAXLEN - 1);
344 break;
345 case 'U':
346 strnz__cpy(init->mapp_user, optarg, MAXLEN - 1);
347 break;
348 case 'W':
349 f_write_config = true;
350 break;
351 case 'X':
352 init->begx = atoi(optarg);
353 break;
354 case 'Y':
355 init->begy = atoi(optarg);
356 break;
357 case 'Z':
358 init->f_stop_on_error = true;
359 break;
360 default:
361 break;
362 }
363 }
364 init->argv[0] = strdup(argv[0]);
365 i = 0;
366 while (i < argc) {
367 init->argv[i] = strdup(argv[i]);
368 i++;
369 }
370 init->argv[i] = nullptr;
371 init->argc = argc;
372 return optind;
373}

671/** @brief Convert prompt type string to integer constant
672 @ingroup init
673 @param s - prompt type string
674 @returns prompt type integer constant
675 @note Valid prompt type strings are "short", "long", and "string"
676 @note Returns PT_NONE for unrecognized prompt type strings
677 */
678int prompt_str_to_int(char *s) {
679 int prompt_type;
680 str_to_lower(s);
681 if (strcmp(s, "short") == 0)
682 prompt_type = PT_SHORT;
683 else if (strcmp(s, "long") == 0)
684 prompt_type = PT_LONG;
685 else if (strcmp(s, "string") == 0)
686 prompt_type = PT_STRING;
687 else
688 prompt_type = PT_NONE;
689 return prompt_type;
690}
691/** @brief Convert prompt type integer constant to string
692 @ingroup init
693 @param s - output prompt type string
694 @param prompt_type - prompt type integer constant
695 @note Valid prompt type integer constants are PT_SHORT, PT_LONG, and
696 PT_STRING
697 @note Returns "none" for unrecognized prompt type integer constants
698 */
699void prompt_int_to_str(char *s, int prompt_type) {
