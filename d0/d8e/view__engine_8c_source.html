814 wmove(view->cmdln.win, view->cmd_line, view->curx);
815 }
816 }
817 break;
818 case '\n':
819 case KEY_ENTER:
820 return c;
821 case '\033':
822 case KEY_F(9):
823 return c;
824 case KEY_MOUSE:
825 continue;
826 default:
827 *cmd_p++ = (char)c;
828 *cmd_p = '\0';
829 if ((char)c < ' ') {
830 waddch(view->cmdln.win, '^');
831 c |= '@';
832 } else if ((uchar)c == 0x7f)
833 c = '?';
834 waddch(view->cmdln.win, (char)c);
835 if (cmd_p >= cmd_e)
836 return 0;
837 if (numeric_arg && (c < '0' || c > '9'))
838 return -1;
839 break;
840 }
841 }
842 return c;
843}

1014bool search(View *view, int *search_cmd, char *regex_pattern) {
1015 char tmp_str[MAXLEN];
1016 int REG_FLAGS = 0;
1017 regmatch_t pmatch[1];
1018 regex_t compiled_regex;
1019 int reti;
1020 int line_offset;
1021 int line_len;
1022 int match_len;
1023 off_t prev_ln;
1024 bool f_page = false;
1025 if (*regex_pattern == '\0')
1026 return false;
1027 if (view->f_ignore_case)
1028 REG_FLAGS = REG_ICASE | REG_EXTENDED;
1029 else
1030 REG_FLAGS = REG_EXTENDED;
1031 reti = regcomp(&compiled_regex, regex_pattern, REG_FLAGS);
1032 if (reti) {
1033 Perror("Invalid pattern");
1034 return false;
1035 }
1037 while (1) {
1039 if (*search_cmd == '/') {
1040 if (view->srch_curr_pos == view->file_size)
1041 view->srch_curr_pos = 0;
1042 } else {
1043 if (view->srch_curr_pos == 0)
1044 view->srch_curr_pos = view->file_size;
1045 }
1046 if (view->srch_curr_pos == view->srch_beg_pos) {
1047 if (view->f_first_iter == true) {
1048 view->f_first_iter = false;
1049 view->f_search_complete = false;
1050 if (*search_cmd == '/')
1051 view->cury = 0;
1052 else
1053 view->cury = view->scroll_lines + 1;
1054 } else {
1055 view->f_search_complete = true;
1056 return true;
1057 }
1058 }
1059 view->file_pos = view->srch_curr_pos;
1060 sync_ln(view);
1062 if (*search_cmd == '/') {
1063 if (view->cury == view->scroll_lines)
1064 return true;
1065 prev_ln = view->ln;
1066 view->srch_curr_pos = get_next_line(view, view->srch_curr_pos);
1067 view->page_bot_pos = view->srch_curr_pos;
1068 } else {
1069 if (view->cury == 0)
1070 return true;
1071 view->srch_curr_pos = get_prev_line(view, view->srch_curr_pos);
1072 prev_ln = view->ln;
1073 view->page_top_pos = view->srch_curr_pos;
1074 }
1075 fmt_line(view);
1076 reti = regexec(&compiled_regex, view->stripped_line_out,
1077 compiled_regex.re_nsub + 1, pmatch, REG_FLAGS);
1078 if (reti == REG_NOMATCH) {
1079 if (f_page) {
1081 if (*search_cmd == '?')
1082 view->cury -= 2;
1083 display_line(view);
1084 if ((*search_cmd == '/' && view->cury == view->scroll_lines) ||
1085 (*search_cmd == '?' && view->cury == 1)) {
1086 break;
1087 }
1088 }
1089 continue;
1090 }
1091 if (reti) {
1092 char err_str[MAXLEN];
1093 regerror(reti, &compiled_regex, err_str, sizeof(err_str));
1094 strnz__cpy(tmp_str, "Regex match failed: ", MAXLEN - 1);
1095 strnz__cat(tmp_str, err_str, MAXLEN - 1);
1096 Perror(tmp_str);
1097 regfree(&compiled_regex);
1098 return false;
1099 }
1101 if (!f_page) {
1102 if (*search_cmd == '/') {
1103 view->page_top_ln = prev_ln;
1104 wmove(view->pad, view->cury, 0);
1105 } else {
1106 view->page_bot_ln = view->ln;
1107 wmove(view->pad, 0, 0);
1108 }
1109 wclrtobot(view->pad);
1110 f_page = true;
1111 }
1112 if (*search_cmd == '?')
1113 view->cury -= 2;
1114 display_line(view);
1118 view->first_match_x = -1;
1119 view->last_match_x = 0;
1120 line_len = strlen(view->stripped_line_out);
1121
1122 line_offset = 0;
1123 while (1) {
1124 view->curx = line_offset + pmatch[0].rm_so;
1125 match_len = pmatch[0].rm_eo - pmatch[0].rm_so;
1126 mvwchgat(view->pad, view->cury - 1, view->curx, match_len,
1127 WA_NORMAL, cp_nt_rev, nullptr);
1128 if (view->first_match_x == -1)
1129 view->first_match_x = pmatch[0].rm_so;
1130 view->last_match_x = line_offset + pmatch[0].rm_eo;
1131 line_offset += pmatch[0].rm_eo;
1132 if (line_offset >= line_len)
1133 break;
1134 view->line_out_p = view->stripped_line_out + line_offset;
1135 reti = regexec(&compiled_regex, view->line_out_p,
1136 compiled_regex.re_nsub + 1, pmatch, REG_FLAGS);
1137 if (reti == REG_NOMATCH)
1138 break;
1139 if (reti) {
1140 char msgbuf[100];
1141 regerror(reti, &compiled_regex, msgbuf, sizeof(msgbuf));
1142 sprintf(tmp_str, "Regex match failed: %s", msgbuf);
1143 Perror(tmp_str);
1144 regfree(&compiled_regex);
1145 return false;
1146 }
1147 if (*search_cmd == '/') {
1148 if (view->cury == view->scroll_lines) {
1149 regfree(&compiled_regex);
1150 return true;
1151 }
1152 } else if (view->cury == 1) {
1153 regfree(&compiled_regex);
1154 return true;
1155 }
1156 }
1157 }
1158 view->file_pos = view->srch_curr_pos;
1159#ifdef DEBUG_SEARCH
1161 ssnprintf(view->tmp_prompt_str, MAXLEN - 1,
1162 "%s|%c%s|Pos %zu-%zu|(%zd) %zu %zu", view->file_name, *search_cmd,
1163 regex_pattern, view->page_top_pos, view->page_bot_pos,
1164 view->file_size, view->srch_beg_pos, view->srch_curr_pos);
1165#else
1166 if (view->last_match_x > view->maxcol)
1167 ssnprintf(view->tmp_prompt_str, MAXLEN - 1,
1168 "%s|%c%s|Match Cols %d-%d of %d-%d|(%zd%%)", view->file_name,
1169 *search_cmd, regex_pattern, view->first_match_x,
1170 view->last_match_x, view->pmincol, view->smaxcol - view->begx,
1171 (view->page_bot_pos * 100 / view->file_size));
1172 else
1173 ssnprintf(view->tmp_prompt_str, MAXLEN - 1,
1174 "%s|%c%s|Pos %zu-%zu|(%zd%%)", view->file_name, *search_cmd,
1175 regex_pattern, view->page_top_pos, view->page_bot_pos,
1176 (view->page_bot_pos * 100 / view->file_size));
1177#endif
1178 regfree(&compiled_regex);
1179 pad_refresh(view);
1180 return true;
1181}
