50/** @brief Initialize form data structure and parse description file
51 @ingroup form_engine
52 @param init A pointer to the Init structure containing form data and state.
53 @param argc The number of command line arguments passed to the form.
54 @param argv The array of command line arguments passed to the form.
55 @param begy The y-coordinate for the top-left corner of the form window.
56 @param begx The x-coordinate for the top-left corner of the form window.
57 @return 0 on success, or a non-zero value if an error occurs during
58*/
59int init_form(Init *init, int argc, char **argv, int begy, int begx) {
60 int rc;
61 char tmp_str[MAXLEN];
62 if (init->form != nullptr)
63 destroy_form(init);
64 init->form = new_form(init, argc, argv, begy, begx);
65 form = init->form;
66 if (!form->f_mapp_spec) {
67 if (form->mapp_spec[0] == '\0') {
68 rc = Perror("Error: No form specification file given");
69 } else {
70 strnz__cpy(tmp_str, "form->mapp_spec: ", MAXLEN - 1);
71 strnz__cat(tmp_str, form->mapp_spec, MAXLEN - 1);
72 strnz__cat(tmp_str, " not found", MAXLEN - 1);
73 rc = Perror(tmp_str);
74 }
75 return rc;
76 }
77 if (begy != 0)
78 form->begy = begy;
79 if (begx != 0)
80 form->begx = begx;
81 if ((form->f_in_spec && (form->in_spec[0] == '\0')) ||
82 (strcmp(form->in_spec, "-") == 0) ||
83 strcmp(form->in_spec, "/dev/stdin") == 0) {
84 strnz__cpy(form->in_spec, "/dev/stdin", MAXLEN - 1);
85 form->f_in_pipe = true;
86 }
87 if (form->title[0] == '\0')
88 strnz__cpy(form->title, form->in_spec, MAXLEN - 1);
89 rc = form_engine(init);
90 if (form->win)
91 win_del();
92 destroy_form(init);
93 return rc;
94}
95/** @brief Form main processing loop
96 @ingroup form_engine
97 @param init A pointer to the Init structure containing form data and state.
98 @return 0 on successful completion, or a non-zero value if the user cancels
99 the form or if an error occurs during processing.
100 1. Parse the form description file to populate the form data structure.
101 2. Read any initial data for the form fields from a specified input
102 source.
103 3. Display the form on the screen with the initial field values.
104 4. Enter a loop to handle user input for field entry, calculation, help,
105 and cancellation:
106 a. If the user selects the accept action, perform any necessary
107 calculations or post-processing, and then either return to field entry
108 or exit the loop if the form is accepted.
109 b. If the user selects the help action, display the help screen and
110 return to the form after the user exits the help screen.
111 c. If the user selects the cancel action, exit the loop and return a
112 cancel status. */
113int form_engine(Init *init) {
114 char tmp_str[MAXLEN];
115 int form_action;
116
117 form = init->form;
118 if (form == nullptr) {
119 Perror("FORM: form data structure is nullptr");
120 }
121 if (form_parse_desc(form)) {
122 destroy_form(init);
123 return 0;
124 }
125 form_read_data(form);
126 form_display_screen(init);
127 form->fidx = 0;
128 form_action = 0;
129 while (1) {
130 if (form_action == 0 || form_action == P_CONTINUE)
131 form_action = form_enter_fields(form);
132 switch (form_action) {
133 case P_ACCEPT:
134 if (form->f_calculate)
135 form_action = form_calculate(init);
136 else
137 form_action = form_end_fields(init);
138 if (form_action == P_HELP || form_action == P_CANCEL ||
139 form_action == P_CONTINUE || form_action == P_END)
140 continue;
141 if (form_action == P_ACCEPT) {
142 form_action = P_END;
143 continue;
144 }
145 break;
146 case P_END:
147 if (form->f_out_spec || form->out_spec[0] != '\0')
148 form_write(form);
149 if (form->f_receiver_cmd) {
150 form_exec_cmd(form);
151 form_action = P_CONTINUE;
152 continue;
153 }
154 return 0;
155 case P_HELP:
156 eargv[0] = strdup("mview");
157 strnz__cpy(tmp_str, "~/menuapp/help/form.help", MAXLEN - 1);
158 expand_tilde(tmp_str, MAXLEN - 1);
159 eargv[1] = strdup(tmp_str);
160 eargv[2] = nullptr;
161 eargc = 2;
162 zero_opt_args(init);
163 parse_opt_args(init, eargc, eargv);
164 init->lines = 30;
165 init->cols = 60;
166 init->begy = menu->begy + 1;
167 init->begx = menu->begx + 1;
168 strnz__cpy(init->title, "Form Help", MAXLEN - 1);
169 mview(init, eargc, eargv);
170 restore_wins();
171 form_action = P_CONTINUE;
172 break;
173 case P_CANCEL:
174 return P_CANCEL;
175 default:
176 form_action = P_CONTINUE;
177 break;
178 }
179 }
180 return 0;
181}
182/** @brief Handle post-processing after field entry, allowing user to edit data,

184 @ingroup form_engine

193 wmove(form->win, form->lines - 1, 0);

196 set_chyron_key(form->chyron, 8, "F8 Edit", KEY_F(8));
197 set_chyron_key(form->chyron, 10, "F10 Commit", KEY_F(10));

201 form_display_chyron(form);

204 }
205 switch (c) {
206 case KEY_F(1):
207 return P_HELP;
208 case KEY_F(8):

210 loop = false;
211 rc = P_CONTINUE;
212 break;
213 }
214 continue;
215 case KEY_F(9):
216 loop = false;
217 rc = P_CANCEL;
218 break;
219 case KEY_F(10):
220 loop = false;
221 rc = P_ACCEPT;
222 break;

226 break;
227 }
228 }
229 unset_chyron_key(form->chyron, 8);
230 form_display_chyron(form);
231 return rc;
232}

234/** @brief Handle integration with an external program for calculation, allowing
235 the user to execute a provider command and read results back into the form
236 fields.
237 @ingroup form_engine
238 @param init A pointer to the Init structure containing form data and state.
239 @return An integer status code indicating the next action for the form
240 processing loop (e.g., P_CONTINUE, P_CANCEL, P_ACCEPT, P_HELP).
241 @details This function provides integration with external programs.
242 The requirements are:
243 1. The form description file must have a line starting with 'C' to
244 indicate that the form supports calculation.
245 2. The form description file must specify the provider command using a
246 line starting with '!' followed by the command and its arguments.
247 3. The external program must be able to accept field data from a file,
248 from standard input, or as command line parameters.
249 4. The external program must output the calculated field values in a
250 format that can be read by the form (e.g., one value per line), either
251 to a file or to standard output.
252
253 The sequence of operations is as follows:
254
255 1. The 'C' option causes Form to pause and display an KEY_F(5) Calculate
256 option on the chyron.
257 2. The user can then cancel the operation by pressing KEY_F(9) or
258 activate the calculate option by pressing KEY_F(5).
259 3. Form outputs its data to file, standard output, or as command line
260 parameters.
261 4. Form executes the external program.
262 5. The external program processes the data and outputs the results.
263 6. Form reads the results and populates the appropriate form fields.
264 7. Form presents the user with an option to edit the data, and the
265 sequence restarts at 1.
266
267 This function forks and executes the provider executable as a child
268 process, creates a pipe to read the output from the provider command,
269 reads the output, and updates the Form fields.
270 */
271int form_calculate(Init *init) {
272 int i, c, rc;
273 char earg_str[MAXLEN + 1];
274 char *eargv[MAXARGS];
275 char file_spec[MAXLEN];
276 bool loop = true;
277 pid_t pid;
278 int pipe_fd[2];
279
280 form = init->form;
281 wmove(form->win, form->lines - 1, 0);
282 wclrtoeol(form->win);
283 unset_chyron_key(form->chyron, 18);
284 unset_chyron_key(form->chyron, 10);
285 set_chyron_key(form->chyron, 5, "F5 Calculate", KEY_F(5));
286
287 while (loop) {
288 form_display_chyron(form);
289 click_y = click_x = -1;
290 tcflush(2, TCIFLUSH);
291 c = xwgetch(form->win, form->chyron);
292 switch (c) {
293 case KEY_F(1):
294 return P_HELP;
295 case KEY_F(5):
296 if (form->f_out_spec)
297 form_write(form);
298 if (form->f_provider_cmd) {
299 strnz__cpy(earg_str, form->provider_cmd, MAXLEN - 1);
300 for (i = 0; i < form->fcnt; i++) {
301 strnz__cat(earg_str, " ", MAXLEN - 1);
302 strnz__cat(earg_str, form->field[i]->accept_s, MAXLEN - 1);
303 }
304 str_to_args(eargv, earg_str, MAXARGS);
305 strnz__cpy(file_spec, eargv[0], MAXLEN - 1);
306 base_name(eargv[0], file_spec);
307 if (pipe(pipe_fd) == -1) {
308 Perror("pipe(pipe_fd) failed in init_form");
309 return (1);
310 }
311 if ((pid = fork()) == -1) {
312 Perror("fork() failed in init_form");
313 return (1);
314 }
315 if (pid == 0) { // Child
316 close(pipe_fd[P_READ]);
317 dup2(pipe_fd[P_WRITE], STDOUT_FILENO);
318 close(pipe_fd[P_WRITE]);
319 execvp(eargv[0], eargv);
320 ssnprintf(em0, MAXLEN, "%s, line: %d", __FILE__,
321 __LINE__ - 2);
322 strnz__cpy(em1, "execvp(", MAXLEN - 1);
323 strnz__cat(em1, eargv[0], MAXLEN - 1);
324 strnz__cat(em1, ", ", MAXLEN - 1);
325 strnz__cat(em1, earg_str, MAXLEN - 1);
326 strnz__cat(em1, ")", MAXLEN - 1);
327 strerror_r(errno, em2, MAXLEN);
328 display_error(em0, em1, em2, nullptr);
329 exit(EXIT_FAILURE);
330 } // Back to parent
331 close(pipe_fd[P_WRITE]);
332 form->in_fp = fdopen(pipe_fd[P_READ], "rb");
333 form->f_in_pipe = true;
334 form_read_data(form);
335 close(pipe_fd[P_READ]);
336 waitpid(pid, nullptr, 0);
337 form_display_fields(form);
338 set_chyron_key(form->chyron, 8, "F5 Edit", KEY_F(5));
339 set_chyron_key(form->chyron, 10, "F10 Commit", KEY_F(10));
340 continue;
341 }
342 break;
343 case KEY_F(8):
344 if (is_set_chyron_key(form->chyron, 8)) {
345 loop = false;
346 rc = P_CONTINUE;
347 break;
348 }
349 continue;
350 case KEY_F(9):
351 loop = false;
352 rc = P_CANCEL;
353 break;
354 case KEY_F(10):
355 loop = false;
356 rc = P_ACCEPT;
357 break;
358 case KEY_MOUSE:
359 break;
360 default:
361 break;
362 }
363 }
364 unset_chyron_key(form->chyron, 8);
365 unset_chyron_key(form->chyron, 5);
366 form_display_chyron(form);
367 return rc;
368}
369/** @brief Handle user input for field entry, allowing navigation between fields
370 and looping until an exit action is selected.
371 @ingroup form_engine
372 @param form A pointer to the Form structure containing form data and state.
373 @return An integer status code indicating the next action for the form
374 processing loop (e.g., P_ACCEPT, P_HELP, P_CALC, P_CANCEL).
375 @note This function manages user input for field entry, including navigation
376 between fields and handling of special keys for accepting, canceling,
377 requesting help, or performing calculations. The function loops until the
378 user selects an exit action (e.g., accept or cancel). */
379int form_enter_fields(Form *form) {
380
381 if (form->fidx < 0)
382 return (-1);
383 while (1) {
384 cmd_key = form_accept_field(form);
385 wrefresh(form->win);
386
387 switch (cmd_key) {
388
389 case KEY_F(10):
390 return (P_ACCEPT);
391
392 case KEY_F(1):
393 return (P_HELP);
394
395 case KEY_F(5):
396 if (form->f_calculate)
397 return (P_CALC);
398 break;
399
400 case KEY_F(9):
401 return (P_CANCEL);
402
403 case KEY_UP:
404 if (form->fidx != 0)
405 form->fidx--;
406 break;
407
408 case '\r':
409 case KEY_ENTER:
410 if (form->fidx < form->fcnt - 1)
411 form->fidx++;
412 else if (form->fidx == form->fcnt - 1)
413 return (P_ACCEPT);
414 break;
415
416 case KEY_DOWN:
417 if (form->fidx < form->fcnt - 1)
418 form->fidx++;
419 else if (form->fidx == form->fcnt - 1)
420 return (P_ACCEPT);
421 break;
422 case KEY_MOUSE:
423 break;
424 default:
425 break;
426 }
427 }
428}
429int form_yx_to_fidx(Form *form, int y, int x) {
430 for (int i = 0; i < form->fcnt; i++) {
431 if (y == form->field[i]->line && x >= form->field[i]->col &&
432 x < form->field[i]->col + form->field[i]->len) {
433 return i;
434 }
435 }
436 return -1; // No field found at the given coordinates
437}

439/** @brief Display the form on the screen, including text elements and fields,
440 and set up the form window based on the form configuration.
441 @ingroup form_engine
442 @param init A pointer to the Init structure containing form data and state.
443 @return 0 on success, or a non-zero value if an error occurs while
444 creating the form window or rendering the form elements. */
445unsigned int form_display_screen(Init *init) {
446 int n;
447 char tmp_str[MAXLEN];
448
449 form = init->form;
450 form->lines = 0;
451 for (n = 0; n < form->dcnt; n++)
452 if (form->text[n]->line > form->lines)
453 form->lines = form->text[n]->line;
454 for (n = 0; n < form->fcnt; n++)
455 if (form->field[n]->line > form->lines)
456 form->lines = form->field[n]->line;
457 form->lines += 3;
458 if (form->lines > (LINES - form->begy))
459 form->lines = LINES - form->begy;
460 for (n = 0; n < form->fcnt; n++) {
461 if (form->field[n]->line >= (form->lines - 2))
462 form->fcnt = n;
463 }
464 form->cols += 2;
465 if (form->cols > (COLS - form->begx - 3))
466 form->cols = COLS - form->begx - 3;
467 if (win_new(form->lines, form->cols, form->begy, form->begx, form->title,
468 0)) {
469 strnz__cpy(tmp_str, "kwin_new failed: ", MAXLEN - 1);
470 strnz__cat(tmp_str, form->title, MAXLEN - 1);
471 Perror(tmp_str);
472 return (1);
473 }
474#ifdef DEBUG
475 immedok(form->win, TRUE);
476#endif
477 form->win = win_win[win_ptr];
478 form->box = win_box[win_ptr];
479 for (n = 0; n < form->dcnt; n++) {
480 strnz(form->text[n]->str, form->cols - 3);
481 mvwaddstr(form->win, form->text[n]->line, form->text[n]->col,
482 form->text[n]->str);
483#ifdef DEBUG
484 wrefresh(form->win);
485#endif
486 }
487 form_display_fields(form);
488 return 0;
489}

532/** @brief Parse the form description file to populate the Form data structure
533 with field definitions, text elements, and other configuration specified in
534 the description file.
535 @ingroup form_engine
536 @param form A pointer to the Form structure containing form data and state.
537 @return 0 on success, or a non-zero value if an error occurs while parsing
538 the description file (e.g., file not found, invalid format, missing
539 directives). */
540int form_parse_desc(Form *form) {
541 FILE *form_desc_fp;
542 char *token;
543 char *s;
544 int cols = 0;
545 int i, l;
546 int in_line_num = 0;
547 char in_buf[BUFSIZ];
548 char tmp_buf[BUFSIZ];
549 char *tmp_buf_p;
550 char tmp_str[BUFSIZ];
551 char delim[5];
552 char directive;
553
554 form_desc_fp = fopen(form->mapp_spec, "r");
555 if (form_desc_fp == nullptr) {
556 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__, __LINE__ - 2);
557 strnz__cpy(em1, "fopen ", MAXLEN - 1);
558 strnz__cat(em1, form->mapp_spec, MAXLEN - 1);
559 strerror_r(errno, em2, MAXLEN);
560 display_error(em0, em1, em2, nullptr);
561 return (1);
562 }
563 for (i = 0; i < FIELD_MAXCNT; i++) {
564 form->field[i] = (Field *)calloc(1, sizeof(Field));
565 if (!form->field[i]) {
566 sprintf(tmp_str, "FORM: calloc failed for fields");
567 abend(EXIT_FAILURE, tmp_str);
568 }
569 }
570 for (i = 0; i < FIELD_MAXCNT; i++) {
571 form->text[i] = (Text *)calloc(1, sizeof(Text));
572 if (!form->field[i]) {
573 sprintf(tmp_str, "FORM: calloc failed for text");
574 abend(EXIT_FAILURE, tmp_str);
575 }
576 }
577 form->didx = 0;
578 form->fidx = 0;
579 form->fcnt = 0;
580 form->cols = 34;
581 while ((fgets(in_buf, MAXLEN, form_desc_fp)) != nullptr) {
582 s = in_buf;
583 in_line_num++;
584 l = trim(in_buf);
585 if (l == 0)
586 continue;
587 if (*s == D_COMMENT)
588 continue;
589 delim[0] = '\n';
590 delim[1] = in_buf[1];
591 delim[2] = '\0';
592 strnz__cpy(tmp_buf, in_buf, MAXLEN - 1);
593 tmp_buf_p = tmp_buf;
594 if (!(token = strtok(tmp_buf_p, delim))) {
595 continue;
596 }
597 directive = *token;
598 switch ((int)directive) {
599 case D_COMMENT:
600 break;
601 case D_CALC:
602 form->f_calculate = true;
603 break;
604 case D_QUERY:
605 form->f_query = true;
606 break;
607 case D_CMD:
608 if (!(token = strtok(nullptr, delim))) {
609 form_desc_error(in_line_num, in_buf,
610 "FORM: receiver_cmd delimiter");
611 continue;
612 }
613 strnz__cpy(form->receiver_cmd, token, MAXLEN - 1);
614 break;
615 case D_HELP:
616 if (!(token = strtok(nullptr, delim))) {
617 form_desc_error(in_line_num, in_buf,
618 "FORM: help_spec delimiter");
619 }
620 strnz__cpy(form->help_spec, token, MAXLEN - 1);
621 break;
622 case D_FIELD:
623 if (form->field[form->fidx] == nullptr) {
624 sprintf(tmp_str, "FORM: calloc failed for fields");
625 abend(EXIT_FAILURE, tmp_str);
626 }
627 if (!(token = strtok(nullptr, delim))) {
628 form_desc_error(in_line_num, in_buf,
629 "FORM: line number delimiter");
630 return 1;
631 }
632 form->field[form->fidx]->line = atoi(token);
633 if (form->field[form->fidx]->line < 0 ||
634 form->field[form->fidx]->line >= FIELD_MAXCNT) {
635 form_desc_error(in_line_num, in_buf,
636 "FORM: invalid line number");
637 return 1;
638 }
639 if (!(token = strtok(nullptr, delim))) {
640 form_desc_error(in_line_num, in_buf,
641 "FORM: column number delimiter");
642 return 1;
643 }
644 form->field[form->fidx]->col = atoi(token);
645 if (form->field[form->fidx]->col < 0 ||
646 form->field[form->fidx]->col >= FIELD_MAXLEN) {
647 form_desc_error(in_line_num, in_buf,
648 "FORM: invalid column number");
649 break;
650 }
651 if (!(token = strtok(nullptr, delim))) {
652 strnz__cpy(tmp_str, in_buf, MAXLEN - 1);
653 form_desc_error(in_line_num, tmp_str, "FORM: length delimiter");
654 break;
655 }
656 form->field[form->fidx]->len = atoi(token);
657 if (form->field[form->fidx]->len < 0 ||
658 form->field[form->fidx]->len > FIELD_MAXLEN) {
659 form_desc_error(in_line_num, in_buf, "FORM: invalid length");
660 break;
661 }
662 if (!(token = strtok(nullptr, delim))) {
