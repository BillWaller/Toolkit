154static error_t parse_opt(int key, char *arg, struct argp_state *state) {
155 Init *init = state->input;
156 SIO *sio = init->sio;
157 switch (key) {
158 case 'a':
159 strnz__cpy(init->minitrc, arg, MAXLEN - 1);
160 break;
161 case 'b':
162 sio->blue_gamma = str_to_double(arg);
163 break;
164 case 'c':
165 strnz__cpy(init->cmd, arg, MAXLEN - 1);
166 break;
167 case 'd':
168 strnz__cpy(init->mapp_spec, arg, MAXLEN - 1);
169 break;
170 case 'e':
171 init->f_erase_remainder = str_to_bool(arg);
172 break;
173 case 'f':
174 strnz__cpy(init->fill_char, arg, 1);
175 break;
176 case 'g':
177 sio->green_gamma = str_to_double(arg);
178 break;
179 case 'i':
180 strnz__cpy(init->in_spec, arg, MAXLEN - 1);
181 break;
182 case 'j':
183 init->f_strip_ansi = true;
184 break;
185 case 'k':
186 strnz__cpy(init->parent_cmd, arg, MAXLEN - 1);
187 break;
188 case 'm':
189 strnz__cpy(init->mapp_home, arg, MAXLEN - 1);
190 break;
191 case 'n':
192 init->select_max = atoi(arg);
193 break;
194 case 'o':
195 strnz__cpy(init->out_spec, arg, MAXLEN - 1);
196 break;
197 case 'r':
198 sio->red_gamma = str_to_double(arg);
199 break;
200 case 's':
201 init->f_squeeze = str_to_bool(arg);
202 break;
203 case 't':
204 init->tab_stop = atoi(arg);
205 if (init->tab_stop < 1)
206 init->tab_stop = 1;
207 break;
208 case 'w':
209 wait_timeout = atoi(arg);
210 wait_timeout = min(wait_timeout, 1);
211 wait_timeout = max(wait_timeout, 29);
212 break;
213 case 'u':
214 strnz__cpy(init->brackets, arg, 2);
215 break;
216 case 'x':
217 init->f_ignore_case = str_to_bool(arg);
218 break;
219 case 'A':
220 strnz__cpy(init->cmd_all, arg, MAXLEN - 1);
221 break;
222 case 'B':
223 strnz__cpy(sio->bg_clr_x, arg, MAXLEN - 1);
224 break;
225 case 'C':
226 init->cols = atoi(arg);
227 break;
228 case 'D':
229 f_dump_config = true;
230 break;
231 case 'F':
232 strnz__cpy(sio->fg_clr_x, arg, MAXLEN - 1);
233 break;
234 case 'G':
235 sio->gray_gamma = str_to_double(arg);
236 break;
237 case 'H':
238 strnz__cpy(init->help_spec, arg, MAXLEN - 1);
239 break;
240 case 'L':
241 init->lines = atoi(arg);
242 break;
243 case 'M':
244 init->f_multiple_cmd_args = str_to_bool(arg);
245 break;
246 case 'N':
247 if (arg)
248 init->f_ln = str_to_bool(arg);
249 else
250 init->f_ln = true;
251 break;
252 case 'O':
253 strnz__cpy(sio->bo_clr_x, arg, MAXLEN - 1);
254 break;
255 case 'R':
256 strnz__cpy(init->receiver_cmd, arg, MAXLEN - 1);
257 break;
258 case 'S':
259 strnz__cpy(init->provider_cmd, arg, MAXLEN - 1);
260 break;
261 case 'T':
262 strnz__cpy(init->title, arg, MAXLEN - 1);
263 break;
264 case 'U':
265 strnz__cpy(init->mapp_user, arg, MAXLEN - 1);
266 break;
267 case 'W':
268 f_write_config = true;
269 break;
270 case 'X':
271 init->begx = atoi(arg);
272 break;
273 case 'Y':
274 init->begy = atoi(arg);
275 break;
276 case MAPP_DATA_DIR:
277 strnz__cpy(init->mapp_data, arg, MAXLEN - 1);
278 break;
279 case MAPP_HELP_DIR:
280 strnz__cpy(init->mapp_help, arg, MAXLEN - 1);
281 break;
282 case MAPP_SPEC:
283 strnz__cpy(init->mapp_spec, arg, MAXLEN - 1);
284 break;
285 case HELP_SPEC:
286 strnz__cpy(init->help_spec, arg, MAXLEN - 1);
287 break;
288 case IN_SPEC:
289 strnz__cpy(init->in_spec, arg, MAXLEN - 1);
290 break;
291 case OUT_SPEC:
292 strnz__cpy(init->out_spec, arg, MAXLEN - 1);
293 break;
294 case MAPP_MSRC:
295 strnz__cpy(init->mapp_msrc, arg, MAXLEN - 1);
296 break;
297 case ARGP_KEY_ARG:
298 if (state->arg_num >= 5)
299 argp_usage(state);
300 init->argv[state->arg_num] = strdup(arg);
301 break;
302 case ARGP_KEY_END:
303 init->argc = state->arg_num;
304 init->argv[state->arg_num + 1] = nullptr;
305 break;
306 default:
307 return ARGP_ERR_UNKNOWN;
308 }
309 return 0;
310}

315/** @brief Main initialization function for MAPP - Menu Application
316 @ingroup init
317 @param init - pointer to Init struct to be initialized
318 @param argc - argument count from main()
319 @param argv - argument vector from main()
320 @code
321 1. Read environment variables and set defaults
322 2. Parse configuration file
323 3. Parse command-line options
324 4. Set up SIO struct with colors and other settings
325 5. Handle special options like help and version
326 @endcode
327 */
328void mapp_initialization(Init *init, int argc, char **argv) {
329 char term[MAXLEN];
330 char tmp_str[MAXLEN];
331 char *e;
332 setlocale(LC_ALL, "en_US.UTF-8");
333 SIO *sio = init->sio;
334 if (!init) {
335 ssnprintf(tmp_str, sizeof(tmp_str), "%s",
336 "init struct not allocated on entry");
337 abend(-1, tmp_str);
338 exit(-1);
339 }
340#ifdef DEBUG_LOG
341 open_cmenu_log();
342#endif
343 e = getenv("CMENU_HOME");
344 if (!e || *e == '\0')
345 strnz__cpy(init->mapp_home, "~/menuapp", MAXLEN);
346 else
347 strnz__cpy(init->mapp_home, e, MAXLEN);
348 e = getenv("CMENU_RC");
349 if (!e || *e == '\0')
350 strnz__cpy(init->minitrc, "~/menuapp/.minitrc", MAXLEN);
351 else
352 strnz__cpy(init->minitrc, e, MAXLEN);
353 if (init->minitrc[0] == '\0')
354 strnz__cpy(init->minitrc, "~/.minitrc", MAXLEN - 1);
355 strnz__cpy(sio->bg_clr_x, "#000007",
356 COLOR_LEN - 1); /**< background color */
357 strnz__cpy(sio->fg_clr_x, "#c0c0c0",
358 COLOR_LEN - 1); /**< foreground color */
359 strnz__cpy(sio->bo_clr_x, "#f00000", COLOR_LEN - 1); /**< bold color */
360 strnz__cpy(sio->ln_clr_x, "#0070ff",
361 COLOR_LEN - 1); /**< line number olor */
362 strnz__cpy(sio->ln_bg_clr_x, "#101010",
363 COLOR_LEN - 1); /**< line number background */
364 init->f_erase_remainder = true; /**< erase remainder on enter */
365 init->brackets[0] = '\0'; /**< field enclosure brackets */
366 strnz__cpy(init->fill_char, "_", MAXLEN - 1); /**< field fill character */
367 init->mapp_spec[0] = '\0'; /**< menu specification file */
368 strnz__cpy(init->mapp_home, "~/menuapp", MAXLEN - 1);
369 strnz__cpy(init->mapp_user, "~/menuapp/user", MAXLEN - 1);
370 strnz__cpy(init->mapp_msrc, "~/menuapp/msrc", MAXLEN - 1);
371 strnz__cpy(init->mapp_data, "~/menuapp/data", MAXLEN - 1);
372 strnz__cpy(init->mapp_help, "~/menuapp/help", MAXLEN - 1);
373
374 e = getenv("TERM");
375 if (e == nullptr || *e == '\0')
376 strnz__cpy(term, "xterm-256color", MAXLEN);
377 else
378 strnz__cpy(term, e, MAXLEN - 1);
379 e = getenv("EDITOR");
380 if (e && *e != '\0')
381 strnz__cpy(init->editor, "vi", MAXLEN - 1);
382 else
383 strnz__cpy(init->editor, e, MAXLEN - 1);
384 parse_config(init);
385 init->argc = argc;
386 argp_parse(&argp, argc, argv, 0, 0, init);
387 if (f_write_config) {
388 write_config(init);
389 exit(EXIT_SUCCESS);
390 }
391 if (f_dump_config) {
392 dump_config(init, "Current Configuration");
393 exit(EXIT_SUCCESS);
394 }
395 if (init->mapp_home[0] != '\0') {
396 expand_tilde(init->mapp_home, MAXLEN - 1);
397 if (!verify_dir(init->mapp_home, R_OK))
398 abend(-1, "MAPP_HOME directory invalid");
399 }
400}
401/** @brief Parse command-line options and set Init struct values accordingly
402 @ingroup init
403 @param init - pointer to Init struct to be populated with option values
404 @param argc - argument count from main()
405 @param argv - argument vector from main()
406 @returns 0 on success, -1 on failure
407 @details This function uses the argp library to parse command-line
408 options defined in the options array. It updates the Init struct with
409 values from the options and handles any special flags for dumping or
410 writing configuration.
411 */
412int parse_opt_args(Init *init, int argc, char **argv) {
413 init->argc = destroy_argv(init->argc, init->argv);
414 argp_parse(&argp, argc, argv, 0, 0, init);
415 return 0;
416}
