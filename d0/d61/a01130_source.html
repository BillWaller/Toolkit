192/** @brief Converts a string into an array of argument strings.
193 @ingroup utility_functions
194 @param argv - array of pointers to arguments
195 @param arg_str - string containing arguments
196 @param max_args - maximum number of arguments to parse
197 @returns argc, a count of allocated vectors in argv
198 @details Handles quoted strings and escaped quotes, preserving text inside
199 quotes as individual arguments. It has been in service for many years without
200 problems.
201 @note The caller is responsible for deallocating the strings in argv. */
202int str_to_args(char **argv, char *arg_str, int max_args) {
203 if (arg_str == nullptr || *arg_str == '\0')
204 return 0;
205 int argc = 0;
206 char *p = arg_str;
207 char tmp_str[MAXLEN];
208 int in_quotes = 0;
209 char *d = tmp_str;
210
211 while (*p != '\0' && argc < max_args) {
212 while (isspace((unsigned char)*p))
213 p++;
214 if (*p == '\0')
215 break;
216 if (*p == '"') {
217 in_quotes = 1;
218 p++;
219 }
220 while (*p != '\0') {
221 if (in_quotes) {
222 if (*p == '\\' && *(p + 1) == '"') {
223 *d++ = '"';
224 p += 2;
225 } else if (*p == '"') {
226 *d++ = '\0';
227 p++;
228 in_quotes = 0;
229 break;
230 } else
231 *d++ = *p++;
232 } else {
233 if (isspace((unsigned char)*p)) {
234 *d++ = '\0';
235 p++;
236 break;
237 } else
238 *d++ = *p++;
239 }
240 }
241 *d = '\0';
242 d = tmp_str;
243 argv[argc++] = strdup(tmp_str);
244 }
245 argv[argc] = nullptr;
246 return argc;
247}
