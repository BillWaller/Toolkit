105/** @brief Fork and exec a command
106 @ingroup exec
107 @param argv - array of arguments for the command to execute
108 @return the return code from the executed command, or -1 on error
109 @note Captures and restores terminal settings around the fork and exec.
110 @note Sets signal handlers to default in the child process.
111 @note Waits for the child process to complete in the parent process.
112 @note Handles errors from fork and execvp, and reports child exit status.
113 @note Restores curses mode and keypad settings after execution.
114 @note Restores window states after execution.
115 @note Uses a temporary string buffer tmp_str for error messages.
116 @note Uses Perror for error reporting.
117 @note Uses sig_dfl_mode and sig_prog_mode for signal handling.
118 @note Uses capture_curses_tioctl and restore_curses_tioctl for terminal
119 settings.
120 @note Uses restore_shell_tioctl for shell terminal settings.
121 @note Uses waitpid to wait for the child process.
122 @note Uses WIFEXITED, WEXITSTATUS, WIFSIGNALED, and WTERMSIG to interpret
123 child status.
124 @note Uses keypad to manage keypad mode in curses.
125 @note Uses restore_wins to restore window states.
126 @note Uses errno for error codes.
127 @note Uses pid_t for process IDs.
128 @note Uses standard file descriptors STDIN_FILENO, STDOUT_FILENO,
129 STDERR_FILENO.
130 @note Uses execvp for executing the command.
131 @note Uses fork for creating a new process.
132 @note Uses ssnprintf for formatting error messages.
133 @note Uses switch-case for handling fork results.
134 @note Uses default shell if SHELL environment variable is not set. */
135int fork_exec(char **argv) {
136 char tmp_str[MAXLEN];
137 pid_t pid;
138 int status;
139 int rc;
140
141 if (argv[0] == 0) {
142 Perror("fork_exec: missing argument for execvp");
143 return (-1);
144 }
145 capture_curses_tioctl();
146 sig_dfl_mode();
147 restore_shell_tioctl();
148
149 tmp_str[0] = '\0';
150 pid = fork();
151 switch (pid) {
152 case -1: // parent fork failed
153 restore_curses_tioctl();
154 sig_prog_mode();
155 keypad(stdscr, true);
156 ssnprintf(tmp_str, sizeof(tmp_str), "fork failed: %s, errno: %d",
157 argv[0], errno);
158 Perror(tmp_str);
159 return (-1);
160 case 0: // child
161 restore_shell_tioctl();
162 execvp(argv[0], argv);
163 restore_curses_tioctl();
164 sig_prog_mode();
165 keypad(stdscr, true);
166 ssnprintf(tmp_str, sizeof(tmp_str), "execvp failed: %s, errno: %d",
167 argv[0], errno);
168 Perror(tmp_str);
169 exit(-1);
170 default: // parent
171 rc = 0;
172 restore_curses_tioctl();
173 sig_prog_mode();
174 waitpid(pid, &status, 0);
175 if (WIFEXITED(status)) {
176 rc = WEXITSTATUS(status);
177 if (rc != 0) {
178 keypad(stdscr, true);
179 ssnprintf(tmp_str, sizeof(tmp_str),
180 "Child %s exited with status %d", argv[0], rc);
181 }
182 } else {
183 if (WIFSIGNALED(status)) {
184 rc = WTERMSIG(status);
185 keypad(stdscr, true);
186 ssnprintf(tmp_str, sizeof(tmp_str),
187 "Child %s terminated by signal %d", argv[0], rc);
188 } else {
189 keypad(stdscr, true);
190 ssnprintf(tmp_str, sizeof(tmp_str),
191 "Child %s terminated abnormally", argv[0]);
192 }
193 }
194 break;
195 }
196 restore_curses_tioctl();
197 sig_prog_mode();
198 keypad(stdscr, true);
199 restore_wins();
200 if (tmp_str[0] != '\0') {
201 Perror(tmp_str);
202 }
203 return (rc);
204}
