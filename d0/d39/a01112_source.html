81static error_t parse_opt(int key, char *arg, struct argp_state *state) {
82 struct LF *lf = state->input;
83 int i = 0;
84
85 switch (key) {
86 case 'a':
87 memset(&tm_info, 0, sizeof(struct tm));
88 if (strptime(arg, "%Y-%m-%dT%H:%M:%S", &tm_info) == NULL) {
89 fprintf(stderr, "-a Failed to parse time string.\n");
90 return 1;
91 }
92 lf->after = mktime(&tm_info);
93 if (lf->after && lf->before && lf->after > lf->before) {
94 fprintf(stderr, "-a time must be before -b time.\n");
95 return 1;
96 }
97 break;
98 case 'b':
99 memset(&tm_info, 0, sizeof(struct tm));
100 if (strptime(arg, "%Y-%m-%dT%H:%M:%S", &tm_info) == NULL) {
101 fprintf(stderr, "-b Failed to parse time string.\n");
102 return 1;
103 }
104 lf->before = mktime(&tm_info);
105 if (lf->after && lf->before && lf->after > lf->before) {
106 fprintf(stderr, "-a time must be before -b time.\n");
107 return 1;
108 }
109 break;
110 case 'd':
111 lf->max_depth = atoi(arg);
112 break;
113 case 'e':
114 lf->exclude = arg;
115 lf->flags |= LF_EXC_REGEX;
116 break;
117 case 'i':
118 lf->flags |= LF_ICASE;
119 break;
120 case 'n':
121 lf->f_hide = true;
122 lf->flags |= LF_HIDE;
123 break;
124 case 'p':
125 for (i = 0; arg[i]; i++) {
126 switch (arg[i]) {
127 case 'g':
128 lf->flags |= LF_SETGID << 16;
129 break;
130 case 'r':
131 lf->flags |= LF_PERM_R << 16;
132 break;
133 case 's':
134 lf->flags |= LF_SETUID << 16;
135 break;
136 case 'w':
137 lf->flags |= LF_PERM_W << 16;
138 break;
139 case 'x':
140 lf->flags |= LF_PERM_X << 16;
141 break;
142 default:
143 break;
144 }
145 }
146 break;
147 case 's':
148 long long ull = a_to_ull(arg);
149 lf->file_size_min = (intmax_t)ull;
150 break;
151 case 't':
152 lf->file_types_p = arg;
153 while (lf->file_types_p[i]) {
154 switch (lf->file_types_p[i++]) {
155 case 'b':
156 lf->flags |= FT_BLK << 8;
157 break;
158 case 'c':
159 lf->flags |= FT_CHR << 8;
160 break;
161 case 'd':
162 lf->flags |= FT_DIR << 8;
163 break;
164 case 'p':
165 lf->flags |= FT_FIFO << 8;
166 break;
167 case 'l':
168 lf->flags |= FT_LNK << 8;
169 break;
170 case 'f': // for regular files, 'f' is more intuitive than 'r'
171 case 'r': // regular files are the most common type, so 'r' is also
172 // accepted
173 lf->flags |= FT_REG << 8;
174 break;
175 case 's':
176 lf->flags |= FT_SOCK << 8;
177 break;
178 case 'u':
179 lf->flags |= FT_UNKNOWN << 8;
180 break;
181 default:
182 break;
183 }
184 }
185 break;
186 case 'u':
187 struct passwd *pwd = getpwnam(arg);
188 if (pwd) {
189 if (pwd->pw_uid > 0xffff) {
190 fprintf(stderr,
191 "User '%s' has UID %d which is too large for this "
192 "program.\n",
193 arg, pwd->pw_uid);
194 exit(EXIT_FAILURE);
195 }
196 lf->flags |= (long)(pwd->pw_uid) << 24;
197 lf->flags |= LF_USER;
198 } else {
199 fprintf(stderr, "User '%s' not found.\n", arg);
200 exit(EXIT_FAILURE);
201 }
202 break;
203 case 'x':
204 strnz__cpy(lf->exec, arg, MAXLEN - 1);
205 break;
206 case 'S':
207 lf->f_sort = true;
208 if (arg && (arg[0] == 'r' || arg[0] == 'R'))
209 lf->f_reverse = true;
210 break;
211 case ARGP_KEY_ARG:
212 if (state->arg_num == 0 || state->arg_num == 1) {
213 lf->args[state->arg_num] = arg;
214 lf->argc = state->arg_num + 1;
215 } else {
216 argp_usage(state);
217 }
218 break;
219 case ARGP_KEY_END:
220 break;
221 default:
222 return ARGP_ERR_UNKNOWN;
223 }
224 return 0;

300/** @brief Execute an external command with the output of lf_find as input.
301 @param lf The LF struct containing the options and flags for lf_find.
302 @param dir The directory to search for files.
303 @param re The regular expression to match file names against.
304 @param eargv The argument vector for the external command to execute.
305 @details This function creates a child process to execute the external
306 command specified by the user. It sets up a pipe to redirect the output of
307 lf_find to the standard input of the external command. The parent process
308 runs lf_find and writes its output to the pipe, while the child process reads
309 from the pipe and executes the external command. After lf_find completes, the
310 parent process waits for the child process to finish before exiting.
311 */
312void external_receiver_command(struct LF lf, char *dir, char *re,
313 char **eargv) {
314 int wstatus;
315 int save_fd = dup(STDOUT_FILENO); // save current STDOUT
316 int fds[2];
317 pipe(fds); // Create the pipes
318 pid_t pid1 = fork();
319 if (pid1 == 0) {
320 close(fds[1]); // Close child write pipe
321 dup2(fds[0], STDIN_FILENO); // Clone child read pipe to STDIN_FILENO
322 close(fds[0]); // Close child read pipe after cloning
323 execvp(eargv[0], eargv); // Execute the external command
324 }
325 close(fds[0]); // Close read end of pipe
326 dup2(fds[1], STDOUT_FILENO); // Clone write pipe to STDOUT_FILENO
327 close(fds[1]); // Close duplicated write pipe
328 setvbuf(stdout, NULL, _IOLBF, 0);
329 lf_find(dir, re, lf.exclude, lf.max_depth, lf.flags, lf.after, lf.before,
330 lf.file_size_min); // Generate the output for external command
331 dup2(save_fd, STDOUT_FILENO); // restore STDOUT
332 waitpid(pid1, &wstatus, 0);
333}
