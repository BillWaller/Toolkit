31/** @brief read the next characater from the virtual file
32 @ingroup view_engine
33 @details Line numbers are tracked when reading forward and stored in a line
34 table for quick access when moving backwards.
35 When reading forward, the End of Data (EOD) flag is set when the
36 position is equal to the file size, and cleared when the position or reading
37 direction changes.
38 Carriage-returns are ignored as they should be.
39 View uses the kernel's demand paged virtual address space to map files
40 directly into memory. This allows for efficient access to file contents
41 without the need for explicit buffering or read system calls, as the kernel
42 handles loading the necessary pages into memory on demand.
43 */
44#define get_next_char()
45 {
46 c = 0;
47 do {
48 if (view->file_pos == view->file_size) {
49 view->f_eod = true;
50 break;
51 } else
52 view->f_eod = false;
53 c = view->buf[view->file_pos++];
54 } while (c == 0x0d);
55 if (c == '\n')
56 increment_ln(view);
57 }
58/** @brief read the previous characater from the virtual file
59 @ingroup view_engine
60 There is no need to track line numbers when moving backwards as they
61 are stored in the line table and accessed as needed.
62 When reading in reverse, the Beginning of Data (BOD) flag is set when
63 the file position is zero, and cleared when the position or reading direction
64 changes.
65 Carriage-returns are ignored as they should be.
66 */
67#define get_prev_char()
68 {
69 c = 0;
70 do {
71 if (view->file_pos == 0) {
72 view->f_bod = true;
73 break;
74 } else
75 view->f_bod = false;
76 c = view->buf[--view->file_pos];
77 } while (c == 0x0d);
78 }

120/** @brief Start view
121 @ingroup view_engine
122 @param init Pointer to the Init structure containing initialization
123 parameters and state for the view application. This structure is used to pass
124 necessary information and maintain state across different functions within
125 the view application.
126 @return Returns 0 on successful completion of the view application, or a

129int view_file(Init *init) {
130 View *view = init->view;
131 if (view->argc < 1) {
132 view->curr_argc = -1;
133 view->argc = 0;
134 view->next_file_spec_ptr = "-";
135 } else
136 view->next_file_spec_ptr = view->argv[0];
137 while (view->curr_argc < view->argc) {
138 if (view->next_file_spec_ptr == nullptr ||
139 *view->next_file_spec_ptr == '\0') {
140 break;
141 }
142 view->file_spec_ptr = view->next_file_spec_ptr;
143 view->next_file_spec_ptr = nullptr;
144 strnz__cpy(view->cur_file_str, view->file_spec_ptr, MAXLEN - 1);
145 if (view_init_input(view, view->cur_file_str) == 0) {
146 if (view->buf) {
147 view->f_eod = 0;
148 view->f_bod = 0;
149 view->maxcol = 0;
150 view->page_top_pos = 0;
151 view->page_top_ln = 0;
152 view->page_bot_ln = 0;
153 view->ln_max_pos = 0;
154 view->ln = 0;
155 view->page_bot_pos = 0;
156 view->file_pos = 0;
157 initialize_line_table(view);
158 next_page(view);
159 view_cmd_processor(init);
160 destroy_line_table(view);
161 munmap(view->buf, view->file_size);
162 }
163 } else {
164 view->curr_argc++;
165 if (view->curr_argc < view->argc) {
166 view->next_file_spec_ptr = view->argv[view->curr_argc];
167 }
168 }
169 }
170 return 0;
171}

734/** @brief Get Command Argument from User Input
735 @ingroup view_engine
736 @param view Pointer to the View structure containing the state and
737 parameters of the view application. This structure is used to access and
738 modify the state of the application as needed.
739 @param prompt A string containing the prompt to be displayed to the user
740 when requesting input for the command argument. This prompt is shown on
741 the command line to guide the user in providing the necessary input for
742 the command being executed.
743 @return Returns the command character entered by the user, or a special
744 value if a mouse event is detected. The command argument entered by the
745 user is stored in the view->cmd_arg buffer for use by the calling
746 function. The function handles user input, including editing keys, and
747 updates the command argument buffer accordingly. If the user enters a
748 numeric argument, it is validated based on the context of the command
749 being executed.
750*/
751int get_cmd_arg(View *view, char *prompt) {
752 int c;
753 int numeric_arg = false;
754 char *cmd_p;
755 char *cmd_e;
756 char prompt_s[PAD_COLS + 1];
757 char *n;
758 int prompt_l;
759 prompt_l = strnz__cpy(prompt_s, prompt, view->cols - 4);
760 if (view->cmd_arg[0] != '\0')
761 return 0;
762 cmd_p = view->cmd_arg;
763 cmd_e = view->cmd_arg + MAXLEN - 2;
764 wmove(view->cmdln.win, view->cmd_line, 0);
765 if (prompt_l == 0)
766 numeric_arg = true;
767 if (prompt_l > 1) {
768 wstandout(view->cmdln.win);
769 waddch(view->cmdln.win, ' ');
770 waddstr(view->cmdln.win, prompt_s);
771 waddch(view->cmdln.win, ' ');
772 wstandend(view->cmdln.win);
773 } else {
774 if (*prompt == ':')
775 numeric_arg = true;
776 else {
777 n = prompt;
778 if (*n >= '0' && *n <= '9') {
779 *cmd_p++ = *n;
780 *cmd_p = '\0';
781 numeric_arg = true;
782 }
783 }
784 waddstr(view->cmdln.win, prompt_s);
785 wmove(view->cmdln.win, view->cmd_line, prompt_l);
786 }
787 wclrtoeol(view->cmdln.win);
788 while (1) {
789 update_panels();
790 doupdate();
791 c = vgetch(view->cmdln.win, -1);
792 switch (c) {
793 /** Basic Editing Keys for Command Line */
794 case KEY_LEFT:
795 case KEY_BACKSPACE:
796 case '\b':
797 if (cmd_p > view->cmd_arg) {
798 cmd_p--;
799 if (*cmd_p < ' ' || *cmd_p == 0x7f) {
800 getyx(view->cmdln.win, view->cury, view->curx);
801 if (view->curx > 0) {
802 view->curx--;
803 wmove(view->cmdln.win, view->cmd_line, view->curx);
804 waddch(view->cmdln.win, ' ');
805 wmove(view->cmdln.win, view->cmd_line, view->curx);
806 }
807 }
808 getyx(view->cmdln.win, view->cury, view->curx);
809 if (view->curx > 0) {
810 view->curx--;
811 wmove(view->cmdln.win, view->cmd_line, view->curx);
812 waddch(view->cmdln.win, ' ');
813 wmove(view->cmdln.win, view->cmd_line, view->curx);
814 }
815 }
816 break;
817 case '\n':
818 case KEY_ENTER:
819 return c;
820 case '\033':
821 case KEY_F(9):
822 return c;
823 case KEY_MOUSE:
824 continue;
825 default:
826 *cmd_p++ = (char)c;
827 *cmd_p = '\0';
828 if ((char)c < ' ') {
829 waddch(view->cmdln.win, '^');
830 c |= '@';
831 } else if ((uchar)c == 0x7f)
832 c = '?';
833 waddch(view->cmdln.win, (char)c);
834 if (cmd_p >= cmd_e)
835 return 0;
836 if (numeric_arg && (c < '0' || c > '9'))
837 return -1;
838 break;
839 }
840 }
841 return c;
842}

994/** @brief Search for Regular Expression Pattern
995 @ingroup view_navigation
996 @param view Pointer to View Structure
997 @param search_cmd Search Command Character ('/' or '?')
998 @param regex_pattern Regular Expression Pattern to Search For
999 @returns true if a match is found and displayed, false if the search
1000 completes without finding a match or if an error occurs
1001 @details The search performs extended regular expression matching, ignoring
1002 ANSI sequences and Unicode characters. Matches are highlighted on the
1003 screen, and the search continues until the page is full or the end of the
1004 file is reached. If the search wraps around the file, a message is
1005 displayed indicating that the search is complete.
1006 The search state is maintained in the view structure, allowing for
1007 repeat searches and tracking of the current search position. This
1008 function highlights all matches in the current ncurses pad, including
1009 those not displayed on the screen, and tracks the first and last match
1010 columns for prompt display.
1011 ANSI sequences and Unicode characters are stripped before
1012 matching, so matching corresponds to the visual display */
1013bool search(View *view, int *search_cmd, char *regex_pattern) {
1014 char tmp_str[MAXLEN];
1015 int REG_FLAGS = 0;
1016 regmatch_t pmatch[1];
1017 regex_t compiled_regex;
1018 int reti;
1019 int line_offset;
1020 int line_len;
1021 int match_len;
1022 off_t prev_ln;
1023 bool f_page = false;
1024 if (*regex_pattern == '\0')
1025 return false;
1026 if (view->f_ignore_case)
1027 REG_FLAGS = REG_ICASE | REG_EXTENDED;
1028 else
1029 REG_FLAGS = REG_EXTENDED;
1030 reti = regcomp(&compiled_regex, regex_pattern, REG_FLAGS);
1031 if (reti) {
1032 Perror("Invalid pattern");
1033 return false;
1034 }
1035 /** */
1036 while (1) {
1037 /** initialize iteration */
1038 if (*search_cmd == '/') {
1039 if (view->srch_curr_pos == view->file_size)
1040 view->srch_curr_pos = 0;
1041 } else {
1042 if (view->srch_curr_pos == 0)
1043 view->srch_curr_pos = view->file_size;
1044 }
1045 if (view->srch_curr_pos == view->srch_beg_pos) {
1046 if (view->f_first_iter == true) {
1047 view->f_first_iter = false;
1048 view->f_search_complete = false;
1049 if (*search_cmd == '/')
1050 view->cury = 0;
1051 else
1052 view->cury = view->scroll_lines + 1;
1053 } else {
1054 view->f_search_complete = true;
1055 return true;
1056 }
1057 }
1058 view->file_pos = view->srch_curr_pos;
1059 sync_ln(view);
1060 /** get line to scan */
1061 if (*search_cmd == '/') {
1062 if (view->cury == view->scroll_lines)
1063 return true;
1064 prev_ln = view->ln; /**< Note placement before get_next_line */
1065 view->srch_curr_pos = get_next_line(view, view->srch_curr_pos);
1066 view->page_bot_pos = view->srch_curr_pos;
1067 } else {
1068 if (view->cury == 0)
1069 return true;
1070 view->srch_curr_pos = get_prev_line(view, view->srch_curr_pos);
1071 prev_ln = view->ln; /**< Note placement after get_prev_line */
1072 view->page_top_pos = view->srch_curr_pos;
1073 }
1074 fmt_line(view);
1075 reti = regexec(&compiled_regex, view->stripped_line_out,
1076 compiled_regex.re_nsub + 1, pmatch, REG_FLAGS);
1077 if (reti == REG_NOMATCH) {
1078 if (f_page) {
1079 /** non-matching page filler */
1080 if (*search_cmd == '?')
1081 view->cury -= 2;
1082 display_line(view);
1083 if ((*search_cmd == '/' && view->cury == view->scroll_lines) ||
1084 (*search_cmd == '?' && view->cury == 1)) {
1085 break;
1086 }
1087 }
1088 continue;
1089 }
1090 if (reti) {
1091 char err_str[MAXLEN];
1092 regerror(reti, &compiled_regex, err_str, sizeof(err_str));
1093 strnz__cpy(tmp_str, "Regex match failed: ", MAXLEN - 1);
1094 strnz__cat(tmp_str, err_str, MAXLEN - 1);
1095 Perror(tmp_str);
1096 regfree(&compiled_regex);
1097 return false;
1098 }
1099 /** Display matching lines */
1100 if (!f_page) {
1101 if (*search_cmd == '/') {
1102 view->page_top_ln = prev_ln;
1103 wmove(view->pad, view->cury, 0);
1104 } else {
1105 view->page_bot_ln = view->ln;
1106 wmove(view->pad, 0, 0);
1107 }
1108 wclrtobot(view->pad);
1109 f_page = true;
1110 }
1111 if (*search_cmd == '?')
1112 view->cury -= 2;
1113 display_line(view);
1114 /** All matches on the current line are highlighted,
1115 including those not displayed on the screen.
1116 Track first and last match columns for prompt display. */
1117 view->first_match_x = -1;
1118 view->last_match_x = 0;
1119 line_len = strlen(view->stripped_line_out);
1120
1121 line_offset = 0;
1122 while (1) {
1123 view->curx = line_offset + pmatch[0].rm_so;
1124 match_len = pmatch[0].rm_eo - pmatch[0].rm_so;
1125 mvwchgat(view->pad, view->cury - 1, view->curx, match_len,
1126 WA_NORMAL, cp_nt_rev, nullptr);
1127 if (view->first_match_x == -1)
1128 view->first_match_x = pmatch[0].rm_so;
1129 view->last_match_x = line_offset + pmatch[0].rm_eo;
1130 line_offset += pmatch[0].rm_eo;
1131 if (line_offset >= line_len)
1132 break;
1133 view->line_out_p = view->stripped_line_out + line_offset;
1134 reti = regexec(&compiled_regex, view->line_out_p,
1135 compiled_regex.re_nsub + 1, pmatch, REG_FLAGS);
1136 if (reti == REG_NOMATCH)
1137 break;
1138 if (reti) {
1139 char msgbuf[100];
1140 regerror(reti, &compiled_regex, msgbuf, sizeof(msgbuf));
1141 sprintf(tmp_str, "Regex match failed: %s", msgbuf);
1142 Perror(tmp_str);
1143 regfree(&compiled_regex);
1144 return false;
1145 }
1146 if (*search_cmd == '/') {
1147 if (view->cury == view->scroll_lines) {
1148 regfree(&compiled_regex);
1149 return true;
1150 }
1151 } else if (view->cury == 1) {
1152 regfree(&compiled_regex);
1153 return true;
1154 }
1155 }
1156 }
1157 view->file_pos = view->srch_curr_pos;
1158#ifdef DEBUG_SEARCH
1159 /** Statistics for debugging */
1160 ssnprintf(view->tmp_prompt_str, MAXLEN - 1,
1161 "%s|%c%s|Pos %zu-%zu|(%zd) %zu %zu", view->file_name, *search_cmd,
1162 regex_pattern, view->page_top_pos, view->page_bot_pos,
1163 view->file_size, view->srch_beg_pos, view->srch_curr_pos);
1164#else
1165 if (view->last_match_x > view->maxcol)
1166 ssnprintf(view->tmp_prompt_str, MAXLEN - 1,
1167 "%s|%c%s|Match Cols %d-%d of %d-%d|(%zd%%)", view->file_name,
1168 *search_cmd, regex_pattern, view->first_match_x,
1169 view->last_match_x, view->pmincol, view->smaxcol - view->begx,
1170 (view->page_bot_pos * 100 / view->file_size));
1171 else
1172 ssnprintf(view->tmp_prompt_str, MAXLEN - 1,
1173 "%s|%c%s|Pos %zu-%zu|(%zd%%)", view->file_name, *search_cmd,
1174 regex_pattern, view->page_top_pos, view->page_bot_pos,
1175 (view->page_bot_pos * 100 / view->file_size));
1176#endif
1177 regfree(&compiled_regex);
1178 pad_refresh(view);
1179 return true;
1180}

1597/** @brief Synchronize Line Table with Current File Position
1598 @ingroup view_navigation
1599 @param view data Structure
1600 @details The line table (view->ln_tbl) is an array that stores the file
1601 position of each line. The index (view->ln + 1) corresponds to the
1602 current line number. (the line number table is 0-based, while line
1603 numbering starts at 1).
1604 @details If the line or position requested is not in the line table, this
1605 function reads forward to sycn.
1606 If the line or positione requested is behind the current line
1607 table index, the line index will be decremented it matches the file
1608 position.
1609 */
1610void sync_ln(View *view) {

1612 off_t idx;
1613 off_t target_pos;
1614 if (view->ln_tbl[view->ln] == view->file_pos)
1615 return;
1616 target_pos = view->file_pos;
1617 view->file_pos = view->ln_tbl[view->ln_tbl_cnt];
1618 if (view->file_pos < target_pos) {
1619 view->ln = view->ln_tbl_cnt;
1620 while (view->ln_max_pos < target_pos) {
1621 get_next_char();
1622 if (view->f_eod)
1623 return;
1624 }
1625 } else if (view->ln_tbl[view->ln] > target_pos) {
1626 idx = view->ln - 1;
1627 while (view->ln_tbl[idx] > target_pos)
1628 idx--;
1629 view->ln = idx;
1630 view->file_pos = view->ln_tbl[view->ln];
1631 } else {
1632 view->ln = view->ln_tbl_cnt;
1633 view->file_pos = view->ln_tbl[view->ln];
1634 }
1635}

1672/** @brief Format Line for Display
1673 @ingroup view_display
1674 @param view pointer to View structure containing line input and output
1675 buffers
1676 @return length of formatted line in characters
1677 @details This function processes the input line from view->line_in_s,
1678 handling ANSI escape sequences for text attributes and colors, as well as
1679 multi-byte characters. It converts the input line into a formatted line
1680 suitable for display in the terminal, storing the result in
1681 view->cmplx_buf and view->stripped_line_out. The function returns the
1682 length of the formatted line in characters, which may be used for
1683 tracking the maximum column width of the displayed content.
1684 */
1685int fmt_line(View *view) {
1686 char ansi_tok[MAXLEN];
1687 int i = 0, j = 0;
1688 int len = 0;
1689 const char *s;
1690 attr_t attr = WA_NORMAL;
1691 int cpx = cp_nt;
1692 cchar_t cc = {0};
1693 wchar_t wstr[2] = {L'\0', L'\0'};
1694
1695 char *in_str = view->line_in_s;
1696 cchar_t *cmplx_buf = view->cmplx_buf;
1697
1698 rtrim(view->line_out_s);
1699 mbstate_t mbstate;
1700 memset(&mbstate, 0, sizeof(mbstate));
1701 while (in_str[i] != '\0') {
1702 if (in_str[i] == '\033' && in_str[i + 1] == '[') {
1703 /** This section calls the decoder for ANSI escape sequences */
1704 len = strcspn(&in_str[i], "mK ") + 1;
1705 memcpy(ansi_tok, &in_str[i], len + 1);
1706 ansi_tok[len] = '\0';
1707 if (ansi_tok[0] == '\0') {
1708 if (i + 2 < MAXLEN)
1709 i += 2;
1710 continue;
1711 }
1712 if (len == 0 || in_str[i + len - 1] == ' ') {
1713 i += 2;
1714 continue;
1715 } else if (in_str[i + len - 1] == 'K') {
1716 i += len;
1717 continue;
1718 }
1719 parse_ansi_str(ansi_tok, &attr, &cpx);
1720 i += len;
1721 } else {
1722 /** This section converts the input string to wide characters,
1723 handling tabs and multi-byte characters, and stores the result
1724 in the complex buffer for display. ANSI escape sequences are
1725 ignored in this section since they are handled separately above.
1726 */
1727 if (in_str[i] == '\033') {
1728 i++;
1729 continue;
1730 }
1731 s = &in_str[i];
1732 if (*s == '\t') {
1733 do {
1734 wstr[0] = L' ';
1735 wstr[1] = L'\0';
1736 setcchar(&cc, wstr, attr, cpx, nullptr);
1737 view->stripped_line_out[j] = ' ';
1738 cmplx_buf[j++] = cc;
1739 } while ((j < PAD_COLS - 2) && (j % view->tab_stop != 0));
1740 i++;
1741 } else {
1742 wstr[1] = L'\0';
1743 len = mbrtowc(wstr, s, MB_CUR_MAX, &mbstate);
1744 if (len <= 0) {
1745 wstr[0] = L'?';
1746 wstr[1] = L'\0';
1747 len = 1;
1748 }
1749 if (setcchar(&cc, wstr, attr, cpx, nullptr) != ERR) {
1750 if (len > 0 && (j + len) < PAD_COLS - 1) {
1751 view->stripped_line_out[j] = *s;
1752 cmplx_buf[j++] = cc;
1753 }
1754 }
1755 i += len;
1756 }
1757 }
1758 }
1759 if (j > view->maxcol)
1760 view->maxcol = j;
1761 wstr[0] = '\0';
1762 wstr[1] = '\0';
1763 setcchar(&cc, wstr, WA_NORMAL, cpx, nullptr);
1764 cmplx_buf[j] = cc;
1765 view->stripped_line_out[j] = '\0';
1766 return j;
1767}
1768/** @brief Parse ANSI SGR Escape Sequence
1769 @ingroup view_display
1770 @param ansi_str is the ANSI escape sequence string to parse
1771 @param attr is a pointer to an attr_t variable where the parsed
1772 attributes will be stored
1773 @param cpx is a pointer to an int variable where the parsed color pair
1774 index will be stored
1775 @details This function parses an ANSI escape sequence and updates the color
1776 pair and color tables according to the attributes specified. Despite the
1777 depth of nested conditionals, with an understanding of the ANSI Select
1778 Graphics Rendition (SGR) scheme, which is defined in the ECMA-48 standard
1779 (ISO/IEC 6429), you will find that it is exceptionally simple and
1780 straightforward.
1781
1782 @verbatim
1783
1784 This function converts the following SGR specification types to the
1785 appropriate curses color pair index for use in the terminal display.
1786
1787 RGB:
1788
1789 foreground \033[38;2;r;g;bm
1790 background \033[48;2;r;g;bm
1791
1792 Where r, g, b are the red, green, and blue color components
1793 (0-255)
1794
1795 XTERM 256-color:
1796
1797 foreground \033[38;5;xm
1798 background \033[48;5;xm
1799
1800 Where x is the 256-color index (0-255)
1801
1802 uses xterm256_idx_to_rgb() to convert the 256-color index to
1803 RGB
1804
1805 8-color:
1806
1807 foreground \033[3cm
1808 background \033[4cm
1809
1810 Where c is the color code (0 for black, 1 for red, 2 for green, 3
1811 for yellow, 4 for blue, 5 for magenta, 6 for cyan, 7 for white).
1812
1813 Attributes:
1814
1815 \033[am
1816
1817 Where a is the attribute code (1 for bold, 2 for dim, 3 for italic,
1818 4 for underline, 5 for blink, 7 for reverse, 8 for invis). The function
1819 also supports resetting attributes and colors to default using \033[0m.
1820
1821 @sa xterm256_idx_to_rgb(), rgb_to_curses_clr(), extended_pair_content(),
1822 get_clr_pair()
1823
1824 @endverbatim
1825*/
1826void parse_ansi_str(char *ansi_str, attr_t *attr, int *cpx) {
1827 char *tok;
1828 char t0, t1;
1829 char tstr[3];
1830 int len, x_idx;
1831 int fg, bg;
1832 int fg_clr, bg_clr;
1833 char *ansi_p = ansi_str + 2;
1834 extended_pair_content(*cpx, &fg_clr, &bg_clr);
1835 fg = fg_clr;
1836 bg = bg_clr;
1837 RGB rgb;
1838 tok = strtok((char *)ansi_p, ";m");
1839 bool a_toi_error = false;
1840 while (1) {
1841 if (tok == nullptr || *tok == '\0')
1842 break;
1843 len = strlen(tok);
1844 if (len == 2) {
1845 t0 = tok[0];
1846 t1 = tok[1];
1847 if (t0 == '3' || t0 == '4') {
1848 if (t1 == '8') {
1849 tok = strtok(nullptr, ";m");
1850 if (tok != nullptr) {
1851 if (*tok == '5') {
1852 tok = strtok(nullptr, ";m");
1853 if (tok != nullptr) {
1854 x_idx = a_toi(tok, &a_toi_error);
1855 rgb = xterm256_idx_to_rgb(x_idx);
1856 }
1857 } else if (*tok == '2') {
1858 tok = strtok(nullptr, ";m");
1859 rgb.r = a_toi(tok, &a_toi_error);
1860 tok = strtok(nullptr, ";m");
1861 rgb.g = a_toi(tok, &a_toi_error);
1862 tok = strtok(nullptr, ";m");
1863 rgb.b = a_toi(tok, &a_toi_error);
1864 }
1865 }
1866 if (t0 == '3')
1867 fg_clr = rgb_to_curses_clr(&rgb);
1868 else if (t0 == '4')
1869 bg_clr = rgb_to_curses_clr(&rgb);
1870 } else if (t1 == '9') {
1871 if (t0 == '3')
1872 fg_clr = CLR_NT_FG;
1873 else if (t0 == '4')
1874 bg_clr = CLR_NT_BG;
1875 } else if (t1 >= '0' && t1 <= '7') {
1876 if (t0 == '3') {
1877 tstr[0] = t1;
1878 tstr[1] = '\0';
1879 x_idx = a_toi(tstr, &a_toi_error);
1880 rgb = xterm256_idx_to_rgb(x_idx);
1881 fg_clr = rgb_to_curses_clr(&rgb);
1882 } else if (t0 == '4') {
1883 tstr[0] = t1;
1884 tstr[1] = '\0';
1885 x_idx = a_toi(tstr, &a_toi_error);
1886 rgb = xterm256_idx_to_rgb(x_idx);
1887 bg_clr = rgb_to_curses_clr(&rgb);
1888 }
1889 }
1890 } else if (t0 == '0') {
1891 *tok = t1;
1892 len = 1;
1893 }
1894 }
1895 if (len == 1) {
1896 if (*tok == '0') {
1897 *attr = WA_NORMAL;
1898 fg_clr = CLR_NT_FG;
1899 bg_clr = CLR_NT_BG;
1900 } else {
1901 switch (a_toi(tok, &a_toi_error)) {
1902 case 1:
1903 *attr |= WA_BOLD;
1904 break;
1905 case 2:
1906 *attr |= WA_DIM;
1907 break;
1908 case 3:
1909 *attr |= WA_ITALIC;
1910 break;
1911 case 4:
1912 *attr |= WA_UNDERLINE;
1913 break;
1914 case 5:
1915 *attr |= WA_BLINK;
1916 break;
1917 case 7:
1918 *attr |= WA_REVERSE;
1919 break;
1920 case 8:
1921 *attr |= WA_INVIS;
1922 break;
1923 default:
1924 break;
1925 }
1926 }
1927 } else if (len == 0) {
1928 *attr = WA_NORMAL;
1929 fg_clr = CLR_NT_FG;
1930 bg_clr = CLR_NT_BG;
1931 }
1932 tok = strtok(nullptr, ";m");
1933 }
1934 if (!a_toi_error && (fg_clr != fg || bg_clr != bg)) {
1935 clr_pair_idx = get_clr_pair(fg_clr, bg_clr);
1936 *cpx = clr_pair_idx;
1937 }
1938 return;
1939}
