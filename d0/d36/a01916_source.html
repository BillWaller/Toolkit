81/** @brief Convert hex color strings to a curses color pair index.
82 *
83 * @param fg Foreground color in hex format (e.g., "#RRGGBB").
84 * @param bg Background color in hex format (e.g., "#RRGGBB").
85 * @return The curses color pair index corresponding to the given foreground and background colors.
86 */
87int ui_pair_from_hex(const char *fg, const char *bg) {
88 RGB rgb;
89 sscanf(fg, "#%02x%02x%02x", &rgb.r, &rgb.g, &rgb.b);
90 int f_idx = ui_color_from_rgb(&rgb);
91 sscanf(bg, "#%02x%02x%02x", &rgb.r, &rgb.g, &rgb.b);
92 int b_idx = ui_color_from_rgb(&rgb);
93 return ui_add_pair(f_idx, b_idx);
94}
95/** @brief Convert a Unicode code point to a curses UiCell with specified foreground and background colors.
96 *
97 * @param ucp Pointer to a wide character string containing the Unicode code point.
98 * @param fg Pointer to a 32-bit integer representing the foreground color in hex format (e.g., 0xRRGGBB).
99 * @param bg Pointer to a 32-bit integer representing the background color in hex format (e.g., 0xRRGGBB).
100 * @return A UiCell structure representing the Unicode character with the specified colors.
101 */
102UiCell ui_cell_from_ucp(const wchar_t *ucp, const uint32_t *fg, const uint32_t *bg) {
103 UiCell cc = {0};
104 RGB rgb;
105 rgb.r = (*fg >> 16) & 0xff;
106 rgb.g = (*fg >> 8) & 0xff;
107 rgb.b = *fg & 0xff;
108 int f_idx = ui_color_from_rgb(&rgb);
109 rgb.r = (*bg >> 16) & 0xff;
110 rgb.g = (*bg >> 8) & 0xff;
111 rgb.b = *bg & 0xff;
112 int b_idx = ui_color_from_rgb(&rgb);
113 short cp = ui_add_pair(f_idx, b_idx);
114 wchar_t wstr[2] = {L'\0', L'\0'};
115 wstr[0] = *ucp;
116 wstr[1] = L'\0';
117 setcchar(&cc, wstr, WA_NORMAL, cp, nullptr);
118 return cc;
119}

449void ui_shutdown() {

458 ui_log(INFO, _("surface destroy: %d"), sfc_ptr);
459 sfc_ptr--;
460 }

462 ui_log(INFO, _("calling del_panel(stdsfc->mpan[BOX])"));
463 hide_panel(stdsfc->mpan[0]);
464 del_panel(stdsfc->mpan[0]);
465 stdsfc->mpan[0] = NULL;
466 }
467 ui_log(INFO, _("calling endwin()"));
468 endwin();
469 f_ncurses_open = false;
470 if (ui->screen != NULL) {
471 ui_log(INFO, _("calling delscreen(ui->screen)"));
472 delscreen(ui->screen);
473 ui->screen = NULL;
474 }
475 if (ui->tty_fp != NULL) {
476 ui_log(INFO, _("closing tty_fp"));
477 fclose(ui->tty_fp);
478 ui->tty_fp = NULL;
479 }
480 if (stdsfc != NULL) {
481 ui_log(INFO, _("freeing stdsfc"));
482 free(stdsfc);
483 stdsfc = NULL;
484 }
485 if (ui != NULL) {
486 if (ui->sio) {
487 free(ui->sio);
488 ui->sio = nullptr;
489 }
490 ui_log(INFO, _("freeing ui"));
491 free(ui);
492 ui = NULL;
493 }
