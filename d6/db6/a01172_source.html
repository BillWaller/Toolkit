100/** @brief Fork and exec a command
101 @ingroup exec
102 @param argv - array of arguments for the command to execute
103 @return the return code from the executed command, or -1 on error
104 @details Captures and restores terminal settings around the fork and exec.
105 Sets signal handlers to default in the child process.
106 Waits for the child process to complete in the parent process.
107 Handles errors from fork and execvp, and reports child exit status.
108 Restores curses mode and keypad settings after execution.
109 Restores window states after execution.
110 Uses a temporary string buffer tmp_str for error messages.
111 Uses Perror for error reporting.
112 Uses sig_dfl_mode and sig_prog_mode for signal handling.
113 Uses capture_curses_tioctl and restore_curses_tioctl for terminal
114 settings.
115 Uses restore_shell_tioctl for shell terminal settings.
116 Uses waitpid to wait for the child process.
117 Uses WIFEXITED, WEXITSTATUS, WIFSIGNALED, and WTERMSIG to interpret
118 child status.
119 Uses keypad to manage keypad mode in curses.
120 Uses restore_wins to restore window states.
121 Uses errno for error codes.
122 Uses pid_t for process IDs.
123 Uses standard file descriptors STDIN_FILENO, STDOUT_FILENO,
124 STDERR_FILENO.
125 Uses execvp for executing the command.
126 Uses fork for creating a new process.
127 Uses ssnprintf for formatting error messages.
128 Uses switch-case for handling fork results.
129 Uses default shell if SHELL environment variable is not set. */
130int fork_exec(char **argv) {
131 char tmp_str[MAXLEN];
132 pid_t pid;
133 int status;
134 int rc;
135
136 if (argv[0] == 0) {
137 Perror("fork_exec: missing argument for execvp");
138 return (-1);
139 }
140 capture_curses_tioctl();
141 curs_set(1);
142 sig_dfl_mode();
143
144 tmp_str[0] = '\0';
145 pid = fork();
146 switch (pid) {
147 case -1: // parent fork failed
148 sig_prog_mode();
149 keypad(stdscr, true);
150 ssnprintf(tmp_str, sizeof(tmp_str), "fork failed: %s, errno: %d",
151 argv[0], errno);
152 Perror(tmp_str);
153 return (-1);
154 case 0: // child
155 restore_shell_tioctl();
156 werase(stdscr);
157 wrefresh(stdscr);
158 execvp(argv[0], argv);
159 restore_curses_tioctl();
160 sig_prog_mode();
161 keypad(stdscr, true);
162 ssnprintf(tmp_str, sizeof(tmp_str), "execvp failed: %s, errno: %d",
163 argv[0], errno);
164 Perror(tmp_str);
165 exit(-1);
166 default: // parent
167 rc = 0;
168 restore_curses_tioctl();
169 sig_prog_mode();
170 waitpid(pid, &status, 0);
171 if (WIFEXITED(status)) {
172 rc = WEXITSTATUS(status);
173 if (rc != 0) {
174 keypad(stdscr, true);
175 ssnprintf(tmp_str, sizeof(tmp_str),
176 "Child %s exited with status %d", argv[0], rc);
177 }
178 } else {
179 if (WIFSIGNALED(status)) {
180 rc = WTERMSIG(status);
181 keypad(stdscr, true);
182 ssnprintf(tmp_str, sizeof(tmp_str),
183 "Child %s terminated by signal %d", argv[0], rc);
184 } else {
185 keypad(stdscr, true);
186 ssnprintf(tmp_str, sizeof(tmp_str),
187 "Child %s terminated abnormally", argv[0]);
188 }
189 }
190 break;
191 }
192 restore_curses_tioctl();
193 sig_prog_mode();
194 touchwin(stdscr);
195 wnoutrefresh(stdscr);
196 wrefresh(stdscr);
197 restore_wins();
198 tmp_str[0] = '\0';
199 if (tmp_str[0] != '\0') {
200 Perror(tmp_str);
201 }
202 return (rc);
203}
204/** @brief Fork and execute a command as a daemon
205 @ingroup exec
206 @param eargv - array of arguments for the command to execute
207 @return EXIT_SUCCESS on success, EXIT_FAILURE on failure
208 @details
209 Sets the session ID
210 Redirects standard file descriptors to /dev/null.
211 Closes all open file descriptors.
212 Executes the command using execvp.
213 Exits with failure if any step fails.
214 @note Tested 2026-06-05 on Linux - appears to be functioning properly */
215int fork_detach_execvp(char **eargv) {
216 pid_t pid = fork();
217 capture_curses_tioctl();
218 curs_set(1);
219 sig_dfl_mode();
220
221 if (pid < 0) {
222 fprintf(stderr, "First fork failed: %s\n", strerror(errno));
223 exit(EXIT_FAILURE);
224 }
225 if (pid == 0) {
226 if (setsid() < 0) {
227 fprintf(stderr, "Set session ID failed: %s\n", strerror(errno));
228 exit(EXIT_FAILURE);
229 }
230 close(STDIN_FILENO);
231 close(STDOUT_FILENO);
232 close(STDERR_FILENO);
233 int dev_null = open("/dev/null", O_RDWR);
234 if (dev_null != -1) {
235 dup2(dev_null, STDIN_FILENO);
236 dup2(dev_null, STDOUT_FILENO);
237 dup2(dev_null, STDERR_FILENO);
238 if (dev_null > 2) {
239 close(dev_null);
240 }
241 }
242 long max_fd = sysconf(_SC_OPEN_MAX);
243 for (long fd = 3; fd < max_fd; fd++)
244 close(fd);
245 execvp(eargv[0], eargv);
246 perror("execvp failed");
247 exit(EXIT_FAILURE);
248 }
249 restore_curses_tioctl();
250 sig_prog_mode();
251 touchwin(stdscr);
252 wnoutrefresh(stdscr);
253 wrefresh(stdscr);
254 restore_wins();
255 return 0;
256}
