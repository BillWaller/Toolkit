28/** @brief The main loop of the menu system.
29 @ingroup menu_engine
30 @param init A pointer to an Init structure containing initialization data for
31 the menu system.
32 @returns an integer indicating the action taken by the user, such as
33 returning to the main menu or exiting the menu system.
34 @note displays the menu and processes user input until the user exits the
35 menu or returns to the main menu.
36 */
37unsigned int menu_engine(Init *init) {
38 int action;
39 int i;
40 char tmp_str[MAXLEN];
41 Menu *menu = init->menu;
42 if (menu == nullptr) {
43 Perror("menu_engine: menu is nullptr");
44 return (1);
45 }
46 if (win_new(menu->lines, menu->cols, menu->begy, menu->begx, menu->title,
47 0)) {
48 ssnprintf(tmp_str, MAXLEN - 1, "win_new(%d, %d, %d, %d, %s, %b) failed",
49 menu->lines, menu->cols, menu->begy, menu->begx, menu->title,
50 0);
51 Perror(tmp_str);
52 return 1;
53 }
54 menu->win = win_win[win_ptr];
55 menu->box = win_box[win_ptr];
56 menu->lines = 0;
57 menu->cols = 0;
58 menu->line_idx = 0;
59 menu->item_count = 0;
60 menu->title[0] = '\0';
61 menu->choice_max_len = 0;
62 menu->text_max_len = 0;
63 parse_menu_description(init);
64
65 action = MA_DISPLAY_MENU;
66 while (action) {
67 switch (action) {
68 case MA_RETURN:
69 win_del();
70 if (win_ptr < 0) {
71 return 0;
72 }
73 menu->win = win_win[win_ptr];
74 menu->box = win_box[win_ptr];
75 restore_wins();
76 return 0;
77 case MA_DISPLAY_MENU:
78 for (menu->line_idx = 0; menu->line_idx < menu->item_count;
79 menu->line_idx++) {
80 mvwaddstr(menu->win, menu->line_idx, 0,
81 menu->line[menu->line_idx]->choice_text);
82 }
83 action = MA_RESET_MENU;
84 break;
85 case MA_RESET_MENU:
86 menu->line_idx = 0;
87 for (i = 0; i < menu->item_count; i++) {
88 if (menu->line[i]->type == MT_CHOICE) {
89 menu->line_idx = i;
90 break;
91 }
92 }
93 action = MA_CONTINUE;
94 break;
95 case MA_CONTINUE:
96 while ((action = menu_cmd_processor(init)) == MA_CONTINUE)
97 ;
98 break;
99 default:
100 break;
101 }
102 }
103 return 0;
104}
105/** @brief Processes user input for the menu system.
106 @ingroup menu_engine
107 @param init A pointer to an Init structure containing initialization data
108 for the menu system.
109 @returns an integer indicating the action taken by the user, such as
110 returning to the main menu, displaying a submenu, or executing a command
111 associated with a menu choice.
112 @note handles navigation through the menu options, executing commands
113 associated with menu choices, and responding to special keys such as
114 function keys and mouse clicks.
115 */
116unsigned int menu_cmd_processor(Init *init) {
117 int i, c;
118 char *d;
119 int in_key;
120 char tmp_str[MAXLEN];
121
122 keypad(menu->win, TRUE);
123 Menu *menu = init->menu;
124 mousemask(BUTTON1_CLICKED | BUTTON1_DOUBLE_CLICKED, nullptr);
125 MEVENT event;
126
127 wattron(menu->win, A_REVERSE);
128 mvwaddstr_fill(menu->win, menu->line_idx, 0,
129 menu->line[menu->line_idx]->choice_text, menu->cols);
130 wattroff(menu->win, A_REVERSE);
131
132 event.y = event.x = -1;
133 // tcflush(2, TCIFLUSH);
134 wmove(menu->win, menu->line_idx, 1);
135 in_key = xwgetch(menu->win, nullptr, -1);
136
137 mvwaddstr_fill(menu->win, menu->line_idx, 0,
138 menu->line[menu->line_idx]->choice_text, menu->cols);
139 switch (in_key) {
140 /** Move up to the previous menu choice */
141 case KEY_UP:
142 i = menu->line_idx;
143 while (i > 0) {
144 i--;
145 if (menu->line[i]->type == MT_CHOICE) {
146 menu->line_idx = i;
147 break;
148 }
149 }
150 return (MA_CONTINUE);
151 /** Move down to the next menu choice */
152 case KEY_DOWN:
153 i = menu->line_idx;
154 while (i < menu->item_count - 1) {
155 i++;
156 if (menu->line[i]->type == MT_CHOICE) {
157 menu->line_idx = i;
158 break;
159 }
160 }
161 return (MA_CONTINUE);
162 /** Select the current menu choice and execute its associated
163 * command */
164 case '\n':
165 case KEY_ENTER:
166 break;
167 /** @brief Display help information for the menu system */
168 case '?':
169 case KEY_F(1):
170 if (menu->f_help_spec && menu->help_spec[0] != '\0')
171 strnz__cpy(tmp_str, menu->help_spec, MAXLEN - 1);
172 else {
173 strnz__cpy(tmp_str, init->mapp_help, MAXLEN - 1);
174 strnz__cat(tmp_str, "/", MAXLEN - 1);
175 strnz__cat(tmp_str, MENU_HELP_FILE, MAXLEN - 1);
176 }
177 eargc = 0;
178 eargv[eargc++] = strdup("view");
179 eargv[eargc++] = strdup("-N");
180 eargv[eargc++] = strdup("f");
181 eargv[eargc++] = strdup(tmp_str);
182 eargv[eargc] = nullptr;
183 init->lines = 30;
184 init->cols = 60;
185 init->begy = menu->begy + 1;
186 init->begx = menu->begx + 1;
187 strnz__cpy(init->title, "Menu Help", MAXLEN - 1);
188 popup_view(init, eargc, eargv, init->lines, init->cols, init->begy,
189 init->begx);
190 destroy_argv(eargc, eargv);
191 return (MA_DISPLAY_MENU);
192 /** Exit the menu and return to the previous menu or exit if at top
193 */
194 case KEY_F(9):
195 case KEY_BREAK:
196 return (MA_RETURN);
197 /** @brief send default printer output file to printer */
198 case KEY_ALTF(9):
199 d = getenv("PRTCMD");
200 if (d == nullptr || *d == '\0')
201 strnz__cpy(earg_str, PRINTCMD, MAXLEN - 1);
202 else
203 strnz__cpy(earg_str, d, MAXLEN - 1);
204 strnz__cat(earg_str, " ", MAXLEN - 1);
205 d = getenv("PRTFILE");
206 if (d == nullptr || *d == '\0') {
207 d = getenv("HOME");
208 if (d == nullptr || *d == '\0')
209 strnz__cat(earg_str, VIEW_PRT_FILE, MAXLEN - 1);
210 else {
211 strnz__cat(earg_str, d, MAXLEN - 1);
212 strnz__cat(earg_str, "/", MAXLEN - 1);
213 strnz__cat(earg_str, VIEW_PRT_FILE, MAXLEN - 1);
214 }
215 } else
216 strnz__cat(earg_str, d, MAXLEN - 1);
217 full_screen_shell(earg_str);
218 return (MA_DISPLAY_MENU);
219 /** @brief open the default editor */
220 case KEY_ALTF(10):
221 restore_wins();
222 return (MA_DISPLAY_MENU);
223 /** @brief process mouse event */
224 case KEY_MOUSE:
225 if (click_y == -1 || click_x == -1)
226 return (MA_CONTINUE);
227 if (click_y < 0 || click_y >= menu->item_count)
228 return (MA_CONTINUE);
229 menu->line_idx = click_y;
230 break;
231 default:
232 /** @brief If the user presses a key that corresponds to a menu
233 * choice's letter, select that menu choice */
234 for (i = 0; i < menu->item_count; i++) {
235 if (menu->line[i]->raw_text[0] == '_') {
236 if (menu->line[i]->choice_letter == (char)in_key) {
237 menu->line_idx = i;
238 break;
239 }
240 } else {
241 if (menu->line[i]->choice_letter == toupper((char)in_key)) {
242 menu->line_idx = i;
243 break;
244 }
245 }
246 }
247 if (i >= menu->item_count)
248 return (MA_CONTINUE);
249 break;
250 }
251 c = (int)menu->line[menu->line_idx]->command_type;
252 switch (c) {
253 /** @brief Execute the command associated with the selected menu
254 * choice
255 */
256 case CT_EXEC:
257 char *s;
258 s = strpbrk(menu->line[menu->line_idx]->command_str, " \t\f\v");
259 strnz__cpy(earg_str, s, MAXLEN - 1);
260 trim(earg_str);
261 eargc = str_to_args(eargv, s, MAX_ARGS);
262 full_screen_fork_exec(eargv);
263 destroy_argv(eargc, eargv);
264 return (MA_DISPLAY_MENU);
265 /** @brief Display help information for the menu system */
266 case CT_HELP:
267 if (menu->f_help_spec && menu->help_spec[0] != '\0')
268 strnz__cpy(tmp_str, menu->help_spec, MAXLEN - 1);
269 else {
270 strnz__cpy(tmp_str, init->mapp_help, MAXLEN - 1);
271 strnz__cat(tmp_str, "/", MAXLEN - 1);
272 strnz__cat(tmp_str, MENU_HELP_FILE, MAXLEN - 1);
273 }
274 eargc = 0;
275 eargv[eargc++] = strdup("view");
276 eargv[eargc++] = strdup("-N");
277 eargv[eargc++] = strdup("f");
278 eargv[eargc++] = strdup(tmp_str);
279 eargv[eargc] = nullptr;
280 init->lines = 66;
281 init->cols = 80;
282 init->begy = menu->begy + 1;
283 init->begx = menu->begx + 1;
284 strnz__cpy(init->title, "Menu Help", MAXLEN - 1);
285 popup_view(init, eargc, eargv, init->lines, init->cols, init->begy,
286 init->begx);
287 destroy_argv(eargc, eargv);
288 return (MA_DISPLAY_MENU);
289 /** @brief Display a submenu or perform an action associated with
290 * the selected menu choice */
291 case CT_MENU:
292 strnz__cpy(earg_str, menu->line[menu->line_idx]->command_str,
293 MAXLEN - 1);
294 eargc = str_to_args(eargv, earg_str, MAX_ARGS);
295 if (eargc == 0)
296 return (MA_DISPLAY_MENU);
297 zero_opt_args(init);
298 parse_opt_args(init, eargc, eargv);
299 if (init->begy == 0)
300 init->begy = menu->begy + 1;
301 if (init->begx == 0)
302 init->begx = menu->begx + 1;
303 Menu *save_menu = init->menu;
304 init->menu = nullptr;
305 init->menu = new_menu(init, eargc, eargv, init->begy, init->begx);
306 if (!init->menu)
307 abend(-1, "menu_cmd_processor: new_menu() failed");
308 // menu = init->menu;
309 parse_menu_description(init);
310 menu_engine(init);
311 destroy_argv(eargc, eargv);
312 init->menu = destroy_menu(init);
313 init->menu = save_menu;
314 menu = init->menu;
315 return (MA_RESET_MENU);
316 /** @brief Display a pick list or form associated with the selected
317 * menu choice */
318 case CT_PICK:
319 strnz__cpy(earg_str, menu->line[menu->line_idx]->command_str,
320 MAXLEN - 1);
321 eargc = str_to_args(eargv, earg_str, MAX_ARGS);
322 popup_pick(init, eargc, eargv, menu->begy + 1, menu->begx + 1);
323 destroy_argv(eargc, eargv);
324 return (MA_DISPLAY_MENU);
325 /** @brief Display a form associated with the selected menu choice
326 */
327 case CT_FORM:
328 strnz__cpy(earg_str, menu->line[menu->line_idx]->command_str,
329 MAXLEN - 1);
330 eargc = str_to_args(eargv, earg_str, MAX_ARGS);
331 popup_form(init, eargc, eargv, menu->begy + 1, menu->begx + 1);
332 destroy_argv(eargc, eargv);
333 return (MA_DISPLAY_MENU);
334 /** @brief Display a view associated with the selected menu choice
335 */
336 case CT_VIEW:
337 strnz__cpy(earg_str, menu->line[menu->line_idx]->command_str,
338 MAXLEN - 1);
339 eargc = str_to_args(eargv, earg_str, MAX_ARGS);
340 init->lines = 66;
341 init->cols = 80;
342 init->begy = menu->begy + 1;
343 init->begx = menu->begx + 1;
344 strnz__cpy(init->title, menu->line[menu->line_idx]->raw_text,
345 MAXLEN - 1);
346 popup_view(init, eargc, eargv, init->lines, init->cols, init->begy,
347 init->begx);
348 destroy_argv(eargc, eargv);
349 return (MA_DISPLAY_MENU);
350 /** @brief open ckeys (test curses keys) */
351 case CT_CKEYS:
352 popup_ckeys();
353 return (MA_DISPLAY_MENU);
354 /** @brief return to calling program */
355 case CT_RETURN:
356 return (MA_RETURN);

358 case CT_WRITE_CONFIG:

360 return (MA_CONTINUE);
361 default:
362 return (MA_CONTINUE);
363 }
364 return 0;
