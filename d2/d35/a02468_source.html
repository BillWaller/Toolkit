206/** @brief Calculate the dimensions and position of the box window for C-Menu
207 View.
208 @ingroup init_view
209 @details This function calculates the dimensions and position of the box
210 window for the C-Menu View based on the screen size and any specified
211 parameters in the Init structure. It ensures that the box window fits within
212 the screen dimensions and adjusts its size and position accordingly.
213 @param init Pointer to the Init structure containing view settings.
214 */
215void view_calc_boxwin_dimensions(Init *init) {
216 View *view = init->view;
217
218 /** Use view->lines and view->cols if set, otherwise calculate based on
219 * screen size with some padding. Ensure the view fits within the screen
220 * dimensions. */
221 uint scr_lines, scr_cols;
222 ui_get_screen_size(&scr_lines, &scr_cols);
223#ifdef DEBUG_RESIZE
224 ssnprintf(em0, MAXLEN - 1,
225 _("%s:%d=%d calc lines=%d, cols=%d, begy=%d, begx=%d"),
226 __FILE__, __LINE__, 526, view->lines, view->cols, view->begy, view->begx);
227 write_cmenu_log(em0);
228#endif
229 if (view->lines == 0 && view->cols == 0 && view->begy == 0 && view->begx == 0) {
230 view->lines = scr_lines - 3;
231 view->cols = scr_cols - 2;
232 view->begy = 0;
233 view->begx = 0;
234 }
235 uint lines = max(scr_lines / 10, 8);
236 view->lines = max(view->lines, lines);
237 if (view->lines > scr_lines - 3)
238 view->lines = scr_lines - 3;
239 if (view->lines + view->begy > scr_lines - 3)
240 view->begy = scr_lines - (view->lines + 3);
241
242 uint cols = max(scr_cols / 3, 40);
243 view->cols = max(view->cols, cols);
244 if (view->cols > scr_cols - 2)
245 view->cols = scr_cols - 2;
246 if (view->cols + view->begx > scr_cols - 2)
247 view->begx = scr_cols - (view->cols + 2);
248
249 view->h_shift = view->cols / 3;
250 view->ln_win_lines = view->lines - 1;
251 if (view->f_ln)
252 view->ln_win_cols = 8;
253 else
254 view->ln_win_cols = 0;
255 view->scroll_lines = view->lines - 1;
256 view->cmd_line = 0;
257 view->pminrow = 0;
258 view->pmincol = 0;
259 view->sminrow = view->begy + 1;
260 view->smincol = view->begx + view->ln_win_cols + 1;
261 view->smaxrow = view->begy + view->lines;
262 view->smaxcol = view->begx + view->cols;
263 view->ln_no = view->page_top_ln_no + view->scroll_lines;
264 view->page_bot_ln_no = view->ln_no;
265#ifdef DEBUG_RESIZE
266 ssnprintf(em0, MAXLEN - 1,
267 _("%s:%d calc BOX lines=%d, cols=%d, begy=%d, begx=%d"),
268 __FILE__, __LINE__, view->lines + 2, view->cols + 2, view->begy, view->begx);
269 write_cmenu_log(em0);
270 ssnprintf(em0, MAXLEN - 1,
271 _("%s:%d calc WIN lines=%d, cols=%d, begy=%d, begx=%d"),

289/** @brief Initialize the input for the C-Menu View.
290 @ingroup init_view
291 @param init Pointer to the Init structure containing view settings.
292 @param file_name Name of the input file or "-" for standard input.
293 @return 0 on success, -1 on failure.
294 @details This function initializes the input for the C-Menu View. It handles
295 both regular files and standard input, setting up pipes if necessary. It also
296 memory-maps the input file for efficient access and sets up the view structure
297 accordingly.
298 */
299int view_init_input(Init *init, char *file_name) {
300 struct stat sb;
301 int idx = 0;
302 pid_t pid = -1;
303 int pipe_fd[2];
304 int s_argc = 0;
305 char *s_argv[MAXARGS];
306 char tmp_str[MAXLEN];
307 View *view = init->view;
308 view->f_in_pipe = false;
309 if (strcmp(file_name, "-") == 0) {
310 file_name = _("/dev/stdin");
311 view->f_in_pipe = true;
312 }
313 if (view->provider_cmd[0] != '\0') {
314 s_argc = str_to_args(s_argv, view->provider_cmd, MAXARGS - 1);
315 if (pipe(pipe_fd) == -1) {
316 ui_perror(_("pipe(pipe_fd) failed in init_view"));
317 return -1;
318 }
319 // endwin();
320 if ((pid = fork()) == -1) {
321 ui_perror(_("fork() failed in init_view"));
322 return -1;
323 }
324 if (pid == 0) { // Child
325 /** Prevent child process from writing to terminal */
326 int dev_null = open(_("/dev/null"), O_WRONLY);
327 if (dev_null == -1) {
328 ui_perror(_("open(/dev/null) failed in init_pick child process"));
329 exit(EXIT_FAILURE);
330 }
331 dup2(dev_null, STDERR_FILENO);
332 close(dev_null);
333 close(pipe_fd[P_READ]);
334 dup2(pipe_fd[P_WRITE], STDOUT_FILENO);
335 close(pipe_fd[P_WRITE]);
336 execvp(s_argv[0], s_argv);
337 strnz__cpy(tmp_str, _("Can't exec view start cmd: "), MAXLEN - 1);
338 strnz__cat(tmp_str, s_argv[0], MAXLEN - 1);
339 ui_perror(tmp_str);
340 exit(EXIT_FAILURE);
341 }
342 // Back to parent
343 destroy_argv(s_argc, s_argv);
344 sig_prog_mode();
345 ui_restore_wins();
346 close(pipe_fd[P_WRITE]);
347 view->in_fd = dup(pipe_fd[P_READ]);
348 view->f_in_pipe = true;
349 } else {
350 if (view->f_in_pipe)
351 view->in_fd = dup(STDIN_FILENO);
352 else {
353 /*----------------------------------------------------------------------*/
354 /* Open the input file for reading and get its size. */
355 expand_tilde(file_name, MAXLEN - 1);
356 view->in_fd = open(file_name, O_RDONLY);
357 if (view->in_fd == -1) {
358 ssnprintf(em0, MAXLEN - 1, _("%s, line: %d"), __FILE__,
359 __LINE__ - 3);
360 ssnprintf(em1, MAXLEN - 1, _("open %s"), file_name);
361 strerror__r(errno, em2, MAXLEN);
362 ui_display_error(em0, em1, em2,
363 nullptr);
364 return -1;
365 }
366 if (fstat(view->in_fd, &sb) == -1) {
367 ssnprintf(em0, MAXLEN - 1, _("%s, line: %d"), __FILE__,
368 __LINE__ - 1);
369 ssnprintf(em1, MAXLEN - 1, _("fstat %s"), file_name);
370 strerror__r(errno, em2, MAXLEN);
371 ui_display_error(em0, em1, em2, nullptr);
372 close(view->in_fd);
373 return -1;
374 }
375 view->file_size = sb.st_size;
376 if (view->file_size == 0) {
377 close(view->in_fd);
378 ssnprintf(em0, MAXLEN - 1, _("%s, line: %d"), __FILE__,
379 __LINE__ - 1);
380 ssnprintf(em1, MAXLEN - 1, _("file %s is empty"), file_name);
381 strerror__r(errno, em2, MAXLEN);
382 ui_display_error(em0, em1, em2, nullptr);
383 return -1;
384 }
385 if (!S_ISREG(sb.st_mode))
386 view->f_in_pipe = true;
387 }
388 /*----------------------------------------------------------------------*/
389 }
390 if (view->f_in_pipe) {
391 errno = 0;
392 view->tmp_fd = memfd_create(_("view_input"), MFD_CLOEXEC);
393 if (view->in_fd < 0 || errno != 0) {
394 ssnprintf(em0, MAXLEN - 1, "memfd_create failed\n");
395 ssnprintf(em1, MAXLEN - 1, "%s", strerror(errno));
396 ssnprintf(em2, MAXLEN - 1, _("%s, line: %d, errno: %d"), __FILE__,
397 __LINE__ - 4, errno);
398 ui_display_error(em0, em1, em2, nullptr);
399 exit(EXIT_FAILURE);
400 }
401 // ------------------------------------------------------
402 // transfer view->in_fd to view->tmp_fd
403 // ------------------------------------------------------
404 char buf[VBUFSIZ];
405 ssize_t bytes_read = 0;
406#ifdef TIMEOUT_POLL
407 // UIKEY_STREAM_DATA = octal 740
408 ui_register_read_fd(view->in_fd, UIKEY_STREAM_DATA);
409 int status = 0;
410 struct timespec start_time, now_time;
411 clock_gettime(CLOCK_MONOTONIC, &start_time);
412 bool active_timeout_prompt = false;
413 int timeout_ms = 300;
414 while (true) {
415 if (active_timeout_prompt) {
416 clock_gettime(CLOCK_MONOTONIC, &now_time);
417 double elapsed = (now_time.tv_sec - start_time.tv_sec) * 1000.0 +
418 (now_time.tv_nsec - start_time.tv_nsec) / 1e9;
419 double remaining = 10.0 - elapsed;
420 if (remaining <= 0)
421 break;
422 timeout_ms = (int)(remaining * 1000.0);
423 }
424 UiEvent ev;
425 memset(&ev, 0, sizeof(UiEvent));
426 if (!active_timeout_prompt) {
427 active_timeout_prompt = true;
428 status = ui_timeout_prompt(_("Waiting for input..."), _("from view_input"), timeout_ms);
429 continue;
430 } else
431 break;
432 if (status == UI_EV_DATA_STREAM) {
433 bytes_read = read(view->in_fd, buf, sizeof(buf));
434 if (bytes_read > 0) {
435 if ((bytes_written = write(view->tmp_fd, buf, bytes_read)) != bytes_read) {
436 ui_abend(-1, _("unable to write view->tmp_fd"));
437 exit(EXIT_FAILURE);
438 }
439 } else if (bytes_read == 0)
440 break;
441 continue;
442 }
443 if (ev.key == UIKEY_F09)
444 exit(EXIT_SUCCESS);
445 }
446 ui_unregister_read_fd(view->in_fd);
447#else
448 while ((bytes_read = read(view->in_fd, buf, sizeof(buf))) > 0) {
449 if (write(view->tmp_fd, buf, bytes_read) != bytes_read) {
450 ui_abend(-1, _("unable to write view->tmp_fd"));
451 exit(EXIT_FAILURE);
452 }
453 }
454#endif
455 // ------------------------------------------------------
456 // end transfer
457 // ------------------------------------------------------
458 if (fstat(view->tmp_fd, &sb) == -1) {
459 ssnprintf(em0, MAXLEN - 1, "fstat(view->in_fd) failed\n");
460 ssnprintf(em1, MAXLEN - 1, "%s", strerror(errno));
461 ssnprintf(em2, MAXLEN - 1, _("%s, line: %d, errno: %d"), __FILE__,
462 __LINE__ - 4, errno);
463 ui_display_error(em0, em1, em2, nullptr);
464 exit(EXIT_FAILURE);
465 }
466 view->file_size = sb.st_size;
467 if (view->file_size == 0) {
468 close(view->tmp_fd);
469 strnz__cpy(tmp_str, _("no standard input"), MAXLEN - 1);
470 ui_abend(-1, tmp_str);
471 exit(EXIT_FAILURE);
472 }
473 waitpid(-1, nullptr, 0);
474 view->in_fd = view->tmp_fd;
475 }
476 view->buf =
477 mmap(nullptr, view->file_size, PROT_READ, MAP_PRIVATE, view->in_fd, 0);
478 if (view->buf == MAP_FAILED) {
479 ssnprintf(em0, MAXLEN - 1, _("%s, line: %d"), __FILE__, __LINE__ - 2);
480 ssnprintf(em1, MAXLEN - 1, _("mmap %s"), file_name);
481 strerror__r(errno, em2, MAXLEN);
482 ui_display_error(em0, em1, em2, nullptr);
483 close(view->in_fd);
484 return -1;
485 }
486 close(view->in_fd);
487 stdio_fdnames(stdio_names_str, _("init_view.c 422"));
488 view->file_size = sb.st_size;
489 view->prev_file_pos = NULL_POSITION;
490 view->buf_curr_ptr = view->buf;
491 if (view->cmd_all[0] != '\0')
492 strnz__cpy(view->cmd, view->cmd_all, MAXLEN - 1);
493 for (idx = 0; idx < NMARKS; idx++)
494 view->mark_tbl[idx] = NULL_POSITION;
495 strnz__cpy(view->cur_file_str, file_name, MAXLEN - 1);
496 base_name(view->file_name, view->cur_file_str);
497 return 0;
498}
