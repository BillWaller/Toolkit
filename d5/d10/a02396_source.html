34/** @brief Initialize C-Menu View in full screen mode.
35 @ingroup init_view
36 @param init Pointer to the Init structure containing view settings.
37 @return 0 on success, -1 on failure.
38 @details This function sets up the view structure for full screen mode and
39 creates a new pad for the view.
40 @verbatim
41 The function creates the following windows:
42 1. view->cmdln_win: Status or Command Line
43 2. view->lnno_win: Line Number Window
44 3. view->pad: Main Content Pad
45 @endverbatim
46 */
47int init_view_full_screen(Init *init) {
48 View *view = init->view;
49 if (view->tab_stop <= 0)
50 view->tab_stop = TABSIZE;
51 set_tabsize(view->tab_stop);
52 view->f_full_screen = true;
53 // -------------------> 1. WIN <-------------------
54 view_calc_full_screen_dimensions(init);
55 view->sfc = ui_surface_new(ui_runtime, NULL, 0, view->lines, view->cols, 0, 0);
56 if (view->sfc->win == nullptr) {
57 ssnprintf(em0, MAXLEN - 1, "newwin(LINES, COLS, 0, 0) failed in init_view_full_screen");
58 Perror(em0);
59 return -1;
60 }
61 ui_bkgrnd_cch(view->sfc, WIN, &CC_NT);
62 // -------------------> 2. LNNO <-------------------
63 ui_surface_addwin(view->sfc, LNNO, WIN, view->scroll_lines, view->ln_win_cols, 0, 0);
64 if (view->sfc->lnno == nullptr) {
65 ssnprintf(em0, MAXLEN - 1, "ui_sfc_addwin(LNNO, LINES - 1, COLS, 0, 0) failed in init_view_full_screen");
66 Perror(em0);
67 return -1;
68 }
69 ui_bkgrnd_cch(view->sfc, LNNO, &CC_LN);
70 ui_render(ui_runtime);
71 ui_scrollok(view->sfc, LNNO, true);
72 ui_setscrreg(view->sfc, LNNO, 0, view->scroll_lines);
73 // -------------------> 3. CMDLN <-------------------
74 ui_surface_addwin(view->sfc, CMDLN, WIN, 1, view->cols, view->scroll_lines, 0);
75 if (view->sfc->cmdln == nullptr) {
76 ssnprintf(em0, MAXLEN - 1, "ui_sfc_addwin(CMDLN, 1, COLS, LINES - 1, 0) failed in init_view_full_screen");
77 Perror(em0);
78 return -1;
79 }
80 ui_bkgrnd_cch(view->sfc, CMDLN, &CC_NT);
81 ui_keypad(view->sfc, CMDLN, true);
82 ui_idlok(view->sfc, CMDLN, false);
83 ui_idcok(view->sfc, CMDLN, false);
84 ui_scrollok(view->sfc, CMDLN, false);
85 // -------------------> 4. PAD <-------------------
86
87 ui_surface_addpad(view->sfc, PAD, view->lines - 1, PAD_COLS - 1);
88
89 // ------------------------------------------------
90 return 0;
91}

191/** @brief Calculate the dimensions and position of the box window for C-Menu
192 View.
193 @ingroup init_view
194 @details This function calculates the dimensions and position of the box
195 window for the C-Menu View based on the screen size and any specified
196 parameters in the Init structure. It ensures that the box window fits within
197 the screen dimensions and adjusts its size and position accordingly.
198 @param init Pointer to the Init structure containing view settings.
199 */
200void view_calc_boxwin_dimensions(Init *init) {
201 int scr_lines, scr_cols;
202 View *view = init->view;
203
204 /** Use view->lines and view->cols if set, otherwise calculate based on
205 * screen size with some padding. Ensure the view fits within the screen
206 * dimensions. */
207
208 getmaxyx(stdscr, scr_lines, scr_cols);
209#ifdef DEBUG_RESIZE
210 ssnprintf(em0, MAXLEN - 1,
211 "%s:%d=%d calc lines=%d, cols=%d, begy=%d, begx=%d",
212 __FILE__, __LINE__, 526, view->lines, view->cols, view->begy, view->begx);
213 write_cmenu_log(em0);
214#endif
215 if (view->lines == 0 && view->cols == 0 && view->begy == 0 && view->begx == 0) {
216 view->lines = scr_lines - 3;
217 view->cols = scr_cols - 2;
218 view->begy = 0;
219 view->begx = 0;
220 }
221 int lines = max(scr_lines / 10, 8);
222 view->lines = max(view->lines, lines);
223 if (view->lines > scr_lines - 3)
224 view->lines = scr_lines - 3;
225 if (view->lines + view->begy > scr_lines - 3)
226 view->begy = scr_lines - (view->lines + 3);
227
228 int cols = max(scr_cols / 3, 40);
229 view->cols = max(view->cols, cols);
230 if (view->cols > scr_cols - 2)
231 view->cols = scr_cols - 2;
232 if (view->cols + view->begx > scr_cols - 2)
233 view->begx = scr_cols - (view->cols + 2);
234
235 view->h_shift = view->cols / 3;
236 view->ln_win_lines = view->lines - 1;
237 if (view->f_ln)
238 view->ln_win_cols = 8;
239 else
240 view->ln_win_cols = 0;
241 view->scroll_lines = view->lines - 1;
242 view->cmd_line = 0;
243 view->pminrow = 0;
244 view->pmincol = 0;
245 view->sminrow = view->begy + 1;
246 view->smincol = view->begx + view->ln_win_cols + 1;
247 view->smaxrow = view->begy + view->lines;
248 view->smaxcol = view->begx + view->cols;
249 view->ln_no = view->page_top_ln_no + view->scroll_lines;
250 view->page_bot_ln_no = view->ln_no;
251#ifdef DEBUG_RESIZE
252 ssnprintf(em0, MAXLEN - 1,
253 "%s:%d calc BOX lines=%d, cols=%d, begy=%d, begx=%d",
254 __FILE__, __LINE__, view->lines + 2, view->cols + 2, view->begy, view->begx);

275/** @brief Initialize the input for the C-Menu View.
276 @ingroup init_view
277 @param init Pointer to the Init structure containing view settings.
278 @param file_name Name of the input file or "-" for standard input.
279 @return 0 on success, -1 on failure.
280 @details This function initializes the input for the C-Menu View. It handles
281 both regular files and standard input, setting up pipes if necessary. It also
282 memory-maps the input file for efficient access and sets up the view structure
283 accordingly.
284 */
285int view_init_input(Init *init, char *file_name) {
286 struct stat sb;
287 int idx = 0;
288 pid_t pid = -1;
289 int pipe_fd[2];
290 int s_argc = 0;
291 char *s_argv[MAXARGS];
292 char tmp_str[MAXLEN];
293 View *view = init->view;
294 view->f_in_pipe = false;
295 if (strcmp(file_name, "-") == 0) {
296 file_name = "/dev/stdin";
297 view->f_in_pipe = true;
298 }
299 if (view->provider_cmd[0] != '\0') {
300 s_argc = str_to_args(s_argv, view->provider_cmd, MAXARGS - 1);
301 if (pipe(pipe_fd) == -1) {
302 Perror("pipe(pipe_fd) failed in init_view");
303 return -1;
304 }
305 // endwin();
306 if ((pid = fork()) == -1) {
307 Perror("fork() failed in init_view");
308 return -1;
309 }
310 if (pid == 0) { // Child
311 /** Prevent child process from writing to terminal */
312 int dev_null = open("/dev/null", O_WRONLY);
313 if (dev_null == -1) {
314 Perror("open(/dev/null) failed in init_pick child process");
315 exit(EXIT_FAILURE);
316 }
317 dup2(dev_null, STDERR_FILENO);
318 close(dev_null);
319 close(pipe_fd[P_READ]);
320 dup2(pipe_fd[P_WRITE], STDOUT_FILENO);
321 close(pipe_fd[P_WRITE]);
322 execvp(s_argv[0], s_argv);
323 strnz__cpy(tmp_str, "Can't exec view start cmd: ", MAXLEN - 1);
324 strnz__cat(tmp_str, s_argv[0], MAXLEN - 1);
325 Perror(tmp_str);
326 exit(EXIT_FAILURE);
327 }
328 // Back to parent
329 destroy_argv(s_argc, s_argv);
330 reset_prog_mode();
331 restore_wins();
332 close(pipe_fd[P_WRITE]);
333 dup2(pipe_fd[P_READ], STDIN_FILENO);
334 view->in_fd = dup(STDIN_FILENO);
335 view->f_in_pipe = true;
336 } else {
337 if (view->f_in_pipe)
338 view->in_fd = dup(STDIN_FILENO);
339 else {
340 /*----------------------------------------------------------------------*/
341 /* Open the input file for reading and get its size. */
342 expand_tilde(file_name, MAXLEN - 1);
343 view->in_fd = open(file_name, O_RDONLY);
344 if (view->in_fd == -1) {
345 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
346 __LINE__ - 3);
347 ssnprintf(em1, MAXLEN - 1, "open %s", file_name);
348 strerror_r(errno, em2, MAXLEN);
349 display_error(em0, em1, em2, nullptr);
350 return -1;
351 }
352 if (fstat(view->in_fd, &sb) == -1) {
353 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
354 __LINE__ - 1);
355 ssnprintf(em1, MAXLEN - 1, "fstat %s", file_name);
356 strerror_r(errno, em2, MAXLEN);
357 display_error(em0, em1, em2, nullptr);
358 close(view->in_fd);
359 return -1;
360 }
361 view->file_size = sb.st_size;
362 if (view->file_size == 0) {
363 close(view->in_fd);
364 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__,
365 __LINE__ - 1);
366 ssnprintf(em1, MAXLEN - 1, "file %s is empty", file_name);
367 strerror_r(errno, em2, MAXLEN);
368 display_error(em0, em1, em2, nullptr);
369 return -1;
370 }
371 if (!S_ISREG(sb.st_mode))
372 view->f_in_pipe = true;
373 }
374 /*----------------------------------------------------------------------*/
375 }
376 if (view->f_in_pipe) {
377 close(view->in_fd);
378 errno = 0;
379 view->in_fd = memfd_create("view_input", MFD_CLOEXEC);
380 if (view->in_fd < 0 || errno != 0) {
381 ssnprintf(em0, MAXLEN - 1, "memfd_create failed\n");
382 ssnprintf(em1, MAXLEN - 1, "%s", strerror(errno));
383 ssnprintf(em2, MAXLEN - 1, "%s, line: %d, errno: %d", __FILE__,
384 __LINE__ - 4, errno);
385 display_error(em0, em1, em2, nullptr);
386 exit(EXIT_FAILURE);
387 }
388 char buf[VBUFSIZ];
389 ssize_t bytes_read = 0;
390 ssize_t bytes_written = 0;
391 while ((bytes_read = read(STDIN_FILENO, buf, sizeof(buf))) > 0) {
392 if ((bytes_written = write(view->in_fd, buf, bytes_read)) != bytes_read) {
393 abend(-1, "unable to write view->in_fd");
394 exit(EXIT_FAILURE);
395 }
396 }
397 if (fstat(view->in_fd, &sb) == -1) {
398 ssnprintf(em0, MAXLEN - 1, "fstat(view->in_fd) failed\n");
399 ssnprintf(em1, MAXLEN - 1, "%s", strerror(errno));
400 ssnprintf(em2, MAXLEN - 1, "%s, line: %d, errno: %d", __FILE__,
401 __LINE__ - 4, errno);
402 display_error(em0, em1, em2, nullptr);
403 exit(EXIT_FAILURE);
404 }
405 view->file_size = sb.st_size;
406 if (view->file_size == 0) {
407 close(view->in_fd);
408 strnz__cpy(tmp_str, "no standard input", MAXLEN - 1);
409 abend(-1, tmp_str);
410 exit(EXIT_FAILURE);
411 }
412 waitpid(-1, nullptr, 0);
413 }
414 view->buf =
415 mmap(nullptr, view->file_size, PROT_READ, MAP_PRIVATE, view->in_fd, 0);
416 if (view->buf == MAP_FAILED) {
417 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__, __LINE__ - 2);
418 ssnprintf(em1, MAXLEN - 1, "mmap %s", file_name);
419 strerror_r(errno, em2, MAXLEN);
420 display_error(em0, em1, em2, nullptr);
421 close(view->in_fd);
422 return -1;
423 }
424 close(view->in_fd);
425 SIO *sio = init->sio;
426 stdio_names(stdio_names_str, "init_view.c 673");
427 stdio_fdnames(stdio_names_str, "init_view.c 674");
428 dup2(sio->stdin_fd, STDIN_FILENO);
429 stdio_names(stdio_names_str, "init_view.c 673");
430 stdio_fdnames(stdio_names_str, "init_view.c 674");
431 view->file_size = sb.st_size;
432 view->prev_file_pos = NULL_POSITION;
433 view->buf_curr_ptr = view->buf;
434 if (view->cmd_all[0] != '\0')
435 strnz__cpy(view->cmd, view->cmd_all, MAXLEN - 1);
436 for (idx = 0; idx < NMARKS; idx++)
437 view->mark_tbl[idx] = NULL_POSITION;
438 strnz__cpy(view->cur_file_str, file_name, MAXLEN - 1);
439 base_name(view->file_name, view->cur_file_str);
440 return 0;
441}
