32/** @brief read the next characater from the virtual file
33 @ingroup view_engine
34 @details Line numbers are tracked when reading forward and stored in a line
35 table for quick access when moving backwards.
36 When reading forward, the End of Data (EOD) flag is set when the
37 position is equal to the file size, and cleared when the position or reading
38 direction changes.
39 @note view_engine.c line numbering
40 The macro, get_next_char() reads lines delimited by '\n', advancing the
41 line number index at the end of each record. The line number index
42 is one record ahead for most operations, including line numbering as lines
43 are displayed. Nevertheless, when providing a line number index to
44 get_line(), you must use the zero origin index, which is one less than
45 the line number index
46 */
47#define get_next_char()
48 {
49 c = 0;
50 do {
51 if (view->file_pos == view->file_size) {
52 view->f_eod = true;
53 break;
54 } else
55 view->f_eod = false;
56 c = view->buf[view->file_pos++];
57 } while (c == 0x0d);
58 if (c == '\n')
59 increment_ln(view);
60 }
61/** @brief read the previous characater from the virtual file
62 @ingroup view_engine
63 There is no need to track line numbers when moving backwards as they
64 are stored in the line table and accessed as needed.
65 When reading in reverse, the Beginning of Data (BOD) flag is set when
66 the file position is zero, and cleared when the position or reading direction
67 changes.
68 Carriage-returns are ignored as they should be.
69 */
70#define get_prev_char()
71 {
72 c = 0;
73 do {
74 if (view->file_pos == 0) {
75 view->f_bod = true;
76 break;
77 } else
78 view->f_bod = false;
79 c = view->buf[--view->file_pos];
80 } while (c == 0x0d);
81 }

148/** @brief Start view
149 @ingroup view_engine
150 @param init Pointer to the Init structure containing initialization
151 parameters and state for the view application. This structure is used to pass
152 necessary information and maintain state across different functions within
153 the view application.
154 @return Returns 0 on successful completion of the view application, or a

157int view_file(Init *init) {
158 View *view = init->view;
159 if (view->argc < 1) {
160 view->curr_argc = -1;
161 view->argc = 0;
162 view->next_file_spec_ptr = "-";
163 } else
164 view->next_file_spec_ptr = view->argv[0];
165 while (view->curr_argc < view->argc) {
166 if (view->next_file_spec_ptr == nullptr ||
167 *view->next_file_spec_ptr == '\0') {
168 break;
169 }
170 view->file_spec_ptr = view->next_file_spec_ptr;
171 view->next_file_spec_ptr = nullptr;
172 strnz__cpy(view->cur_file_str, view->file_spec_ptr, MAXLEN - 1);
173 if (view_init_input(init, view->cur_file_str) == 0) {
174 if (view->buf) {
175 view->f_eod = 0;
176 view->f_bod = 0;
177 view->maxcol = 0;
178 view->page_top_pos = 0;
179 view->page_top_ln_no = 0;
180 view->page_bot_ln_no = 0;
181 view->ln_max_pos = 0;
182 view->page_bot_pos = 0;
183 view->file_pos = 0;
184 strnz__cpy(view->title, view->cur_file_str, MAXLEN - 1);
185 border_title(view->sfc, view->title);
186 initialize_line_table(view);
187 next_page(view);
188 view_cmd_processor(init);
189 destroy_line_table(view);
190 munmap(view->buf, view->file_size);
191 }
192 } else {
193 view->curr_argc++;
194 if (view->curr_argc < view->argc) {
195 view->next_file_spec_ptr = view->argv[view->curr_argc];
196 }
197 }
198 }
199 ui_surface_destroy(view->sfc);
200 destroy_view(init);
201 return 0;
202}

907/** @brief Build Prompt String
908 * @ingroup view_engine
909 @param view Pointer to the View structure containing the state and

913 This function constructs a prompt string that provides information about

1092/** @brief Search for Regular Expression Pattern
1093 @ingroup view_navigation
1094 @param view Pointer to View Structure
1095 @param search_cmd Search Command Character ('/' or '?')
1096 @param regex_pattern Regular Expression Pattern to Search For
1097 @returns true if a match is found and displayed, false if the search
1098 completes without finding a match or if an error occurs
1099 @details The search performs extended regular expression matching, ignoring
1100 ANSI sequences and Unicode characters. Matches are highlighted on the
1101 screen, and the search continues until the page is full or the end of the
1102 file is reached. If the search wraps around the file, a message is
1103 displayed indicating that the search is complete.
1104 The search state is maintained in the view structure, allowing for
1105 repeat searches and tracking of the current search line. This
1106 function highlights all matches in the current ncurses pad, including
1107 those not displayed on the screen, and tracks the first and last match
1108 columns for prompt display.
1109 ANSI sequences and Unicode characters are stripped before
1110 matching, so matching corresponds to the visual display */
1111bool search(View *view, int search_cmd, char *regex_pattern) {
1112 char tmp_str[MAXLEN];
1113 int REG_FLAGS = 0;
1114 regmatch_t pmatch[1];
1115 regex_t compiled_regex;
1116 int reti;
1117 int line_offset;
1118 int line_len;
1119 int match_idx;
1120 int match_len;
1121 mbstate_t mbstate;
1122 memset(&mbstate, 0, sizeof(mbstate));
1123 wchar_t wstr[2] = {L'\0', L'\0'};
1124 attr_t attr;
1125 short cpx;
1126 cchar_t cc = {0};
1127 off_t prev_ln_no;
1128 bool f_page = false;
1129 UiSurface *sfc = view->sfc;
1130 if (*regex_pattern == '\0')
1131 return false;
1132 if (view->f_ignore_case)
1133 REG_FLAGS = REG_ICASE | REG_EXTENDED;
1134 else
1135 REG_FLAGS = REG_EXTENDED;
1136 reti = regcomp(&compiled_regex, regex_pattern, REG_FLAGS);
1137 if (reti) {
1138 Perror("Invalid pattern");
1139 return false;
1140 }
1141 bool rc = false;
1142 while (1) {
1143 /** initialize iteration */
1144 if (search_cmd == '/') {
1145 if (view->srch_curr_ln_no == view->file_size)
1146 view->srch_curr_ln_no = 0;
1147 } else {
1148 if (view->srch_curr_ln_no == 0)
1149 view->srch_curr_ln_no = view->file_size;
1150 }
1151 if (view->srch_curr_ln_no == view->srch_beg_ln_no) {
1152 if (view->f_first_iter == true) {
1153 view->f_first_iter = false;
1154 view->f_search_complete = false;
1155 if (search_cmd == '/')
1156 view->cury = 0;
1157 else
1158 view->cury = view->scroll_lines + 1;
1159 } else {
1160 view->f_search_complete = true;
1161 goto cleanup;
1162 }
1163 }
1164 view->ln_no = view->srch_curr_ln_no;
1165 sync_ln(view);
1166 /** get line to scan */
1167 if (search_cmd == '/') {
1168 if (view->cury == view->scroll_lines)
1169 goto cleanup;
1170 prev_ln_no = view->srch_curr_ln_no;
1171 get_line(view, view->srch_curr_ln_no);
1172 view->page_bot_ln_no = view->srch_curr_ln_no;
1173 } else {
1174 if (view->cury == 0)
1175 goto cleanup;
1176 get_line(view, view->srch_curr_ln_no);
1177 prev_ln_no = view->srch_curr_ln_no;
1178 view->page_top_ln_no = view->srch_curr_ln_no;
1179 }
1180 fmt_line(view);
1181 reti = regexec(&compiled_regex, view->stripped_line_out,
1182 compiled_regex.re_nsub + 1, pmatch, REG_FLAGS);
1183 if (reti == REG_NOMATCH) {
1184 if (f_page) {
1185 /** non-matching page filler */
1186 if (search_cmd == '?')
1187 view->cury -= 2;
1188 display_line(view);
1189 if ((search_cmd == '/' && view->cury == view->scroll_lines) ||
1190 (search_cmd == '?' && view->cury == 1)) {
1191 break;
1192 }
1193 }
1194 view->srch_curr_ln_no += (search_cmd == '/') ? 1 : -1;
1195 continue;
1196 }
1197 if (reti) {
1198 char err_str[MAXLEN];
1199 regerror(reti, &compiled_regex, err_str, sizeof(err_str));
1200 strnz__cpy(tmp_str, "Regex match failed: ", MAXLEN - 1);
1201 strnz__cat(tmp_str, err_str, MAXLEN - 1);
1202 Perror(tmp_str);
1203 rc = false; /* Set status */
1204 goto cleanup;
1205 }
1206 rc = true;
1207 /** Display matching lines */
1208 if (!f_page) {
1209 if (search_cmd == '/') {
1210 view->page_top_ln_no = prev_ln_no;
1211 ui_cursor_move(sfc, PAD, view->cury, 0);
1212 } else {
1213 view->page_bot_ln_no = view->ln_no;
1214 ui_cursor_move(sfc, PAD, 0, 0);
1215 }
1216 ui_wclrtobot(sfc, CMDLN);
1217 f_page = true;
1218 }
1219 if (search_cmd == '?')
1220 view->cury -= 2;
1221 view->first_match_x = -1;
1222 view->last_match_x = 0;
1223 line_len = strlen(view->stripped_line_out);
1224 line_offset = 0;
1225 while (1) {
1226 match_idx = line_offset + pmatch[0].rm_so;
1227 match_len = pmatch[0].rm_eo - pmatch[0].rm_so;
1228 for (int i = match_idx; i < match_idx + match_len; i++) {
1229 cc = view->cmplx_buf[i];
1230 getcchar(&cc, wstr, &attr, &cpx, nullptr);
1231 cpx = cp_nt_rev;
1232 ui_setcchar(&cc, wstr, attr, cpx, nullptr);
1233 view->cmplx_buf[i] = cc;
1234 }
1235 if (view->first_match_x == -1)
1236 view->first_match_x = pmatch[0].rm_so;
1237 view->last_match_x = line_offset + pmatch[0].rm_eo;
1238 line_offset += pmatch[0].rm_eo;
1239 if (line_offset >= line_len)
1240 break;
1241 view->line_out_p = view->stripped_line_out + line_offset;
1242 reti = regexec(&compiled_regex, view->line_out_p,
1243 compiled_regex.re_nsub + 1, pmatch, REG_FLAGS);
1244 if (reti == REG_NOMATCH)
1245 break;
1246 if (reti) {
1247 char msgbuf[100];
1248 regerror(reti, &compiled_regex, msgbuf, sizeof(msgbuf));
1249 sprintf(tmp_str, "Regex match failed: %s", msgbuf);
1250 Perror(tmp_str);
1251 rc = false; /* Set status */
1252 goto cleanup;
1253 }
1254 if (search_cmd == '/') {
1255 if (view->cury == view->scroll_lines - 1) {
1256 break;
1257 }
1258 } else if (view->cury == 1) {
1259 break;
1260 }
1261 }
1262 display_line(view);
1263 view->srch_curr_ln_no += (search_cmd == '/') ? 1 : -1;
1264 }

1901/** @brief Get Line Number for a Given File Position
1902 @ingroup view_navigation
1903 @param view data structure
1904 @param target file position
1905 @returns line number corresponding to the given file position
1906 @details This function performs a binary search on the line table (view->ln_tbl)
1907 to find the line number corresponding to the specified file position (target).
1908 It returns the index of the line in the line table that is closest to the target
1909 position without exceeding it. If the target position is not found, it returns
1910 the previous line number.
1911 */
1912off_t line_number(View *view, off_t position) {
1913 off_t low = 0;
1914 off_t high = view->ln_tbl_size - 1;
1915 off_t prev_ln_no = 0;
1916 while (low <= high) {
1917 off_t guess = low + (high - low) / 2;
1918 if (view->ln_tbl[guess] == position)
1919 return guess;
1920 if (view->ln_tbl[guess] < position) {
1921 prev_ln_no = guess;
1922 low = guess + 1;
1923 } else
1924 high = guess - 1;
1925 }
1926 return prev_ln_no;
1927}
