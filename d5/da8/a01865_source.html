27UiRuntime *ui_init(const UiConfig *cfg) {
28 setlocale(LC_ALL, "");
29
30 UiRuntime *ui = calloc(1, sizeof(*ui));
31 if (!ui)
32 return NULL;
33
34 if (cfg) {
35 ui->mouse_enabled = cfg->enable_mouse;
36 ui->alt_screen = cfg->enable_alt_screen;
37 ui->cursor_visible = cfg->cursor_visible;
38 } else {
39 ui->cursor_visible = true;
40 ui->alt_screen = true;
41 }
42
43 FILE *tty = NULL;
44 if (cfg && cfg->tty_path) {
45 tty = fopen(cfg->tty_path, "r+");
46 if (!tty) {
47 free(ui);
48 return NULL;
49 }
50 }
51
52 struct notcurses_options opts = {
53 .flags = NCOPTION_SUPPRESS_BANNERS | (ui->alt_screen ? 0 : NCOPTION_NO_ALTERNATE_SCREEN),
54 };
55
56 ui->nc = notcurses_init(&opts, tty);
57 if (!ui->nc) {
58 if (tty)
59 fclose(tty);
60 free(ui);
61 return NULL;
62 }
63
64 if (ui->mouse_enabled)
65 notcurses_mice_enable(ui->nc, NCMICE_ALL_EVENTS);
66
67 if (!ui->cursor_visible)
68 notcurses_cursor_disable(ui->nc);
69
70 unsigned int r = 0, c = 0;
71 notcurses_stddim_yx(ui->nc, &r, &c);
72 ui->lines = (int)r;
73 ui->cols = (int)c;
74
75 return ui;
76}

150void ui_get_caps(const UiRuntime *ui, UiCaps *caps) {
151 if (!caps)
152 return;
153 memset(caps, 0, sizeof(*caps));
154 if (!ui)
155 return;
156 caps->truecolor = notcurses_cantruecolor(ui->nc);
157 caps->palette256 = true; /* NotCurses always manages its own palette */
158 caps->mouse = ui->mouse_enabled;
159 caps->unicode = notcurses_canutf8(ui->nc);
160 caps->resize = true; /* NCKEY_RESIZE is delivered as a normal event */
161 caps->color_pairs = 0; /* unlimited — NotCurses uses direct RGB */

168UiSurface *ui_surface_new(UiRuntime *ui, UiSurface *parent, UiRect rect) {
169 if (!ui)
170 return NULL;
171
172 UiSurface *s = calloc(1, sizeof(*s));
173 if (!s)
174 return NULL;
175
176 s->runtime = ui;
177 s->parent = parent;
178 s->y = rect.y;
179 s->x = rect.x;
180 s->lines = rect.lines;
181 s->cols = rect.cols;
182
183 struct ncplane *parent_plane =
184 parent ? parent->plane : notcurses_stdplane(ui->nc);
185
186 struct ncplane_options plane_opts = {
187 .y = rect.y,
188 .x = rect.x,
189 .lines = (unsigned int)rect.lines,
190 .cols = (unsigned int)rect.cols,
191 .name = NULL,
192 };
193
194 s->plane = ncplane_create(parent_plane, &plane_opts);
195 if (!s->plane) {
196 free(s);
197 return NULL;
198 }
199
200 return s;
201}

331int ui_surface_set_base(UiSurface *s, const UiStyle *style, uint32_t fill_ch) {
332 if (!s)
333 return -1;
334 uint64_t channels = ui_notcurses_channels_from_style(style);
335 uint32_t attrs = ui_notcurses_attrs_from_style(style);
336 char utf8[5] = " ";
337 if (fill_ch >= 0x20) {
338 /* Encode the Unicode codepoint as UTF-8. */
339 if (fill_ch < 0x80) {
340 utf8[0] = (char)fill_ch;
341 utf8[1] = '\0';
342 } else if (fill_ch < 0x800) {
343 utf8[0] = (char)(0xC0 | (fill_ch >> 6));
344 utf8[1] = (char)(0x80 | (fill_ch & 0x3F));
345 utf8[2] = '\0';
346 } else if (fill_ch < 0x10000) {
347 utf8[0] = (char)(0xE0 | (fill_ch >> 12));
348 utf8[1] = (char)(0x80 | ((fill_ch >> 6) & 0x3F));
349 utf8[2] = (char)(0x80 | (fill_ch & 0x3F));
350 utf8[3] = '\0';
351 } else {
352 utf8[0] = (char)(0xF0 | (fill_ch >> 18));
353 utf8[1] = (char)(0x80 | ((fill_ch >> 12) & 0x3F));
354 utf8[2] = (char)(0x80 | ((fill_ch >> 6) & 0x3F));
355 utf8[3] = (char)(0x80 | (fill_ch & 0x3F));
356 utf8[4] = '\0';
357 }
358 }
359 ncplane_set_base(s->plane, utf8, attrs, channels);
360 return 0;
361}
