18/** @brief Translate a NotCurses key code into a UiKey.
19 @param id NotCurses key code.
20 @param ni NotCurses input event (for context).
21 @return Corresponding UiKey value.
22*/
23static UiKey translate_nckey(uint32_t id, const ncinput *ni) {
24 if (id == NCKEY_INVALID || id == (uint32_t)NCKEY_EOF)
25 return UIKEY_NONE;
26
27 /* Mouse buttons — handled separately in ui_get_event(). */
28 if (nckey_mouse_p(id))
29 return UIKEY_MOUSE;
30
31 switch (id) {
32 case NCKEY_RESIZE:
33 return UIKEY_RESIZE;
34 case NCKEY_ENTER:
35 return UIKEY_ENTER;
36 case NCKEY_ESC:
37 return UIKEY_ESCAPE;
38 case NCKEY_BACKSPACE:
39 return UIKEY_BACKSPACE;
40 case NCKEY_TAB:
41 return UIKEY_TAB;
42 case NCKEY_UP:
43 return UIKEY_UP;
44 case NCKEY_DOWN:
45 return UIKEY_DOWN;
46 case NCKEY_LEFT:
47 return UIKEY_LEFT;
48 case NCKEY_RIGHT:
49 return UIKEY_RIGHT;
50 case NCKEY_HOME:
51 return UIKEY_HOME;
52 case NCKEY_END:
53 return UIKEY_END;
54 case NCKEY_PGUP:
55 return UIKEY_PPAGE;
56 case NCKEY_PGDOWN:
57 return UIKEY_NPAGE;
58 case NCKEY_INS:
59 return UIKEY_IC;
60 case NCKEY_DEL:
61 return UIKEY_DC;
62 case NCKEY_F01:
63 return UIKEY_F01;
64 case NCKEY_F02:
65 return UIKEY_F02;
66 case NCKEY_F03:
67 return UIKEY_F03;
68 case NCKEY_F04:
69 return UIKEY_F04;
70 case NCKEY_F05:
71 return UIKEY_F05;
72 case NCKEY_F06:
73 return UIKEY_F06;
74 case NCKEY_F07:
75 return UIKEY_F07;
76 case NCKEY_F08:
77 return UIKEY_F08;
78 case NCKEY_F09:
79 return UIKEY_F09;
80 case NCKEY_F10:
81 return UIKEY_F10;
82 case NCKEY_F11:
83 return UIKEY_F11;
84 case NCKEY_F12:
85 return UIKEY_F12;
86 default:
87 if (ni && ni->evtype == NCTYPE_UNKNOWN)
88 return UIKEY_NONE;
89 /* Printable Unicode codepoint. */
90 if (id >= 0x20 && id < NCKEY_INVALID)
91 return UIKEY_CHAR;
92 return UIKEY_NONE;
93 }
94}

100/** @brief Wait for an input event from the NotCurses context.
101 @param ui UI runtime context.
102 @param target Unused for NotCurses (events are global to the context).
103 @param ev Output UiEvent structure.
104 @param timeout_ms Milliseconds to wait; -1 = block indefinitely.
105 @return 0 on success, -1 if @p ui or @p ev is NULL.
106*/
107int ui_get_event(UiSurface *s, ss_t w, UiChyron *chyron, UiEvent *ev, int timeout_ms) {
108 if (!ui || !ev)
109 return -1;
110 memset(ev, 0, sizeof(*ev));
111 ncinput ni;
112 ui_render();
113 if (timeout_ms < 0) {
114 do {
115 notcurses_get_blocking(ui->nc, &ni);
116 } while (ni.evtype == NCTYPE_RELEASE ||
117 ni.id == NCKEY_INVALID || ni.id == NCKEY_MOTION || ni.id == NCKEY_SIGNAL);
118
119 } else {
120 struct timespec ts = {
121 .tv_sec = timeout_ms / 1000,
122 .tv_nsec = (long)(timeout_ms % 1000) * 1000000L,
123 };
124 do {
125 notcurses_get(ui->nc, &ts, &ni);
126 } while (ni.evtype == NCTYPE_RELEASE || ni.id == NCKEY_INVALID);
127 }
128 ev->key = translate_nckey(ni.id, &ni);
129 ev->alt = ncinput_alt_p(&ni);
130 ev->ctrl = ncinput_ctrl_p(&ni);
131 ev->shift = ncinput_shift_p(&ni);
132 if (ev->key == UIKEY_CHAR)
133 ev->ch = ni.id; /* Unicode codepoint */
134 else if (ev->key == UIKEY_MOUSE) {
135 if (ni.id == NCKEY_BUTTON4)
136 ev->mouse_action = UIKEY_SCROLL_UP;
137 else if (ni.id == NCKEY_BUTTON5)
138 ev->mouse_action = UIKEY_SCROLL_DOWN;
139 else if (ni.id == NCKEY_BUTTON1 && ni.evtype == NCTYPE_PRESS) {
140 ev->mouse_action = UIKEY_BUTTON1_PRESSED;
141 struct ncplane *clicked = ui_ncplane_clicked(s, w, &ni);
142 ev->y = ni.y;
143 ev->x = ni.x;
144 ev->in_win = ui_get_plane_idx(s, clicked);
145 if (chyron) {
146 if (ev->in_win == chyron->w && ev->y == chyron->y) {
147 ev->mouse_action = UIKEY_BUTTON1_CLICKED;
148 ev->key = ui_get_chyron_key(chyron, ev->x);
149 return ev->key;
150 } else
151 return 0;
152 } else {
153 if (ev->in_win == w) {
154 ev->mouse_action = UIKEY_BUTTON1_CLICKED;
155 ev->key = ev->y;
156 return ev->key;
157 } else
158 return 0;
159 }
160 }
161 }
162 return ev->ch;
163}
164/** @brief Wait for a single input character from the NotCurses context.
165 @return The NotCurses key code, or -1 if @p ui is NULL.

167int ui_getch() {
168 if (!ui)
169 return -1;

172 notcurses_get(ui->nc, NULL, &ni);

174 ni.id == NCKEY_INVALID ||
175 ni.id == NCKEY_MOTION ||
176 ni.id == NCKEY_SIGNAL);

179/** @brief Wait for an input event from the NotCurses context, ignoring mouse events.
180 @param target Unused for NotCurses (events are global to the context).
181 @param ev Output UiEvent structure.
182 @return The NotCurses key code, or -1 if @p ui or @p ev is NULL.
183*/
184int ui_get_event_no_mouse(UiSurface *target, ss_t w, UiEvent *ev) {
185 (void)target;
186 (void)w;
187 if (!ui || !ev)
188 return -1;
189 memset(ev, 0, sizeof(*ev));
190 ncinput ni;
191 // notcurses_mice_disable(ui->nc);
192 do {
193 notcurses_get(ui->nc, NULL, &ni);
194 } while (ni.id == NCKEY_INVALID);
195 ev->key = translate_nckey(ni.id, &ni);
196 ev->alt = ncinput_alt_p(&ni);
197 ev->ctrl = ncinput_ctrl_p(&ni);
198 ev->shift = ncinput_shift_p(&ni);
199 if (ev->key == UIKEY_CHAR)
200 ev->ch = ni.id;
201 return ni.id;
202}

215/** @brief Determine which ncplane was clicked based on the mouse event coordinates.
216 @param s The UiSurface containing the planes.
217 @param w The index of the plane to check for clicks.
218 @param ni The NotCurses input event containing mouse coordinates.
219 @return The ncplane that was clicked, or NULL if no plane was clicked.
220*/
221NcPlane *ui_ncplane_clicked(UiSurface *s, ss_t w, ncinput *ni) {
222 NcPlane *pile_member = s->mplane[w];
223 struct ncplane *cur = ncpile_top(pile_member);
224 while (cur != NULL) {
225 int y = ni->y, x = ni->x;
226 if (ncplane_translate_abs(cur, &y, &x)) {
227 ni->y = y;
228 ni->x = x;
229 return cur;
230 }
231 cur = ncplane_below(cur);
232 }
233 return NULL;
234}
