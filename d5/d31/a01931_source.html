21/** @brief clear the window from the current cursor position to the end of the window
22 * @param s surface
23 * @param w window

28 return -1;

30 ncplane_cursor_yx(s->mplane[w], &y, &x);
31 ncplane_dim_yx(s->mplane[w], &maxy, &maxx);

49 ylen = 1;

456/** @brief display an image on the terminal using NotCurses
457 * @param nc NotCurses context
458 * @param mm UiMultiMedia structure to hold the image and surface
459 * @param image_file path to the image file
460 * @param y height of the image display area (or -1 for full height)
461 * @param x width of the image display area (or -1 for full width)
462 * @param begy starting row for the image display area (or -1 for top)
463 * @param begx starting column for the image display area (or -1 for left)
464 * @return pointer to the ncvisual structure on success, nullptr on error
465 */
466struct ncvisual *ui_display_image(struct notcurses *nc, UiMultiMedia *mm, const char *image_file, int y, int x, int begy, int begx) {
467 struct ncplane *stdn = notcurses_stdplane(nc);
468 unsigned term_rows, term_cols;
469 ncplane_dim_yx(stdn, &term_rows, &term_cols);
470 mm->ncv = ncvisual_from_file(image_file);
471 if (!mm->ncv) {
472 fprintf(stderr, "Error: Could not load image file.\n");
473 return nullptr;
474 }
475 struct ncvgeom geom;
476 if (ncvisual_geom(nc, mm->ncv, NULL, &geom) < 0) {
477 ncvisual_destroy(mm->ncv);
478 return nullptr;
479 }
480 if (begy == -1)
481 begy = 0;
482 if (begx == -1)
483 begx = 0;
484 if (y == -1)
485 y = (int)term_rows - begy;
486 if (x == -1)
487 x = (int)term_cols - begx;
488 if (begy + y > (int)term_rows)
489 y = (int)term_rows - begy;
490 if (begx + x > (int)term_cols)
491 x = (int)term_cols - begx;
492 int max_rows = y - 2;
493 int max_cols = x - 2;
494 if (max_rows <= 0) {
495 ncvisual_destroy(mm->ncv);
496 return nullptr;
497 }
498 struct ncvisual_options vopts_calc = {
499 .scaling = NCSCALE_SCALE_HIRES,
500 .blitter = NCBLIT_PIXEL,
501 };
502 struct ncplane_options nopts = {
503 .y = begy,
504 .x = x,
505 .rows = max_rows,
506 .cols = max_cols,
507 };
508 struct ncplane *tmp_bound_plane = ncplane_create(stdn, &nopts);
509 if (!tmp_bound_plane) {
510 ncvisual_destroy(mm->ncv);
511 return nullptr;
512 }
513 vopts_calc.n = tmp_bound_plane;
514 if (ncvisual_geom(nc, mm->ncv, &vopts_calc, &geom) < 0) {
515 ncplane_destroy(tmp_bound_plane);
516 ncvisual_destroy(mm->ncv);
517 return nullptr;
518 }
519 unsigned rows = geom.rcelly;
520 unsigned cols = geom.rcellx;
521 // dimensions
522 ncplane_destroy(tmp_bound_plane);
523 mm->sfc = ui_surface_box(stdsfc, BOX, rows + 2, cols + 2, begy, 0, image_file);
524 ui_surface_addwin(mm->sfc, WIN, BOX, rows, cols, 1, 1);
525 struct ncvisual_options vopts = {
526 .n = mm->sfc->mplane[WIN],
527 .scaling = NCSCALE_SCALE_HIRES,
528 .blitter = NCBLIT_PIXEL,
529 };
530 ncvisual_blit(nc, mm->ncv, &vopts);
531 notcurses_render(nc);
532 return mm->ncv;
533}
