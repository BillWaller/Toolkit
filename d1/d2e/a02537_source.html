320 @param maxlen - size of buffer

322 @details This function uses getifaddrs to retrieve a linked list of network interfaces on the local machine. It iterates through the list and checks for interfaces with IPv4 addresses (AF_INET). 352 }

402/** @brief Converts a string into an array of argument strings.
403 @ingroup utility_functions
404 @param argv - array of pointers to arguments
405 @param arg_str - string containing arguments
406 @param max_args - maximum number of arguments to parse
407 @returns argc, a count of allocated vectors in argv
408 @details Handles quoted strings and escaped quotes, preserving text inside

412int str_to_args(char **argv, char *arg_str, uint max_args) {
413 if (argv == nullptr || arg_str == nullptr || max_args == 0)
414 return 0;
415 if (*arg_str == '\0')
416 return 0;

421 char *d = tmp_str;
422
423 while (*p != '\0' && argc < max_args) {
424 while (isspace((unsigned char)*p))
425 p++;
426 if (*p == '\0')
427 break;
428 if (*p == '"') {

431 }

433 if (in_quotes) {
434 if (*p == '\\' && *(p + 1) == '"') {
435 *d++ = '"';
436 p += 2;
437 } else if (*p == '"') {
438 *d++ = '\0';
439 p++;
440 in_quotes = 0;
441 break;
442 } else
443 *d++ = *p++;
444 } else {

447 p++;

451 }
