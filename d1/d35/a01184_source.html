56 @param by Beginning y coordinate for pick window

58 @details If provider_cmd is specified, it takes precedence over in_spec and

60 provider_cmd is executed and its output is read as pick input
61 If provider_cmd is not specified, in_spec is used to read pick

63 If provider_cmd is specified, it is executed and its output is

65int init_pick(Init *init, int argc, char **argv, int by, int bx) {

78 if (pipe(pipe_fd) == -1) {
79 Perror("pipe(pipe_fd) failed in init_pick");
80 return (1);

82 if ((pid = fork()) == -1) {
83 Perror("fork() failed in init_pick");
84 return (1);

86 if (pid == 0) {
87 /** Prevent child process from writing to terminal */
88 int dev_null = open("/dev/null", O_WRONLY);
89 if (dev_null == -1) {

91 exit(EXIT_FAILURE);
92 }
93 dup2(dev_null, STDERR_FILENO);
94 close(dev_null);
95 /** Close read end of pipe as Child only needs to write to pipe */
96 close(pipe_fd[P_READ]);
97 /** Connect CHILD STDOUT to write end of pipe */
98 dup2(pipe_fd[P_WRITE], STDOUT_FILENO);
99 /** STDOUT attached to write end of pipe, so close pipe fd */
100 close(pipe_fd[P_WRITE]);
101 execvp(s_argv[0], s_argv);
102 strnz__cpy(tmp_str, "Can't exec pick start cmd: ", MAXLEN - 1);
103 strnz__cat(tmp_str, s_argv[0], MAXLEN - 1);
104 Perror(tmp_str);
105 exit(EXIT_FAILURE);
106 }
107 /** Return to Parent
108 Close write end of pipe as Parent only needs to read from pipe */
109 close(pipe_fd[P_WRITE]);
110 /** Open a file pointer on read end of pipe */
111 pick->in_fp = fdopen(pipe_fd[P_READ], "rb");
112 pick->f_in_pipe = true;
113 destroy_argv(s_argc, s_argv);
114 } else {
115 if ((pick->in_spec[0] == '\0') || strcmp(pick->in_spec, "-") == 0 || strcmp(pick->in_spec, "/dev/stdin") == 0) {
116 strnz__cpy(pick->in_spec, "/dev/stdin", MAXLEN - 1);
117 pick->in_fp = fdopen(STDIN_FILENO, "rb");
118 pick->f_in_pipe = true;
119 }
120 }
121 if (!pick->f_in_pipe) {
122 /** No provider_cmd specified, so read pick input from file or stdin */
123 if (lstat(pick->in_spec, &sb) == -1) {
124 strnz__cpy(tmp_str, "Can\'t stat pick input file: ", MAXLEN - 1);
125 strnz__cat(tmp_str, pick->in_spec, MAXLEN - 1);
126 Perror(tmp_str);

128 }

218 /** If unexpected error occurs while waiting for pick input, kill

220 kill(pid, SIGKILL);
221 waitpid(pid, nullptr, 0);
222 close(pipe_fd[P_READ]);

224 return (1);

228 if (pick->f_in_pipe && pid > 0) {
229 /** Wait for provider_cmd child process to finish before proceeding */
230 waitpid_with_timeout(pid, wait_timeout);
231 close(pipe_fd[P_READ]);
232 dup2(sio->stdin_fd, STDIN_FILENO);
233 dup2(sio->stdout_fd, STDOUT_FILENO);
234 restore_curses_tioctl();
235 sig_prog_mode();
236 keypad(pick->win, true);
237 }

239 Perror("No pick objects available");
240 return (1);
241 }

243 pick->m_idx = 0;
244 pick->d_idx = 0;
245 while (pick->m_idx < pick->m_cnt)
246 pick->d_object[pick->d_idx++] = pick->m_object[pick->m_idx++];
247 pick->d_cnt = pick->d_idx;
248 pick_engine(init);
249 if (pick->p_view_files)
250 destroy_pick_view(init);
251 win_del();
252 return 0;

295/** @brief Initializes pick interface, calculates window size and position, and
296 enters picker loop
297 * @ingroup pick_engine
298 @param init Pointer to Init structure containing pick information
299 @return Count of selected objects on success, -1 if user cancels
300 @details Initializes key command strings for chyron display and calculates
301 pick window size and position based on terminal size and pick parameters. Opens
302 pick window and displays first page of objects. Enters picker loop to handle
303 user input and interactions. If user cancels selection, returns -1. If user
304 accepts selection, returns count of selected objects. */
305int pick_engine(Init *init) {
306 int rc;
307 int maxy, maxx;
308 int whitespace_ratio = 20;
309 int usable_lines;
310 int pick_ratio = 50;
311 int tbl_max_cols, pg_max_objs;
312 bool f_processed = false;
313
314 Pick *pick = init->pick;
315 getmaxyx(stdscr, maxy, maxx);
316 // Screen Geometry
317 // Calculate pick window size and position based on terminal size and pick
318 // parameters
319 //
320 // Calculate line usage:
321 if (pick->begy == 0)
322 pick->begy = (maxy * whitespace_ratio) / 200;
323 usable_lines = maxy - (maxy * whitespace_ratio) / 100;
324 usable_lines -= pick->begy;
325 usable_lines -= 5; // Pick overhead lines
326 if (pick->p_view_files) {
327 if (pick->lines == 0) {
328 pick->lines = usable_lines * pick_ratio / 100;
329 usable_lines -= pick->lines;
330 }
331 init->begy = pick->begy + 5 + pick->lines;
332 init->begx = 0;
333 usable_lines -= 3; // View overhead lines
334 init->lines = usable_lines;
335 init->cols = maxx - 2;
336 } else if (pick->lines == 0)
337 pick->lines = usable_lines;
338
339 pick->tbl_col_width = max(pick->tbl_col_width, 4);
340 pick->tbl_col_width = min(pick->tbl_col_width, maxx - 2);
341 if (pick->d_cnt <= pick->lines) {
342 pick->tbl_lines = pick->d_cnt;
343 pick->tbl_cols = 1;
344 } else {
345 tbl_max_cols = (maxx / (pick->tbl_col_width + 1));
346 pg_max_objs = pick->lines * tbl_max_cols;
347 if (pick->d_cnt > pg_max_objs)
348 pick->tbl_cols = tbl_max_cols;
349 else
350 pick->tbl_cols = pick->d_cnt / pick->lines;
351 pick->tbl_lines = pick->d_cnt / tbl_max_cols;
352 }
353 pick->tbl_pages = (pick->tbl_lines / (pick->lines - 1)) + 1;
354 // pick->lines = (pick->tbl_lines + pick->tbl_pages - 1) / pick->tbl_pages;
355 pick->tbl_page = 0;
356 // if (pick->begy == 0)
357 // pick->begy = (LINES - pick->lines) / 5;
358 // else if (pick->begy + pick->lines > LINES - 4)
359 // pick->begy = LINES - pick->lines - 2;
360 pick->width = (pick->tbl_col_width + 1) * pick->tbl_cols;
361
362 pick->chyron = new_chyron();
363 set_chyron_key(pick->chyron, 1, "F1 Help", KEY_F(1));
364 set_chyron_key(pick->chyron, 2, "F9 Cancel", KEY_F(9));
365 set_chyron_key(pick->chyron, 3, "F10 Accept", KEY_F(10));
366 set_chyron_key(pick->chyron, 4, "Sp Toggle", KEY_F(13));
367 set_chyron_key(pick->chyron, 5, "Tab Edit", '\t');
368 if (pick->tbl_pages > 1) {
369 set_chyron_key(pick->chyron, 11, "PgUp", KEY_PPAGE);
370 set_chyron_key(pick->chyron, 2, "PgDn", KEY_NPAGE);
371 }
372 compile_chyron(pick->chyron);
373 if (pick->chyron->l > pick->width)
374 pick->chyron->l = strnz(pick->chyron->s, pick->width);
375 if (pick->width < pick->chyron->l)
376 pick->width = pick->chyron->l;
377 pick->width = max(pick->width, pick->chyron->l) + 5;
378 if (pick->begx + pick->width > COLS - 2)
379 pick->begx = COLS - pick->width - 2;
380 else if (pick->begx == 0)
381 pick->begx = (COLS - pick->width) / 2;
382 rc = open_pick_win(init);
383 if (rc)
384 return (rc);
385 /** Enter picker loop to handle user input and interactions */
386 pick->d_idx = 0;
387 pick->x = 1;
388 char field[MAXLEN]; /**< Buffer for user input in the field */
389 field[0] = '\0';
390 display_page(pick);
391
392 do {
393 rc = picker(init, field);
394 if (rc == KEY_F(9))
395 break;
396 if (rc == -1)
397 break;
398 else {
399 if (pick->select_cnt > 0) {
400 if (pick->f_out_spec && pick->out_spec[0]) {
401 output_objects(pick);
402 f_processed = true;
403 }
404 if (pick->f_cmd && pick->cmd[0]) {
405 exec_objects(init);
406 f_processed = true;
407 }
408 if (pick->f_read_theme) {
409 read_theme(init);
410 f_processed = true;
411 }
412 if (f_processed) {
413 mvwaddstr(pick->win2, 0, 0, "Selection Processed");
414 wclrtoeol(pick->win2);
415 update_panels();
416 }
417 }
418 }
419 deselect_object(pick);
420 } while (1);
421 update_panels();
422 doupdate();
423 destroy_chyron(pick->chyron);
424 return (rc);
425}

447 }
