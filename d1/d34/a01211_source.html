138/** @brief Extract the next directory path from the PATH string
139 @param dp A buffer to store the extracted directory path
140 @param sp A pointer to the current position in the PATH string
141 @return The length of the extracted directory path
142 @details This function takes a buffer and a pointer to the current position
143 in the PATH string, and extracts the next directory path. If the next
144 character in the PATH string is a colon, it treats it as an empty path and
145 uses the current working directory. Otherwise, it copies characters until it
146 reaches a colon or the end of the string, and returns the length of the
147 extracted path.
148 */
149int next_path(char *dp, char **pp) {
150 int dl;
151
152 if (**pp == ':') {
153 (*pp)++;
154 getcwd(dp, PATH_MAX);
155 return (strlen(dp));
156 } else {
157 dl = 0;
158 while (**pp != '\0' && **pp != '\n' && **pp != '\r' && **pp != ':') {
159 *dp++ = *(*pp)++;
160 dl++;
161 }
162 *dp = '\0';
163 if (**pp == ':' && *++(*pp) == '\0')
164 (*pp)--;
165 return (dl);
166 }
167}
168/** @brief Split a file specification into directory and file name components

173 @details This function takes a file specification, checks if it is a

199 file_spec_l = strlen(tmp_file_spec);
200 if (file_spec_l > 0) {
201 i = 0;

213 file_name[0] = '\0';

216 strnz__cpy(file_spec, file_path, MAXLEN - 1);
217 strnz__cat(file_spec, file_name, MAXLEN - 1);
218 } else {
219 tmp_file_spec[last_slash] = '\0';
220 strnz__cpy(file_path, tmp_file_spec, MAXLEN - 1);
221 strnz__cat(file_path, "/", MAXLEN - 1);
222 if (last_slash < file_spec_l)
223 last_slash++;
224 strnz__cpy(file_name, tmp_file_spec + last_slash, MAXLEN - 1);
225 }
226 return (0);
