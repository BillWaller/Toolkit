206/** open_curses
207 @brief Initialize NCurses and color settings
208 @ingroup window_support
209 @param sio Pointer to SIO struct with terminal and color settings
210 @return true if successful, false if error
211 @details This function initializes NCurses and sets up color pairs based on
212 the settings in the SIO struct. It also applies gamma correction to colors.
213 Use this function to initialize NCurses if you don't want NCurses to receive
214 data from the stdin pipe
215 @code
216 1. saves stdin and stdout file descriptors in SIO
217 2. opens a terminal device for NCurses screen IO
218 3. replaces STDERR_FILENO with terminal file descriptor
219 @endcode */
220bool open_curses(SIO *sio) {
221 char tmp_str[MAXLEN];
222 char emsg0[MAXLEN];
223
224 // Get the name of the terminal device
225 sio->stdin_fd = dup(STDIN_FILENO);
226 sio->stdout_fd = dup(STDOUT_FILENO);
227 sio->stderr_fd = dup(STDERR_FILENO);
228 if (ttyname_r(STDERR_FILENO, sio->tty_name, sizeof(sio->tty_name)) != 0) {
229 strerror_r(errno, tmp_str, MAXLEN - 1);
230 strnz__cpy(emsg0, "ttyname_r failed ", MAXLEN - 1);
231 strnz__cat(emsg0, tmp_str, MAXLEN - 1);
232 fprintf(stderr, "%s\n", tmp_str);
233 exit(0);
234 }
235 // open the terminal device for NCurses input and output
236 tty_fp = fopen(sio->tty_name, "r+");
237 if (tty_fp == nullptr) {
238 strerror_r(errno, tmp_str, MAXLEN - 1);
239 strnz__cpy(emsg0, "fopen(sio->tty_name) failed ", MAXLEN - 1);
240 strnz__cat(emsg0, tmp_str, MAXLEN - 1);
241 fprintf(stderr, "%s\n", tmp_str);
242 exit(0);
243 }
244 // newterm() allows us to specify a tty device for NCurses input and
245 // output.
246 screen = newterm(nullptr, tty_fp, tty_fp);
247 if (screen == nullptr) {
248 strerror_r(errno, tmp_str, MAXLEN - 1);
249 strnz__cpy(emsg0, "newterm failed ", MAXLEN - 1);
250 strnz__cat(emsg0, tmp_str, MAXLEN - 1);
251 fprintf(stderr, "%s\n", tmp_str);
252 exit(0);
253 }
254 set_term(screen);
255 f_curses_open = true;
256 if (!has_colors()) {
257 destroy_curses();
258 abend(-1, "Terminal color support required");
259 }
260 start_color();
261 if (!can_change_color()) {
262 destroy_curses();
263 fprintf(stderr, "Terminal cannot change colors\n");
264 fprintf(stderr, "Check TERM environment variable\n");
265 fprintf(stderr, "Check terminfo for missing \"ccc\"\n");
266 abend(-1, "fatal error");
267 }
268 panel_main = new_panel(stdscr);
269 noecho();
270 keypad(stdscr, true);
271 idlok(stdscr, false);
272 idcok(stdscr, false);
273 wbkgrnd(stdscr, &CC_NT);
274 for (sfc_ptr = 0; sfc_ptr < MAXWIN; sfc_ptr++) {

276 ui_surface[sfc_ptr]->mpan[i] = nullptr;
277 ui_surface[sfc_ptr]->mwin[i] = nullptr;
278 }

280 sfc_ptr = -1;

282}

442/** rgb_to_xterm256_idx
443 @brief Convert RGB color to XTerm 256 color index
444 @ingroup color_management
445 @param rgb RGB color
446 @return XTerm 256 color index
447 @details This function converts an RGB color to the nearest XTerm 256 color
448 index. It first checks if the color is a shade of gray, and if so, it uses
449 the gray ramp. Otherwise, it calculates the nearest color in the 6x6x6 color
450 cube. */
451int rgb_to_xterm256_idx(RGB *rgb) {
452 if (rgb->r == rgb->g && rgb->g == rgb->b) {
453 if (rgb->r < 8)
454 return 16;
455 if (rgb->r > 248)
456 return 231;
457 return ((rgb->r - 8) / 10) + 231;
458 } else {
459 int r_index = (rgb->r < 45) ? 0 : (rgb->r - 60) / 40 + 1;
460 int g_index = (rgb->g < 45) ? 0 : (rgb->g - 60) / 40 + 1;
461 int b_index = (rgb->b < 45) ? 0 : (rgb->b - 60) / 40 + 1;
462 return 16 + (36 * r_index) + (6 * g_index) + b_index;
463 }
464}
465/** xterm256_idx_to_rgb
466 @brief Convert XTerm 256 color index to RGB
467 @ingroup color_management
468 @param idx XTerm 256 color index
469 @return RGB color
470 @details This function converts an XTerm 256 color index to an RGB color. It
471 first checks if the index is in the standard 16 colors, then checks if it's
472 in the 6x6x6 color cube, and finally checks if it's in the gray ramp. */
473RGB xterm256_idx_to_rgb(int idx) {
474 /** Convert XTerm 256 color index to RGB
475 @param idx - XTerm 256 color index
476 @return RGB struct */
477 RGB rgb;
478 if (idx > 255)
479 idx = 255;
480 if (idx < 0)
481 idx = 0;
482 rgb.r = rgb.g = rgb.b = 0;
483 if (idx < 16) {
484 rgb.r = StdColors[idx].r;
485 rgb.g = StdColors[idx].g;
486 rgb.b = StdColors[idx].b;
487 } else if (idx >= 16 && idx <= 231) {
488 idx -= 16;
489 rgb.r = (idx / 36) % 6 * 51;
490 rgb.g = (idx / 6) % 6 * 51;
491 rgb.b = (idx % 6) * 51;
492 } else if (idx >= 232 && idx <= 255) {
493 int gray = (idx - 232) * 11;
494 rgb.r = rgb.g = rgb.b = gray;
495 }
496 return rgb;
497}

499/** apply_gamma
500 @brief Apply gamma correction to RGB color
501 @ingroup color_management
502 @param rgb Pointer to RGB color
503 @details This function modifies the RGB color in place. It applies gamma
504 correction to the RGB color based on the gamma values set in the SIO struct.
505 If the color is a shade of gray, it applies the gray gamma correction.
506 Otherwise, it applies the individual red, green, and blue gamma corrections.
507 */
508void apply_gamma(RGB *rgb) {
509 if (rgb->r == rgb->g && rgb->r == rgb->b) {
510 if (GRAY_GAMMA > 0.0f && GRAY_GAMMA != 1.0f) {
511 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / GRAY_GAMMA) * 255.0f);
512 rgb->g = rgb->r;
513 rgb->b = rgb->r;
514 }
515 return;
516 }
517 if (rgb->r != 0 && RED_GAMMA > 0.0f && RED_GAMMA != 1.0f)
518 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / RED_GAMMA) * 255.0f);
519 if (rgb->g != 0 && GREEN_GAMMA > 0.0f && GREEN_GAMMA != 1.0f)
520 rgb->g = (int)(pow((rgb->g / 255.0f), 1.0f / GREEN_GAMMA) * 255.0f);
521 if (rgb->b != 0 && BLUE_GAMMA > 0.0f && BLUE_GAMMA != 1.0f)
522 rgb->b = (int)(pow((rgb->b / 255.0f), 1.0f / BLUE_GAMMA) * 255.0f);
523}

694 @return Number of bytes processed from the input string
695 @details This function converts a multibyte string to an array of complex

697 multibyte characters and applies the specified color pair to each character.
698 The pos parameter is updated to reflect the current position in the output
699 buffer, and the function ensures that it does not exceed the maximum length.
700*/
701int mb_to_cc(cchar_t *cmplx_buf, char *str, attr_t attr, int cpx, int *pos, int maxlen) {
702 int i = 0, len = 0;
703 const char *s;
704 cchar_t cc = {0};

766/** @brief Converts a Unicode code point to a UTF-8 encoded string.
767 @ingroup utility_functions
768 @param cp - Unicode code point to convert
769 @param buffer - buffer to receive UTF-8 encoded string (must be at least 4 bytes)
770 @returns number of bytes written to buffer, or 0 if cp is invalid
771 @details This function encodes the given Unicode code point into its UTF-8 representation and stores it in the provided buffer. The caller must ensure that the buffer has enough space to hold the resulting UTF-8 string (up to 4 bytes for code points up to U+10FFFF). If the code point is invalid (e.g., greater than U+10FFFF), this function returns 0 and does not modify the buffer. */
772int wccp_to_str(wchar_t cp, uint8_t *buffer) {
773 if (cp <= 0x7F) {
774 // 1-byte sequence: 0xxxxxxx
775 buffer[0] = (uint8_t)cp;
776 return 1;
777 } else if (cp <= 0x7FF) {
778 // 2-byte sequence: 110xxxxx 10xxxxxx
779 buffer[0] = (uint8_t)(0xC0 | ((cp >> 6) & 0x1F));
780 buffer[1] = (uint8_t)(0x80 | (cp & 0x3F));
781 return 2;
782 } else if (cp <= 0xFFFF) {
783 // 3-byte sequence: 1110xxxx 10xxxxxx 10xxxxxx
784 buffer[0] = (uint8_t)(0xE0 | ((cp >> 12) & 0x0F));
785 buffer[1] = (uint8_t)(0x80 | ((cp >> 6) & 0x3F));
786 buffer[2] = (uint8_t)(0x80 | (cp & 0x3F));
787 return 3;
788 } else if (cp <= 0x10FFFF) {
789 // 4-byte sequence: 11110xxx 10xxxxxx 10xxxxxx 10xxxxxx
790 buffer[0] = (uint8_t)(0xF0 | ((cp >> 18) & 0x07));
791 buffer[1] = (uint8_t)(0x80 | ((cp >> 12) & 0x3F));
792 buffer[2] = (uint8_t)(0x80 | ((cp >> 6) & 0x3F));
793 buffer[3] = (uint8_t)(0x80 | (cp & 0x3F));
794 return 4;
795 }
796 return 0; // Invalid Unicode code point
797}

1514/** xwgetch
1515 @brief Wrapper for wgetch that handles signals, mouse events, checks for
1516 clicks on the chyron line, and accepts a sinigle character answer
1517 @ingroup window_support
1518 @param win Pointer to window
1519 @param chyron Pointer to chyron struct
1520 @param n Number of seconds to wait before timing out
1521 @verbatim
1522
1523 0: Wait indefinitely for user input (raw mode)
1524 accept a single character answer, and don't wait for Enter key
1525 1: Wait for 1 decisecond
1526 n > 1: Wait for n/10 seconds
1527
1528 @endverbatim
1529 @return Key code or ERR if interrupted by signal
1530 @details Get mouse event and check if it's a left click or double click. If
1531 the click is outside the window, ignore it. If it's on the chyron line, get
1532 the corresponding key command. Otherwise, store the click coordinates as
1533 click_y and click_x for later use. */
1534int xwgetch(WINDOW *win, Chyron *chyron, int n) {
1535 int c;
1536 MEVENT event;
1537 mousemask(BUTTON1_CLICKED | BUTTON1_DOUBLE_CLICKED | BUTTON4_PRESSED | BUTTON5_PRESSED,
1538 nullptr);
1539 click_y = event.y = -1;
1540 click_x = event.x = -1;
1541
1542 if (n == -1) {
1543 struct termios raw_tioctl;
1544 raw_tioctl = curses_tioctl;
1545 mk_raw_tioctl(&raw_tioctl);
1546 } else if (n == 0)
1547 halfdelay(1);
1548 else
1549 halfdelay(min(255, max(0, n * 10)));
1550 tcflush(2, TCIFLUSH);
1551 do {
1552 curs_set(1);
1553 c = wgetch(win);
1554 curs_set(0);
1555 if (sig_received != 0) {
1556 if (handle_signal(sig_received))
1557 c = display_error(em0, em1, em2, nullptr);
1558 if (c == 'q' || c == 'Q' || c == KEY_F(9))
1559 exit(EXIT_FAILURE);
1560 }
1561 if (n > 0 && c == ERR) {
1562 c = 0;
1563 break;
1564 }
1565 if (c == ERR)
1566 continue;
1567 if (c == KEY_MOUSE) {
1568 if (getmouse(&event) != OK) {
1569 c = 0;
1570 continue;
1571 }
1572 if (event.bstate & BUTTON4_PRESSED) {
1573 return KEY_UP;
1574 } else if (event.bstate & BUTTON5_PRESSED) {
1575 return KEY_DOWN;
1576 }
1577 if (event.bstate & BUTTON1_CLICKED || event.bstate & BUTTON1_DOUBLE_CLICKED) {
1578 if (wenclose(win, event.y, event.x)) {
1579 wmouse_trafo(win, &event.y, &event.x, false);
1580 click_y = event.y;
1581 click_x = event.x;
1582 if (chyron && event.y == getmaxy(win) - 1) {
1583 c = get_chyron_key(chyron, event.x);
1584 break;
1585 } else
1586 break;
1587 }
1588 c = ERR;
1589 continue;
1590 }
1591 }
1592 } while (c == ERR);
1593 restore_curses_tioctl();
1594 return c;
1595}
