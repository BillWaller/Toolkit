293/** mbc_to_wc
294 @brief Convert a multibyte character to a wide character
295 @ingroup window_support
296 @param wc Output array for wide characters (size 2)
297 @param mbc Input multibyte character
298 @details This function converts a single multibyte character to a wide
299 character. It uses the mbrtowc function to perform the conversion and handles
300 any errors by setting the output to a question mark ('?') if the conversion
301 fails. The output array is null-terminated. */
302void ui_mbc_to_wc(wchar_t wc[2], const char mbc) {
303 wc[0] = wc[1] = L'\0';
304 mbstate_t state = {};
305 size_t len = mbrtowc(wc, &mbc, MB_CUR_MAX, &state);
306 if (len <= 0) {
307 wc[0] = L'?';
308 wc[1] = L'\0';
309 len = 1;
310 }
311}
312/** ui_mbstr_to_wcstr
313 @brief Convert a multibyte string to a wide character string
314 @ingroup window_support
315 @param mb_str Input multibyte string
316 @return Pointer to newly allocated wide character string, or nullptr on error
317 @details This function converts a null-terminated multibyte string to a wide
318 character string. It first calculates the required length for the wide
319 character string, allocates memory for it, and then performs the conversion.
320 The caller is responsible for freeing the allocated memory. If the conversion
321 fails, it returns nullptr. */
322wchar_t *ui_mbstr_to_wcstr(const char *mb_str) {
323 const char *src_ptr = mb_str;
324 mbstate_t state = {};
325 size_t wc_count = mbsrtowcs(NULL, &src_ptr, 0, &state);
326 if (wc_count == (size_t)-1)
327 return nullptr;
328 src_ptr = mb_str;
329 wmemset((wchar_t *)&state, 0, sizeof(state) / sizeof(wchar_t));
330 wchar_t *wc_str = malloc((wc_count + 1) * sizeof(wchar_t));
331 if (!wc_str)
332 return nullptr;
333 mbsrtowcs(wc_str, &src_ptr, wc_count + 1, &state);
334 return wc_str;
335}

772/** compile_chyron
773 @brief construct the chyron string from the chyron structure
774 @ingroup UiChyron
775 @param chyron
776 @details The chyron string is constructed by concatenating the labels of the

780 clicked based on the X position of the click.

783 uint end_pos = 0;

785 uint pos = 0;
786 UiCell cell_base = cell_nt_rev;
787 UiCell *cx;
788 char tmp_str[MAXLEN];
789 while (k < CHYRON_KEYS) {
790 if (chyron->key[k]->text[0] == '\0' || !chyron->key[k]->active) {
791 k++;
792 continue;
793 }
794 cell_base = chyron->key[k]->cell_base;
795 if (end_pos == 0) {
796 cx = chyron->cmplx_buf;
797 ui_mbstr_to_cellstr(chyron->sfc, chyron->w,
798 cx,
799 " ",
800 &cell_base,
801 &pos,
802 MAXLEN - 1);
803 } else {
804 ui_mbstr_to_cellstr(chyron->sfc, chyron->w,
805 chyron->cmplx_buf,
806 "|",
807 &cell_base,
808 &pos,
809 MAXLEN - 1);
810 }
811 cx = chyron->cmplx_buf;
812 ui_mbstr_to_cellstr(chyron->sfc, chyron->w, cx, chyron->key[k]->text, &cell_base, &pos, MAXLEN - 1);
813 end_pos = pos;
814 chyron->l = end_pos;
815 chyron->key[k]->end_pos = end_pos;
816 ssnprintf(tmp_str, MAXLEN - 1, _("k=%d, text=%s, end_pos=%d"), k,
817 chyron->key[k]->text, chyron->key[k]->end_pos);
818 k++;
819 }
820 ui_mbstr_to_cellstr(chyron->sfc, chyron->w, chyron->cmplx_buf, " ", &cell_base, &pos, MAXLEN - 1);
821 chyron->l = end_pos;
822}

877/** ui_rgb_to_xterm256_idx
878 @brief Convert RGB color to XTerm 256 color index
879 @ingroup color_management
880 @param rgb RGB color
881 @return XTerm 256 color index
882 @details This function converts an RGB color to the nearest XTerm 256 color
883 index. It first checks if the color is a shade of gray, and if so, it uses
884 the gray ramp. Otherwise, it calculates the nearest color in the 6x6x6 color
885 cube. */
886uint ui_rgb_to_xterm256_idx(RGB *rgb) {
887 if (rgb->r == rgb->g && rgb->g == rgb->b) {
888 if (rgb->r < 8)
889 return 16;
890 if (rgb->r > 248)
891 return 231;
892 return ((rgb->r - 8) / 10) + 231;
893 } else {
894 uint r_index = (rgb->r < 45) ? 0 : (rgb->r - 60) / 40 + 1;
895 uint g_index = (rgb->g < 45) ? 0 : (rgb->g - 60) / 40 + 1;
896 uint b_index = (rgb->b < 45) ? 0 : (rgb->b - 60) / 40 + 1;
897 return 16 + (36 * r_index) + (6 * g_index) + b_index;
898 }
899}
900/** ui_xterm256_idx_to_rgb
901 @brief Convert XTerm 256 color index to RGB
902 @ingroup color_management
903 @param idx XTerm 256 color index
904 @return RGB color
905 @details This function converts an XTerm 256 color index to an RGB color. It
906 first checks if the index is in the standard 16 colors, then checks if it's
907 in the 6x6x6 color cube, and finally checks if it's in the gray ramp. */
908RGB ui_xterm256_idx_to_rgb(uint idx) {
909 /** Convert XTerm 256 color index to RGB
910 @param idx - XTerm 256 color index
911 @return RGB struct */
912 RGB rgb;
913 if (idx > 255)
914 idx = 255;
915 rgb.r = rgb.g = rgb.b = 0;
916 if (idx < 16) {
917 rgb.r = std_color[idx].r;
918 rgb.g = std_color[idx].g;
919 rgb.b = std_color[idx].b;
920 } else if (idx >= 16 && idx <= 231) {
921 idx -= 16;
922 rgb.r = (idx / 36) % 6 * 51;
923 rgb.g = (idx / 6) % 6 * 51;
924 rgb.b = (idx % 6) * 51;
925 } else if (idx >= 232 && idx <= 255) {
926 uint gray = (idx - 232) * 11;
927 rgb.r = rgb.g = rgb.b = gray;
928 }
929 return rgb;
930}

933 @brief Apply gamma correction to RGB color
934 @ingroup color_management
935 @param rgb Pointer to RGB color
936 @details This function modifies the RGB color in place. It applies gamma
937 correction to the RGB color based on the gamma values set in the SIO struct.
938 If the color is a shade of gray, it applies the gray gamma correction.
939 Otherwise, it applies the individual red, green, and blue gamma corrections.
940 */
941void ui_apply_gamma(RGB *rgb) {
942 if (rgb->r == rgb->g && rgb->r == rgb->b) {

946 rgb->b = rgb->r;
947 }
948 return;
949 }
950 if (rgb->r != 0 && RED_GAMMA > 0.0f && RED_GAMMA != 1.0f)
951 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / RED_GAMMA) * 255.0f);
952 if (rgb->g != 0 && GREEN_GAMMA > 0.0f && GREEN_GAMMA != 1.0f)
953 rgb->g = (int)(pow((rgb->g / 255.0f), 1.0f / GREEN_GAMMA) * 255.0f);
954 if (rgb->b != 0 && BLUE_GAMMA > 0.0f && BLUE_GAMMA != 1.0f)
955 rgb->b = (int)(pow((rgb->b / 255.0f), 1.0f / BLUE_GAMMA) * 255.0f);
956}
