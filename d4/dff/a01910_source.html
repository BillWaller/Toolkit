305/** ui_mbstr_to_wcstr
306 @brief Convert a multibyte string to a wide character string
307 @ingroup window_support
308 @param mb_str Input multibyte string
309 @return Pointer to newly allocated wide character string, or nullptr on error
310 @details This function converts a null-terminated multibyte string to a wide
311 character string. It first calculates the required length for the wide
312 character string, allocates memory for it, and then performs the conversion.
313 The caller is responsible for freeing the allocated memory. If the conversion
314 fails, it returns nullptr. */
315wchar_t *ui_mbstr_to_wcstr(const char *mb_str) {
316 const char *src_ptr = mb_str;
317 mbstate_t state = {};
318 size_t wc_count = mbsrtowcs(NULL, &src_ptr, 0, &state);
319 if (wc_count == (size_t)-1)
320 return nullptr;
321 src_ptr = mb_str;
322 wmemset((wchar_t *)&state, 0, sizeof(state) / sizeof(wchar_t));
323 wchar_t *wc_str = malloc((wc_count + 1) * sizeof(wchar_t));
324 if (!wc_str)
325 return nullptr;
326 mbsrtowcs(wc_str, &src_ptr, wc_count + 1, &state);
327 return wc_str;
328}

996/** ui_rgb_to_xterm256_idx
997 @brief Convert RGB color to XTerm 256 color index
998 @ingroup color_management
999 @param rgb RGB color
1000 @return XTerm 256 color index
1001 @details This function converts an RGB color to the nearest XTerm 256 color
1002 index. It first checks if the color is a shade of gray, and if so, it uses
1003 the gray ramp. Otherwise, it calculates the nearest color in the 6x6x6 color
1004 cube. */
1005uint ui_rgb_to_xterm256_idx(RGB *rgb) {
1006 if (rgb->r == rgb->g && rgb->g == rgb->b) {
1007 if (rgb->r < 8)
1008 return 16;
1009 if (rgb->r > 248)
1010 return 231;
1011 return ((rgb->r - 8) / 10) + 231;
1012 } else {
1013 uint r_index = (rgb->r < 45) ? 0 : (rgb->r - 60) / 40 + 1;
1014 uint g_index = (rgb->g < 45) ? 0 : (rgb->g - 60) / 40 + 1;
1015 uint b_index = (rgb->b < 45) ? 0 : (rgb->b - 60) / 40 + 1;
1016 return 16 + (36 * r_index) + (6 * g_index) + b_index;
1017 }
1018}
1019/** ui_xterm256_idx_to_rgb
1020 @brief Convert XTerm 256 color index to RGB
1021 @ingroup color_management
1022 @param idx XTerm 256 color index
1023 @return RGB color
1024 @details This function converts an XTerm 256 color index to an RGB color. It
1025 first checks if the index is in the standard 16 colors, then checks if it's
1026 in the 6x6x6 color cube, and finally checks if it's in the gray ramp. */
1027RGB ui_xterm256_idx_to_rgb(uint idx) {
1028 /** Convert XTerm 256 color index to RGB
1029 @param idx - XTerm 256 color index
1030 @return RGB struct */
1031 RGB rgb;
1032 if (idx > 255)
1033 idx = 255;
1034 rgb.r = rgb.g = rgb.b = 0;
1035 if (idx < 16) {
1036 rgb.r = std_color[idx].r;
1037 rgb.g = std_color[idx].g;
1038 rgb.b = std_color[idx].b;
1039 } else if (idx >= 16 && idx <= 231) {
1040 idx -= 16;
1041 rgb.r = (idx / 36) % 6 * 51;
1042 rgb.g = (idx / 6) % 6 * 51;
1043 rgb.b = (idx % 6) * 51;
1044 } else if (idx >= 232 && idx <= 255) {
1045 uint gray = (idx - 232) * 11;
1046 rgb.r = rgb.g = rgb.b = gray;
1047 }
1048 return rgb;
1049}

1051/** ui_apply_gamma
1052 @brief Apply gamma correction to RGB color
1053 @ingroup color_management
1054 @param rgb Pointer to RGB color
1055 @details This function modifies the RGB color in place. It applies gamma
1056 correction to the RGB color based on the gamma values set in the SIO struct.
1057 If the color is a shade of gray, it applies the gray gamma correction.
1058 Otherwise, it applies the individual red, green, and blue gamma corrections.
1059 */
1060void ui_apply_gamma(RGB *rgb) {
1061 if (rgb->r == rgb->g && rgb->r == rgb->b) {
