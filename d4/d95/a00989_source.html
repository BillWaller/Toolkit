26 @ingroup parse_menu
27 @param init Pointer to Init structure containing menu information
28 @return 0 on success, non-zero on failure

30unsigned int parse_menu_description(Init *init) {
31 FILE *fp;
32 char tmp_str[MAXLEN];
33 char tmp_buf[MAXLEN];
34 char in_buf[MAXLEN];
35 char *in_buf_p;
36 int Pos;
37 unsigned char ltr;
38 unsigned char fltr[127];
39 int directive;
40 int l;
41 char *s, *d, *e;
42 int actions = 0;
43 int choices = 0;
44 int in_fp_line = 0;
45 menu = init->menu;
46 fp = fopen(menu->mapp_spec, "r");
47 if (fp == nullptr) {
48 strnz__cat(tmp_buf, "file not found", MAXLEN);
49 abend(-1, tmp_buf);
50 exit(-1);
51 }
52 while ((fgets(in_buf, MAXLEN, fp)) != nullptr) {
53 if (in_buf[0] == '\0')
54 continue;
55 in_fp_line++;
56 chrep(in_buf, '\r', '\0');
57 chrep(in_buf, '\n', '\0');
58 chrep(in_buf, '\t', ' ');
59 in_buf_p = in_buf;
60 directive = *in_buf_p;
61 in_buf_p++;
62 strnz__cpy(tmp_buf, in_buf_p, MAXLEN);
63 trim(tmp_buf);
64 l = strlen(tmp_buf);
65 if (!l)
66 continue;
67 switch (directive) {
68 /** '#' Comment line, ignore */
69 case '#':
70 break;
71 /** '!' Choice line, parse and add to menu */
72 case '!':
73 if (!menu->line_idx)
74 break;
75 if (menu->line[menu->line_idx - 1]->type != MT_TEXT)
76 break;
77 menu->line_idx--;
78 l = strlen(tmp_buf);
79 if (l + 1 > MAXLEN)
80 break;
81 menu->line[menu->line_idx]->command_str =
82 (char *)malloc(MAXLEN + 1);
83 if (!menu->line[menu->line_idx]->command_str) {
84 sprintf(tmp_str,
85 "0-malloc(%d bytes) failed M-L[%d]->command_str",
86 MAXLEN + 1, menu->line_idx);
87 abend(-1, tmp_str);
88 }
89 menu->line[menu->line_idx]->command_str = strdup(tmp_buf);
90 menu->line[menu->line_idx]->command_type =
91 get_command_type(tmp_buf);
92 s = menu->line[menu->line_idx]->raw_text;
93 if (*s == '-' || *s == '_') {
94 s++;
95 ltr = *s++;
96 } else
97 ltr = *s;
98 l = 0;
99 while (*s++ != '\0')
100 l++;
101 if (l + 1 > MAXLEN)
102 l = MAXLEN - 1;
103 menu->line[menu->line_idx]->choice_text =
104 (char *)malloc(MAXLEN + 1);
105 if (!menu->line[menu->line_idx]->choice_text) {
106 sprintf(tmp_str,
107 "1-malloc(%d bytes) failed M-L[%d]->choice_text",
108 MAXLEN + 1, menu->line_idx);
109 abend(-1, tmp_str);
110 }
111 d = menu->line[menu->line_idx]->choice_text;
112 e = d + l;
113 s = menu->line[menu->line_idx]->raw_text;
114 if (*s == '-' || *s == '_')
115 s += 2;
116 while (*s != '\0' && d < e)
117 *d++ = *s++;
118 *d = (char)'\0';
119 if (l > menu->choice_max_len)
120 menu->choice_max_len = l;
121 if (menu->line[menu->line_idx]->command_type == CT_RETURN)
122 menu->line[menu->line_idx]->choice_letter = 'Q';
123 else
124 menu->line[menu->line_idx]->choice_letter = ltr;
125 menu->line[menu->line_idx]->type = MT_CHOICE;
126 menu->line_idx++;
127 actions++;
128 break;
129 /** ':' Text line, parse and add to menu */
130 case ':':
131 if (choices > actions) {
132 ssnprintf(em0, MAXLEN - 1,
133 "More choices than actions at line %d of",
134 in_fp_line);
135 strnz__cpy(em1, menu->mapp_spec, MAXLEN - 1);
136 strnz__cat(em2, in_buf, MAXLEN - 1);
137 display_error(em0, em1, em2, nullptr);
138 abend(-1, "unrecoverable error");
139 }
140 chrep(tmp_buf, '\t', ' ');
141 l = strlen(tmp_buf);
142 if (l > menu->text_max_len)
143 menu->text_max_len = l;
144 if (!menu->title[0]) { /**< in_buf -> Title */
145 if (l + 5 > MAXLEN)
146 l = MAXLEN - 5;
147 strnz__cpy(menu->title, tmp_buf, l);
148 l += 4;
149 if (l > menu->text_max_len)
150 menu->text_max_len = l;
151 } else {
152 menu->line[menu->line_idx] = (Line *)malloc(sizeof(Line));
153 if (menu->line[menu->line_idx] == (Line *)0) {
154 sprintf(tmp_str,
155 "2-malloc(%ld bytes) failed menu->line[%d]",
156 sizeof(Line), menu->line_idx);
157 abend(-1, tmp_str);
158 }
159 menu->line[menu->line_idx]->type = MT_TEXT;
160 menu->line[menu->line_idx]->raw_text = strdup(tmp_buf);
161 menu->line[menu->line_idx]->choice_text = nullptr;
162 menu->line[menu->line_idx]->choice_letter = '\0';
163 menu->line[menu->line_idx]->letter_pos = 0;
164 menu->line[menu->line_idx]->command_type = '\0';
165 menu->line[menu->line_idx]->command_str = nullptr;
166 menu->line_idx++;
167 choices++;
168 }
169 break; /** '?' Help line, ignore for now */
170 case '?':
171 break; /** ' ' Empty line, ignore */
172 case ' ':
173 case '\0':
174 case '\n':
175 break;
176 default:
177
178 // strnz__cpy(tmp_buf, "unrecognized operator in ", MAXLEN);
179 // abend(-1, tmp_buf);
180 }
181 }
182 fclose(fp);
183 menu->item_count = menu->line_idx;
184 for (ltr = '0'; ltr < 'z'; ltr++)
185 fltr[ltr] = FALSE;
186 fltr['Q'] = TRUE;
187 for (menu->line_idx = 0; menu->line_idx < menu->item_count;
188 menu->line_idx++) {
189 ltr = menu->line[menu->line_idx]->choice_letter;
190 if (ltr < '0' || ltr > 'z')
191 ltr = '0';
192 if (!fltr[ltr] ||
193 (menu->line[menu->line_idx]->command_type == CT_RETURN &&
194 ltr == 'Q')) {
195 fltr[ltr] = TRUE;
196 s = menu->line[menu->line_idx]->choice_text;
197 Pos = 0;
198 while (*s != '\0') {
199 if (*s == ltr)
200 break;
201 s++;
202 Pos++;
203 }
204 } else {
205 Pos = 0;
206 while (ltr != '\0') {
207 Pos++;
208 ltr = toupper(menu->line[menu->line_idx]->choice_text[Pos]);
209 if (ltr >= '0' && ltr <= 'Z')
210 if (!fltr[ltr])
211 break;
212 }
213 if (ltr == '\0') {
214 for (ltr = '0'; ltr < 'Z'; ltr++)
215 if (!fltr[ltr])
216 break;
217 if (ltr > 'z') {
218 Perror("Ran out of letters");
219 return 0;
220 }
221 }
222 fltr[ltr] = TRUE;
223 menu->line[menu->line_idx]->choice_letter = ltr;
224 }
225 menu->line[menu->line_idx]->letter_pos = Pos;
226 }
227 menu->lines = menu->item_count;
228 if (menu->text_max_len > (menu->choice_max_len + 6))
229 menu->cols = menu->text_max_len;
230 else
231 menu->cols = menu->choice_max_len + 6;
232 if (menu->cols >= MAXLEN)
233 Perror("line too long");
234 for (menu->line_idx = 0; menu->line_idx < menu->item_count;
235 menu->line_idx++) {
236 strnz__cpy(tmp_buf, " x - ", MAXLEN - 1);
237 tmp_buf[1] = menu->line[menu->line_idx]->choice_letter;
238 strnz__cat(tmp_buf, menu->line[menu->line_idx]->choice_text,
239 MAXLEN - 1);
240 strnz__cpy(menu->line[menu->line_idx]->choice_text, tmp_buf,
241 MAXLEN - 1);
242 }
243
244 return 0;
245}
246/** @brief Get command type from command string
247 @ingroup parse_menu
248 @param t Command string
249 @return Command type as an unsigned int
250 */
251unsigned int get_command_type(char *t) {
252 char *s, *p;
253
254 s = p = t;
255 while (*s != ' ' && *s != '\0') {
256 if (*s == '/')
257 p = s + 1;
258 s++;
259 }
260 *s = '\0';
261 if (!strcmp(p, "returnmain"))
262 return (CT_RETURNMAIN);
263 else if (!strcmp(p, "ckeys"))
264 return (CT_CKEYS);
265 else if (!strcmp(p, "exec"))
266 return (CT_EXEC);
267 else if (!strcmp(p, "help"))
268 return (CT_HELP);
269 else if (!strcmp(p, "about"))
270 return (CT_ABOUT);
271 else if (!strcmp(p, "menu"))
272 return (CT_MENU);
273 else if (!strcmp(p, "form"))
274 return (CT_FORM);
275 else if (!strcmp(p, "pick"))
276 return (CT_PICK);
277 else if (!strcmp(p, "return"))
278 return (CT_RETURN);
279 else if (!strcmp(p, "view"))
280 return (CT_VIEW);
281 else if (!strcmp(p, "?"))
282 return (CT_HELP);
283 else if (!strcmp(p, "write_config"))
284 return (CT_WRITE_CONFIG);
285 return (CT_UNDEFINED);
286}
