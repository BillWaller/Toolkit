207/** @brief Create a new UI surface with the specified parameters.
208 * @param w The index of the surface to create.
209 * @param parent Pointer to the parent UiSurface, or NULL for no parent.
210 * @param p The index of the parent plane to attach to.
211 * @param lines The number of lines (rows) for the new surface.
212 * @param cols The number of columns for the new surface.
213 * @param y The y-coordinate for the new surface's position.
214 * @param x The x-coordinate for the new surface's position.
215 * @return Pointer to the newly created UiSurface, or NULL on failure.
216 *
217 * This function creates a new UI surface and initializes its properties based
218 * on the provided parameters. It allocates memory for the surface, sets up its
219 * metadata, and creates a new ncplane for rendering. If any step fails, it
220 * cleans up and returns NULL.
221 */
222UiSurface *ui_surface_new(ss_t w, UiSurface *parent, uint p, uint lines, uint cols, uint y, uint x) {
223 if (!ui)
224 return NULL;
225 UiSurface *s = calloc(1, sizeof(*s));
226 if (!s) {
227 ui_log(ERROR, "Failed to allocate memory for UiSurface\n");
228 return NULL;
229 }
230 ui_log(INFO, _("Memory allocated for new surface"));
231 s->runtime = ui;
232 s->parent = parent;
233 ncplane_options plane_opts = {
234 .y = y,
235 .x = x,
236 .rows = lines,
237 .cols = cols};

239 s->meta[w].x = x;
240 s->meta[w].lines = lines;
241 s->meta[w].cols = cols;
242 s->meta[w].hidden = false;
243 if (parent && parent->mplane[p]) {
244 s->mplane[w] = ncplane_create(parent->mplane[p], &plane_opts);
245 if (!s->mplane[w]) {

247 }
248 } else {
249 stdplane = notcurses_stdplane(ui->nc);

251 if (!s->mplane[w]) {

255 ui_log(ERROR, _("mplane[%d] created"), w);
256 if (!s->mplane[w]) {
257 notcurses_stop(ui->nc);
258 f_notcurses_open = false;
259 return NULL;
260 }
261 return s;
262}

771/** @brief Scroll the contents of the specified window up or down by the specified
772 * number of lines.
773 * @param s The surface containing the window to scroll.
774 * @param w The index of the window to scroll.
775 * @param r The number of lines to scroll. Positive values scroll up, negative
776 * values scroll down.
777 * @return 0 on success, -1 on error.
778 *
779 * If r is positive, the contents of the plane are moved up, leaving blank lines
780 * at the bottom. In the view application, the view scope is moved toward end of
781 * file (eof). - scope_toward_eof()
782 *
783 * @note ui_wscrl(-n) - If r is negative, the contents of the plane are moved down,
784 * leaving blank lines at the top. In the view application, the view scope is
785 * moved toward beginning of file (bof). - scope_toward_bof()
786 */
787int ui_wscrl(UiSurface *s, ss_t w, int r) {
788 uint rows, cols;
789 ncplane_dim_yx(s->mplane[w], &rows, &cols);
790
791 if (r > 0) {
792 // Scroll up (toward EOF): notcurses has this natively
793 ncplane_scrollup(s->mplane[w], r);
794 return 0;
795 }
796
797 // Scroll down (toward BOF): shift content down by |r| rows
798 r = -r;
799 if ((uint)r >= rows) {
800 ncplane_erase(s->mplane[w]);
801 return 0;
802 }
803
804 // Copy rows bottom-up to avoid overwriting source before reading
805 for (int src_row = (int)(rows - r) - 1; src_row >= 0; src_row--) {
806 int dst_row = src_row + r;
807 for (uint col = 0; col < cols; col++) {
808 nccell c = NCCELL_TRIVIAL_INITIALIZER;
809 ncplane_at_yx_cell(s->mplane[w], src_row, col, &c);
810 ncplane_putc_yx(s->mplane[w], dst_row, col, &c);
811 nccell_release(s->mplane[w], &c);
812 }
813 }
814 // Blank the top r rows
815 for (int row = 0; row < r; row++) {
816 for (uint col = 0; col < cols; col++) {
817 ncplane_putchar_yx(s->mplane[w], row, col, ' ');
818 }
819 }
820 return 0;
821}

1225/** @brief Create a UiCell from a Unicode codepoint and specified foreground and background colors.
1226 * @param ucp Pointer to a wide character string representing the Unicode codepoint.
1227 * @param fg Pointer to a 32-bit unsigned integer representing the foreground color.
1228 * @param bg Pointer to a 32-bit unsigned integer representing the background color.
1229 * @return A UiCell structure representing the created cell.
1230 *
1231 * This function creates a UiCell from a Unicode codepoint and specified foreground
1232 * and background colors. It initializes the cell, sets its width based on the
1233 * Unicode codepoint, converts the codepoint to UTF-8, and sets the styles and
1234 * channels for the cell. The resulting UiCell can be used for rendering text
1235 * in a UI surface.
1236 */
1237UiCell ui_cell_from_ucp(const wchar_t *ucp, const uint32_t *fg, const uint32_t *bg) {
1238 nccell cell;
1239 nccell_init(&cell);
1240 GCluster *gc = (GCluster *)&cell.gcluster;
1241 wchar_t wstr[2] = {ucp[0], L'\0'};
1242 // wcwidth expects Unicode codepoints, not UTF-8 encoded bytes
1243 cell.width = wcwidth(wstr[0]);
1244 // unicode_to_utf8_gcluster converts Unicode codepoints to UTF-8
