773/** @brief Scroll the contents of the specified window up or down by the specified
774 * number of lines.
775 * @param s The surface containing the window to scroll.
776 * @param w The index of the window to scroll.
777 * @param r The number of lines to scroll. Positive values scroll up, negative
778 * values scroll down.
779 * @return 0 on success, -1 on error.
780 *
781 * If r is positive, the contents of the plane are moved up, leaving blank lines
782 * at the bottom. In the view application, the view scope is moved toward end of
783 * file (eof). - scope_toward_eof()
784 *
785 * @note ui_wscrl(-n) - If r is negative, the contents of the plane are moved down,
786 * leaving blank lines at the top. In the view application, the view scope is
787 * moved toward beginning of file (bof). - scope_toward_bof()
788 */
789int ui_wscrl(UiSurface *s, ss_t w, int r) {
790 uint rows, cols;
791 ncplane_dim_yx(s->mplane[w], &rows, &cols);
792
793 if (r > 0) {
794 // Scroll up (toward EOF): notcurses has this natively
795 ncplane_scrollup(s->mplane[w], r);
796 return 0;
797 }
798
799 // Scroll down (toward BOF): shift content down by |r| rows
800 r = -r;
801 if ((uint)r >= rows) {
802 ncplane_erase(s->mplane[w]);
803 return 0;
804 }
805
806 // Copy rows bottom-up to avoid overwriting source before reading
807 for (int src_row = (int)(rows - r) - 1; src_row >= 0; src_row--) {
808 int dst_row = src_row + r;
809 for (uint col = 0; col < cols; col++) {
810 nccell c = NCCELL_TRIVIAL_INITIALIZER;
811 ncplane_at_yx_cell(s->mplane[w], src_row, col, &c);
812 ncplane_putc_yx(s->mplane[w], dst_row, col, &c);
813 nccell_release(s->mplane[w], &c);
814 }
815 }
816 // Blank the top r rows
817 for (int row = 0; row < r; row++) {
818 for (uint col = 0; col < cols; col++) {
819 ncplane_putchar_yx(s->mplane[w], row, col, ' ');
820 }
821 }
822 return 0;
823}
