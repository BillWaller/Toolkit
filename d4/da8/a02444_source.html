30/** @brief The main loop of the menu system.
31 @ingroup menu_engine
32 @param init A pointer to an Init structure containing initialization data for
33 the menu system.
34 @returns an integer indicating the action taken by the user, such as
35 returning to the main menu or exiting the menu system.
36 @details displays the menu and processes user input until the user exits the
37 menu or returns to the main menu.
38 */
39unsigned int menu_engine(Init *init) {
40 int action;
41 int i;
42 char tmp_str[MAXLEN];
43 Menu *menu = init->menu;
44 if (menu == nullptr) {
45 Perror("menu_engine: menu is nullptr");
46 return (1);
47 }
48 menu->lines = 0;
49 menu->cols = 0;
50 menu->line_idx = 0;
51 menu->item_count = 0;
52 menu->title[0] = '\0';
53 menu->choice_max_len = 0;
54 menu->text_max_len = 0;
55 parse_menu_description(init);
56 box_win_new(menu->lines, menu->cols, menu->begy, menu->begx, menu->title);
57 UiSurface *sfc = ui_surface[sfc_ptr];
58 if (sfc == nullptr) {
59 ssnprintf(tmp_str, MAXLEN - 1, "box_win_new(%d, %d, %d, %d, %s) failed",
60 menu->lines, menu->cols, menu->begy, menu->begx, menu->title);
61 Perror(tmp_str);
62 return 1;
63 }
64 mbstate_t mbstate;
65 memset(&mbstate, 0, sizeof(mbstate));
66
67 action = MA_DISPLAY_MENU;
68 while (action) {
69 switch (action) {
70 case MA_RETURN:
71 cm_surface_destroy(sfc);
72 if (sfc_ptr < 0) {
73 destroy_menu(init);
74 return 0;
75 }
76 restore_wins();
77 return 0;
78 case MA_DISPLAY_MENU:
79 for (menu->line_idx = 0; menu->line_idx < menu->item_count;
80 menu->line_idx++) {
81 ui_bkgrndset_cch(sfc, WIN, &CC_NT);
82 ui_draw_text_fill(sfc, WIN, menu->line_idx, 0, NULL,
83 menu->line[menu->line_idx]->choice_text,
84 menu->cols);
85 ui_bkgrndset_cch(sfc, WIN, &CC_NT_HL);
86 ui_mvwaddch(sfc, WIN, menu->line_idx,
87 menu->line[menu->line_idx]->letter_pos,
88 menu->line[menu->line_idx]->choice_letter);
89 ui_bkgrndset_cch(sfc, WIN, &CC_NT);
90 }
91 action = MA_RESET_MENU;
92 break;
93 case MA_RESET_MENU:
94 menu->line_idx = 0;
95 for (i = 0; i < menu->item_count; i++) {
96 if (menu->line[i]->type == MT_CHOICE) {
97 menu->line_idx = i;
98 break;
99 }
100 }
101 action = MA_CONTINUE;
102 break;
103 case MA_CONTINUE:
104 action = menu_cmd_processor(init);
105 break;
106 default:
107 break;
108 }
109 }
110 destroy_menu(init);
111 return 0;
112}
