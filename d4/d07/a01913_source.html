80/** @brief Convert hex color strings to a curses color pair index.
81 *
82 * @param fg Foreground color in hex format (e.g., "#RRGGBB").
83 * @param bg Background color in hex format (e.g., "#RRGGBB").
84 * @return The curses color pair index corresponding to the given foreground and background colors.
85 */
86int ui_pair_from_hex(const char *fg, const char *bg) {
87 RGB rgb;
88 sscanf(fg, "#%02x%02x%02x", &rgb.r, &rgb.g, &rgb.b);
89 int f_idx = ui_color_from_rgb(&rgb);
90 sscanf(bg, "#%02x%02x%02x", &rgb.r, &rgb.g, &rgb.b);
91 int b_idx = ui_color_from_rgb(&rgb);
92 return ui_add_pair(f_idx, b_idx);
93}
94/** @brief Convert a Unicode code point to a curses UiCell with specified foreground and background colors.
95 *
96 * @param ucp Pointer to a wide character string containing the Unicode code point.
97 * @param fg Pointer to a 32-bit integer representing the foreground color in hex format (e.g., 0xRRGGBB).
98 * @param bg Pointer to a 32-bit integer representing the background color in hex format (e.g., 0xRRGGBB).
99 * @return A UiCell structure representing the Unicode character with the specified colors.
100 */
101UiCell ui_cell_from_ucp(const wchar_t *ucp, const uint32_t *fg, const uint32_t *bg) {
102 UiCell cc = {0};
103 RGB rgb;
104 rgb.r = (*fg >> 16) & 0xff;
105 rgb.g = (*fg >> 8) & 0xff;
106 rgb.b = *fg & 0xff;
107 int f_idx = ui_color_from_rgb(&rgb);
108 rgb.r = (*bg >> 16) & 0xff;
109 rgb.g = (*bg >> 8) & 0xff;
110 rgb.b = *bg & 0xff;
111 int b_idx = ui_color_from_rgb(&rgb);
112 short cp = ui_add_pair(f_idx, b_idx);
113 wchar_t wstr[2] = {L'\0', L'\0'};
114 wstr[0] = *ucp;
115 wstr[1] = L'\0';
116 setcchar(&cc, wstr, WA_NORMAL, cp, nullptr);
117 return cc;
118}

218/** @brief Create a new UI surface with the specified parameters.

432 hide_panel(stdsfc->mpan[0]);

435 }

438 f_ncurses_open = false;
439 if (ui->screen != NULL) {
440 ui_log(INFO, _("calling delscreen(ui->screen)"));

453 }
