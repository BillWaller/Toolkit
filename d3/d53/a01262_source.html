25/** * @brief Create a new UI surface.
26 * @ingroup ui_ncurses
27 * @param ui The UI runtime instance.
28 * @param parent The parent surface, or NULL for a top-level surface.
29 * @param rect The rectangle defining the position and size of the surface.
30 * @return A pointer to the newly created UiSurface, or NULL on failure.
31 */
32UiSurface *ui_surface_new(UiRuntime *ui, UiSurface *parent, UiRect rect) {
33 UiSurface *s = calloc(1, sizeof(*s));
34 if (!s)
35 return NULL;
36
37 s->runtime = ui;
38 s->parent = parent;
39 s->y = rect.y;
40 s->x = rect.x;
41 s->rows = rect.rows;
42 s->cols = rect.cols;
43
44 if (parent && parent->win) {
45 s->win = derwin(parent->win, rect.rows, rect.cols, rect.y, rect.x);
46 } else {
47 s->win = newwin(rect.rows, rect.cols, rect.y, rect.x);
48 }
49
50 if (!s->win) {
51 free(s);
52 return NULL;
53 }
54
55 s->pan = new_panel(s->win);
56 if (!s->pan) {
57 delwin(s->win);
58 free(s);
59 return NULL;
60 }
61
62 return s;
63}

170/** @brief Initialize NCUrses
171 * @ingroup ui_ncurses
172 * @param cfg The UI runtime instance.
173 * @return 0 on success, or -1 on failure.
174 */
175UiRuntime *ui_init(const UiConfig *cfg) {
176 UiRuntime *ui = calloc(1, sizeof(*ui));
177 if (!ui)
178 return NULL;
179
180 initscr();
181 cbreak();
182 noecho();
183 keypad(stdscr, TRUE);
184
185 if (has_colors()) {
186 start_color();
187 use_default_colors();
188 }
189
190 if (cfg) {
191 ui->mouse_enabled = cfg->enable_mouse;
192 ui->alt_screen = cfg->enable_alt_screen;
193 ui->cursor_visible = cfg->cursor_visible;
194 } else {
195 ui->cursor_visible = true;
196 }
197
198 if (ui->mouse_enabled) {
199 mousemask(ALL_MOUSE_EVENTS | REPORT_MOUSE_POSITION, NULL);
200 }
201
202 curs_set(ui->cursor_visible ? 1 : 0);
203 getmaxyx(stdscr, ui->rows, ui->cols);
204
205 return ui;
206}
