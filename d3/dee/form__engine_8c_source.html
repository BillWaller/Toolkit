165 form_exec_receiver(init);

606int form_parse_desc(Form *form) {
607 FILE *form_desc_fp;
608 char *token;
609 char *s;
610 int cols = 0;
611 int i, l;
612 int in_line_num = 0;
613 char in_buf[BUFSIZ];
614 char tmp_buf[BUFSIZ];
615 char *tmp_buf_p;
616 char tmp_str[BUFSIZ];
617 char delim[5];
618 char directive;
619
620 form_desc_fp = fopen(form->mapp_spec, "r");
621 if (form_desc_fp == nullptr) {
622 ssnprintf(em0, MAXLEN - 1, "%s, line: %d", __FILE__, __LINE__ - 2);
623 strnz__cpy(em1, "fopen ", MAXLEN - 1);
624 strnz__cat(em1, form->mapp_spec, MAXLEN - 1);
625 strerror_r(errno, em2, MAXLEN);
626 display_error(em0, em1, em2, nullptr);
627 return (1);
628 }
629 for (i = 0; i < FIELD_MAXCNT; i++) {
630 form->field[i] = calloc(1, sizeof(Field));
631 if (!form->field[i]) {
632 sprintf(tmp_str, "FORM: calloc failed for fields");
633 abend(EXIT_FAILURE, tmp_str);
634 }
635 }
636 for (i = 0; i < FIELD_MAXCNT; i++) {
637 form->text[i] = calloc(1, sizeof(Text));
638 if (!form->text[i]) {
639 sprintf(tmp_str, "FORM: calloc failed for text");
640 abend(EXIT_FAILURE, tmp_str);
641 }
642 }
643 form->didx = 0;
644 form->fidx = 0;
645 form->fcnt = 0;
646 form->cols = 34;
647 while ((fgets(in_buf, MAXLEN, form_desc_fp)) != nullptr) {
648 s = in_buf;
649 in_line_num++;
650 l = trim(in_buf);
651 if (l == 0)
652 continue;
653 if (*s == D_COMMENT)
654 continue;
655 delim[0] = '\n';
656 delim[1] = in_buf[1];
657 delim[2] = '\0';
658 strnz__cpy(tmp_buf, in_buf, MAXLEN - 1);
659 tmp_buf_p = tmp_buf;
660 if (!(token = strtok(tmp_buf_p, delim))) {
661 continue;
662 }
663 directive = *token;
664 switch ((int)directive) {
665 case D_COMMENT:
666 break;
667 case D_CALC:
668 form->f_calculate = true;
669 break;
670 case D_QUERY:
671 form->f_query = true;
672 break;
673 case D_GETTER:
674 form->f_process = true;
675 break;
676 case D_CMD:
677 if (!(token = strtok(nullptr, delim))) {
678 form_desc_error(form, in_line_num, in_buf,
679 "FORM: receiver_cmd delimiter");
680 continue;
681 }
682 strnz__cpy(form->receiver_cmd, token, MAXLEN - 1);
683 break;
684 case D_HELP:
685 if (!(token = strtok(nullptr, delim))) {
686 form_desc_error(form, in_line_num, in_buf,
687 "FORM: help_spec delimiter");
688 }
689 strnz__cpy(form->help_spec, token, MAXLEN - 1);
690 break;
691 case D_FIELD:
692 if (form->field[form->fidx] == nullptr) {
693 sprintf(tmp_str, "FORM: calloc failed for fields");
694 abend(EXIT_FAILURE, tmp_str);
695 }
696 if (!(token = strtok(nullptr, delim))) {
697 form_desc_error(form, in_line_num, in_buf,
698 "FORM: line number delimiter");
699 return 1;
700 }
701 form->field[form->fidx]->line = atoi(token);
702 if (form->field[form->fidx]->line < 0 ||
703 form->field[form->fidx]->line >= FIELD_MAXCNT) {
704 form_desc_error(form, in_line_num, in_buf,
705 "FORM: invalid line number");
706 return 1;
707 }
708 if (!(token = strtok(nullptr, delim))) {
709 form_desc_error(form, in_line_num, in_buf,
710 "FORM: column number delimiter");
711 return 1;
712 }
713 form->field[form->fidx]->col = atoi(token);
714 if (form->field[form->fidx]->col < 0 ||
715 form->field[form->fidx]->col >= FIELD_MAXLEN) {
716 form_desc_error(form, in_line_num, in_buf,
717 "FORM: invalid column number");
718 break;
719 }
720 if (!(token = strtok(nullptr, delim))) {
721 strnz__cpy(tmp_str, in_buf, MAXLEN - 1);
722 form_desc_error(form, in_line_num, tmp_str, "FORM: length delimiter");
723 break;
724 }
725 form->field[form->fidx]->len = atoi(token);
726 if (form->field[form->fidx]->len < 0 ||
727 form->field[form->fidx]->len > FIELD_MAXLEN) {
728 form_desc_error(form, in_line_num, in_buf, "FORM: invalid length");
729 break;
730 }
731 if (!(token = strtok(nullptr, delim))) {
732 form_desc_error(form, in_line_num, in_buf,
733 "FORM: validation code delimiter");
734 break;
735 }
736 form->field[form->fidx]->ff = -1;
737 for (i = 0; i < FF_INVALID; i++) {
738 str_to_lower(token);
739 str_to_lower(ff_tbl[i]);
740 if (!strcmp(token, ff_tbl[i])) {
741 form->field[form->fidx]->ff = i;
742 break;
743 }
744 }
745 if (form->field[form->fidx]->ff < 0 ||
746 form->field[form->fidx]->ff >= FF_INVALID) {
747 form_desc_error(form, in_line_num, in_buf,
748 "FORM: invalid format code");
749 break;
750 }
751 cols =
752 form->field[form->fidx]->col + form->field[form->fidx]->len + 1;
753 if (cols > form->cols)
754 form->cols = cols;
755 form->fidx++;
756 form->fcnt = form->fidx;
757 break;
758 case D_TEXT:
759 if (form->text[form->didx] == nullptr) {
760 sprintf(tmp_str, "FORM: calloc failed for text");
761 abend(EXIT_FAILURE, tmp_str);
