50/** @brief Accept input for a field
51 @ingroup fields
52 @param form Pointer to Form structure
53 @return int Key code of action taken
54 @note Handles character input, navigation keys, and mouse events for field
55 editing. Validates input based on field format and updates display.
56 Returns key code of action taken (e.g., accept field, tab to next field,
57 break out of input loop).
58 */
59int field_editor(Form *form) {
60 bool f_insert = FALSE;
61 int in_key;
62 char *s, *d;
63 WINDOW *win = form->win;
64
65 int flin = form->field[form->fidx]->line;
66 int fcol = form->field[form->fidx]->col;
67 int flen = form->field[form->fidx]->len;
68 int ff = form->field[form->fidx]->ff;
69 char *accept_s = form->field[form->fidx]->accept_s;
70 char *filler_s = form->field[form->fidx]->filler_s;
71 form_fmt_field(form, accept_s);
72 click_x = click_y = -1;
73 char *fstart = accept_s;
74 char *fend = fstart + flen;
75 int x = fcol;
76 char *p = fstart = accept_s;
77 char *str_end = p + strlen(p);
78 in_key = 0;
79 if (f_insert)
80 set_chyron_key_cp(form->chyron, 18, "INS", KEY_IC,
81 cp_reverse_highlight);
82 else
83 set_chyron_key_cp(form->chyron, 18, "INS", KEY_IC, cp_reverse);
84 compile_chyron(form->chyron);
85 display_chyron(form->win, form->chyron, form->lines - 1, form->chyron->l);
86
87 while (1) {
88 if (in_key == 0) {
89 mvwaddstr(win, flin, fcol, filler_s);
90 mvwaddstr(win, flin, fcol, accept_s);
91 wmove(win, flin, x);
92 tcflush(0, TCIFLUSH);
93 wmove(win, flin, x);
94 in_key = xwgetch(win, form->chyron, -1);
95 }
96 switch (in_key) {
97 /** KEY_F(10) is the default key for accepting the field and moving
98 to the next field */
99 case KEY_F(10):
100 form_fmt_field(form, accept_s);
101 form_display_field(form);
102 if (form_validate_field(form) != 0)
103 continue;
104 return (in_key);
105 /** KEY_F(9) Cancels the current operation */
106 case KEY_BREAK:
107 case KEY_F(9):
108 form_display_field(form);
109 in_key = KEY_F(9);
110 return (in_key);
111 /** KEY_UP, KEY_BTAB moves to the previous field */
112 case 'H':
113 case KEY_F(1):
114 return (in_key);
115 case KEY_BTAB:
116 case KEY_UP:
117 form_fmt_field(form, accept_s);
118 form_display_field(form);
119 in_key = KEY_UP;
120 return (in_key);
121 /** KEY_DOWN, TAB moves to the next field */
122 case '\t':
123 case KEY_DOWN:
124 form_fmt_field(form, accept_s);
125 form_display_field(form);
126 in_key = KEY_DOWN;
127 return (in_key);
128 /** KEY_END moves cursor to end of field */
129 case KEY_END:
130 case Ctrl('e'):
131 while (*p != '\0')
132 p++;
133 x = fcol + (p - fstart);
134 in_key = 0;
135 continue;
136 /**< Enter accepts the field and moves to the next field if
137 * form->f_erase_remainder is set, it will clear the remaining
138 * characters above and after the current cursor location */
139 case '\n':
140 case KEY_ENTER:
141 if (form->f_erase_remainder)
142 *p = '\0';
143 form_fmt_field(form, accept_s);
144 form_display_field(form);
145 in_key = KEY_ENTER;
146 return (in_key);
147 /** KEY_IC toggles insert mode */
148 case KEY_IC:
149 if (f_insert) {
150 f_insert = FALSE;
151 set_chyron_key_cp(form->chyron, 18, "INS", KEY_IC, cp_reverse);
152 } else {
153 f_insert = TRUE;
154 set_chyron_key_cp(form->chyron, 18, "INS", KEY_IC,
155 cp_reverse_highlight);
156 }
157 compile_chyron(form->chyron);
158 display_chyron(form->win, form->chyron, form->lines - 1,
159 form->chyron->l);
160 in_key = 0;
161 continue;
162 /** KEY_DC deletes character at cursor */
163 case KEY_DC:
164 s = p + 1;
165 d = p;
166 while (*s != '\0')
167 *d++ = *s++;
168 *d = '\0';
169 str_end = d;
170 f_insert = FALSE;
171 in_key = 0;
172 continue;
173 /** KEY_HOME moves cursor to start of field */
174 case KEY_HOME:
175 case Ctrl('a'):
176 p = fstart;
177 x = fcol;
178 in_key = 0;
179 continue;
180 /** KEY_BACKSPACE deletes character before cursor */
181 case KEY_BACKSPACE:
182 if (p > fstart) {
183 p--;
184 x--;
185 }
186 s = p + 1;
187 d = p;
188 while (*s != '\0')
189 *d++ = *s++;
190 *d = '\0';
191 str_end = d;
192 in_key = 0;
193 continue;
194 /** KEY_LEFT moves cursor left one character */
195 case KEY_LEFT:
196 if (p > fstart) {
197 p--;
198 x--;
199 }
200 in_key = 0;
201 continue;
202 /** KEY_RIGHT moves cursor right one character */
203 case KEY_RIGHT:
204 if (p < fend && p < str_end) {
205 p++;
206 x++;
207 }
208 in_key = 0;
209 continue;
210 /** Handles mouse events for field editing */
211 case KEY_MOUSE:
212 form->fidx = form_yx_to_fidx(form, click_y, click_x);
213 x = click_x;
214 flin = click_y;
215 flen = form->field[form->fidx]->len;
216 ff = form->field[form->fidx]->ff; /* ff - field forat */
217 accept_s = form->field[form->fidx]->accept_s;
218 filler_s = form->field[form->fidx]->filler_s;
219 form_fmt_field(form, accept_s);
220 fstart = accept_s;
221 fend = fstart + flen;
222 str_end = fstart + strlen(fstart);
223 p = fstart + (x - form->field[form->fidx]->col);
224 if (p > str_end)
225 x -= p - str_end;
226 p = min(p, str_end);
227 in_key = 0;
228 continue;
229 default:
230 if (p >= fend) {
231 in_key = 0;
232 continue;
233 }
234 /** Validates fields based on format */
235 switch (ff) {
236 /** FF_STRING accepts all printable characters and spaces */
237 case FF_STRING:
238 break;
239 /** FF_DECIMAL_INT accepts digits 0 through 9 and decimal point
240 * ('.') */
241 case FF_DECIMAL_INT:
242 if ((in_key >= '0' && in_key <= '9') || in_key == '.')
243 break;
244 beep();
245 in_key = 0;
246 continue;
247 /** FF_HEX_INT accepts digits 0 through 9 and letters A through
248 * F (case-insensitive) */
249 case FF_HEX_INT:
250 if ((in_key >= '0' && in_key <= '9') ||
251 (in_key >= 'A' && in_key <= 'F') ||
252 (in_key >= 'a' && in_key <= 'f'))
253 break;
254 beep();
255 in_key = 0;
256 continue;
257 /** FF_FLOAT accepts digits 0 through 9 and decimal point ('.')
258 * and negative operator ('-') at the start of the field */
259 case FF_FLOAT:
260 if ((in_key >= '0' && in_key <= '9') || in_key == '.' ||
261 (in_key == '-' && p == fstart))
262 break;
263 beep();
264 in_key = 0;
265 continue;
266 /** FF_DOUBLE accepts digits 0 through 9 and decimal point ('.')
267 * and negative operator ('-') at the start of the field */
268 case FF_DOUBLE:
269 if ((in_key >= '0' && in_key <= '9') || in_key == '.' ||
270 (in_key == '-' && p == fstart))
271 break;
272 beep();
273 in_key = 0;
274 continue;
275 /** FF_CURRENCY accepts digits 0 through 9 and decimal point
276 * ('.') and negative operator ('-') at the start of the field
277 */
278 case FF_CURRENCY:
279 if ((in_key >= '0' && in_key <= '9') || in_key == '.' ||
280 (in_key == '-' && p == fstart))
281 break;
282 beep();
283 in_key = 0;
284 continue;
285 /** FF_YYYYMMDD accepts digits 0 through 9 */
286 case FF_YYYYMMDD:
287 if (in_key >= '0' && in_key <= '9')
288 break;
289 beep();
290 in_key = 0;
291 continue;
292 /** FF_HHMMSS accepts digits 0 through 9 */
293 case FF_HHMMSS:
294 if (in_key >= '0' && in_key <= '9')
295 break;
296 beep();
297 in_key = 0;
298 continue;
299 /** FF_APR accepts digits 0 through 9 and decimal point ('.') */
300 case FF_APR:
301 if ((in_key >= '0' && in_key <= '9') || in_key == '.')
302 break;
303 beep();
304 in_key = 0;
305 continue;
306 default:
307 Perror("field_editor() invalid format");
308 break;
309 }
310 if (in_key >= ' ') {
311 if (f_insert) {
312 if (str_end < fend) {
313 s = str_end - 1;
314 d = str_end;
315 while (s >= p)
316 *d-- = *s--;
317 *p++ = in_key;
318 str_end++;
319 x++;
320 }
321 } else {
322 if (p < fend) {
323 if (p < str_end) {
324 *p++ = in_key;
325 x++;
326 } else if (p == str_end) {
327 *p++ = in_key;
328 *p = '\0';
329 str_end = p;
330 x++;
331 }
332 }
333 }
334 in_key = 0;
335 continue;
336 }
337 }
338 }
339}

379/** @brief Display brackets around current field if set
380 @ingroup fields
381 @param form Pointer to Form structure
382 @return 0 on success, non-zero on error
383 @note This function checks if the form's brackets array has non-empty values
384 for both the left and right brackets. If so, it retrieves the line and column
385 information for the current field and uses the form's box window to display
386 the left bracket at the start of the field and the right bracket at the end

389int form_display_field_brackets(Form *form) {

392 WINDOW *box = form->box;

403/** @brief Format field according to its format type
404@ingroup fields
405 @param form Pointer to Form structure
406 @param s Input string to format
407 @return 0 on success, non-zero on error
408 @note takes the input string for the current field and formats it
409 according to the field's specified format type (ff). It updates the
410 accept_s and display_s strings for the field based on the formatted
411 value.
412 @note handles various format types, including strings, decimal integers,
413 hexadecimal integers, floating-point numbers, currency, dates, and times.
414 @note uses helper functions for validation and formatting, such as
415 is_valid_date(), is_valid_time(), numeric(), right_justify(),
416 left_justify(), and strnzcpy(). The function ensures that the formatted
417 output fits within the field's length and creates a filler string for the
418 field as needed. The function also handles error cases, such as invalid
419 formats, and provides feedback through error messages. It is designed to
420 be extensible, allowing for additional format types to be added in the
421 future as needed.
422 @note assumes that the input string is well-formed and does not contain
423 malicious content. Input validation and sanitization should be performed
424 at a higher level in the application to ensure security and robustness.
425 @note The function currently does not handle localization or
426 internationalization of number formats, such as different decimal
427 separators or currency symbols. Future enhancements may include support
428 for locale-specific formatting. @note Error handling is basic; future
429 versions may include more detailed error reporting and logging
430 mechanisms. @note Performance optimizations may be considered for
431 handling large volumes of data or high-frequency updates in real-time
432 applications. @note The following variables and structures are used in
433 this function:
434 @code
435 char field_s[FIELD_MAXLEN];
436 int decimal_int_n = 0;
437 int hex_int_n = 0;
438 float float_n = 0.0;
439 double double_n = 0.0;
440 double currency_n = 0.0;
441 struct Date { int yyyy; int mm; int dd; };
442 struct Time { int hh; int mm; int ss; };
443 @endcode
444 @note supports format types: FF_STRING, FF_DECIMAL_INT, FF_HEX_INT,
445 FF_FLOAT, FF_DOUBLE, FF_CURRENCY, FF_YYYYMMDD, FF_HHMMSS, FF_APR with
446 appropriate formatting and validation for each type. @note Handles field
447 length and filler string creation. @note As the roadmap indicates, these
448 data types are just a start and more complex types and formats may be
449 added in the future. @note These data types are not suitable for
450 financial, scientific, or high-precision applications. They are intended
451 for basic data entry and display in a text-based user interface.
452 @note In the future, we will aqdd support for additional data types and
453 formats such as: dates with time zones, timestamps, percentages,
454 scientific notation, and custom user-defined formats.
455 @note For high precision applications, we will be integrating support
456 for 128-bit binary coded decimal (BCD) types and arbitrary precision
457 decimal types using libraries such as the GNU MPFR library, IBM's
458 decNumber, Rust Decimal, or the Intel Decimal Floating-Point Math
459 Library.
460 @note C-Menu Form converts each field's input string into internal
461 numeric binary based on the field's specified format. The internal
462 numberic binary is then formated and displayed, verifying the user's
463 input and Form's interpretation of the user's input.
464 */
465int form_fmt_field(Form *form, char *s) {
466 strnz__cpy(form->field[form->fidx]->input_s, s, FIELD_MAXLEN - 1);
467 char *input_s = form->field[form->fidx]->input_s;
468 char *accept_s = form->field[form->fidx]->accept_s;
469 char *display_s = form->field[form->fidx]->display_s;
470 char *filler_s = form->field[form->fidx]->filler_s;
471 int ff = form->field[form->fidx]->ff;
472 int fl = form->field[form->fidx]->len;
473
474 char field_s[FIELD_MAXLEN];
475 int decimal_int_n = 0;
476 int hex_int_n = 0;
477 float float_n = 0.0;
478 double double_n = 0.0;
479 double currency_n = 0.0;
480
481 struct {
482 int yyyy;
483 int mm;
484 int dd;
485 } Date;
486 Date.yyyy = Date.mm = Date.dd = 0;
487 struct {
488 int hh;
489 int mm;
490 int ss;
491 } Time;
492 Time.hh = Time.mm = Time.ss = 0;
493 strnz(accept_s, fl);
494 switch (ff) {
495 case FF_STRING:
496 left_justify(s);
497 trim(s);
498 strnz__cpy(input_s, s, FIELD_MAXLEN - 1);
499 strnz__cpy(accept_s, s, FIELD_MAXLEN - 1);
500 strnz__cpy(display_s, s, FIELD_MAXLEN - 1);
501 break;
502 case FF_DECIMAL_INT:
503 sscanf(input_s, "%d", &decimal_int_n);
504 sprintf(accept_s, "%d", decimal_int_n);
505 sprintf(display_s, "%d", decimal_int_n);
506 right_justify(display_s, fl);
507 break;
508 case FF_HEX_INT:
509 sscanf(input_s, "%x", &hex_int_n);
510 sprintf(accept_s, "%d", hex_int_n);
511 sprintf(display_s, "%x", hex_int_n);
512 right_justify(display_s, fl);
513 break;
514 case FF_FLOAT:
515 sscanf(input_s, "%f", &float_n);
516 sprintf(accept_s, "%f", float_n);
517 sprintf(display_s, "%f", float_n);
518 right_justify(display_s, fl);
519 break;
520 case FF_DOUBLE:
521 sscanf(input_s, "%lf", &double_n);
522 sprintf(accept_s, "%lf", double_n);
523 sprintf(display_s, "%lf", double_n);
524 right_justify(display_s, fl);
525 break;
526 case FF_CURRENCY:
527 numeric(field_s, input_s);
528 sscanf(field_s, "%lf", &currency_n);
529 sprintf(accept_s, "%.2lf", currency_n);
530 sprintf(display_s, "%'.2lf", currency_n);
531 right_justify(display_s, fl);
532 break;
533 case FF_YYYYMMDD:
534 Date.yyyy = Date.mm = Date.dd = 0;
535 strnz__cpy(field_s, input_s, FIELD_MAXLEN - 1);
536 sscanf(field_s, "%4d%2d%2d", &Date.yyyy, &Date.mm, &Date.dd);
537 sprintf(accept_s, "%04d%02d%02d", Date.yyyy, Date.mm, Date.dd);
538 if (is_valid_date(Date.yyyy, Date.mm, Date.dd))
539 sprintf(display_s, "%04d-%02d-%02d", Date.yyyy, Date.mm, Date.dd);
540 break;
541 case FF_HHMMSS:
542 Time.hh = Time.mm = Time.ss = 0;
543 strnz__cpy(field_s, input_s, FIELD_MAXLEN - 1);
544 sscanf(field_s, "%2d%2d%2d", &Time.hh, &Time.mm, &Time.ss);
545 sprintf(accept_s, "%02d%02d%02d", Time.hh, Time.mm, Time.ss);
546 if (is_valid_time(Time.hh, Time.mm, Time.ss))
547 sprintf(display_s, "%02d:%02d:%02d", Time.hh, Time.mm, Time.ss);
548 break;
549 case FF_APR:
550 sscanf(input_s, "%lf", &double_n);
551 sprintf(accept_s, "%lf", double_n);
552 sprintf(display_s, "%0.3lf", double_n);
553 right_justify(display_s, fl);
554 break;
555 default:
556 Perror("form_fmt_field() invalid format");
557 break;
558 }
559 strnz(accept_s, fl);
560 left_justify(accept_s);
561 mk_filler(filler_s, fl);
562 return 0;
563}
564/** @brief Validate current field based on flags
565 @ingroup fields
566 @param form Pointer to Form structure
567 @return 0 if valid, 1 if invalid
568 @note Very underdeveloped - only checks F_NOTBLANK and F_NOMETAS
569 */
570int form_validate_field(Form *form) {
571 int n = form->fidx;
572 char *p = form->field[n]->accept_s;
573 if (form->field[n]->ff & F_NOTBLANK) {
574 char *s = form->field[n]->accept_s;
575 while (*s++ == ' ')
576 ;
577 if (*s == '\0') {
578 Perror("blank field not allowed");
579 return (1);
580 }
581 }
582 if (form->field[n]->ff & F_NOMETAS) {
583 if (strpbrk(p, "*?[]") != 0) {
584 Perror("metacharacters not allowed");
585 return (1);
586 }
587 }
588 return (0);
589}

608/** @brief Left justify string by removing leading spaces
609 @ingroup fields
610 @param s String to left justify
611 @note This function takes a string s and removes any leading spaces,
612 effectively left-justifying the text. It does this by finding the first
613 non-space character and shifting the string to the left, overwriting the
614 leading spaces. The resulting string is null-terminated.
615 */
616void left_justify(char *s) { trim(s); }
617/** @brief Right justify string by removing trailing spaces and adding leading
618 @ingroup fields
619 spaces
620 @param s String to right justify
621 @param fl Field length
622 @note This function takes a string s and right-justifies it within a field
623 of length fl. It first removes any trailing spaces from the string, then
624 shifts the characters to the right end of the field, filling the left side
625 with spaces. The resulting string is null-terminated and fits within the
626 specified field length.
627 */
628void right_justify(char *s, int fl) {
629 char *p = s;
630 char *d = s + fl;
631 trim(s);
632 *d = '\0';
633 while (*s != '\0') {
634 s++;
635 }
636 while (s != p) {
637 *(--d) = *(--s);
638 }
639 while (d != p) {
640 *(--d) = ' ';
641 }
642}
643/** @brief Check if a given date is valid, including leap years
644 @ingroup fields
645 @param yyyy Year
646 @param mm Month
647 @param dd Day
648 @return true if the date is valid, false otherwise
649 @note This function checks if the provided year, month, and day constitute a
650 valid date. It accounts for leap years when determining the number of days in
651 February. The function returns true if the date is valid and false if it is
652 not.
653 */
654bool is_valid_date(int yyyy, int mm, int dd) {
655 if (yyyy < 1 || mm < 1 || mm > 12 || dd < 1)
656 return false;
657 int days_in_month[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
658 if ((yyyy % 4 == 0 && yyyy % 100 != 0) || (yyyy % 400 == 0))
659 days_in_month[2] = 29;
660 if (dd > days_in_month[mm])
661 return false;
