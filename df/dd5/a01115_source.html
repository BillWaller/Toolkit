119static error_t parse_opt(int key, char *arg, struct argp_state *state) {
120 SearchFilters *f = state->input;
121 int i = 0;
122 bool a_toi_error = false;
123
124 switch (key) {
125 case 'a':
126 memset(&tm_info, 0, sizeof(struct tm));
127 if (strptime(arg, "%Y-%m-%dT%H:%M:%S", &tm_info) == NULL) {
128 fprintf(stderr, "-a Failed to parse time string.\n");
129 return 1;
130 }
131 f->after = mktime(&tm_info);
132 if (f->after && f->before && f->after > f->before) {
133 fprintf(stderr, "-a time must be before -b time.\n");
134 return 1;
135 }
136 break;
137 case 'b':
138 memset(&tm_info, 0, sizeof(struct tm));
139 if (strptime(arg, "%Y-%m-%dT%H:%M:%S", &tm_info) == NULL) {
140 fprintf(stderr, "-b Failed to parse time string.\n");
141 return 1;
142 }
143 f->before = mktime(&tm_info);
144 if (f->after && f->before && f->after > f->before) {
145 fprintf(stderr, "-a time must be before -b time.\n");
146 return 1;
147 }
148 break;
149 case 'd':
150 if (arg && arg[0] != '\0')
151 f->max_depth = a_toi(arg, &a_toi_error);
152 break;
153 case 'D':
154 if (arg && arg[0] != '\0')
155 f->debug = a_toi(arg, &a_toi_error);
156 else
157 f->debug = 1;
158 break;
159 case 'e':
160 strnz__cpy(f->ere, arg, MAXLEN - 1);
161 f->flags |= LF_EXC_REGEX;
162 break;
163 case 'H':
164 f->f_hidden = true; // Include hidden files
165 f->flags &= ~(LF_HIDE); // Don't hide hidden files
166 break;
167 case 'L':
168 f->f_lnk_dir = true; // follow symbolic links
169 break;
170 case 'i':
171 f->flags |= LF_ICASE;
172 break;
173 case 'p':
174 for (i = 0; arg[i]; i++) {
175 switch (arg[i]) {
176 case 'g':
177 f->flags |= LF_SETGID << 16;
178 break;
179 case 'r':
180 f->flags |= LF_PERM_R << 16;
181 break;
182 case 's':
183 f->flags |= LF_SETUID << 16;
184 break;
185 case 'w':
186 f->flags |= LF_PERM_W << 16;
187 break;
188 case 'x':
189 f->flags |= LF_PERM_X << 16;
190 break;
191 default:
192 break;
193 }
194 }
195 break;
196 case 's':
197 long long ull = a_to_ull(arg);
198 f->file_size_min = (intmax_t)ull;
199 break;
200 case 't':
201 f->file_types_p = arg;
202 while (f->file_types_p[i]) {
203 switch (f->file_types_p[i++]) {
204 case 'b':
205 f->flags |= FT_BLK << 8;
206 break;
207 case 'c':
208 f->flags |= FT_CHR << 8;
209 break;
210 case 'd':
211 f->flags |= FT_DIR << 8;
212 break;
213 case 'p':
214 f->flags |= FT_FIFO << 8;
215 break;
216 case 'l':
217 f->flags |= FT_LNK << 8;
218 break;
219 case 'f': // for regular files, 'f' is more intuitive than 'r'
220 case 'r': // regular files are the most common type, so 'r' is also
221 // accepted
222 f->flags |= FT_REG << 8;
223 break;
224 case 's':
225 f->flags |= FT_SOCK << 8;
226 break;
227 case 'u':
228 f->flags |= FT_UNKNOWN << 8;
229 break;
230 default:
231 break;
232 }
233 }
234 break;
235 case 'u':
236 struct passwd *pwd = getpwnam(arg);
237 if (pwd) {
238 if (pwd->pw_uid > 0xffff) {
239 fprintf(stderr,
240 "User '%s' has UID %d which is too large for this "
241 "program.\n",
242 arg, pwd->pw_uid);
243 exit(EXIT_FAILURE);
244 }
245 f->flags |= (long)(pwd->pw_uid) << 24;
246 f->flags |= LF_USER;
247 } else {
248 fprintf(stderr, "User '%s' not found.\n", arg);
249 exit(EXIT_FAILURE);
250 }
251 break;
252 case 'x':
253 strnz__cpy(f->exec, arg, MAXLEN - 1);
254 break;
255 case 'S':
256 f->f_sort = true;
257 if (arg && (arg[0] == 'r' || arg[0] == 'R'))
258 f->f_reverse = true;
259 break;
260 case ARGP_KEY_ARG:
261 if (state->arg_num == 0 || state->arg_num == 1) {
262 f->args[state->arg_num] = arg;
263 f->argc = state->arg_num + 1;
264 } else {
265 argp_usage(state);
266 }
267 break;
268 case ARGP_KEY_END:
269 break;
270 default:
271 return ARGP_ERR_UNKNOWN;
272 }
273 return 0;

338 * to the sort command. We can achieve this by creating a child process
339 * that runs the sort command, and redirecting the output of the file
340 * finder to the input of the sort command using a pipe. The parent
341 * process will run the file finder and write its output to the pipe,
342 * while the child process will read from the pipe and execute the sort

357 dup2(fds[0], STDIN_FILENO); // Clone child read pipe to STDIN_FILENO
358 close(fds[0]); // Close child read pipe after cloning
359 execvp(eargv[0], eargv); // Execute the external command
360 }
361 // Parent process
362 close(fds[0]); // Close read end of pipe
363 dup2(fds[1], STDOUT_FILENO); // Clone write pipe to STDOUT_FILENO
364 close(fds[1]); // Close duplicated write pipe
365 setvbuf(stdout, NULL, _IOLBF, 0);
366 // line buffering is essential to ensure that output is flushed to the
367 // sort process in a timely manner, preventing deadlocks and ensuring
368 // that the sort process can start processing input as soon as it is
369 // available. Without line buffering, the output may be buffered until
370 // the buffer is full, which could lead to delays in processing and
371 // potential deadlocks if the sort process is waiting for input that
372 // hasn't been flushed yet.
373 init_find(f); // Initialize and transfer control to the file finder
374 dup2(save_fd, STDOUT_FILENO); // restore STDOUT
375 waitpid(pid1, &wstatus, 0);
376 } else

378 exit(EXIT_SUCCESS);

380/** @brief Initialize the file search based on the provided SearchFilters and
381 start finder threads.
382 @param f A pointer to a SearchFilters struct containing the options and
383 flags for filtering.
384 @return true if the search was initialized successfully, false if an error
385 occurred during initialization (e.g., regex compilation failure).
386 @details This function processes the flags and options specified in the
387 SearchFilters struct to set up the search criteria. It compiles any regular
388 expressions provided by the user and initializes the queue with the base
389 directory. It then creates a number of finder threads equal to the number of
390 CPU cores to perform the directory traversal and file scanning concurrently.
391 The function waits for all finder threads to complete before cleaning up
392 resources and returning.
393 */
394bool init_find(SearchFilters *f) {
395 if (f->flags & LF_USER)
396 f->user_id = f->flags >> 24 & 0xffff;
397 f->include_types = f->flags >> 8 & 0xff;
398 /** suppress file types that aren't included */
399 if (f->include_types)
400 f->suppress_types = f->include_types ^ 0xff;
401 f->permissions = f->flags >> 16 & 0xff;
402 f->suppress_hidden = f->flags & LF_HIDE ? true : false;
403 f->blk = f->suppress_types & FT_BLK ? true : false;
404 f->chr = f->suppress_types & FT_CHR ? true : false;
405 f->dir = f->suppress_types & FT_DIR ? true : false;
406 f->fifo = f->suppress_types & FT_FIFO ? true : false;
407 f->lnk = f->suppress_types & FT_LNK ? true : false;
408 f->reg = f->suppress_types & FT_REG ? true : false;
409 f->sock = f->suppress_types & FT_SOCK ? true : false;
410 f->unknown = f->suppress_types & FT_UNKNOWN ? true : false;
411 int reti = 0;
412 f->reg_flags = REG_EXTENDED;
413 if (f->flags & LF_ICASE)
414 f->reg_flags |= REG_ICASE;
415 if (f->flags & LF_REGEX) {
416 reti = regcomp(&f->compiled_re, f->re, f->reg_flags);
417 if (reti) {
418 fprintf(stderr, "lf: '%s' Invalid pattern\n", f->re);
419 regfree(&f->compiled_re);
420 return false;
421 }
422 }
423 if (f->flags & LF_EXC_REGEX) {
424 reti = regcomp(&f->compiled_ere, f->ere, f->reg_flags);
425 if (reti) {
426 fprintf(stderr, "lf: '%s' Invalid exclude pattern\n", f->ere);
427 regfree(&f->compiled_ere);
428 return false;
429 }
430 }
431 int thread_count = get_nprocs();
432 if (f->debug) {
433 fprintf(stderr, "Starting search in: %s\n", f->base_path);
434 fprintf(stderr, "Using %d threads\n", thread_count);
435 fprintf(stderr, " include_types=%08b\n", f->include_types);
436 fprintf(stderr, "suppress_types=%08b\n", f->suppress_types);
437 if (f->flags & LF_REGEX)
438 fprintf(stderr, "Include regex: %s\n", f->re);
439 if (f->flags & LF_EXC_REGEX)
440 fprintf(stderr, "Exclude regex: %s\n", f->ere);
441 if (f->flags & LF_USER)
442 fprintf(stderr, "User ID filter: %ju\n", f->user_id);
443 if (f->after)
444 fprintf(stderr, "Modified after: %s", ctime(&f->after));
445 if (f->before)
446 fprintf(stderr, "Modified before: %s", ctime(&f->before));
447 if (f->file_size_min)
448 fprintf(stderr, "Minimum file size: %jd bytes\n",
449 (intmax_t)f->file_size_min);
450 if (f->max_depth)
451 fprintf(stderr, "Max depth: %d\n", f->max_depth);
452 if (f->f_hidden)
453 fprintf(stderr, "Include hidden files.\n");
454 if (f->f_lnk_dir)
455 fprintf(stderr, "Follow symbolic links.\n");
456 if (f->include_types) {
457 fprintf(stderr, "Included file types:");
458 if (f->include_types & FT_BLK)
459 fprintf(stderr, " block");
460 if (f->include_types & FT_CHR)
461 fprintf(stderr, " character");
462 if (f->include_types & FT_CHR)
463 fprintf(stderr, " character");
464 if (f->include_types & FT_DIR)
465 fprintf(stderr, " directory");
466 if (f->include_types & FT_FIFO)
467 fprintf(stderr, " pipe");
468 if (f->include_types & FT_LNK)
469 fprintf(stderr, " link");
470 if (f->include_types & FT_REG)
471 fprintf(stderr, " regular");
472 if (f->include_types & FT_SOCK)
473 fprintf(stderr, " socket");
474 if (f->include_types & FT_UNKNOWN)
475 fprintf(stderr, " unknown");
476 fprintf(stderr, "\n");
477 }
478 }
479 enqueue_dir(f->base_path, 0);
480 pthread_t threads[thread_count];
481 for (int i = 0; i < thread_count; i++)
482 pthread_create(&threads[i], NULL, finder, f);
483 pthread_mutex_lock(&queue_mutex);
484 while (!shutdown) {
485 pthread_cond_wait(&cond_var, &queue_mutex);
486 }
487 pthread_mutex_unlock(&queue_mutex);
488 for (int i = 0; i < thread_count; i++)
489 pthread_join(threads[i], NULL);
490 // End Program
491 if (f->flags & LF_REGEX)
492 regfree(&f->compiled_re);
493 if (f->flags & LF_EXC_REGEX)
494 regfree(&f->compiled_ere);
495 free(f);
496 if (reti)
497 return false;
498 return true;
499}
500/** @brief Enqueue a directory path for processing by finder threads.
501 @param path The directory path to enqueue.
502 @param depth The current depth in the directory tree for the given path.
503 @details This function creates a new Node containing the specified directory
504 path and depth, and adds it to the global queue. It uses a mutex to ensure
505 thread-safe access to the queue, and signals finder threads that new work is
506 available. The function also checks for shutdown conditions to prevent
507 enqueuing new work when the program is exiting.
508 */
509void enqueue_dir(const char *path, int depth) {
510 Node *new_node = malloc(sizeof(Node));
511 strnz__cpy(new_node->path, path, PATH_MAX - 1);
512 new_node->depth = depth;
513 new_node->next = NULL;
514
515 pthread_mutex_lock(&queue_mutex);
516 if (tail)
517 tail->next = new_node;
518 else
519 head = new_node;
520 tail = new_node;
521 pthread_cond_signal(&cond_var);
522 pthread_mutex_unlock(&queue_mutex);
523}
524/** @brief Dequeue a directory path for processing by finder threads.
525 @return A pointer to a Node containing the directory path and depth, or NULL
526 if the queue is empty and shutdown has been initiated.
527 @details This function removes and returns the next Node from the global
528 queue. It uses a mutex to ensure thread-safe access to the queue, and waits
529 on a condition variable if the queue is empty. The function also checks for
530 shutdown conditions to allow finder threads to exit gracefully when there is
531 no more work to process.
532 */
533Node *dequeue_dir() {
534 pthread_mutex_lock(&queue_mutex);
535 while (head == NULL && !shutdown) {
536 // Exit condition: No work in queue AND no threads busy
537 if (atomic_load(&active_tasks) == 0) {
538 shutdown = 1;
539 pthread_cond_broadcast(&cond_var);
540 break;
541 }
542 pthread_cond_wait(&cond_var, &queue_mutex);
543 }
544 if (shutdown && head == NULL) {
545 pthread_mutex_unlock(&queue_mutex);
546 return NULL;
547 }
548 Node *temp = head;
549 head = head->next;
550 if (!head)
551 tail = NULL;
552 pthread_mutex_unlock(&queue_mutex);
553 return temp;
554}
555/** @brief Worker thread function to process directories from the queue.
556 @param arg Pointer to the SearchFilters struct containing the options and
557 flags for filtering.
558 @return NULL
559 @details This function continuously dequeues directory paths from the global
560 queue and processes them. For each directory, it lists its contents and
561 applies the specified filters to each file. If a subdirectory is found and it
562 meets the criteria for further searching (e.g., not hidden if hidden files
563 are suppressed, and within max depth), it is enqueued for processing. The
564 function uses atomic operations to track active tasks and condition variables
565 to manage thread synchronization and shutdown when all work is complete.
566 */
567void *finder(void *arg) {
568 SearchFilters *f = (SearchFilters *)arg;
569 struct stat st;
570 // regmatch_t pmatch;
571
572 while (1) {
573 Node *current = dequeue_dir();
574 if (!current)
575 break;
576
577 atomic_fetch_add(&active_tasks, 1);
578
579 DIR *dir = opendir(current->path);
580 if (dir) {
581 struct dirent *entry;
582 while ((entry = readdir(dir)) != NULL) {
583 if (strcmp(entry->d_name, ".") == 0 ||
584 strcmp(entry->d_name, "..") == 0)
585 continue;
586 char full_path[PATH_MAX];
587 strnz__cpy(full_path, current->path, PATH_MAX - 1);
588 strnz__cat(full_path, "/", PATH_MAX - 1);
589 strnz__cat(full_path, entry->d_name, PATH_MAX - 1);
590 int isdir = 0;
591 bool suppress = false;
592 if (entry->d_type == DT_LNK) {
593 if (stat(full_path, &st) == 0) {
594 isdir = S_ISDIR(st.st_mode);
595 if (isdir && f->f_lnk_dir)
596 entry->d_type = DT_DIR;
597 else
598 suppress = true;
599 }
600 }
601 if (entry->d_type == DT_DIR) {
602 if (!((f->flags & LF_HIDE) && (entry->d_name[0] == '.'))) {
603 if (f->max_depth == 0 ||
604 current->depth + 1 < f->max_depth) {
605 enqueue_dir(full_path, current->depth + 1);
606 }
607 }
608 } else {
609 if (!suppress)
610 scan_file(current->path, f, entry);
611 }
612 }
613 closedir(dir);
614 }
615
616 free(current);
617 atomic_fetch_sub(&active_tasks, 1);
618 pthread_cond_broadcast(
619 &cond_var); // Wake up threads checking for idle exit
620 }
621 return NULL;
622}
623/** @brief Scan a single file against the search filters and print if it
624 matches.
625 @param current_path The directory path of the file being scanned.
626 @param f The SearchFilters struct containing the options and flags for
627 filtering.
628 @param dir_st The dirent struct representing the file being scanned.
629 @return true if the file was processed successfully, false if an error
630 occurred.
631 @details This function checks the specified file against the various filters
632 defined in the SearchFilters struct, including file type, regex patterns,
633 ownership, permissions, modification time, and size. If the file matches all
634 criteria, its path is printed to standard output. If any errors occur during
635 processing (e.g., regex compilation failure), an error message is printed to
636 standard error and the function returns false.
637 @note This function is called by finder threads for each file encountered
638 during the directory traversal. It is designed to be thread-safe and
639 efficient, minimizing redundant system calls. For example, it only calls
640 stat() when necessary for filters that require file metadata, and it caches
641 the results to avoid multiple stat() calls for the same file.
642 @note regex matching is opttimized by pre-compiling the regular expressions
643 in init_find() and using regexec() for each file, which is more efficient
644 than compiling the regex for each file. Additionally, the function minimizes
645 calls to regexec() by first checking simpler filters (like file type and
646 hidden status) before performing regex matching, thus avoiding unnecessary

765 suppress_file = false;
766 }
767 if (!suppress_file && f->before) {

769 if (stat(file_spec, &sb) == 0)
770 f_stat = true;
771 }
772 if (f_stat && sb.st_mtime > f->before)
773 suppress_file = true;
774 }
775 if (!suppress_file && f->after) {
776 if (!f_stat) {

778 f_stat = true;
779 }
780 if (f_stat && sb.st_mtime < f->after)
781 suppress_file = true;
782 }

786 f_stat = true;
787 }

789 suppress_file = true;
790 }
791 if (!suppress_file) {

793 printf("%s\n", &file_spec[2]);
794 else
795 printf("%s\n", file_spec);
796 }
