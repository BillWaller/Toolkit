49/** @brief Create and initialize Init structure
50 @ingroup mem
51 @note calloc initializes all fields to zero/nullptr
52 @param argc, argv - arguments
53 idiomatic directory usage:
54 @code
55 init->mapp_msrc description files
56 init->mapp_help help files
57 init->mapp_data in, out, data files
58 init->mapp_user executable scripts
59 init->mapp_bin binary executables
60 @endcode
61 @note Initialize file specifications in priority order:
62 @code
63 1 - Default values
64 2 - Configuration file
65 3 - Environment variables
66 4 - Command line positional arguments
67 5 - Command line option arguments
68 @endcode
69 */
70Init *new_init(int argc, char **argv) {
71 int i = 0;
72 Init *init = calloc(1, sizeof(Init));
73 if (init == nullptr) {
74 abend(-1, "calloc init failed");
75 return nullptr;
76 }
77 init->argv = calloc(MAXARGS + 1, sizeof(char *));
78 if (init->argv == nullptr) {
79 free(init);
80 ssnprintf(em0, MAXLEN - 1, "%s, line: %d, errno: %d", __FILE__,
81 __LINE__ - 4, errno);
82 ssnprintf(em1, MAXLEN - 1, "%s", strerror(errno));
83 ssnprintf(em2, MAXLEN - 1, "view->argv = calloc(%d, %d) failed\n",
84 (MAXARGS + 1), sizeof(char *));
85 display_error(em0, em1, em2, nullptr);
86 abend(-1, "calloc init->argv failed");
87 return nullptr;
88 }
89 init->argc = argc;
90 for (i = 0; i < init->argc; i++) {
91 init->argv[i] = strdup(argv[i]);
92 }
93 init->argv[i] = nullptr;
94 init->sio = (SIO *)calloc(1, sizeof(SIO));
95 if (!init->sio) {
96 abend(-1, "calloc sio failed");
97 return nullptr;
98 }
99 init_cnt++;
100 return init;
101}
102/** @brief Destroy Init structure
103 @ingroup mem
104 @param init structure
105 @returns nullptr
106 */
107Init *destroy_init(Init *init) {
108 int i;
109 if (!init)
110 return nullptr;
111 if (init->sio) {
112 free(init->sio);
113 init->sio = nullptr;
114 }
115 if (init->menu) {
116 init->menu = destroy_menu(init);
117 init->menu = nullptr;
118 }
119 if (init->view) {
120 init->view = destroy_view(init);
121 init->view = nullptr;
122 }
123 if (init->form) {
124 init->form = destroy_form(init);
125 init->form = nullptr;
126 }
127 if (init->pick) {
128 init->pick = destroy_pick(init);
129 init->pick = nullptr;
130 }
131 for (i = 0; i <= init->argc; i++) {
132 if (init->argv[i]) {
133 if (init->argv[i] != nullptr)
134 free(init->argv[i]);
135 init->argv[i] = nullptr;
136 }
137 }
138 if (init->argv) {
139 if (init->argv != nullptr)
140 free(init->argv);
141 init->argv = nullptr;
142 }
143 if (init != nullptr) {
144 free(init);
145 init = nullptr;
146 }
147 init_cnt--;
148 return init;
149}

364/** @brief Verify file specification argument
365 @ingroup mem
366 @param spec - menu->spec, form->spec, etc.
367 @param org_spec - init->._spec | argv[optind]
368 @param dir - init->._. directory
369 @param alt_dir - literal, "~/menuapp/data", etc.
370 @param mode - R_OK, W_OK, X_OK, WC_OK, S_QUIET
371 @note mode is a bitwise OR of the following flags:
372 S_QUIET - suppress error messages
373 WC_OK - write create ok
374 @return bool - true if file verified
375 */
376bool verify_spec_arg(char *spec, char *org_spec, char *dir, char *alt_dir,
377 int mode) {
378 bool f_dir = false;
379 bool f_spec = false;
380 bool f_quote = true;
381 char *s1;
382 char *s2;
383 char s1_s[MAXLEN];
384 char s2_s[MAXLEN];
385 char file_name[MAXLEN];
386 char try_spec[MAXLEN];
387 char idio_spec[MAXLEN];
388
389 if (!org_spec[0])
390 return false;
391 idio_spec[0] = '\0';
392 strnz__cpy(try_spec, org_spec, MAXLEN - 1);
393 f_quote = stripz_quotes(try_spec);
394 s1 = strtok(try_spec, " \t\n");
395 strnz__cpy(s1_s, s1, MAXLEN - 1);
396 s2 = strtok(nullptr, "\n");
397 strnz__cpy(s2_s, s2, MAXLEN - 1);
398 strnz__cpy(file_name, s1, MAXLEN - 1);
399 strnz__cpy(try_spec, file_name, MAXLEN - 1);
400 canonicalize_file_spec(try_spec);
401 if (try_spec[0]) {
402 expand_tilde(try_spec, MAXLEN - 1);
403 if (try_spec[0] == '/') {
404 f_spec = verify_file(try_spec, mode);
405 if (f_quote)
406 /** preserve quotes */
407 strnz__cpy(spec, org_spec, MAXLEN - 1);
408 else
409 strnz__cpy(spec, try_spec, MAXLEN - 1);
410 return f_spec;
411 } else {
412 if (!f_dir && dir[0]) {
413 if (strcmp(dir, "$PATH") == 0) {
414 strnz__cpy(try_spec, file_name, MAXLEN - 1);
415 f_spec = locate_file_in_path(try_spec, file_name);
416 } else {
417 strnz__cpy(try_spec, dir, MAXLEN - 1);
418 expand_tilde(try_spec, MAXLEN - 1);
419 f_dir = verify_dir(try_spec, mode);
420 if (f_dir) {
421 strnz__cat(try_spec, "/", MAXLEN - 1);
422 strnz__cat(try_spec, file_name, MAXLEN - 1);
423 strnz__cpy(idio_spec, try_spec, MAXLEN - 1);
424 if (mode & S_WCOK)
425 f_spec = true;
426 else
427 f_spec = verify_file(idio_spec, mode | S_QUIET);
428 }
429 }
430 }
431 if (!f_spec && alt_dir && alt_dir[0] != '\0') {
432 if (strcmp(alt_dir, "$PATH") == 0) {
433 strnz__cpy(try_spec, file_name, MAXLEN - 1);
434 f_spec = locate_file_in_path(try_spec, file_name);
435 } else {
436 strnz__cpy(try_spec, alt_dir, MAXLEN - 1);
437 expand_tilde(try_spec, MAXLEN - 1);
438 f_dir = verify_dir(try_spec, mode);
439 if (f_dir) {
440 strnz__cat(try_spec, "/", MAXLEN - 1);
441 strnz__cat(try_spec, file_name, MAXLEN - 1);
442 if (mode & S_WCOK)
443 f_spec = true;
444 else
445 f_spec = verify_file(try_spec, mode | S_QUIET);
446 }
447 }
448 }
449 if (!f_spec) {
450 strnz__cpy(try_spec, ".", MAXLEN - 1);
451 strnz__cat(try_spec, "/", MAXLEN - 1);
452 strnz__cat(try_spec, file_name, MAXLEN - 1);
453 f_spec = verify_file(try_spec, mode | S_QUIET);
454 }
455 if (!f_spec && mode == W_OK) {
456 strnz__cpy(try_spec, idio_spec, MAXLEN - 1);
457 FILE *fp = fopen(try_spec, "a");
458 if (fp) {
459 fclose(fp);
460 f_spec = true;
461 }
462 }
463 if (f_quote)
464 /** preserve quotes */
465 strnz__cpy(spec, org_spec, MAXLEN - 1);
466 else if (f_spec)
467 strnz__cpy(spec, try_spec, MAXLEN - 1);
468 if (try_spec[0] == '\0' && idio_spec[0] != '\0')
469 strnz__cpy(spec, idio_spec, MAXLEN - 1);
470 else
471 strnz__cpy(spec, try_spec, MAXLEN - 1);
472 if (s2_s[0] != '\0') {
473 strnz__cat(spec, " ", MAXLEN - 1);
474 strnz__cat(spec, s2_s, MAXLEN - 1);
475 }
476 return f_spec;
477 }
478 }
479 return false;
480}
