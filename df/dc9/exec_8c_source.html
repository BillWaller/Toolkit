126int fork_exec(char **argv) {
127 char tmp_str[MAXLEN];
128 pid_t pid;
129 int status;
130 int rc;
131
132 if (argv[0] == 0) {
133 Perror("fork_exec: missing argument for execvp");
134 return (-1);
135 }
136 capture_curses_tioctl();
137 curs_set(1);
138 sig_dfl_mode();
139 endwin();
140 tmp_str[0] = '\0';
141 pid = fork();
142 if (pid < 0) {
143 sig_prog_mode();
144 keypad(stdscr, true);
145 ssnprintf(tmp_str, sizeof(tmp_str), "fork failed: %s, errno: %d",
146 argv[0], errno);
147 Perror(tmp_str);
148 return (-1);
149 } else if (pid == 0) {
150 restore_shell_tioctl();
151 sig_dfl_mode();
152 execvp(argv[0], argv);
153 fprintf(stderr, "execvp failed: %s, errno: %d\n", argv[0], errno);
154 exit(EXIT_FAILURE);
155 }
156 waitpid(pid, &status, 0);
157 if (WIFEXITED(status)) {
158 rc = WEXITSTATUS(status);
159 } else if (WIFSIGNALED(status)) {
160 ssnprintf(tmp_str, sizeof(tmp_str), "Child process terminated by signal: %d",
161 WTERMSIG(status));
162 Perror(tmp_str);
163 rc = -1;
164 } else {
165 ssnprintf(tmp_str, sizeof(tmp_str), "Child process terminated abnormally");
166 Perror(tmp_str);
167 rc = -1;
168 }
169 reset_prog_mode();
170 restore_wins();
171 return (rc);
172}

184int fork_detach_execvp(char **eargv) {
185 pid_t pid = fork();
186 capture_curses_tioctl();
187 curs_set(1);
188 sig_dfl_mode();
189
190 if (pid < 0) {
191 fprintf(stderr, "First fork failed: %s\n", strerror(errno));
192 exit(EXIT_FAILURE);
193 }
194 if (pid == 0) {
195 if (setsid() < 0) {
196 fprintf(stderr, "Set session ID failed: %s\n", strerror(errno));
197 exit(EXIT_FAILURE);
198 }
199 close(STDIN_FILENO);
200 close(STDOUT_FILENO);
201 close(STDERR_FILENO);
202 int dev_null = open("/dev/null", O_RDWR);
203 if (dev_null != -1) {
204 dup2(dev_null, STDIN_FILENO);
205 dup2(dev_null, STDOUT_FILENO);
206 dup2(dev_null, STDERR_FILENO);
207 if (dev_null > 2) {
208 close(dev_null);
209 }
210 }
211 long max_fd = sysconf(_SC_OPEN_MAX);
212 for (long fd = 3; fd < max_fd; fd++)
213 close(fd);
214 execvp(eargv[0], eargv);
215 perror("execvp failed");
216 exit(EXIT_FAILURE);
217 }
218 restore_curses_tioctl();
219 sig_prog_mode();
220 restore_wins();
221 return 0;
