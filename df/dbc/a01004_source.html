180/** @brief Create and initialize Chyron structure
181 @ingroup Chyron
182 @return pointer to new Chyron structure
183 @details This function allocates memory for a new Chyron structure and
184 initializes the key pointers. Each key pointer is allocated memory for a
185 ChyronKey structure. The Chyron structure is used to manage function key
186 labels and their associated keycodes for mouse click handling in the chyron
187 area of the interface.
188 @note The use of calloc ensures that the allocated memory is initialized to
189 zero, which means that the text for each key will be initialized to an empty
190 string and the keycodes will be initialized to zero. This allows the
191 is_set_chyron_key function to check if a key is set by checking if the first
192 character of the text is not '\0'. If any memory allocation fails, the
193 function will call abend to handle the error and return nullptr.
194 */
195Chyron *new_chyron() {
196 Chyron *chyron = (Chyron *)calloc(1, sizeof(Chyron));
197 if (!chyron) {
198 abend(-1, "calloc chyron failed");
199 return nullptr;
200 }
201 for (int i = 0; i < CHYRON_KEYS; i++)
202 chyron->key[i] = (ChyronKey *)calloc(1, sizeof(ChyronKey));
203 return chyron;
204}
205/** @brief Destroy Chyron structure
206 @ingroup Chyron
207 @param chyron pointer to Chyron structure
208 @return nullptr
209 */
210Chyron *destroy_chyron(Chyron *chyron) {
211 int i;
212
213 if (!chyron)
214 return nullptr;
215 for (i = 0; i < CHYRON_KEYS; i++) {
216 if (chyron->key[i])
217 free(chyron->key[i]);
218 chyron->key[i] = nullptr;
219 }
220 free(chyron);
221 chyron = nullptr;
222 return chyron;
223}

235/** @brief Set chyron key
236 @ingroup Chyron
237 @param chyron structure
238 @param k chyron key index (0-19)
239 @param s chyron key label
240 @param kc chyron key code
241 @details This function sets the label and keycode for a chyron key. The
242 label is copied into the chyron structure, and the keycode is stored for
243 later retrieval when theire is a mouse click. The compile_chyron function
244 uses the keycode values to determine which key was clicked based on the mouse
245 X position. If the label string is empty, the key is unset by setting the
246 first character of the text to '\0'.
247*/
248void set_chyron_key(Chyron *chyron, int k, char *s, int kc) {
249 if (*s != '\0')
250 ssnprintf(chyron->key[k]->text, CHYRON_KEY_MAXLEN - 1, "%s", s);
251 else
252 chyron->key[k]->text[0] = '\0';
253 chyron->key[k]->keycode = kc;
254}
255/** @brief Unset chyron key
256 @ingroup Chyron
257 @param chyron structure
258 @param k chyron_key index
259*/
260void unset_chyron_key(Chyron *chyron, int k) { chyron->key[k]->text[0] = '\0'; }
261/** @brief construct the chyron string from the chyron structure
262 @ingroup Chyron
263 @param chyron
264 @details The chyron string is constructed by concatenating the labels of the
265 set keys, separated by " | ". The end_pos values for each key are set to
266 determine the zones for mouse clicks. When a mouse click occurs, the
267 get_chyron_key function uses the end_pos values to determine which key was
268 clicked based on the X position of the click.
269*/
270void compile_chyron(Chyron *chyron) {
271 int end_pos = 0;
272 int i = 0;
273 chyron->s[0] = '\0';
274 chyron->l = 0;
275 while (i < CHYRON_KEYS) {
276 if (chyron->key[i]->text[0] == '\0') {
277 i++;
278 continue;
279 }
280 if (end_pos == 0)
281 strnz__cat(chyron->s, " ", MAXLEN - 1);
282 else
283 strnz__cat(chyron->s, " | ", MAXLEN - 1);
284 strnz__cat(chyron->s, chyron->key[i]->text, MAXLEN - 1);
285 end_pos = strlen(chyron->s) + 1;
286 chyron->key[i]->end_pos = end_pos;
287 i++;
288 }
289 if (end_pos > 0)
290 chyron->l = strnz__cat(chyron->s, " ", MAXLEN - 1);
291}
292/** @brief Get keycode from chyron
293 @ingroup Chyron
294 @param chyron structure
295 @param x Mouse X position
296 @return Keycode
297 @note This function uses the end_pos values set in compile_chyron
298 to determine which key was clicked
299 @note The chyron functions provide xwgetch() with a mechanism to translate
300 mouse click positions into key codes based on the labels set in the chyron
301 structure. When a mouse click occurs, xwgetch() can call get_chyron_key()
302 with the X position of the click to determine which function key was clicked,
303 allowing for dynamic and customizable function key behavior in the chyron
304 area of the interface.
305*/
306int get_chyron_key(Chyron *chyron, int x) {
307 int i;
308 for (i = 0; chyron->key[i]->end_pos != -1; i++)
309 if (x < chyron->key[i]->end_pos)
310 break;
311 return chyron->key[i]->keycode;
312}
313/** @brief Initialize NCurses and color settings
314 @ingroup window_support
315 @param sio Pointer to SIO struct with terminal and color settings
316 @return true if successful, false if error
317 note This function initializes NCurses and sets up color pairs based on the
318 settings in the SIO struct. It also applies gamma correction to colors.
319 Use this function to initialize NCurses if you don't want NCurses to receive
320 data from the stdin pipe
321 @code
322 1. saves stdin and stdout file descriptors in SIO
323 2. opens a terminal device for NCurses screen IO
324 3. replaces STDERR_FILENO with terminal file descriptor
325 @endcode */
326bool open_curses(SIO *sio) {
327 char tmp_str[MAXLEN];
328 char emsg0[MAXLEN];
329 int rc;
330 RGB rgb;
331
332 if (ttyname_r(STDERR_FILENO, sio->tty_name, sizeof(sio->tty_name)) != 0) {
333 strerror_r(errno, tmp_str, MAXLEN - 1);
334 strnz__cpy(emsg0, "ttyname_r failed ", MAXLEN - 1);
335 strnz__cat(emsg0, tmp_str, MAXLEN - 1);
336 fprintf(stderr, "%s\n", tmp_str);
337 exit(0);
338 }
339 /** save stdin and stdout file descriptors */
340 sio->stdin_fd = dup(STDIN_FILENO);
341 sio->stdout_fd = dup(STDOUT_FILENO);
342 /** open the terminal device for reading and writing */
343 sio->tty_fp = fopen(sio->tty_name, "r+");
344 if (sio->tty_fp == nullptr) {
345 strerror_r(errno, tmp_str, MAXLEN - 1);
346 strnz__cpy(emsg0, "fopen(sio->tty_name) failed ", MAXLEN - 1);
347 strnz__cat(emsg0, tmp_str, MAXLEN - 1);
348 fprintf(stderr, "%s\n", tmp_str);
349 exit(0);
350 }
351 /** Attach the terminal descriptor to the STDERR_FILENO */
352 dup2(fileno(sio->tty_fp), STDERR_FILENO);
353 /** We use SCREEN and newterm because this allows us to */
354 /** specify the terminal FILE */
355 SCREEN *screen = newterm(nullptr, sio->tty_fp, sio->tty_fp);
356 if (screen == nullptr) {
357 strerror_r(errno, tmp_str, MAXLEN - 1);
358 strnz__cpy(emsg0, "newterm failed ", MAXLEN - 1);
359 strnz__cat(emsg0, tmp_str, MAXLEN - 1);
360 fprintf(stderr, "%s\n", tmp_str);
361 exit(0);
362 }
363 set_term(screen);
364 f_curses_open = true;
365 if (!has_colors()) {
366 destroy_curses();
367 abend(-1, "Terminal color support required");
368 }
369 start_color();
370 if (!can_change_color()) {
371 destroy_curses();
372 fprintf(stderr, "Terminal cannot change colors\n");
373 fprintf(stderr, "Check TERM environment variable\n");
374 fprintf(stderr, "Check terminfo for missing \"ccc\"\n");
375 abend(-1, "fatal error");
376 }
377 init_clr_palette(sio);
378 /** Set gamma correction values */
379 /** These are read from ~/.minitrc */
380 /** We need these values when initializing colors */
381 RED_GAMMA = sio->red_gamma;
382 GREEN_GAMMA = sio->green_gamma;
383 BLUE_GAMMA = sio->blue_gamma;
384 GRAY_GAMMA = sio->gray_gamma;
385
386 cp_norm = get_clr_pair(CLR_FG, CLR_BG);
387 cp_reverse = get_clr_pair(CLR_BG, CLR_FG);
388 cp_box = get_clr_pair(CLR_BO, CLR_BG);
389 cp_ln = get_clr_pair(CLR_LN, CLR_LN_BG);
390
391 CCC_NORM = mkccc(cp_norm);
392 CCC_REVERSE = mkccc(cp_reverse);
393 CCC_BOX = mkccc(cp_box);
394 CCC_LN = mkccc(cp_ln);
395 noecho();
396 cbreak();
397 keypad(stdscr, true);
398 idlok(stdscr, false);
399 idcok(stdscr, false);
400#ifdef DEBUG
401 immedok(stdscr, true);
402#endif
403 wbkgrndset(stdscr, &CCC_NORM);
404 rc = extended_color_content(CLR_FG, &rgb.r, &rgb.g, &rgb.b);
405 rc = extended_color_content(CLR_BG, &rgb.r, &rgb.g, &rgb.b);
406 rc = extended_color_content(CLR_BO, &rgb.r, &rgb.g, &rgb.b);
407 rc = extended_color_content(CLR_LN, &rgb.r, &rgb.g, &rgb.b);
408 rc = extended_color_content(CLR_LN_BG, &rgb.r, &rgb.g, &rgb.b);
409 if (rc == ERR) {
410 destroy_curses();
411 abend(-1, "extended_color_content failed");
412 }
413 return sio;
414}

477/** @brief Convert RGB color to XTerm 256 color index
478 @ingroup color_management
479 @param rgb RGB color
480 @return XTerm 256 color index
481 note This function converts an RGB color to the nearest XTerm 256 color
482 index. It first checks if the color is a shade of gray, and if so, it uses
483 the gray ramp. Otherwise, it calculates the nearest color in the 6x6x6 color
484 cube. */
485int rgb_to_xterm256_idx(RGB *rgb) {
486 if (rgb->r == rgb->g && rgb->g == rgb->b) {
487 if (rgb->r < 8)
488 return 16;
489 if (rgb->r > 248)
490 return 231;
491 return ((rgb->r - 8) / 10) + 231;
492 } else {
493 int r_index = (rgb->r < 45) ? 0 : (rgb->r - 60) / 40 + 1;
494 int g_index = (rgb->g < 45) ? 0 : (rgb->g - 60) / 40 + 1;
495 int b_index = (rgb->b < 45) ? 0 : (rgb->b - 60) / 40 + 1;
496 return 16 + (36 * r_index) + (6 * g_index) + b_index;
497 }
498}

500/** @brief Convert XTerm 256 color index to RGB color
501 @ingroup color_management
502 @param idx XTerm 256 color index
503 @return RGB color
504 note This function converts an XTerm 256 color index to an RGB color. It
505 first checks if the index is in the standard 16 colors, then checks if it's
506 in the 6x6x6 color cube, and finally checks if it's in the gray ramp. */
507RGB xterm256_idx_to_rgb(int idx) {
508 /** Convert XTerm 256 color index to RGB
509 @param idx - XTerm 256 color index
510 @return RGB struct */
511 RGB rgb;
512 if (idx > 255)
513 idx = 255;
514 if (idx < 0)
515 idx = 0;
516 rgb.r = rgb.g = rgb.b = 0;
517 if (idx < 16) {
518 rgb.r = StdColors[idx].r;
519 rgb.g = StdColors[idx].g;
520 rgb.b = StdColors[idx].b;
521 } else if (idx >= 16 && idx <= 231) {
522 idx -= 16;
523 rgb.r = (idx / 36) % 6 * 51;
524 rgb.g = (idx / 6) % 6 * 51;
525 rgb.b = (idx % 6) * 51;
526 } else if (idx >= 232 && idx <= 255) {
527 int gray = (idx - 232) * 11;
528 rgb.r = rgb.g = rgb.b = gray;
529 }
530 return rgb;
531}

533/** @brief Apply gamma correction to RGB color
534 @ingroup color_management
535 @param rgb Pointer to RGB color
536 @note This function modifies the RGB color in place. It applies gamma
537 correction to the RGB color based on the gamma values set in the SIO struct.
538 If the color is a shade of gray, it applies the gray gamma correction.
539 Otherwise, it applies the individual red, green, and blue gamma corrections.
540 */
541void apply_gamma(RGB *rgb) {
542 if (rgb->r == rgb->g && rgb->r == rgb->b) {
543 if (GRAY_GAMMA > 0.0f && GRAY_GAMMA != 1.0f) {
544 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / GRAY_GAMMA) * 255.0f);
545 rgb->g = rgb->r;
546 rgb->b = rgb->r;
547 }
548 return;
549 }
550 if (rgb->r != 0 && RED_GAMMA > 0.0f && RED_GAMMA != 1.0f)
551 rgb->r = (int)(pow((rgb->r / 255.0f), 1.0f / RED_GAMMA) * 255.0f);
552 if (rgb->g != 0 && GREEN_GAMMA > 0.0f && GREEN_GAMMA != 1.0f)
553 rgb->g = (int)(pow((rgb->g / 255.0f), 1.0f / GREEN_GAMMA) * 255.0f);
554 if (rgb->b != 0 && BLUE_GAMMA > 0.0f && BLUE_GAMMA != 1.0f)
555 rgb->b = (int)(pow((rgb->b / 255.0f), 1.0f / BLUE_GAMMA) * 255.0f);
556}

1145/** @brief Wrapper for wgetch that handles signals and mouse events
1146 @ingroup window_support
1147 @param win Pointer to window
1148 @param chyron Pointer to chyron for handling chyron line clicks
1149 @return Key code or ERR if interrupted by signal
1150 @note This, of course, will be expanded into an event loop for message
1151 queuing
1152 @details Get mouse event and check if it's a left click or double click. If
1153 the click is outside the window, ignore it. If it's on the chyron line, get
1154 the corresponding key command. Otherwise, store the click coordinates as
1155 click_y and click_x for later use. */
1156int xwgetch(WINDOW *win, Chyron *chyron) {
1157 int c;
1158 MEVENT event;
1159 mousemask(BUTTON1_CLICKED | BUTTON1_DOUBLE_CLICKED | BUTTON4_PRESSED |
1160 BUTTON5_PRESSED,
1161 nullptr);
1162 event.y = event.x = -1;
1163 click_y = click_x = -1;
1164 halfdelay(1);
1165 tcflush(2, TCIFLUSH);
1166 do {
1167 c = wgetch(win);
1168 if (sig_received != 0) {
1169 if (handle_signal(sig_received))
1170 c = display_error(em0, em1, em2, nullptr);
1171 if (c == 'q' || c == KEY_F(9))
1172 exit(EXIT_FAILURE);
1173 continue;
1174 }
1175 if (c == ERR)
1176 continue;
1177 if (c == KEY_MOUSE) {
1178 if (getmouse(&event) != OK) {
1179 c = 0;
1180 continue;
1181 }
1182 if (event.bstate & BUTTON4_PRESSED)
1183 return (KEY_UP);
1184 else if (event.bstate & BUTTON5_PRESSED)
1185 return (KEY_DOWN);
1186 if (event.bstate & BUTTON1_CLICKED ||
1187 event.bstate & BUTTON1_DOUBLE_CLICKED) {
1188 if (!wenclose(win, event.y, event.x)) {
1189 c = 0;
1190 continue;
1191 }
1192 wmouse_trafo(win, &event.y, &event.x, false);
1193 if (event.y < 0 || event.x < 0 || event.x >= getmaxx(win) ||
1194 event.y >= getmaxy(win)) {
1195 c = 0;
1196 continue;
1197 }
1198 click_y = event.y;
1199 click_x = event.x;
1200 if (chyron && event.y == getmaxy(win) - 1)
1201 return get_chyron_key(chyron, event.x);
1202 else
1203 return KEY_MOUSE;
1204 }
1205 }
1206 } while (c == ERR);
1207 return c;
