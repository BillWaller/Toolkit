19/** @brief Translate a raw NCurses key code into a UiKey.

23static UiKey translate_key(int ch) {

26 return UIKEY_NONE;
27 case '\n':
28 case '\r':
29 return UIKEY_ENTER;
30 case 27:
31 return UIKEY_ESCAPE;
32 case '\t':
33 return UIKEY_TAB;

36 return UIKEY_BTAB;

53 return UIKEY_NPAGE;

108/** @brief Read input events
109 @param s Surface - may contain multiple widgets (planes/panels)
110 @param w Widget index (A surface may contain multilple widgets)
111 @param chyron A zoned group of Actions or Commands
112 @param ev Event data structure
113 @param timeout_ms Timeout in milliseconds (0 for no wait, -1 for infinite wait).
114 @return 0 on success, -1 if @p ev is NULL.
115*/
116int ui_get_event(UiSurface *s, ss_t w, UiChyron *chyron, UiEvent *ev, int timeout_ms) {
117 if (!ev)
118 return -1;
119 int ch;
120 ui_flush_input();
121 ui_render();
122 if (timeout_ms < 0)
123 timeout_ms = -1; // Block waiting for input
124 if (timeout_ms > 0) {
125 // ------------------------------------------------------------
126 // Poll for input on the TTY and any registered external files
127 // ------------------------------------------------------------
128 struct pollfd fds[1 + MAX_EXT_FDS];
129 int token_id;
130 int reg_idx;
131 memset(fds, 0, sizeof(fds));
132 fds[0].fd = ui->tty_fd;
133 fds[0].events = POLLIN; // also POLLERR | POLLHUP | POLLNVAL
134 nfds_t nfds = 1;
135 nfds_t ext_start_idx = nfds;
136 for (nfds_t i = 0; i < ui_ctx.ext_nfds; i++) {
137 fds[nfds].fd = ui_ctx.ext_fds[i].fd;
138 fds[nfds].events = POLLIN; // also POLLERR | POLLHUP | POLLNVAL
139 nfds++;
140 }
141 int poll_rc = poll(fds, nfds, timeout_ms);
142 if (poll_rc < 0) {
143 if (errno == EINTR)
144 return 0; // continue
145 return -1;
146 }
147 if (poll_rc == 0) { // timed out with no input
148 token_id = UIKEY_TIMEOUT;
149 return token_id;
150 }
151 bool ui_ready = (fds[0].revents & POLLIN);
152 if (!ui_ready) {
153 for (nfds_t i = ext_start_idx; i < nfds; i++) {
154 if (fds[i].revents & (POLLIN | POLLHUP | POLLERR)) {
155 reg_idx = i - ext_start_idx;
156 token_id = ui_ctx.ext_fds[reg_idx].token_id;
157 return token_id;
158 }
159 }
160 } else
161 timeout_ms = -1;
162 }
163 memset(ev, 0, sizeof(*ev));
164 mousemask(ALL_MOUSE_EVENTS | REPORT_MOUSE_POSITION, NULL);
165 wtimeout(s->mwin[w], timeout_ms);
166 do {
167 curs_set(2);
168 ch = wgetch(s->mwin[w]);
169 if (sig_received != 0) {
170 if (handle_signal(sig_received))
171 ch = ui_display_error(em0, em1, em2, nullptr);
172 if (ch == 'q' || ch == 'Q' || ch == UIKEY_F09)
173 exit(EXIT_FAILURE);
174 sig_received = 0;
175 return 0;
176 }
177 if (timeout_ms > 0 && ch == ERR) {
178 ch = 0;
179 break;
180 }
181 if (ch == ERR)
182 continue;
183 ev->key = translate_key(ch);
184 if (ev->key == UIKEY_CHAR)
185 ev->ch = (uint32_t)ch; /* Unicode codepoint */
186 else if (ev->key == UIKEY_MOUSE) {
187 MEVENT me;
188 if (getmouse(&me) != OK) {
189 ch = 0;
190 continue;
191 }
192 ev->bstate = me.bstate;
193 ev->y = me.y;
194 ev->x = me.x;
195 if (me.bstate == UIKEY_BUTTON4_PRESSED)
196 ev->mouse_action = UIKEY_SCROLL_UP;
197 else if (me.bstate == UIKEY_BUTTON5_PRESSED)
198 ev->mouse_action = UIKEY_SCROLL_DOWN;
199 else if (me.bstate == UIKEY_BUTTON1_CLICKED)
200 ev->mouse_action = UIKEY_BUTTON1_CLICKED;
201 for (ss_t i = WIN; i < SUB_SFC_MAX; i++) {
202 if (s->mwin[i] != NULL &&
203 wenclose(s->mwin[i], me.y, me.x) &&
204 wmouse_trafo(s->mwin[i], &me.y, &me.x, false)) {
205 ch = ev->mouse_action;
206 ev->bstate = me.bstate;
207 ev->in_win = i;
208 ev->y = me.y;
209 ev->x = me.x;
210 break;
211 }
212 }
213 if (ev->in_win == w)
214 ev->mouse_inside = true;
215 if (chyron && ev->mouse_action == UIKEY_BUTTON1_CLICKED) {
216 if (ev->in_win == chyron->w && ev->y == chyron->y) {
217 ev->key = ui_get_chyron_key(chyron, ev->x);
218 return ev->key;
219 } else
220 return 0;
221 }
222 }
223 } while (ch == ERR);
224 curs_set(0);
225 return ch;
226}
227/** @brief
228 * Wait for an input event on @p target (or stdscr if NULL) without mouse support.
229 @param ui UI runtime context (unused — event comes from the window).
230 @param target Surface to read from, or NULL for stdscr.
231 @param ev Output UiEvent structure.
232 @return 0 on success, -1 if @p ev is NULL.
233*/
234int ui_get_event_no_mouse(UiSurface *s, ss_t w, UiEvent *ev) {
235 int ch;
236 mousemask(0, NULL);
237 curs_set(2);
238 ui_flush_input();
239 cbreak();
240 do {
241 curs_set(2);
242 ch = wgetch(s->mwin[w]);
243 curs_set(0);
244 ev->key = translate_key(ch);
245 if (ev->key == UIKEY_CHAR) {
246 ev->ch = (uint32_t)ch;
247 break;
248 }
249 } while (ch == ERR);
250 curs_set(0);
251 return ch;
252}
