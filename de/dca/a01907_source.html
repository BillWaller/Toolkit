811/** @brief Scroll the contents of the specified window up or down by the specified
812 * number of lines.
813 * @param s The surface containing the window to scroll.
814 * @param w The index of the window to scroll.
815 * @param r The number of lines to scroll. Positive values scroll up, negative
816 * values scroll down.
817 * @return 0 on success, -1 on error.
818 *
819 * If r is positive, the contents of the plane are moved up, leaving blank lines
820 * at the bottom. In the view application, the view scope is moved toward end of
821 * file (eof). - scope_toward_eof()
822 *
823 * @note ui_wscrl(-n) - If r is negative, the contents of the plane are moved down,
824 * leaving blank lines at the top. In the view application, the view scope is
825 * moved toward beginning of file (bof). - scope_toward_bof()
826 */
827int ui_wscrl(UiSurface *s, ss_t w, int r) {
828 uint rows, cols;
829 ncplane_dim_yx(s->mplane[w], &rows, &cols);
830
831 if (r > 0) {
832 // Scroll up (toward EOF): notcurses has this natively
833 ncplane_scrollup(s->mplane[w], r);
834 return 0;
835 }
836
837 // Scroll down (toward BOF): shift content down by |r| rows
838 r = -r;
839 if ((uint)r >= rows) {
840 ncplane_erase(s->mplane[w]);
841 return 0;
842 }
843
844 // Copy rows bottom-up to avoid overwriting source before reading
845 for (int src_row = (int)(rows - r) - 1; src_row >= 0; src_row--) {
846 int dst_row = src_row + r;
847 for (uint col = 0; col < cols; col++) {
848 nccell c = NCCELL_TRIVIAL_INITIALIZER;
849 ncplane_at_yx_cell(s->mplane[w], src_row, col, &c);
850 ncplane_putc_yx(s->mplane[w], dst_row, col, &c);
851 nccell_release(s->mplane[w], &c);
852 }
853 }
854 // Blank the top r rows
855 for (int row = 0; row < r; row++) {
856 for (uint col = 0; col < cols; col++) {
857 ncplane_putchar_yx(s->mplane[w], row, col, ' ');
858 }
859 }
860 return 0;
861}
