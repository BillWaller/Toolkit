27/* -------------------------------------------------------------------------
28 Internal color-pair cache — independent of dwin.c.
29 Keeps a simple linear map of (fg_color_idx, bg_color_idx) → pair_id.
30 ------------------------------------------------------------------------- */
31#define NC_MAX_COLORS 512
32#define NC_MAX_PAIRS 512
33
34typedef struct {
35 int fg;
36 int bg;
37 int pair_id;
38} NcColorCache;
39
40static NcColorCache nc_color_cache[NC_MAX_PAIRS];
41static int nc_color_cache_cnt = 0;
42static int nc_color_cnt = 0; /* colors allocated via init_extended_color */
43
44/** Translate an 8-bit channel value (0-255) to the 1000-based NCurses scale. */
45static inline int nc_scale_1000(uint8_t v) {
46 return (int)((v * 1000) / 255);
47}
48
49/**
50 * Allocate (or find) an extended NCurses color index for an RGB triple.
51 * Returns the color index, or -1 on failure.
52 */
53static int nc_alloc_color(uint8_t r, uint8_t g, uint8_t b) {
54 int r1000 = nc_scale_1000(r);
55 int g1000 = nc_scale_1000(g);
56 int b1000 = nc_scale_1000(b);
57 /* Search for an existing allocation. */
58 for (int i = 0; i < nc_color_cnt && i < NC_MAX_COLORS; i++) {
59 int cr, cg, cb;
60 extended_color_content(i, &cr, &cg, &cb);
61 if (cr == r1000 && cg == g1000 && cb == b1000)
62 return i;
63 }
64 if (nc_color_cnt >= NC_MAX_COLORS || nc_color_cnt >= COLORS)
65 return -1;
66 init_extended_color(nc_color_cnt, r1000, g1000, b1000);
67 return nc_color_cnt++;
68}
69/**
70 * Find or allocate an NCurses extended color pair for (fg, bg).
71 * Returns the pair index.
72 */
73static int nc_alloc_pair(int fg, int bg) {
74 for (int i = 0; i < nc_color_cache_cnt; i++) {
75 if (nc_color_cache[i].fg == fg && nc_color_cache[i].bg == bg)
76 return nc_color_cache[i].pair_id;
77 }
78 if (nc_color_cache_cnt >= NC_MAX_PAIRS)
79 return 0;
80 int pair_id = nc_color_cache_cnt + 1;
81 init_extended_pair(pair_id, fg, bg);
82 nc_color_cache[nc_color_cache_cnt].fg = fg;
83 nc_color_cache[nc_color_cache_cnt].bg = bg;
84 nc_color_cache[nc_color_cache_cnt].pair_id = pair_id;
85 nc_color_cache_cnt++;
86 return pair_id;
87}

97UiRuntime *ui_init(const UiConfig *cfg) {
98 UiRuntime *ui = calloc(1, sizeof(*ui));
99 if (!ui)
100 return NULL;
101
102 /* Determine the TTY device to use. */
103 char tty_name[XLEN];
104 if (cfg && cfg->tty_path) {
105 strncpy(tty_name, cfg->tty_path, sizeof(tty_name) - 1);
106 tty_name[sizeof(tty_name) - 1] = '\0';
107 } else {
108 if (ttyname_r(STDERR_FILENO, tty_name, sizeof(tty_name)) != 0) {
109 free(ui);
110 return NULL;
111 }
112 }
113
114 ui->tty_fp = fopen(tty_name, "r+");
115 if (!ui->tty_fp) {
116 free(ui);
117 return NULL;
118 }
119
120 ui->screen = newterm(NULL, ui->tty_fp, ui->tty_fp);
121 if (!ui->screen) {
122 fclose(ui->tty_fp);
123 free(ui);
124 return NULL;
125 }
126 set_term(ui->screen);
127
128 /* Keep legacy globals in sync when libcm (dwin.c) is linked. */
129#ifdef UAL_LEGACY_COMPAT
130 screen = ui->screen;
131 tty_fp = ui->tty_fp;
132 f_curses_open = true;
133#endif
134
135 if (!has_colors() || !can_change_color()) {
136 ui_shutdown(ui);
137 return NULL;
138 }
139 start_color();
140 use_default_colors();
141 cbreak();
142 noecho();
143 keypad(stdscr, TRUE);
144
145 if (cfg) {
146 ui->mouse_enabled = cfg->enable_mouse;
147 ui->alt_screen = cfg->enable_alt_screen;
148 ui->cursor_visible = cfg->cursor_visible;
149 } else {
150 ui->cursor_visible = false;
151 }
152 ui->panel_main = new_panel(stdscr);
153
154 wbkgrnd(stdscr, &CC_NT);
155 if (ui->mouse_enabled)
156 mousemask(ALL_MOUSE_EVENTS | REPORT_MOUSE_POSITION, NULL);
157
158 curs_set(ui->cursor_visible ? 1 : 0);
159 getmaxyx(stdscr, ui->lines, ui->cols);
160
161#ifdef UAL_LEGACY_COMPAT
162 sfc_ptr = -1;
163#endif
164 curs_set(ui->cursor_visible ? 1 : 0);
165 return ui;
166}

168void ui_shutdown(UiRuntime *ui) {
169 if (!ui)
170 return;
171 if (sfc_ptr >= 0) {
172 for (int i = sfc_ptr; i >= 0; i--) {
173 UiSurface *sfc = ui_surface[i];
174 if (sfc) {
175 ui_surface_destroy(sfc);
176 sfc = NULL;
177 }
178 }
179 }
180 sfc_ptr = -1;
181 endwin();
182 if (ui->screen) {
183 delscreen(ui->screen);
184#ifdef UAL_LEGACY_COMPAT
185 screen = NULL;
186#endif
187 }
188 if (ui->tty_fp) {
189 fclose(ui->tty_fp);
190#ifdef UAL_LEGACY_COMPAT
191 tty_fp = NULL;
192#endif
193 }
194#ifdef UAL_LEGACY_COMPAT
195 f_curses_open = false;
196#endif
197 free(ui);
198}

591 if ((ssize_t)n > 0)

593 } else {
594 style.wc = L' ';

609UiStyle *ui_style_from_cch(const cchar_t *cch) {
610 UiStyle *style = calloc(1, sizeof(*style));
611 if (!style)
612 return NULL;
613 wchar_t wc[2] = {L'\0', L'\0'};
614 attr_t attrs;
615 short cpx;
616 int fg, bg;
617 getcchar(cch, wc, &attrs, &cpx, NULL);
618 style->attrs = attrs;
619 extended_pair_content(cpx, &fg, &bg);
620 int r, g, b;
621 extended_color_content(fg, &r, &g, &b);
622 style->fg.r = (uint8_t)((r * 255) / 1000);
623 style->fg.g = (uint8_t)((g * 255) / 1000);
624 style->fg.b = (uint8_t)((b * 255) / 1000);
625 extended_color_content(bg, &r, &g, &b);
626 style->bg.r = (uint8_t)((r * 255) / 1000);
627 style->bg.g = (uint8_t)((g * 255) / 1000);
628 style->bg.b = (uint8_t)((b * 255) / 1000);
629 return style;
630}

632cchar_t ui_style_to_cch(const UiStyle *style, const char *c) {
633 attr_t attrs = 0;
634 uint32_t cpx = 0;
635
636 if (style) {
637 attrs = style->attrs;
638 int fg = nc_alloc_color(style->fg.r, style->fg.g, style->fg.b);
639 int bg = nc_alloc_color(style->bg.r, style->bg.g, style->bg.b);
640 if (fg >= 0 && bg >= 0)
641 cpx = (uint32_t)nc_alloc_pair(fg, bg);
642 }
643 /* Encode the first character (or space) as a cchar_t. */
644 mbstate_t mbst;
645 memset(&mbst, 0, sizeof(mbst));
646 wchar_t wstr[2] = {L' ', L'\0'};
647 if (c && *c) {
648 wchar_t wc = L'\0';
649 size_t n = mbrtowc(&wc, c, MB_CUR_MAX, &mbst);
650 if ((ssize_t)n > 0)
651 wstr[0] = wc;
652 }
653 cchar_t cc;
654 setcchar(&cc, wstr, attrs, (short)cpx, NULL);
655 return cc;
656}
